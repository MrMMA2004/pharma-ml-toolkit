   631→def _fail(message: str) -> None:
   632→    raise SystemExit(f"DRIVER FAILED: {message}")

   635→def apply_edits() -> int:
   636→    """Replace the fabrication cells + patch the prose.  Idempotent."""
   637→    changed = 0
   638→    notebooks: dict[str, object] = {}
   639→    for name in NOTEBOOKS:
   640→        notebooks[name] = nbformat.read(EXAMPLES / name, as_version=4)
   641→
   642→    for name, index, anchor, new_source, done in CODE_EDITS:
   643→        nb = notebooks[name]
   644→        cell = nb.cells[index]
   645→        source = cell.source
   646→        if done in source:
   647→            continue  # already converted (idempotent re-run)
   648→        if cell.cell_type != "code":
   649→            _fail(f"{name} cell {index} is {cell.cell_type}, expected code")
   650→        anchors = anchor if isinstance(anchor, tuple) else (anchor,)
   651→        if not any(a in source for a in anchors):
   652→            _fail(f"{name} cell {index} lost its anchor {anchors!r} -- "
   653→                  "cell numbering drifted; update the driver's edit table")
   654→        compile(new_source, f"{name}#{index}", "exec")  # never write broken code
   655→        cell.source = new_source
   656→        cell.outputs = []
   657→        cell.execution_count = None
   658→        changed += 1
   659→
   660→    for name, index, old, new in PROSE_EDITS:
   661→        nb = notebooks[name]
   662→        cell = nb.cells[index]
   663→        if new in cell.source:
   664→            continue  # already patched
   665→        if cell.cell_type != "markdown":
   666→            _fail(f"{name} cell {index} is {cell.cell_type}, expected markdown")
   667→        if old not in cell.source:
   668→            _fail(f"{name} cell {index} lost its prose anchor -- "
   669→                  f"update the driver's PROSE_EDITS entry starting {old[:48]!r}")
   670→        cell.source = cell.source.replace(old, new, 1)
   671→        changed += 1
   672→
   673→    for name, nb in notebooks.items():
   674→        nbformat.write(nb, EXAMPLES / name)
   675→    return changed

   678→def guard() -> None:
   679→    """Zero banned fabrication tokens across all six executed notebooks."""
   680→    hits = []
   681→    for name in NOTEBOOKS:
   682→        nb = nbformat.read(EXAMPLES / name, as_version=4)
   683→        for i, cell in enumerate(nb.cells):
   684→            if cell.cell_type != "code":
   685→                continue
   686→            for token in BANNED_TOKENS:
   687→                if token in cell.source:
   688→                    hits.append((name, i, token))
   689→    if hits:
   690→        for name, i, token in hits:
   691→            print(f"  BANNED {name} cell {i}: {token!r}")
   692→        _fail(f"{len(hits)} banned fabrication token hit(s) remain")
   693→    print(f"guard: 0 hits for all {len(NOTEBOOKS) * len(BANNED_TOKENS)} "
   694→          "notebook x token combinations")

   697→def execute_notebooks(names: tuple[str, ...] = NOTEBOOKS) -> None:
   698→    """Re-run notebooks top-to-bottom with nbclient (cwd=examples/)."""
   699→    from nbclient import NotebookClient
   700→
   701→    for name in names:
   702→        path = EXAMPLES / name
   703→        nb = nbformat.read(path, as_version=4)
   704→        for cell in nb.cells:
   705→            if cell.cell_type == "code":
   706→                cell.outputs = []
   707→                cell.execution_count = None
   708→        started = time.perf_counter()
   709→        client = NotebookClient(
   710→            nb,
   711→            timeout=2400,
   712→            kernel_name="python3",
   713→            resources={"metadata": {"path": str(EXAMPLES)}},
   714→        )
   715→        client.execute()
   716→        nbformat.write(nb, path)
   717→
   718→        code_cells = [c for c in nb.cells if c.cell_type == "code"]
   719→        executed = sum(1 for c in code_cells if c.get("execution_count"))
   720→        errors = [o for c in code_cells for o in c.get("outputs", [])
   721→                  if o.output_type == "error"]
   722→        figures = sum(1 for c in code_cells for o in c.get("outputs", [])
   723→                      if o.get("data", {}).get("image/png"))
   724→        if errors or executed != len(code_cells):
   725→            _fail(f"{name}: {len(errors)} error output(s), "
   726→                  f"{executed}/{len(code_cells)} cells executed")
   727→        print(f"executed {name}: {len(code_cells)} code cells, {figures} figures, "
   728→              f"{time.perf_counter() - started:.0f} s")
