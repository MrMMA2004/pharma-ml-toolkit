    37→def atom_line(
    38→    serial: int,
    39→    name: str,
    40→    resname: str,
    41→    chain: str,
    42→    resseq: int,
    43→    x: float,
    44→    y: float,
    45→    z: float,
    46→    element: str,
    47→    record: str = "ATOM  ",
    48→    altloc: str = " ",
    49→) -> str:
    50→    """Column-exact PDB ATOM/HETATM record (altloc at col 17)."""
    51→    return (
    52→        f"{record}{serial:5d} {name:>4s}{altloc}{resname:>3s} {chain}"
    53→        f"{resseq:4d}    {x:8.3f}{y:8.3f}{z:8.3f}"
    54→        f"{1.0:6.2f}{0.0:6.2f}          {element:>2s}"
    55→    )
    56→
    57→
    58→def pdbqt_line(serial: int, name: str, resname: str, chain: str, resseq: int,
    59→               charge: float, atom_type: str) -> str:
    60→    """Column-exact PDBQT record (charge at cols 71-76, type last)."""
    61→    return (
    62→        f"ATOM  {serial:5d} {name:>4s} {resname:>3s} {chain}{resseq:4d}    "
    63→        f"{1.0:8.3f}{2.0:8.3f}{3.0:8.3f}{1.0:6.2f}{0.0:6.2f}"
    64→        f"    {charge:+6.3f}{atom_type:>3s}"
    65→    )

   305→class TestProteinPdbqtPassthrough:
   306→    def _receptor(self, tmp_path: Path, extra: list[str] | None = None) -> Path:
   307→        lines = [
   308→            "REMARK  prepared receptor",
   309→            pdbqt_line(1, "CA", "ALA", "A", 1, 0.24, "C"),
   310→            pdbqt_line(2, "O", "ALA", "A", 1, -0.27, "OA"),
   311→            pdbqt_line(3, "HD", "ALA", "A", 1, 0.42, "HD"),
   312→            "TER",
   313→            "END",
   314→        ]
   315→        if extra:
   316→            lines = lines[:1] + extra + lines[1:]
   317→        path = tmp_path / "r.pdbqt"
   318→        path.write_text("\n".join(lines) + "\n", encoding="utf-8")
   319→        return path

   328→        assert report.residues == 1
   329→        assert report.chains == ["A"]
   330→        assert report.charge_min == pytest.approx(-0.27)
   331→        assert report.charge_max == pytest.approx(0.42)
   332→        text = (tmp_path / "out.pdbqt").read_text()
   333→        assert "+0.240" in text  # charges preserved verbatim
