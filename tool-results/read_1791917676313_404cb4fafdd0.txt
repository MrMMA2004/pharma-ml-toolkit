    88→/* 1. Creatinine clearance (Cockcroft–Gault)                           */
    89→/* ------------------------------------------------------------------ */
    90→
    91→const CRCL_BANDS: Array<{ min: number; label: string; className: string }> = [
    92→  { min: 90, label: "Normal renal function", className: "bg-primary/10 text-primary border-primary/30" },
    93→  { min: 60, label: "Mild reduction", className: "bg-teal-500/10 text-teal-700 dark:text-teal-400 border-teal-500/30" },
    94→  { min: 30, label: "Moderate reduction", className: "bg-amber-500/10 text-amber-700 dark:text-amber-400 border-amber-500/30" },
    95→  { min: 0, label: "Severe reduction", className: "bg-rose-500/10 text-rose-700 dark:text-rose-400 border-rose-500/30" },
    96→];
    97→
    98→function CrclCard() {
    99→  const [age, setAge] = useState("68");
   100→  const [weight, setWeight] = useState("70");
   101→  const [scr, setScr] = useState("1.1");
   102→  const [sex, setSex] = useState<"male" | "female">("male");
   103→
   104→  const a = parsePositiveNumber(age);
   105→  const w = parsePositiveNumber(weight);
   106→  const s = parsePositiveNumber(scr);
   107→
   108→  const err = (parsed: number | null, fallback: string) =>
   109→    parsed === null ? (fallback.length === 0 ? "Enter a number" : "Must be a positive number") : null;
   110→
   111→  const valid = a !== null && w !== null && s !== null;
   112→  const crcl = valid ? ((140 - (a as number)) * (w as number)) / (72 * (s as number)) : null;
   113→  const female = sex === "female" ? 0.85 : 1;
   114→  const crclFinal = crcl !== null ? crcl * female : null;
   115→
   116→  const band =
   117→    crclFinal !== null ? (CRCL_BANDS.find((b) => crclFinal >= b.min) ?? CRCL_BANDS[3]) : null;
   118→
   119→  return (
   120→    <CalcCard
   121→      icon={Activity}
   122→      title="Creatinine clearance"
   123→      description="Cockcroft–Gault estimate of renal function"
   124→    >
   125→      <div className="grid grid-cols-2 gap-3">
   126→        <NumberField id="crcl-age" label="Age" unit="years" value={age} onChange={setAge} error={err(a, age)} placeholder="68" />
   127→        <NumberField id="crcl-wt" label="Weight" unit="kg" value={weight} onChange={setWeight} error={err(w, weight)} placeholder="70" />
   128→        <NumberField id="crcl-scr" label="Serum creatinine" unit="mg/dL" value={scr} onChange={setScr} error={err(s, scr)} placeholder="1.1" />
   129→        <div className="space-y-1.5">
   130→          <Label htmlFor="crcl-sex" className="text-xs font-medium">
   131→            Sex
   132→          </Label>
   133→          <Select value={sex} onValueChange={(v) => setSex(v === "female" ? "female" : "male")}>
   134→            <SelectTrigger id="crcl-sex" className="h-10">
   135→              <SelectValue />
   136→            </SelectTrigger>
   137→            <SelectContent>
   138→              <SelectItem value="male">Male</SelectItem>
   139→              <SelectItem value="female">Female (× 0.85)</SelectItem>
   140→            </SelectContent>
   141→          </Select>
   142→        </div>
   143→      </div>
   144→      <ResultBlock
   145→        label="CrCl"
   146→        value={crclFinal !== null ? fmtNum(crclFinal, 4) : null}
   147→        unit="mL/min"
   148→        copyText={crclFinal !== null ? `CrCl = ${fmtNum(crclFinal, 4)} mL/min` : null}
   149→        formula={`CrCl = (140 − age) × wt / (72 × SCr)${female === 1 ? "" : " × 0.85 (female)"}`}
   150→      >
   151→        {band && crclFinal !== null ? (
   152→          <div className="mt-2">
   153→            <Badge variant="outline" className={`border font-normal ${band.className}`}>
   154→              {band.label}
   155→            </Badge>
   156→          </div>
   157→        ) : null}
   158→      </ResultBlock>
   159→    </CalcCard>
   160→  );
   161→}

   196→
   197→const FREQUENCIES: Array<{ value: string; label: string; hours: number }> = [
   198→  { value: "6", label: "q6h (4×/day)", hours: 6 },
   199→  { value: "8", label: "q8h (3×/day)", hours: 8 },

   203→
   204→function PediatricDoseCard() {
   205→  const [mgPerKg, setMgPerKg] = useState("15");
   206→  const [weight, setWeight] = useState("22");
   207→  const [maxAdult, setMaxAdult] = useState("500");
   208→  const [freq, setFreq] = useState("8");
   209→
   210→  const d = parsePositiveNumber(mgPerKg);
   211→  const w = parsePositiveNumber(weight);
   212→  const max = maxAdult.trim() === "" ? null : parsePositiveNumber(maxAdult);
   213→  const hours = FREQUENCIES.find((f) => f.value === freq)?.hours ?? 8;
   214→
   215→  const valid = d !== null && w !== null && (maxAdult.trim() === "" || max !== null);
   216→  const rawDose = valid ? (d as number) * (w as number) : null;
   217→  const capped = rawDose !== null && max !== null ? Math.min(rawDose, max) : rawDose;
   218→  const perDay = capped !== null ? (capped * 24) / hours : null;
   219→
   220→  return (
   221→    <CalcCard icon={Pill} title="Pediatric dose" description="mg/kg scaling with optional adult cap">
   222→      <div className="grid grid-cols-2 gap-3">

   224→        <NumberField id="ped-wt" label="Weight" unit="kg" value={weight} onChange={setWeight} error={w === null ? "Must be a positive number" : null} placeholder="22" />
   225→        <NumberField
   226→          id="ped-max"
   227→          label="Max adult dose"
   228→          unit="mg"
   229→          optional
   230→          value={maxAdult}
   231→          onChange={setMaxAdult}
   232→          error={maxAdult.trim() !== "" && max === null ? "Must be a positive number" : null}
   233→          placeholder="500"
   234→        />
   235→        <div className="space-y-1.5">
   236→          <Label htmlFor="ped-freq" className="text-xs font-medium">
   237→            Frequency
   238→          </Label>
   239→          <Select value={freq} onValueChange={setFreq}>
   240→            <SelectTrigger id="ped-freq" className="h-10">
   241→              <SelectValue />
   242→            </SelectTrigger>
   243→            <SelectContent>
   244→              {FREQUENCIES.map((f) => (
   245→                <SelectItem key={f.value} value={f.value}>
   246→                  {f.label}
   247→                </SelectItem>
   248→              ))}
   249→            </SelectContent>
   250→          </Select>
   251→        </div>
   252→      </div>
   253→      <div className="grid gap-3 sm:grid-cols-2">
   254→        <ResultBlock
   255→          label="Per dose"
   256→          value={capped !== null ? fmtNum(capped, 4) : null}
   257→          unit="mg"
   258→          copyText={capped !== null ? `Dose = ${fmtNum(capped, 4)} mg` : null}

   262→          label="Per day"
   263→          value={perDay !== null ? fmtNum(perDay, 4) : null}
   264→          unit="mg/day"

   266→          formula={`daily = dose × 24 / τ (τ = ${hours} h)`}
   267→        />
   268→      </div>
   269→    </CalcCard>
   270→  );
   271→}

   276→
   277→function LoadingMaintenanceCard() {
   278→  const [targetConc, setTargetConc] = useState("12");
   279→  const [vd, setVd] = useState("45");
   280→  const [clearance, setClearance] = useState("3.2");
   281→  const [css, setCss] = useState("6");
   282→  const [tau, setTau] = useState("8");
   283→  const [f, setF] = useState("1.0");
   284→
   285→  const tc = parsePositiveNumber(targetConc);
   286→  const v = parsePositiveNumber(vd);
   287→  const cl = parsePositiveNumber(clearance);
   288→  const cs = parsePositiveNumber(css);
   289→  const t = parsePositiveNumber(tau);
   290→  const fv = parsePositiveNumber(f);
   291→
   292→  const fError =
   293→    fv === null
   294→      ? "Must be > 0"
   295→      : (fv as number) > 1
   296→        ? "Bioavailability cannot exceed 1"
   297→        : null;
   298→
   299→  const valid = tc !== null && v !== null && cl !== null && cs !== null && t !== null && fError === null;
   300→
   301→  const ld = valid ? (tc as number) * (v as number) : null;
   302→  const md = valid ? ((cl as number) * (cs as number) * (t as number)) / (fv as number) : null;
   303→
   304→  return (
   305→    <CalcCard
   306→      icon={Timer}
   307→      title="Loading & maintenance dose"
   308→      description="One-compartment regimen design"
   309→    >
   310→      <div className="grid grid-cols-2 gap-3 sm:grid-cols-3">
   311→        <NumberField id="ld-tc" label="Target conc." unit="mg/L" value={targetConc} onChange={setTargetConc} error={tc === null ? "Must be > 0" : null} />
   312→        <NumberField id="ld-vd" label="Vd" unit="L" value={vd} onChange={setVd} error={v === null ? "Must be > 0" : null} />
   313→        <NumberField id="ld-cl" label="Clearance" unit="L/h" value={clearance} onChange={setClearance} error={cl === null ? "Must be > 0" : null} />
   314→        <NumberField id="ld-css" label="Target Css" unit="mg/L" value={css} onChange={setCss} error={cs === null ? "Must be > 0" : null} />
   315→        <NumberField id="ld-tau" label="Interval τ" unit="h" value={tau} onChange={setTau} error={t === null ? "Must be > 0" : null} />
   316→        <NumberField id="ld-f" label="Bioavail. F" unit="0–1" value={f} onChange={setF} error={fError} />
   317→      </div>
   318→      <div className="grid gap-3 sm:grid-cols-2">
   319→        <ResultBlock
   320→          label="Loading dose"
   321→          value={ld !== null ? fmtNum(ld, 4) : null}
   322→          unit="mg"
   323→          copyText={ld !== null ? `Loading dose = ${fmtNum(ld, 4)} mg` : null}
   324→          formula="LD = C_target × Vd"
   325→        />
   326→        <ResultBlock
   327→          label="Maintenance dose"
   328→          value={md !== null ? fmtNum(md, 4) : null}
   329→          unit="mg per τ"
   330→          copyText={md !== null ? `Maintenance dose = ${fmtNum(md, 4)} mg per ${fmtNum(t as number, 2)} h` : null}
   331→          formula="MD = CL × Css × τ / F"
   332→        />
   333→      </div>
   334→    </CalcCard>
   335→  );
   336→}

   398→/* ------------------------------------------------------------------ */
   399→/* 6. AUC₀→last (trapezoidal) with concentration–time chart            */
   400→/* ------------------------------------------------------------------ */
   401→
   402→interface AucRow {
   403→  id: number;
   404→  t: string;
   405→  c: string;
   406→}
   407→
   408→const SAMPLE_ROWS: Array<{ t: string; c: string }> = [
   409→  { t: "0", c: "0" },
   410→  { t: "1", c: "8.2" },
   411→  { t: "2", c: "6.1" },
   412→  { t: "4", c: "3.4" },
   413→  { t: "8", c: "1.1" },
   414→  { t: "12", c: "0.35" },
   415→];
   416→
   417→function AucCard() {
   418→  const [rows, setRows] = useState<AucRow[]>(() =>
   419→    SAMPLE_ROWS.map((r, i) => ({ id: i + 1, ...r }))
   420→  );
   421→
   422→  const parsed = useMemo(
   423→    () => rows.map((r) => ({ t: parseNonNegativeNumber(r.t), c: parseNonNegativeNumber(r.c) })),
   424→    [rows]
   425→  );
   426→
   427→  const rowError = (v: number | null) => (v === null ? "≥ 0" : null);
   428→
   429→  const timesIncreasing = useMemo(() => {
   430→    const ts = parsed.map((p) => p.t).filter((v): v is number => v !== null);
   431→    if (ts.length !== parsed.length || ts.length < 2) return false;
   432→    for (let i = 1; i < ts.length; i++) {
   433→      if (ts[i] <= ts[i - 1]) return false;
   434→    }
   435→    return true;
   436→  }, [parsed]);
   437→
   438→  const allValid = parsed.length >= 2 && parsed.every((p) => p.t !== null && p.c !== null) && timesIncreasing;
   439→
   440→  const auc = useMemo(() => {
   441→    if (!allValid) return null;
   442→    let acc = 0;
   443→    for (let i = 1; i < parsed.length; i++) {
   444→      const t0 = parsed[i - 1]?.t as number;
   445→      const c0 = parsed[i - 1]?.c as number;
   446→      const t1 = parsed[i]?.t as number;
   447→      const c1 = parsed[i]?.c as number;
   448→      acc += (t1 - t0) * ((c0 + c1) / 2);
   449→    }
   450→    return acc;
   451→  }, [allValid, parsed]);
   452→
   453→  const chartData = useMemo(
   454→    () =>
   455→      parsed
   456→        .map((p) => ({ t: p.t, c: p.c }))
   457→        .filter((p): p is { t: number; c: number } => p.t !== null && p.c !== null),
   458→    [parsed]
   459→  );
   460→
   461→  const nextId = rows.length > 0 ? Math.max(...rows.map((r) => r.id)) + 1 : 1;
   462→
   463→  function updateRow(id: number, field: "t" | "c", value: string) {
   464→    setRows((prev) => prev.map((r) => (r.id === id ? { ...r, [field]: value } : r)));
   465→  }
   466→
   467→  return (
   468→    <CalcCard
   469→      icon={Sigma}
   470→      title="AUC₀→last — trapezoidal rule"
   471→      description="Editable concentration–time pairs with a live chart"
   472→      className="xl:col-span-2"
   473→    >
   474→      <div className="grid gap-4 lg:grid-cols-[minmax(280px,380px)_minmax(0,1fr)]">
   475→        {/* Pair editor */}
   476→        <div className="space-y-2">
   477→          <div className="grid grid-cols-[28px_1fr_1fr_36px] items-center gap-2 px-1 text-[10px] font-medium uppercase tracking-wide text-muted-foreground">
   478→            <span>#</span>
   479→            <span>time (h)</span>
   480→            <span>conc (µg/mL)</span>
   481→            <span className="sr-only">Remove row</span>
   482→          </div>
   483→          <div className="thin-scrollbar max-h-64 space-y-1.5 overflow-y-auto pr-1">
   484→            {rows.map((r, i) => (
   485→              <div key={r.id} className="grid grid-cols-[28px_1fr_1fr_36px] items-start gap-2">
   486→                <span className="flex h-10 items-center justify-center rounded-md bg-muted font-mono text-xs text-muted-foreground tabular-nums">
   487→                  {i + 1}
   488→                </span>
   489→                <div>
   490→                  <Input
   491→                    aria-label={`Row ${i + 1} time in hours`}
   492→                    inputMode="decimal"
   493→                    value={r.t}
   494→                    onChange={(e) => updateRow(r.id, "t", e.target.value)}
   495→                    className="h-10 font-mono text-sm tabular-nums"
   496→                  />
   497→                  {rowError(parsed[i]?.t ?? null) ? (
   498→                    <p className="mt-0.5 text-[10px] text-destructive">Must be ≥ 0</p>
   499→                  ) : null}
   500→                </div>
   501→                <div>
   502→                  <Input
   503→                    aria-label={`Row ${i + 1} concentration`}
   504→                    inputMode="decimal"
   505→                    value={r.c}
   506→                    onChange={(e) => updateRow(r.id, "c", e.target.value)}
   507→                    className="h-10 font-mono text-sm tabular-nums"
   508→                  />
   509→                  {rowError(parsed[i]?.c ?? null) ? (
   510→                    <p className="mt-0.5 text-[10px] text-destructive">Must be ≥ 0</p>
   511→                  ) : null}
   512→                </div>
   513→                <Button
   514→                  type="button"
   515→                  variant="ghost"
   516→                  size="icon"
   517→                  className="h-10 w-9 text-muted-foreground hover:text-destructive"

   618→/* 7. PD Emax / Hill simulator                                         */
   619→/* ------------------------------------------------------------------ */
   620→
   621→function PdCard() {
   622→  const [emax, setEmax] = useState(80);
   623→  const [ec50, setEc50] = useState(5);
   624→  const [hill, setHill] = useState(2.5);
   625→
   626→  const data = useMemo(() => {
   627→    const pts: Array<{ conc: number; plain: number; sigmoid: number }> = [];
   628→    const N = 80;
   629→    for (let i = 0; i < N; i++) {
   630→      const conc = 0.01 * Math.pow(100 / 0.01, i / (N - 1)); // 0.01 → 100
   631→      const plain = (emax * conc) / (ec50 + conc);
   632→      const sigmoid = (emax * Math.pow(conc, hill)) / (Math.pow(ec50, hill) + Math.pow(conc, hill));
   633→      pts.push({ conc, plain, sigmoid });
   634→    }
   635→    return pts;
   636→  }, [emax, ec50, hill]);
   637→
   638→  const copyText = `Emax = ${emax}, EC50 = ${ec50}, Hill = ${hill.toFixed(1)}`;
   639→
   640→  return (
   641→    <CalcCard
   642→      icon={LineChartIcon}
   643→      title="PD Emax / sigmoid Hill simulator"
   644→      description="Dose–response curves on a log concentration axis"

   647→      <div className="grid gap-4 lg:grid-cols-[minmax(260px,320px)_minmax(0,1fr)]">

   663→              aria-label="Emax maximum effect"
   664→            />
   665→          </div>
   666→          <div className="space-y-2">

   688→                Hill coefficient
