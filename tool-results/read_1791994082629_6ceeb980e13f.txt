     1→"""Command-line interface — the scaffolding entry point of PharmaML.
     2→
     3→Replaces the legacy workflow of "open a per-algorithm .py file, edit
     4→the hardcoded CSV filename at the bottom, run it".  One command covers
     5→training, comparison, splitting, conversion, PK analysis and docking
     6→scans::
     7→
     8→    pharmaml ml train    --data qsar.csv --target Target --model rf --plots
     9→    pharmaml ml train    --data qsar.csv --task auto        # infer classification/regression
    10→    pharmaml ml train    --data qsar.csv --keep-categorical  # one-hot string features
    11→    pharmaml ml compare  --data qsar.csv --target Target --task classification --plot
    12→    pharmaml data inspect --data qsar.csv                  # audit your own file
    13→    pharmaml data split  --data qsar.csv --test-size 0.2 --val-size 0.1
    14→    pharmaml data convert --data ligands.csv --to sdf      # .smi/.csv/.sdf round-trip
    15→    pharmaml chem pic50  --data activity.xlsx --column "IC50 (nM)"
    16→    pharmaml pharma pk   --data profile.csv --time time_h --conc conc_ng_ml
    17→    pharmaml docking prep-ligands --ligands ligands.csv --out ligand_folders/
    18→    pharmaml docking prep-protein --protein protein.pdb --out r.pdbqt
    19→    pharmaml docking scan --directory docking_run/
    20→    pharmaml models list
    21→
    22→``--model`` accepts registry names and their aliases (``rf``,
    23→``logistic``, ``xgb`` ... — see ``pharmaml.ml.MODEL_ALIASES``).

   281→def _load_cli_dataset(args: argparse.Namespace):
   282→    """Load the ``--data`` file via :mod:`pharmaml.data` with task inference.
   283→
   284→    Both ``ml train`` and ``ml compare`` accept ``--task auto`` (the
   285→    default), which lets :func:`pharmaml.data.load_dataset` decide
   286→    classification (2+ classes) vs regression from the target column,
   287→    and ``--keep-categorical`` to keep string feature columns for
   288→    one-hot encoding.
   289→    """
   290→    from pharmaml.data import load_dataset
   291→
   292→    task = None if getattr(args, "task", "auto") == "auto" else args.task
   293→    return load_dataset(
   294→        args.data,
   295→        target=getattr(args, "target", None),
   296→        task=task,
   297→        keep_categorical=getattr(args, "keep_categorical", False),
   298→    )

   330→def _smi_bridge_frame(path: Path):
   331→    """Read a ``.smi`` file into a SMILES/ID table (CLI-internal bridge).
   332→
   333→    Lets ``--data x.smi --to sdf|smi`` reuse the table converters
   334→    without a temporary CSV on disk (the legacy two-step workflow).
   335→    """
   336→    import pandas as pd
   337→
   338→    from pharmaml.exceptions import DataValidationError
   339→
   340→    rows: list[tuple[str, str]] = []
   341→    with open(path, "r", encoding="utf-8", errors="ignore") as handle:
   342→        for line in handle:
   343→            parts = line.strip().split()
   344→            if not parts:
   345→                continue
   346→            rows.append((parts[0], " ".join(parts[1:])))
   347→    frame = pd.DataFrame(rows, columns=["SMILES", "ID"])
   348→    if frame.empty:
   349→        raise DataValidationError(f"no SMILES lines found in {path}")
   350→    return frame

   353→def _cmd_data_convert(args: argparse.Namespace) -> int:
   354→    from pharmaml.chem.converters import (
   355→        sdf_to_table,
   356→        smi_to_csv,
   357→        table_to_sdf,
   358→        table_to_smi,
   359→    )
   360→    from pharmaml.exceptions import DataValidationError
   361→    from pharmaml.utils.io import read_table, write_table
   362→
   363→    src = Path(args.data)
   364→    target = args.to
   365→    suffix = src.suffix.lower()
   366→    out = Path(args.out) if args.out else src.with_suffix(f".{target}")
   367→
   368→    if suffix == ".smi":
   369→        if target == "csv":
   370→            written = smi_to_csv(src, out)
   371→            print(f"converted {src.name} -> {written}")
   372→            return 0
   373→        if target == "smi":
   374→            raise DataValidationError(
   375→                f"input {src.name} is already a .smi file -- nothing to convert"
   376→            )
   377→        frame = _smi_bridge_frame(src)
   378→        report = table_to_sdf(
   379→            frame,
   380→            out,
   381→            smiles_col=args.smiles_col,
   382→            name_col=args.name_col,
   383→            embed3d=not args.no_3d,
   384→        )
   385→    elif suffix == ".sdf":
   386→        if target == "sdf":
   387→            raise DataValidationError(
   388→                f"input {src.name} is already an SDF file -- use --to csv or --to smi"
   389→            )
   390→        if target == "csv":
   391→            frame = sdf_to_table(src, out=out, smiles_col=args.smiles_col)
   392→            print(f"converted {src.name} -> {out} ({len(frame)} molecules)")
   393→            return 0
   394→        frame = sdf_to_table(src, smiles_col=args.smiles_col)
   395→        name_col = args.name_col or ("name" if "name" in frame.columns else None)
   396→        written = table_to_smi(
   397→            frame, out, smiles_col=args.smiles_col, name_col=name_col
   398→        )
   399→        print(f"converted {src.name} -> {written} ({len(frame)} molecules)")
   400→        return 0
   401→    else:  # CSV/TSV/Excel table input
   402→        if target == "smi":
   403→            written = table_to_smi(
   404→                src, out, smiles_col=args.smiles_col, name_col=args.name_col
   405→            )
   406→            print(f"converted {src.name} -> {written}")
   407→            return 0
   408→        if target == "csv":
   409→            frame = read_table(src)
   410→            written = write_table(frame, out)
   411→            print(f"converted {src.name} -> {written} ({len(frame)} rows)")
   412→            return 0
   413→        report = table_to_sdf(
   414→            src,
   415→            out,
   416→            smiles_col=args.smiles_col,
   417→            name_col=args.name_col,
   418→            embed3d=not args.no_3d,
   419→        )
   420→
   421→    # --to sdf: full audit + output path; invalid SMILES -> exit 1
   422→    print(report)
   423→    print(f"output -> {report.output_sdf}")
   424→    if report.failures:
   425→        print(
   426→            f"{report.failures} invalid SMILES skipped: "
   427→            f"{report.failed_smiles}",
   428→            file=sys.stderr,
   429→        )
   430→        return 1
   431→    return 0

   434→def _train_figure_bundle(result, report, dataset, X_test, y_test):
   435→    """Collect the standard figure/table set for ``ml train --plots``."""
   436→    import pandas as pd
   437→
   438→    from pharmaml.viz import metrics_to_dataframe, ml_plots
   439→
   440→    figures: dict[str, Any] = {}
   441→    tables: dict[str, Any] = {}
   442→
   443→    if dataset.task == "classification":
   444→        figures["confusion_matrix"] = ml_plots.plot_confusion_matrix(report)
   445→        if hasattr(result.model, "predict_proba"):
   446→            figures["roc_curves"] = ml_plots.plot_roc_curves(
   447→                result.model, X_test, y_test
   448→            )
   449→        if dataset.class_counts:
   450→            figures["class_distribution"] = ml_plots.plot_class_distribution(
   451→                dataset.class_counts
   452→            )
   453→            tables["class_counts"] = pd.DataFrame(
   454→                {"class": list(dataset.class_counts),
   455→                 "count": list(dataset.class_counts.values())}
   456→            )
   457→    else:
   458→        y_pred = result.model.predict(X_test)
   459→        figures["predicted_vs_actual"] = ml_plots.plot_predicted_vs_actual(
   460→            y_test, y_pred, metrics=report.as_dict()
   461→        )
   462→        figures["residuals"] = ml_plots.plot_residuals(y_test, y_pred)
   463→
   464→    if result.cv_scores is not None:
   465→        figures["cv_scores"] = ml_plots.plot_cv_scores(result)
   466→    try:
   467→        figures["feature_importance"] = ml_plots.plot_feature_importance(result.model)
   468→    except Exception:  # noqa: BLE001 - estimator exposes no importances
   469→        pass  # e.g. kernel SVM / KNN — importance plot is simply skipped
   470→
   471→    tables.setdefault(
   472→        "held_out_metrics", metrics_to_dataframe(report)
   473→    )
   474→    return figures, tables

   477→def _cmd_ml_train(args: argparse.Namespace) -> int:
   478→    from pharmaml.config import PipelineConfig
   479→    from pharmaml.ml.evaluation import evaluate_model
   480→    from pharmaml.ml.persistence import save_model
   481→    from pharmaml.ml.preprocessing import make_train_test_split
   482→    from pharmaml.ml.training import train_model
   483→
   484→    dataset = _load_cli_dataset(args)
   485→    X_train, X_test, y_train, y_test = make_train_test_split(
   486→        dataset.X, dataset.y, test_size=args.test_size
   487→    )
   488→
   489→    result = train_model(
   490→        args.model, dataset.task, X_train, y_train, tune=not args.no_tune, cv=args.cv
   491→    )
   492→    report = evaluate_model(result.model, X_test, y_test, dataset.task)
   493→
   494→    config = PipelineConfig()
   495→    model_path = config.output_path(f"{result.name}_{dataset.task}.joblib")
   496→    save_model(
   497→        result.model,
   498→        model_path,
   499→        name=result.name,
   500→        task=dataset.task,
   501→        features=dataset.feature_names,
   502→        best_params=result.best_params,
   503→        metrics=report.as_dict(),
   504→    )

   712→def main(argv: list[str] | None = None) -> int:
   713→    """CLI entry point (returns a process exit code)."""
   714→    parser = _build_parser()
   715→    args = parser.parse_args(argv)
   716→
   717→    configure_logging(
   718→        level=10 if args.verbose else 30,  # DEBUG / INFO
   719→        log_file=args.log_file,
   720→    )
   721→
   722→    sub = getattr(args, f"{args.command}_command", None)
   723→    handler = _HANDLERS.get((args.command, sub))
   724→    if handler is None:  # pragma: no cover - argparse guards this
   725→        parser.error("unknown command combination")
   726→    try:
   727→        return handler(args)
   728→    except Exception as exc:
   729→        _LOG.error("%s", exc)
   730→        if args.verbose:
   731→            raise
   732→        print(f"error: {exc}", file=sys.stderr)
   733→        return 1
   734→
   735→
   736→if __name__ == "__main__":
   737→    raise SystemExit(main())
