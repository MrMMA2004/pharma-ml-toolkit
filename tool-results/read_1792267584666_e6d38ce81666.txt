    91→def _unwrap_confusion(confusion, labels: list[Any] | None):
    92→    """Accept a raw matrix, a ``ClassificationReport`` or a ``Dataset``."""
    93→    if hasattr(confusion, "confusion") and hasattr(confusion, "labels"):
    94→        # ClassificationReport
    95→        return np.asarray(confusion.confusion), list(confusion.labels)
    96→    matrix = np.asarray(confusion)
    97→    if matrix.ndim != 2 or matrix.shape[0] != matrix.shape[1]:
    98→        raise DataValidationError(
    99→            f"confusion matrix must be square, got shape {matrix.shape}"
   100→        )
   101→    if labels is None:
   102→        labels = [str(i) for i in range(matrix.shape[0])]
   103→    if len(labels) != matrix.shape[0]:
   104→        raise DataValidationError(
   105→            f"{len(labels)} labels do not match a {matrix.shape[0]}x{matrix.shape[0]} matrix"
   106→        )
   107→    return matrix, list(labels)

   110→def plot_confusion_matrix(
   111→    confusion,
   112→    labels: list[Any] | None = None,
   113→    normalize: bool = False,
   114→    title: str = "Confusion matrix",
   115→) -> Any:
   116→    """Heat-map of predicted-vs-true class counts.
   117→
   118→    Mirrors the legacy ``Classification/*.py`` confusion-matrix PNG but
   119→    annotates each cell with both the count and its row-share, works for
   120→    binary **and multiclass** problems, and colour-scales by correctness
   121→    when ``normalize=True``.
   122→
   123→    Parameters
   124→    ----------
   125→    confusion:
   126→        A square count matrix, a
   127→        :class:`~pharmaml.ml.evaluation.ClassificationReport` (labels
   128→        are read automatically), or any object exposing
   129→        ``.confusion``/``.labels``.
   130→    labels:
   131→        Class labels in matrix order (optional when a report is passed).
   132→    normalize:
   133→        Scale colours by row-normalized shares instead of raw counts.
   134→
   135→    Returns
   136→    -------
   137→    matplotlib.figure.Figure
   138→    """
   139→    matrix, labels = _unwrap_confusion(confusion, labels)
   140→    n = matrix.shape[0]
   141→    display = matrix / matrix.sum(axis=1, keepdims=True) if normalize else matrix
   142→
   143→    fig, ax = new_figure(figsize=(2.2 + 1.1 * n, 2.2 + 1.0 * n))
   144→    image = ax.imshow(display, cmap="YlGnBu", vmin=0)
   145→    fig.colorbar(image, ax=ax, fraction=0.046, pad=0.04)
   146→
   147→    threshold = display.max() / 2.0 if display.max() else 0.0
   148→    for i in range(n):
   149→        for j in range(n):
   150→            share = matrix[i, j] / matrix[i].sum() if matrix[i].sum() else 0.0
   151→            # a float-dtype matrix (e.g. re-plotted normalized counts) must
   152→            # not hit the ":d" format code — cast integer annotations
   153→            cell = f"{display[i, j]:.2f}" if normalize else f"{round(float(display[i, j])):d}"
   154→            annotation = cell + f"\n({share:.0%})"
   155→            ax.text(
   156→                j, i, annotation, ha="center", va="center", fontsize=9,
   157→                color="white" if display[i, j] > threshold else "#1F2937",
   158→            )
   159→    ax.set_xticks(range(n), [str(label) for label in labels], rotation=45, ha="right")
   160→    ax.set_yticks(range(n), [str(label) for label in labels])
   161→    ax.set_xlabel("Predicted label")
   162→    ax.set_ylabel("True label")
   163→    ax.set_title(title + (" (row-normalized)" if normalize else ""))
   164→    ax.grid(False)
   165→    return fig

   168→def plot_roc_curves(
   169→    model: Any,
   170→    X_test: np.ndarray | pd.DataFrame,
   171→    y_test: np.ndarray | pd.Series | list,
   172→    title: str = "ROC curves",
   173→) -> Any:
   174→    """Receiver-operating curves — binary or one-vs-rest multiclass.
   175→
   176→    Reproduces the legacy per-class ROC PNGs (``AUC {cls}={auc:.3f}``
   177→    legend entries) for any classifier with ``predict_proba``:
   178→
   179→    * binary targets    → one curve for the positive class;
   180→    * 3+ class targets  → one curve per class (one-vs-rest), each in a
   181→      different house colour, sharing the diagonal reference.
   182→
   183→    Parameters
   184→    ----------
   185→    model:
   186→        Fitted estimator or Pipeline exposing ``predict_proba``.
   187→    X_test, y_test:
   188→        Held-out features and labels.
   189→
   190→    Returns
   191→    -------
   192→    matplotlib.figure.Figure
   193→
   194→    Raises
   195→    ------
   196→    ModelError
   197→        If the model lacks ``predict_proba``.
   198→    """
   199→    from sklearn.metrics import auc as sk_auc
   200→    from sklearn.metrics import roc_curve
   201→
   202→    if not hasattr(model, "predict_proba"):
   203→        raise ModelError(
   204→            "plot_roc_curves needs a model with predict_proba; "
   205→            "fit a probabilistic classifier (logistic_regression, random_forest, ...)"
   206→        )
   207→
   208→    proba = np.asarray(model.predict_proba(X_test))
   209→    classes = list(model.classes_)
   210→    y_true = np.asarray(y_test)
   211→
   212→    fig, ax = new_figure(figsize=(7.0, 6.0))
   213→    ax.plot([0, 1], [0, 1], linestyle="--", color="#9CA3AF", label="chance")
   214→
   215→    if len(classes) == 2:
   216→        scores = proba[:, 1]
   217→        fpr, tpr, _ = roc_curve(y_true, scores, pos_label=classes[1])
   218→        area = sk_auc(fpr, tpr)
   219→        ax.plot(fpr, tpr, color=PALETTE[0], lw=2.2,
   220→                label=f"AUC {classes[1]}={area:.3f}")
   221→    else:
   222→        for index, cls in enumerate(classes):
   223→            binary_truth = (y_true == cls).astype(int)
   224→            scores = proba[:, index]
   225→            try:
   226→                fpr, tpr, _ = roc_curve(binary_truth, scores)
   227→            except ValueError as exc:
   228→                _LOG.warning("ROC for class %r skipped: %s", cls, exc)
   229→                continue
   230→            area = sk_auc(fpr, tpr)
   231→            ax.plot(fpr, tpr, lw=2.0, color=PALETTE[index % len(PALETTE)],
   232→                    label=f"AUC {cls}={area:.3f}")
   233→
   234→    ax.set_xlabel("False positive rate")
   235→    ax.set_ylabel("True positive rate")
   236→    ax.set_title(title)
   237→    ax.set_xlim(-0.02, 1.02)
   238→    ax.set_ylim(-0.02, 1.05)
   239→    ax.legend(loc="lower right", fontsize=9)
   240→    return fig
   241→

   243→def plot_calibration_curve(
   244→    model: Any,
   245→    X_test: np.ndarray | pd.DataFrame,
   246→    y_test: np.ndarray | pd.Series | list,
   247→    n_bins: int = 10,
   248→    title: str = "Calibration (reliability diagram)",
   249→) -> Any:
   250→    """Reliability diagram — are the predicted probabilities honest?
   251→
   252→    A diagnostic the legacy scripts never drew: discrimination (ROC)
   253→    says *whether* the model ranks actives above inactives, calibration
   254→    says whether a predicted ``0.8`` really means 80 % positives. Poorly
   255→    calibrated probabilities quietly corrupt threshold choices, expected
   256→    cost calculations and consensus rescoring.
   257→
   258→    * binary targets    → one curve (positive class) with its Brier score;
   259→    * 3+ class targets  → one one-vs-rest curve per class, each with its
   260→      Brier score, sharing the diagonal reference.
   261→
   262→    Parameters
   263→    ----------
   264→    model:
   265→        Fitted estimator or Pipeline exposing ``predict_proba``.
   266→    X_test, y_test:
   267→        Held-out features and labels.
   268→    n_bins:
   269→        Probability bins for the diagram (classic: 10).
   270→
   271→    Returns
   272→    -------
   273→    matplotlib.figure.Figure
   274→
   275→    Raises
   276→    ------
   277→    ModelError
   278→        If the model lacks ``predict_proba``.
   279→    """
   280→    from sklearn.calibration import calibration_curve
   281→    from sklearn.metrics import brier_score_loss
   282→
   283→    if not hasattr(model, "predict_proba"):
   284→        raise ModelError(
   285→            "plot_calibration_curve needs a model with predict_proba; "
   286→            "fit a probabilistic classifier (logistic_regression, random_forest, ...)"
   287→        )
   288→    if n_bins < 2:
   289→        raise DataValidationError(f"n_bins must be >= 2, got {n_bins}")
   290→
   291→    proba = np.asarray(model.predict_proba(X_test))
   292→    classes = list(model.classes_)
   293→    y_true = np.asarray(y_test)
   294→
   295→    if len(classes) == 2:
   296→        curves = [(classes[1], (y_true == classes[1]).astype(int), proba[:, 1])]
   297→    else:
   298→        curves = [
   299→            (cls, (y_true == cls).astype(int), proba[:, index])
   300→            for index, cls in enumerate(classes)
   301→        ]
   302→
   303→    fig, ax = new_figure(figsize=(6.6, 6.0))
   304→    ax.plot([0, 1], [0, 1], linestyle="--", color="#9CA3AF",
   305→            label="perfectly calibrated")
   306→
   307→    for index, (cls, truth, scores) in enumerate(curves):
   308→        fraction_positive, mean_predicted = calibration_curve(
   309→            truth, scores, n_bins=n_bins, strategy="uniform"
   310→        )
   311→        brier = brier_score_loss(truth, scores)
   312→        ax.plot(
   313→            mean_predicted, fraction_positive, marker="o", lw=1.9,
   314→            color=PALETTE[index % len(PALETTE)],
   315→            label=f"{cls} — Brier {brier:.3f}",
   316→        )
   317→
   318→    ax.set_xlabel("mean predicted probability")
   319→    ax.set_ylabel("observed fraction of positives")
   320→    ax.set_title(title)
   321→    ax.set_xlim(-0.02, 1.02)
   322→    ax.set_ylim(-0.02, 1.05)
   323→    ax.legend(loc="upper left", fontsize=9)
   324→    return fig
   325→

   327→def plot_learning_curve(
   328→    model: Any,
   329→    X: np.ndarray | pd.DataFrame,
   330→    y: np.ndarray | pd.Series | list,
   331→    cv: int = 5,
   332→    scoring: str | None = None,
   333→    train_sizes: np.ndarray | list[float] | None = None,
   334→    title: str = "Learning curve",
   335→) -> Any:
   336→    """Training-set size vs CV score — is more data still buying accuracy?
   337→
   338→    The honest answer to "should we assay 500 more compounds?" that a
   339→    single CV number cannot give:
   340→
   341→    * both curves converging to a plateau → more data will not help much;
   342→    * validation still climbing at the right edge → more data *will* help;
   343→    * a persistent gap between the curves → variance/overfitting, not a
   344→      data problem.
   345→
   346→    Parameters
   347→    ----------
   348→    model:
   349→        Any scikit-learn compatible estimator (cloned per fold).
   350→    X, y:
   351→        Full training features and labels.
   352→    cv:
   353→        Fold count for the cross-validation at each size.
   354→    scoring:
   355→        Scorer name (default: estimator's ``score``).
   356→    train_sizes:
   357→        Absolute or fractional sizes; default ``10 % … 100 %`` in 8 steps.
   358→
   359→    Returns
   360→    -------
   361→    matplotlib.figure.Figure
   362→    """
   363→    from sklearn.model_selection import learning_curve as sklearn_learning_curve
   364→
   365→    if cv < 2:
   366→        raise DataValidationError(f"cv must be >= 2, got {cv}")
   367→    if train_sizes is None:
   368→        train_sizes = np.linspace(0.1, 1.0, 8)
   369→
   370→    sizes, train_scores, val_scores = sklearn_learning_curve(
   371→        model, X, y, cv=cv, scoring=scoring, train_sizes=train_sizes, n_jobs=None,
   372→    )
   373→    train_mean = train_scores.mean(axis=1)
   374→    train_std = train_scores.std(axis=1)
   375→    val_mean = val_scores.mean(axis=1)
   376→    val_std = val_scores.std(axis=1)
   377→
   378→    fig, ax = new_figure(figsize=(8.4, 5.4))
   379→    ax.fill_between(sizes, train_mean - train_std, train_mean + train_std,
   380→                    alpha=0.15, color=PALETTE[0], linewidth=0)
   381→    ax.plot(sizes, train_mean, marker="o", lw=1.9, color=PALETTE[0],
   382→            label=f"training score ({train_mean[-1]:.3g})")
   383→    ax.fill_between(sizes, val_mean - val_std, val_mean + val_std,
   384→                    alpha=0.15, color=PALETTE[1], linewidth=0)
   385→    ax.plot(sizes, val_mean, marker="s", lw=1.9, color=PALETTE[1],
   386→            label=f"CV score ({val_mean[-1]:.3g})")
   387→
   388→    gap = float(train_mean[-1] - val_mean[-1])
   389→    verdict = (
   390→        f"gap ≈ {gap:.2f} — overfitting dominates, more data alone will not close it"
   391→        if gap > 0.15
   392→        else "curves converging — additional data has diminishing returns"
   393→    )
   394→    ax.set_xlabel("training samples")
   395→    ax.set_ylabel(scoring or "score")
   396→    ax.set_title(f"{title}\n{verdict}")
   397→    ax.legend(loc="lower right", fontsize=9)
   398→    return fig
   399→

   401→def plot_predicted_vs_actual(
   402→    y_true: np.ndarray | pd.Series | list,
   403→    y_pred: np.ndarray | list,
   404→    metrics: Mapping[str, float] | None = None,
   405→    title: str = "Predicted vs actual",
   406→) -> Any:
   407→    """Parity scatter for regression, with the identity reference line.
   408→
   409→    Upgrades the legacy ``Regression/*.py`` figure: the diagonal
   410→    ``ŷ = y`` line is now labelled, and an optional ``metrics`` mapping
   411→    (e.g. ``{"RMSE": 0.38, "R²": 0.77}``) is rendered as a summary box.
   412→    """
   413→    truth = np.asarray(y_true, dtype=float)
   414→    prediction = np.asarray(y_pred, dtype=float)
   415→    if truth.shape != prediction.shape:
   416→        raise DataValidationError(
   417→            f"y_true {truth.shape} and y_pred {prediction.shape} must match"
   418→        )
   419→
   420→    fig, ax = new_figure(figsize=(6.4, 6.0))
   421→    low = float(min(truth.min(), prediction.min()))
   422→    high = float(max(truth.max(), prediction.max()))
   423→    pad = 0.05 * (high - low or 1.0)
   424→    limits = (low - pad, high + pad)
   425→    ax.plot(limits, limits, linestyle="--", color="#9CA3AF", label="perfect prediction (ŷ = y)")
   426→    ax.scatter(truth, prediction, s=28, alpha=0.75, color=PALETTE[0], edgecolor="white",
   427→               linewidth=0.4)
   428→    ax.set_xlim(*limits)
   429→    ax.set_ylim(*limits)
   430→    ax.set_xlabel("actual")
   431→    ax.set_ylabel("predicted (ŷ)")
   432→    ax.set_title(title)
   433→
   434→    if metrics:
   435→        text = "  ".join(f"{key}={value:.4g}" for key, value in metrics.items())
   436→        ax.text(0.03, 0.97, text, transform=ax.transAxes, va="top", ha="left", fontsize=9,
   437→                bbox={"boxstyle": "round,pad=0.35", "facecolor": "white",
   438→                      "edgecolor": "#D1D5DB", "alpha": 0.9})
   439→    ax.legend(loc="lower right", fontsize=9)
   440→    return fig
   441→

   443→def plot_residuals(
   444→    y_true: np.ndarray | pd.Series | list,
   445→    y_pred: np.ndarray | list,
   446→    title: str = "Residual diagnostics",
   447→) -> Any:
   448→    """Residuals vs predicted (left) and residual histogram (right).
   449→
   450→    A residual view the legacy scripts never produced — heteroscedastic
   451→    fans and skewed error distributions become visible at a glance.
   452→    """
   453→    truth = np.asarray(y_true, dtype=float)
   454→    prediction = np.asarray(y_pred, dtype=float)
   455→    residuals = truth - prediction
   456→
   457→    fig, (ax_res, ax_hist) = new_figure(nrows=1, ncols=2, figsize=(11.0, 4.8))
   458→    ax_res.axhline(0.0, linestyle="--", color="#9CA3AF")
   459→    ax_res.scatter(prediction, residuals, s=26, alpha=0.75, color=PALETTE[1],
   460→                   edgecolor="white", linewidth=0.4)
   461→    ax_res.set_xlabel("predicted (ŷ)")
   462→    ax_res.set_ylabel("residual (y − ŷ)")
   463→    ax_res.set_title("Residuals vs predicted")
   464→
   465→    ax_hist.hist(residuals, bins=min(30, max(8, len(residuals) // 4)), color=PALETTE[0],
   466→                 alpha=0.85, edgecolor="white")
   467→    ax_hist.axvline(0.0, linestyle="--", color="#9CA3AF")
   468→    ax_hist.set_xlabel("residual (y − ŷ)")
   469→    ax_hist.set_ylabel("count")
   470→    ax_hist.set_title("Residual distribution")
   471→    fig.suptitle(title, y=1.02)
   472→    return fig

   475→def plot_model_comparison(
   476→    leaderboard: pd.DataFrame,
   477→    metric: str | None = None,
   478→    title: str = "Cross-validated model leaderboard",
   479→) -> Any:
   480→    """Leaderboard as horizontal bars with per-fold std error bars.
   481→
   482→    Turns the :func:`~pharmaml.ml.training.compare_models` DataFrame
   483→    into a figure: models sorted by their mean CV metric, error bars
   484→    showing fold-to-fold variability, values annotated at the bar ends.
   485→
   486→    Parameters
   487→    ----------
   488→    leaderboard:
   489→        DataFrame from ``compare_models`` — a ``model`` column plus
   490→        ``<metric>_mean`` / ``<metric>_std`` columns.
   491→    metric:
   492→        Metric name when the frame contains several; defaults to the
   493→        first ``*_mean`` column found.
   494→    """
   495→    if "model" not in leaderboard.columns:
   496→        raise DataValidationError("leaderboard needs a 'model' column (compare_models output)")
   497→    if metric is None:
   498→        mean_cols = [c for c in leaderboard.columns if c.endswith("_mean")]
   499→        if not mean_cols:
   500→            raise DataValidationError(
   501→                "no '<metric>_mean' column found in the leaderboard frame"
   502→            )
   503→        metric = mean_cols[0][: -len("_mean")]
   504→
   505→    mean_col = f"{metric}_mean"
   506→    std_col = f"{metric}_std"
   507→    if mean_col not in leaderboard.columns:
   508→        raise DataValidationError(f"column {mean_col!r} not found in the leaderboard")
   509→
   510→    frame = leaderboard.sort_values(mean_col, ascending=True)
   511→    values = frame[mean_col].to_numpy(dtype=float)
   512→    errors = frame[std_col].to_numpy(dtype=float) if std_col in frame.columns else None
   513→
   514→    fig, ax = new_figure(figsize=(8.6, 0.62 * len(frame) + 1.6))
   515→    positions = np.arange(len(frame))
   516→    ax.barh(positions, values, xerr=errors, color=PALETTE[0], alpha=0.88,
   517→            edgecolor="white", height=0.62, error_kw={"elinewidth": 1.1, "capsize": 3})
   518→    for pos, value in zip(positions, values):
   519→        if np.isfinite(value):
   520→            ax.text(value, pos, f" {value:.4g}", va="center", ha="left", fontsize=9)
   521→    ax.set_yticks(positions, [str(name) for name in frame["model"]])
   522→    ax.set_xlabel(f"mean CV {metric} (± std across folds)" if errors is not None
   523→                  else f"mean CV {metric}")
   524→    ax.set_title(title)
   525→    return fig

   528→def plot_class_distribution(
   529→    class_counts: Mapping[Any, int] | Any,
   530→    title: str = "Class distribution",
   531→) -> Any:
   532→    """Bar chart of class balance (accepts a mapping or a ``Dataset``)."""
   533→    if hasattr(class_counts, "class_counts") and class_counts.class_counts is not None:
   534→        class_counts = class_counts.class_counts
   535→    if not isinstance(class_counts, Mapping):
   536→        raise DataValidationError(
   537→            "class_counts must be a mapping {label: count} or a classification Dataset"
   538→        )
   539→    if not class_counts:
   540→        raise DataValidationError("class_counts is empty — nothing to plot")
   541→
   542→    labels = [str(key) for key in class_counts]
   543→    counts = [int(value) for value in class_counts.values()]
   544→    total = sum(counts)

   546→    fig, ax = new_figure(figsize=(1.9 * len(labels) + 3.4, 5.0))
   547→    bars = ax.bar(labels, counts, color=PALETTE[: len(labels)], alpha=0.9, width=0.62,
   548→                  edgecolor="white")
   549→    for bar, count in zip(bars, counts):
   550→        ax.text(bar.get_x() + bar.get_width() / 2.0, bar.get_height(),
   551→                f"{count}\n({count / total:.0%})", ha="center", va="bottom", fontsize=9)
   552→    ax.set_ylabel("samples")
   553→    ax.set_title(f"{title} — {total} sample(s), {len(labels)} class(es)")

   557→def _extract_importances(model_or_mapping, feature_names: list[str] | None):
   558→    """Normalise ``model`` / ``{name: value}`` inputs into (names, values)."""
   559→    if isinstance(model_or_mapping, Mapping):
   560→        names = [str(key) for key in model_or_mapping]
   561→        values = np.asarray(list(model_or_mapping.values()), dtype=float)
   562→        return names, values
   563→
   564→    estimator = model_or_mapping
   565→    if hasattr(estimator, "named_steps"):  # sklearn Pipeline
   566→        for step in reversed(list(estimator.named_steps.values())):
   567→            if hasattr(step, "feature_importances_") or hasattr(step, "coef_"):
   568→                estimator = step
   569→                break
   570→    if hasattr(estimator, "feature_importances_"):
   571→        values = np.asarray(estimator.feature_importances_, dtype=float)
   572→    elif hasattr(estimator, "coef_"):
   573→        coef = np.asarray(estimator.coef_, dtype=float)
   574→        values = np.abs(coef).max(axis=0) if coef.ndim > 1 else np.abs(coef)
   575→    else:
   576→        raise DataValidationError(
   577→            "could not extract feature importances: pass a tree/linear model, "
   578→            "a Pipeline containing one, or an explicit {feature: importance} mapping"
   579→        )
   580→
   581→    # A Pipeline's preprocessing steps (impute/scale/one-hot) know the
   582→    # transformed feature names — use them so one-hot expanded columns
   583→    # stay readable instead of "x0, x1, ...".
   584→    if (
   585→        feature_names is None
   586→        and hasattr(model_or_mapping, "named_steps")
   587→        and len(model_or_mapping.named_steps) > 1
   588→    ):
   589→        try:
   590→            expanded = list(model_or_mapping[:-1].get_feature_names_out())
   591→        except Exception:  # noqa: BLE001 — capability probe; transformers vary
   592→            expanded = []
   593→        if expanded and len(expanded) == len(values):
   594→            feature_names = expanded
   595→    if feature_names is None:
   596→        feature_names = [f"x{i}" for i in range(len(values))]
   597→    if len(feature_names) != len(values):
   598→        raise DataValidationError(
   599→            f"{len(feature_names)} feature names for {len(values)} importance values"
   600→        )
   601→    return [str(name) for name in feature_names], values
   602→
   603→
   604→def plot_feature_importance(
   605→    model_or_mapping: Any,
   606→    feature_names: list[str] | None = None,
   607→    top_n: int = 15,
   608→    title: str = "Feature importance",
   609→) -> Any:
   610→    """Top-N feature importances as horizontal bars.
   611→
   612→    Works with tree ensembles (``feature_importances_``), linear models
   613→    (``|coef_|``), wrapped :class:`~sklearn.pipeline.Pipeline` models,
   614→    or a plain ``{feature: importance}`` mapping when you computed the
   615→    scores yourself.
   616→    """
   617→    names, values = _extract_importances(model_or_mapping, feature_names)
   618→    order = np.argsort(values)[::-1][: min(top_n, len(values))][::-1]
   619→
   620→    fig, ax = new_figure(figsize=(8.4, 0.42 * len(order) + 1.7))
   621→    positions = np.arange(len(order))
   622→    ax.barh(positions, values[order], color=PALETTE[0], alpha=0.88, height=0.62,
   623→            edgecolor="white")
   624→    for pos, value in zip(positions, values[order]):
   625→        ax.text(value, pos, f" {value:.4g}", va="center", ha="left", fontsize=9)
   626→    ax.set_yticks(positions, [names[i] for i in order])
   627→    ax.set_xlabel("importance")
   628→    ax.set_title(title)
   629→    return fig

   632→def plot_cv_scores(
   633→    cv_scores: Any,
   634→    model_name: str | None = None,
   635→    metric: str | None = None,
   636→    title: str = "Cross-validation fold scores",
   637→) -> Any:
   638→    """Per-fold CV scores as bars with the mean line and ±std band.
   639→
   640→    Accepts a :class:`~pharmaml.ml.training.TrainingResult` or a plain
   641→    array-like of fold scores — the spread across folds is the honest
   642→    uncertainty the legacy scripts hid behind a single mean.
   643→    """
   644→    if hasattr(cv_scores, "cv_scores") and cv_scores.cv_scores is not None:
   645→        model_name = model_name or getattr(cv_scores, "name", None)
   646→        scores = np.asarray(cv_scores.cv_scores, dtype=float)
   647→    else:
   648→        scores = np.asarray(cv_scores, dtype=float)
   649→    scores = scores.reshape(-1)
   650→    if scores.size == 0:
   651→        raise DataValidationError("no CV fold scores to plot")
   652→
   653→    mean = float(scores.mean())
   654→    std = float(scores.std())
   655→    label = model_name or "model"
   656→    metric_label = metric or "score"
   657→
   658→    fig, ax = new_figure(figsize=(7.6, 4.6))
   659→    positions = np.arange(1, len(scores) + 1)
   660→    ax.bar(positions, scores, color=PALETTE[0], alpha=0.85, width=0.6, edgecolor="white")
   661→    ax.axhline(mean, color=PALETTE[1], lw=2.0,
   662→               label=f"mean = {mean:.4g}")
   663→    ax.axhspan(mean - std, mean + std, color=PALETTE[1], alpha=0.14,
   664→               label=f"± 1 std = {std:.4g}")
   665→    ax.set_xticks(positions, [f"fold {i}" for i in positions])
   666→    ax.set_ylabel(metric_label)
   667→    ax.set_title(f"{title} — {label}")
   668→    ax.legend(fontsize=9, loc="best")
   669→    return fig
