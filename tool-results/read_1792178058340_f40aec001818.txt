     1→"""Drug-response screens: GDSC/CCLE-style pharmacology tables.
     2→
     3→A *response screen* is a long-format table with one row per
     4→(cell line, compound) pair and one or more numeric metric columns --
     5→the layout popularised by the GDSC and CCLE sensitivity datasets
     6→(``LN_IC50``, ``AUC``, ``RMSE`` ...).  This module loads, audits and
     7→reshapes such tables:
     8→
     9→* :func:`load_response_table` accepts a DataFrame or a csv/xlsx path,
    10→  validates the identifier columns, auto-detects the metric columns
    11→  (every remaining numeric column) and coerces stray non-numeric
    12→  entries to NaN *with a warning* (round-11/16 semantics: warn, keep
    13→  NaN -- never drop rows silently),
    14→* :class:`ResponseScreen` is the immutable container carrying the
    15→  frame plus the column configuration, with a ``summary()`` in the
    16→  package's audit style,
    17→* :func:`audit_response_table` reports the classic screen hygiene
    18→  issues (missing identifiers, missing metrics, duplicate pairs, cell
    19→  lines claimed by two tissues, under-tested compounds),
    20→* :func:`collapse_replicates` resolves duplicated pairs by the median
    21→  per metric (rows with a missing identifier are kept as their own
    22→  NaN-keyed group, never silently dropped),
    23→* :func:`ln_ic50_to_pic50` / :func:`pic50_to_ln_ic50` convert between
    24→  GDSC's ``LN_IC50`` (natural log of the IC50 in micromolar) and the
    25→  chemist's pIC50 scale,
    26→* :func:`response_matrix` pivots one metric into the classic
    27→  cell-line x compound sensitivity matrix (untested pairs stay NaN),
    28→* :func:`summarize_compounds` gives per-compound potency statistics
    29→  across the panel, and
    30→* :func:`call_sensitive_resistant` thresholds one compound's response
    31→  vector into sensitive / intermediate / resistant calls.
    32→
    33→Direction conventions matter everywhere here: for ``LN_IC50``-style
    34→metrics *lower* means more sensitive, for pIC50-style metrics *higher*
    35→means more sensitive.  Every direction-sensitive function documents
    36→its assumption and exposes a switch.
    37→"""
    38→
    39→from __future__ import annotations
    40→
    41→import math
    42→from dataclasses import dataclass, field
    43→from pathlib import Path
    44→from typing import TypeAlias
    45→
    46→import numpy as np
    47→import pandas as pd
    48→
    49→from pharmaml.exceptions import DataValidationError
    50→from pharmaml.logger import get_logger
    51→from pharmaml.utils.io import read_table
    52→from pharmaml.utils.validation import require_columns
    53→
    54→_LOG = get_logger(__name__)
    55→
    56→__all__ = [
    57→    "ResponseScreen",
    58→    "audit_response_table",
    59→    "call_sensitive_resistant",
    60→    "collapse_replicates",
    61→    "ln_ic50_to_pic50",
    62→    "load_response_table",
    63→    "pic50_to_ln_ic50",
    64→    "response_matrix",
    65→    "summarize_compounds",
    66→]
    67→
    68→#: What a table argument may be: an in-memory frame or a csv/xlsx path.
    69→ResponseTableSource: TypeAlias = pd.DataFrame | str | Path
    70→
    71→#: ``ln(10)`` -- the natural-log/decade conversion factor.
    72→_LN10 = math.log(10.0)

    74→
    75→def _detect_metric_columns(
    76→    frame: pd.DataFrame, exclude: list[str]
    77→) -> list[str]:
    78→    """Return the remaining columns that hold numeric measurements.
    79→
    80→    A column qualifies when its dtype is numeric or when at least one
    81→    value parses as a number (``pd.to_numeric`` with coercion) -- the
    82→    latter keeps a ``LN_IC50`` column that went ``object`` dtype
    83→    because of a few stray strings in an otherwise numeric export.
    84→
    85→    Boolean columns never qualify: ``pandas`` counts ``bool`` as a
    86→    numeric dtype, but an ``msi_status``-style flag is an annotation,
    87→    not a measurement.  The rule covers both ``bool`` dtype and an
    88→    ``object`` column holding only ``True``/``False`` (with missing
    89→    entries -- what a csv round-trip of a flag column produces).
    90→    Pass such a column explicitly via ``metrics=[...]`` to treat its
    91→    ``True``/``False`` values as 1/0.
    92→    """
    93→    metrics: list[str] = []
    94→    for column in frame.columns:
    95→        if column in exclude:
    96→            continue
    97→        series = frame[column]
    98→        if pd.api.types.is_bool_dtype(series):
    99→            continue
   100→        if pd.api.types.is_numeric_dtype(series):
   101→            metrics.append(column)
   102→            continue
   103→        non_missing = series.dropna()
   104→        if not non_missing.empty and non_missing.map(
   105→            lambda value: isinstance(value, (bool, np.bool_))
   106→        ).all():
   107→            continue
   108→        coerced = pd.to_numeric(series, errors="coerce")
   109→        if coerced.notna().any():
   110→            metrics.append(column)
   111→    return metrics
   112→
   113→
   114→def _coerce_metrics(frame: pd.DataFrame, metrics: list[str]) -> pd.DataFrame:
   115→    """Coerce metric columns to numeric dtype, warning per column.
   116→
   117→    Values that fail to parse become NaN *after* a warning names the
   118→    column and the affected count (round-11/16 semantics: warn, keep
   119→    NaN).  Already-numeric columns pass through unchanged.  Boolean
   120→    metric columns (only reachable through the explicit ``metrics=``
   121→    path -- auto-detection skips them) are converted to 0/1 floats:
   122→    ``pd.to_numeric`` passes ``bool`` dtype through unchanged, but a
   123→    metric column must support arithmetic.
   124→    """
   125→    for metric in metrics:
   126→        series = frame[metric]
   127→        if pd.api.types.is_bool_dtype(series):
   128→            frame[metric] = series.astype("float64")
   129→            continue
   130→        if pd.api.types.is_numeric_dtype(series):
   131→            continue
   132→        coerced = pd.to_numeric(series, errors="coerce")
   133→        lost = int((series.notna() & coerced.isna()).sum())
   134→        if lost:
   135→            _LOG.warning(
   136→                "%d non-numeric value(s) in metric column %r coerced to NaN",
   137→                lost,
   138→                metric,
   139→            )
   140→        frame[metric] = coerced
   141→    return frame

   144→@dataclass(frozen=True, slots=True)
   145→class ResponseScreen:
   146→    """One loaded drug-response screen.
   147→
   148→    Attributes
   149→    ----------
   150→    frame:
   151→        The long-format table: one row per (cell line, compound) pair
   152→        with the metric columns already coerced to numeric dtype.
   153→    cell_col:
   154→        Name of the cell-line identifier column.
   155→    compound_col:
   156→        Name of the compound identifier column.
   157→    tissue_col:
   158→        Name of the optional tissue/lineage annotation column.
   159→    metrics:
   160→        Names of the numeric response-metric columns (``LN_IC50``,
   161→        ``AUC`` ...).  When empty at construction time the numeric
   162→        non-identifier columns are auto-detected.
   163→
   164→    Raises
   165→    ------
   166→    DataValidationError
   167→        If no metric column can be auto-detected.
   168→    """
   169→
   170→    frame: pd.DataFrame
   171→    cell_col: str
   172→    compound_col: str
   173→    tissue_col: str | None = None
   174→    metrics: list[str] = field(default_factory=list)
   175→
   176→    def __post_init__(self) -> None:
   177→        if not self.metrics:
   178→            detected = _detect_metric_columns(
   179→                self.frame,
   180→                exclude=[
   181→                    column
   182→                    for column in (self.cell_col, self.compound_col, self.tissue_col)
   183→                    if column is not None
   184→                ],
   185→            )
   186→            if not detected:
   187→                raise DataValidationError(
   188→                    "no metric columns detected in the response table -- "
   189→                    "pass metrics=[...] explicitly. Available columns: "
   190→                    f"{list(self.frame.columns)}"
   191→                )
   192→            object.__setattr__(self, "metrics", detected)
   193→
   194→    @property
   195→    def cell_lines(self) -> list[str]:
   196→        """Sorted unique cell-line identifiers (missing ones excluded)."""
   197→        return sorted(self.frame[self.cell_col].dropna().astype(str).unique().tolist())
   198→
   199→    @property
   200→    def compounds(self) -> list[str]:
   201→        """Sorted unique compound identifiers (missing ones excluded)."""
   202→        return sorted(
   203→            self.frame[self.compound_col].dropna().astype(str).unique().tolist()
   204→        )
   205→
   206→    @property
   207→    def n_pairs(self) -> int:
   208→        """Number of unique (cell line, compound) pairs."""
   209→        return len(self.frame[[self.cell_col, self.compound_col]].drop_duplicates())
   210→
   211→    def _duplicate_pairs(self) -> list[tuple[str, str]]:
   212→        """(cell, compound) pairs that occupy more than one row.
   213→
   214→        Uses ``dropna=False`` so pairs keyed by a missing identifier
   215→        are counted too -- silently dropping them would hide
   216→        duplicated unusable rows from every audit.
   217→        """
   218→        counts = self.frame.groupby(
   219→            [self.cell_col, self.compound_col], sort=True, dropna=False
   220→        ).size()
   221→        return [pair for pair, count in counts.items() if count > 1]
   222→
   223→    def summary(self) -> str:
   224→        """Multi-line audit summary (the package's ``inspect`` style)."""
   225→        lines = [
   226→            (f"screen        : {len(self.cell_lines)} cell line(s) x "
   227→            f"{len(self.compounds)} compound(s) -> {self.n_pairs} (cell, compound) pair(s)")
   228→        ]
   229→        if len(self.frame) != self.n_pairs:
   230→            lines.append(
   231→                f"rows          : {len(self.frame)} "
   232→                f"({len(self.frame) - self.n_pairs} duplicate row(s))"
   233→            )
   234→        if self.tissue_col is not None:
   235→            tissues = sorted(
   236→                self.frame[self.tissue_col].dropna().astype(str).unique().tolist()
   237→            )
   238→            shown = ", ".join(tissues[:8])
   239→            extra = len(tissues) - 8
   240→            more = f" (+{extra} more)" if extra > 0 else ""
   241→            lines.append(
   242→                f"tissue        : {len(tissues)} value(s) in {self.tissue_col!r}: "
   243→                f"{shown}{more}"
   244→            )
   245→        lines.append(f"metrics       : {', '.join(self.metrics) or '(none)'}")
   246→        for metric in self.metrics:
   247→            present = int(self.frame[metric].notna().sum())
   248→            total = len(self.frame)
   249→            missing = total - present
   250→            note = f" ({missing} missing)" if missing else ""
   251→            lines.append(f"  {metric:<13s}: {present}/{total} value(s) present{note}")
   252→        duplicates = self._duplicate_pairs()
   253→        if duplicates:
   254→            shown = ", ".join(f"({cell} + {compound})" for cell, compound in duplicates[:3])
   255→            extra = len(duplicates) - 3
   256→            more = f" (+{extra} more)" if extra > 0 else ""
   257→            lines.append(f"dup. pairs    : {len(duplicates)}: {shown}{more}")
   258→        else:
   259→            lines.append("dup. pairs    : 0")
   260→        return "\n".join(lines)
   261→

   263→def load_response_table(
   264→    data: ResponseTableSource,
   265→    cell_column: str = "cell_line",
   266→    compound_column: str = "compound",
   267→    tissue_column: str | None = None,
   268→    metrics: list[str] | None = None,
   269→) -> ResponseScreen:
   270→    """Load a drug-response screen from a DataFrame or csv/xlsx path.
   271→
   272→    Parameters
   273→    ----------
   274→    data:
   275→        Long-format table (one row per cell line x compound pair) as a
   276→        DataFrame or a path read via
   277→        :func:`pharmaml.utils.io.read_table`.
   278→    cell_column, compound_column:
   279→        Names of the identifier columns.
   280→    tissue_column:
   281→        Optional tissue annotation column (kept out of the metrics).
   282→    metrics:
   283→        Response-metric column names.  When ``None`` (the default) all
   284→        remaining numeric columns are used -- including ``object``
   285→        columns that hold at least one parseable number, so a stray
   286→        string in an otherwise numeric export degrades to NaN instead
   287→        of hiding the whole metric.  Boolean columns (flags like
   288→        ``msi_status``) are *never* auto-detected; pass one explicitly
   289→        to have its ``True``/``False`` values treated as 1/0.
   290→
   291→    Returns
   292→    -------
   293→    ResponseScreen
   294→        The validated screen.  Metric columns are coerced to numeric
   295→        dtype on a *copy* of the frame (the caller's frame is never
   296→        mutated); non-numeric values become NaN after a warning that
   297→        names the column and the count.
   298→
   299→    Raises
   300→    ------
   301→    DataValidationError
   302→        If an identifier column is missing (all offenders listed at
   303→        once), an explicit metric column is missing, or no metric
   304→        column can be detected at all.
   305→    """
   306→    if isinstance(data, pd.DataFrame):
   307→        frame = data.copy()
   308→    elif isinstance(data, (str, Path)):
   309→        frame = read_table(data)
   310→    else:
   311→        raise DataValidationError(
   312→            "data must be a pandas DataFrame or a path to a csv/xlsx file, "
   313→            f"got {type(data).__name__}"
   314→        )
   315→
   316→    required = [cell_column, compound_column]
   317→    if tissue_column is not None:
   318→        required.append(tissue_column)
   319→    require_columns(frame, required)
   320→
   321→    if metrics is None:
   322→        detected = _detect_metric_columns(frame, exclude=required)
   323→        if not detected:
   324→            raise DataValidationError(
   325→                "no metric columns detected in the response table -- pass "
   326→                "metrics=[...] explicitly. Available columns: "
   327→                f"{list(frame.columns)}"
   328→            )
   329→        metrics = detected
   330→        _LOG.debug("Auto-detected metric column(s): %s", metrics)
   331→    else:
   332→        require_columns(frame, list(metrics))
   333→
   334→    frame = _coerce_metrics(frame, list(metrics))
   335→    _LOG.info(
   336→        "Loaded response screen: %d row(s), %d cell line(s), %d compound(s), "
   337→        "metric(s) %s",
   338→        len(frame),
   339→        frame[cell_column].nunique(),
   340→        frame[compound_column].nunique(),
   341→        metrics,
   342→    )
   343→    return ResponseScreen(
   344→        frame=frame,
   345→        cell_col=cell_column,
   346→        compound_col=compound_column,
   347→        tissue_col=tissue_column,
   348→        metrics=list(metrics),
   349→    )

   352→def audit_response_table(screen: ResponseScreen, n_min: int = 3) -> pd.DataFrame:
   353→    """Audit a response screen for the classic hygiene issues.
   354→
   355→    One row per concrete issue with columns ``issue``, ``detail`` and
   356→    ``count``:
   357→
   358→    * ``missing identifier`` -- per identifier column (cell line /
   359→      compound) holding NaN entries; ``count`` is the number of rows
   360→      without the identifier,
   361→    * ``missing metric values`` -- per metric column with missing
   362→      entries; ``count`` is the number of missing values,
   363→    * ``duplicate pairs`` -- per (cell line, compound) pair occupying
   364→      more than one row (pairs keyed by a missing identifier included);
   365→      ``count`` is the number of rows for the pair,
   366→    * ``ambiguous tissue`` -- per cell line mapped to more than one
   367→      tissue; ``count`` is the number of distinct tissues,
   368→    * ``under-tested compound`` -- per compound measured on fewer than
   369→      ``n_min`` cell lines; ``count`` is the number of cell lines.
   370→
   371→    A clean screen returns the single row ``("ok", "no issues found",
   372→    0)``.  ``to_string()`` on the result is print-friendly.
   373→
   374→    Parameters
   375→    ----------
   376→    screen:
   377→        The screen to audit.
   378→    n_min:
   379→        Minimum number of cell lines a compound should be measured on
   380→        (GDSC-style panels use 3 as the floor for any per-compound
   381→        statistic).
   382→    """
   383→    rows: list[dict[str, object]] = []
   384→
   385→    for column in (screen.cell_col, screen.compound_col):
   386→        missing_ids = int(screen.frame[column].isna().sum())
   387→        if missing_ids:
   388→            rows.append(
   389→                {
   390→                    "issue": "missing identifier",
   391→                    "detail": f"column {column!r}",
   392→                    "count": missing_ids,
   393→                }
   394→            )
   395→
   396→    for metric in screen.metrics:
   397→        missing = int(screen.frame[metric].isna().sum())
   398→        if missing:
   399→            rows.append(
   400→                {
   401→                    "issue": "missing metric values",
   402→                    "detail": f"metric {metric!r}",
   403→                    "count": missing,
   404→                }
   405→            )
   406→
   407→    pair_counts = screen.frame.groupby(
   408→        [screen.cell_col, screen.compound_col], sort=True, dropna=False
   409→    ).size()
   410→    for (cell, compound), count in pair_counts.items():
   411→        if count > 1:
   412→            rows.append(
   413→                {
   414→                    "issue": "duplicate pairs",
   415→                    "detail": f"{cell} / {compound}",
   416→                    "count": int(count),
   417→                }
   418→            )
   419→
   420→    if screen.tissue_col is not None:
   421→        tissues = (
   422→            screen.frame[[screen.cell_col, screen.tissue_col]]
   423→            .dropna(subset=[screen.tissue_col])
   424→            .groupby(screen.cell_col, sort=True, dropna=False)[screen.tissue_col]
   425→            .agg(lambda values: sorted(set(values.astype(str))))
   426→        )
   427→        for cell, tissue_list in tissues.items():
   428→            if len(tissue_list) > 1:
   429→                rows.append(
   430→                    {
   431→                        "issue": "ambiguous tissue",
   432→                        "detail": f"{cell}: {', '.join(tissue_list)}",
   433→                        "count": len(tissue_list),
   434→                    }
   435→                )
   436→
   437→    n_cells = screen.frame.groupby(screen.compound_col, sort=True, dropna=False)[
   438→        screen.cell_col
   439→    ].nunique()
   440→    for compound, count in n_cells.items():
   441→        if count < n_min:
   442→            rows.append(
   443→                {
   444→                    "issue": "under-tested compound",
   445→                    "detail": f"{compound}",
   446→                    "count": int(count),
   447→                }
   448→            )
   449→
   450→    if not rows:
   451→        rows.append({"issue": "ok", "detail": "no issues found", "count": 0})
   452→    report = pd.DataFrame(rows, columns=["issue", "detail", "count"])
   453→    _LOG.info("Response-table audit: %d issue row(s)", len(report))
   454→    return report

   457→def collapse_replicates(screen: ResponseScreen) -> ResponseScreen:
   458→    """Collapse duplicated (cell line, compound) pairs by the median.
   459→
   460→    Every pair is reduced to a single row: metric columns take the
   461→    median over the replicates, annotation columns (tissue, study ...)
   462→    keep the first value.  Pairs without duplicates pass through
   463→    untouched.  Rows whose cell-line or compound identifier is missing
   464→    are kept as their own (NaN-keyed) group instead of being dropped
   465→    (pandas' ``groupby`` default would silently discard them); a
   466→    warning names the count of such rows.  The collapsed pairs are
   467→    logged at DEBUG level.
   468→
   469→    Returns
   470→    -------
   471→    ResponseScreen
   472→        A new screen (the input is never mutated) with at most one row
   473→        per pair, preserving first-appearance row order.
   474→    """
   475→    frame = screen.frame
   476→    missing_ids = frame[screen.cell_col].isna() | frame[screen.compound_col].isna()
   477→    if missing_ids.any():
   478→        _LOG.warning(
   479→            "%d row(s) have a missing cell-line/compound identifier -- kept "
   480→            "as their own (NaN) group instead of being dropped "
   481→            "(audit_response_table reports them)",
   482→            int(missing_ids.sum()),
   483→        )
   484→    aggregations: dict[str, str] = {}
   485→    for column in frame.columns:
   486→        aggregations[column] = (
   487→            "median" if column in screen.metrics else "first"
   488→        )
   489→    collapsed = frame.groupby(
   490→        [screen.cell_col, screen.compound_col],
   491→        sort=False,
   492→        as_index=False,
   493→        dropna=False,
   494→    ).agg(aggregations)
   495→
   496→    duplicate_pairs = screen._duplicate_pairs()
   497→    if duplicate_pairs:
   498→        shown = ", ".join(f"({cell} + {compound})" for cell, compound in duplicate_pairs[:10])
   499→        extra = len(duplicate_pairs) - 10
   500→        more = f" (+{extra} more)" if extra > 0 else ""
   501→        _LOG.debug(
   502→            "Collapsed %d duplicate pair(s) by median: %s%s",
   503→            len(duplicate_pairs),
   504→            shown,
   505→            more,
   506→        )
   507→    _LOG.info(
   508→        "collapse_replicates: %d row(s) -> %d row(s)", len(frame), len(collapsed)
   509→    )
   510→    return ResponseScreen(
   511→        frame=collapsed,
   512→        cell_col=screen.cell_col,
   513→        compound_col=screen.compound_col,
   514→        tissue_col=screen.tissue_col,
   515→        metrics=list(screen.metrics),
   516→    )

   519→def ln_ic50_to_pic50(ln_ic50):
   520→    """Convert GDSC ``LN_IC50`` values to pIC50.
   521→
   522→    GDSC reports the natural log of the IC50 in micromolar, so
   523→    ``LN_IC50 = 0`` (IC50 = 1 uM) maps to ``pIC50 = 6`` via
   524→    ``pIC50 = 6 - ln(IC50_uM) / ln(10)``.
   525→
   526→    Parameters
   527→    ----------
   528→    ln_ic50:
   529→        Scalar, list/array or :class:`pandas.Series` of ``LN_IC50``
   530→        values.  NaN passes through; a Series keeps its index.
   531→
   532→    Returns
   533→    -------
   534→    Same kind as the input (float for scalars, ndarray for arrays,
   535→    Series for Series) with the converted values.
   536→    """
   537→    if isinstance(ln_ic50, pd.Series):
   538→        numeric = pd.to_numeric(ln_ic50, errors="coerce")
   539→        return 6.0 - numeric / _LN10
   540→    values = np.asarray(ln_ic50, dtype=float)
   541→    result = 6.0 - values / _LN10
   542→    if result.ndim == 0:
   543→        return float(result)
   544→    return result
   545→
   546→
   547→def pic50_to_ln_ic50(pic50):
   548→    """Convert pIC50 values back to GDSC ``LN_IC50`` (the exact inverse).
   549→
   550→    ``ln(IC50_uM) = (6 - pIC50) * ln(10)``; a 1 uM IC50 (pIC50 6)
   551→    maps back to ``LN_IC50 = 0``.  Accepts the same input kinds as
   552→    :func:`ln_ic50_to_pic50` with NaN passthrough.
   553→    """
   554→    if isinstance(pic50, pd.Series):
   555→        numeric = pd.to_numeric(pic50, errors="coerce")
   556→        return (6.0 - numeric) * _LN10
   557→    values = np.asarray(pic50, dtype=float)
   558→    result = (6.0 - values) * _LN10
   559→    if result.ndim == 0:
   560→        return float(result)
   561→    return result

   564→def response_matrix(screen: ResponseScreen, metric: str) -> pd.DataFrame:
   565→    """Pivot one metric into a cell-line x compound sensitivity matrix.
   566→
   567→    Rows are cell lines, columns are compounds (both sorted); untested
   568→    pairs hold NaN.  Duplicated pairs are *not* silently averaged --
   569→    collapse them first with :func:`collapse_replicates`.
   570→
   571→    Parameters
   572→    ----------
   573→    screen:
   574→        The screen to pivot.
   575→    metric:
   576→        Metric column to place in the matrix.
   577→
   578→    Returns
   579→    -------
   580→    pandas.DataFrame
   581→        The pivoted matrix.
   582→
   583→    Raises
   584→    ------
   585→    DataValidationError
   586→        If ``metric`` is not one of the screen's metrics, or duplicate
   587→        (cell line, compound) pairs are present.
   588→    """
   589→    if metric not in screen.metrics:
   590→        raise DataValidationError(
   591→            f"metric {metric!r} is not part of the screen -- available "
   592→            f"metrics: {list(screen.metrics)}"
   593→        )
   594→    duplicates = screen._duplicate_pairs()
   595→    if duplicates:
   596→        shown = ", ".join(f"({cell} + {compound})" for cell, compound in duplicates[:5])
   597→        extra = len(duplicates) - 5
   598→        more = f" (+{extra} more)" if extra > 0 else ""
   599→        raise DataValidationError(
   600→            f"{len(duplicates)} duplicate (cell, compound) pair(s) prevent "
   601→            f"pivoting: {shown}{more} -- collapse_replicates() first"
   602→        )
   603→    matrix = screen.frame.pivot(
   604→        index=screen.cell_col, columns=screen.compound_col, values=metric
   605→    )
   606→    matrix.index.name = screen.cell_col
   607→    matrix.columns.name = screen.compound_col
   608→    return matrix.sort_index().sort_index(axis=1)

   611→def summarize_compounds(
   612→    screen: ResponseScreen,
   613→    metric: str,
   614→    higher_is_more_sensitive: bool = False,
   615→) -> pd.DataFrame:
   616→    """Per-compound potency statistics across the cell-line panel.
   617→
   618→    Rows are compounds (sorted) with columns ``n_cell_lines`` (number of
   619→    distinct cell lines measured, replicates counted once -- missing
   620→    metric values are the audit's business), ``median``, ``min``,
   621→    ``max``, ``std`` (sample standard deviation, ddof=1) and
   622→    ``sensitive_spread`` = ``max - min`` -- the potency range across
   623→    the panel, i.e. how selectively the compound hits the panel
   624→    regardless of direction.  Rows whose compound identifier is missing
   625→    form their own trailing ``NaN`` row instead of being dropped
   626→    (:func:`audit_response_table` reports them).
   627→
   628→    Parameters
   629→    ----------
   630→    screen:
   631→        The screen to summarise.
   632→    metric:
   633→        Metric column to summarise.
   634→    higher_is_more_sensitive:
   635→        Direction of ``metric``: ``False`` (default) for LN_IC50-style
   636→        metrics where *lower* means more potent; ``True`` for
   637→        pIC50-style metrics where *higher* means more potent.  It only
   638→        affects the ``most_sensitive_cell_line`` column -- the cell
   639→        line with the strongest response under that direction.
   640→
   641→    Raises
   642→    ------
   643→    DataValidationError
   644→        If ``metric`` is not one of the screen's metrics.
   645→    """
   646→    if metric not in screen.metrics:
   647→        raise DataValidationError(
   648→            f"metric {metric!r} is not part of the screen -- available "
   649→            f"metrics: {list(screen.metrics)}"
   650→        )
   651→    grouped = screen.frame.groupby(
   652→        screen.compound_col, sort=True, dropna=False
   653→    )[metric]
   654→    summary = pd.DataFrame(
   655→        {
   656→            # distinct cell lines (replicates counted once); missing
   657→            # metric values stay the audit's job to report.
   658→            "n_cell_lines": screen.frame.groupby(
   659→                screen.compound_col, sort=True, dropna=False
   660→            )[screen.cell_col].nunique(),
   661→            "median": grouped.median(),
   662→            "min": grouped.min(),
   663→            "max": grouped.max(),
   664→            "std": grouped.std(),
   665→        }
   666→    )
   667→    summary["sensitive_spread"] = summary["max"] - summary["min"]
   668→
   669→    # Direction-aware pick of the strongest-responding cell line per
   670→    # compound (NaN metric rows cannot win by construction).
   671→    lookup = screen.frame[
   672→        [screen.compound_col, screen.cell_col, metric]
   673→    ].dropna(subset=[metric])
   674→    if higher_is_more_sensitive:
   675→        positions = lookup.groupby(
   676→            screen.compound_col, sort=True, dropna=False
   677→        )[metric].idxmax()
   678→    else:
   679→        positions = lookup.groupby(
   680→            screen.compound_col, sort=True, dropna=False
   681→        )[metric].idxmin()
   682→    # ``positions`` maps compound -> winning row label; re-align the picked
   683→    # cell lines onto the compound index before assigning into the summary.
   684→    summary["most_sensitive_cell_line"] = pd.Series(
   685→        lookup.loc[positions, screen.cell_col].to_numpy(), index=positions.index
   686→    )
   687→    summary = summary[
   688→        [
   689→            "n_cell_lines",
   690→            "median",
   691→            "min",
   692→            "max",
   693→            "std",
   694→            "sensitive_spread",
   695→            "most_sensitive_cell_line",
   696→        ]
   697→    ]
   698→    summary.index.name = screen.compound_col
   699→    return summary

   702→def call_sensitive_resistant(
   703→    response_series,
   704→    sensitive_quantile: float = 0.25,
   705→    resistant_quantile: float = 0.75,
   706→    lower_is_sensitive: bool = True,
   707→):
   708→    """Threshold a response vector into sensitivity calls.
   709→
   710→    The lowest ``sensitive_quantile`` fraction of cells is called
   711→    ``"sensitive"``, the highest ``resistant_quantile`` fraction
   712→    ``"resistant"`` and everything in between ``"intermediate"``
   713→    (thresholds are inclusive, so tied values stay on the boundary
   714→    side).  NaN entries are labelled ``"not tested"``.
   715→
   716→    Parameters
   717→    ----------
   718→    response_series:
   719→        Response values, typically one compound's column of a
   720→        :func:`response_matrix` pivot.
   721→    sensitive_quantile, resistant_quantile:
   722→        Quantile cuts; must satisfy
   723→        ``0 < sensitive_quantile < resistant_quantile < 1``.
   724→    lower_is_sensitive:
   725→        Direction of the metric: ``True`` (default) for LN_IC50-style
   726→        metrics where *lower* means more sensitive; ``False`` for
   727→        pIC50-style metrics where *higher* means more sensitive (the
   728→        quantile cuts are mirrored accordingly).
   729→
   730→    Returns
   731→    -------
   732→    pandas.Series
   733→        Labels ``"sensitive"`` / ``"resistant"`` / ``"intermediate"``
   734→        / ``"not tested"``, indexed like the input.
   735→
   736→    Raises
   737→    ------
   738→    DataValidationError
   739→        If the quantile ordering is violated.
   740→    """
   741→    if not (
   742→        0.0 < sensitive_quantile < resistant_quantile < 1.0
   743→    ):
   744→        raise DataValidationError(
   745→            "quantiles must satisfy 0 < sensitive_quantile < "
   746→            f"resistant_quantile < 1, got {sensitive_quantile} and "
   747→            f"{resistant_quantile}"
   748→        )
   749→    series = (
   750→        response_series
   751→        if isinstance(response_series, pd.Series)
   752→        else pd.Series(response_series)
   753→    )
   754→    numeric = pd.to_numeric(series, errors="coerce")
   755→    labels = pd.Series("not tested", index=numeric.index, dtype=object)
   756→
   757→    valid = numeric.dropna()
   758→    if valid.empty:
   759→        _LOG.warning(
   760→            "call_sensitive_resistant: no non-NaN values -- every cell line "
   761→            "labelled 'not tested'"
   762→        )
   763→        return labels
   764→
   765→    if lower_is_sensitive:
   766→        sensitive_cut = valid.quantile(sensitive_quantile)
   767→        resistant_cut = valid.quantile(resistant_quantile)
   768→        is_sensitive = numeric <= sensitive_cut
   769→        is_resistant = numeric >= resistant_cut
   770→    else:
   771→        # High values are the sensitive ones: mirror the quantile cuts.
   772→        sensitive_cut = valid.quantile(1.0 - sensitive_quantile)
   773→        resistant_cut = valid.quantile(1.0 - resistant_quantile)
   774→        is_sensitive = numeric >= sensitive_cut
   775→        is_resistant = numeric <= resistant_cut
   776→
   777→    labels[is_sensitive] = "sensitive"
   778→    labels[is_resistant & ~is_sensitive] = "resistant"
   779→    intermediate = numeric.notna() & ~is_sensitive & ~is_resistant
   780→    labels[intermediate] = "intermediate"
   781→    _LOG.debug(
   782→        "Sensitivity calls: %d sensitive, %d resistant, %d intermediate, "
   783→        "%d not tested",
   784→        int((labels == "sensitive").sum()),
   785→        int((labels == "resistant").sum()),
   786→        int((labels == "intermediate").sum()),
   787→        int((labels == "not tested").sum()),
   788→    )
   789→    return labels
