   126→/** Qualitative label for a Pearson correlation coefficient. */
   127→function corrStrength(r: number): string {
   128→  const a = Math.abs(r);
   129→  if (a >= 0.9) return "very strong";
   130→  if (a >= 0.7) return "strong";
   131→  if (a >= 0.5) return "moderate";
   132→  return "weak";
   133→}

   135→/** Compact "time ago" label for run-history rows (client-side only). */
   136→function timeAgo(iso: string): string {
   137→  const t = Date.parse(iso);
   138→  if (!Number.isFinite(t)) return "";
   139→  const s = Math.max(0, Math.floor((Date.now() - t) / 1000));
   140→  if (s < 60) return `${s}s ago`;
   141→  if (s < 3600) return `${Math.floor(s / 60)}m ago`;
   142→  if (s < 86400) return `${Math.floor(s / 3600)}h ago`;
   143→  return `${Math.floor(s / 86400)}d ago`;
   144→}

   271→  /** Export the full deterministic dataset (all rows) as a CSV file. */
   272→  async function downloadCsv() {
   273→    if (!seedValid) {
   274→      setError("Seed must be a non-negative integer (0 – 2147483647).");
   275→      return;
   276→    }
   277→    setCsvLoading(true);
   278→    setError(null);
   279→    const seedNum = Number.parseInt(seed, 10) || 0;
   280→    const fileName = `pharmaml_${task}_n${samples}_seed${seedNum}.csv`;
   281→    try {
   282→      const qs = new URLSearchParams({
   283→        task,
   284→        samples: String(samples),
   285→        noise: String(noise),
   286→        seed: String(seedNum),
   287→      });
   288→      const res = await fetch(`/api/ml/dataset/csv?${qs.toString()}`);
   289→      if (!res.ok) {
   290→        let message = `HTTP ${res.status}`;
   291→        try {
   292→          const data: unknown = await res.json();
   293→          if (typeof data === "object" && data !== null && "error" in data) {
   294→            message = String((data as { error: unknown }).error);
   295→          }
   296→        } catch {
   297→          /* body wasn't JSON — keep the status message */
   298→        }
   299→        throw new Error(message);
   300→      }
   301→      const blob = await res.blob();
   302→      const url = URL.createObjectURL(blob);
   303→      const anchor = document.createElement("a");
   304→      anchor.href = url;
   305→      anchor.download = fileName;
   306→      document.body.appendChild(anchor);
   307→      anchor.click();
   308→      document.body.removeChild(anchor);
   309→      URL.revokeObjectURL(url);
   310→      toast.success("Dataset exported", {
   311→        description: `${fileName} · ${samples} rows · ${fmtBytes(blob.size)}`,
   312→      });
   313→    } catch (err) {
   314→      setError(err instanceof Error ? err.message : "CSV export failed.");
   315→    } finally {
   316→      setCsvLoading(false);
   317→    }

   348→  /** Load CSV text into the store, reset column picks, drop stale results.
   349→   *  Returns false (with a toast) when the text is unusable. */
   350→  function loadCsv(fileName: string, text: string): boolean {
   351→    if (text.trim().length < 4 || text.length > MAX_CSV_TEXT) {
   352→      toast.error("CSV not loaded", {
   353→        description: "Expected a non-empty file of at most 256 KB.",
   354→      });
   355→      return false;
   356→    }
   357→    const table = parseCsv(text);
   358→    if (table.columns.length === 0 || table.rows.length === 0) {
   359→      toast.error("Could not parse the CSV", {
   360→        description: "Expected a header row followed by data rows.",
   361→      });
   362→      return false;
   363→    }
   364→    setMlCsv({ fileName, text });
   365→    setMlConfig({ featureColumn: "", featureColumn2: "", targetColumn: "" });
   366→    setStatus("idle");
   367→    setResult(null);
   368→    setPreview(null);
   369→    setError(null);
   370→    setXZoom(null);
   371→    toast.success("CSV loaded", {
   372→      description: `${fileName} — ${table.rows.length} rows × ${table.columns.length} columns · ${numericColumnNames(table).length} numeric`,
   373→    });
   374→    return true;
   375→  }

   390→      });
   391→    reader.readAsText(file);

   437→  async function run() {
   438→    if (source === "synthetic" && !seedValid) {
   439→      setError("Seed must be a non-negative integer (0 – 2147483647).");
   440→      return;
   441→    }
   442→    if (source === "csv" && (!mlCsv || !csvUsable)) {
   443→      setError(
   444→        `Load a CSV with at least ${MIN_CSV_ROWS} rows and 2 numeric columns before training.`
   445→      );
   446→      return;
   447→    }
   448→    if (task === "regression" && regModel === "ridge" && !alphaValid) {
   449→      setError("Ridge alpha must be a number ≥ 0.");
   450→      return;
   451→    }
   452→    if (task === "classification" && !l2Valid) {
   453→      setError("L2 strength must be a number ≥ 0.");
   454→      return;
   455→    }
   456→    setStatus("running");
   457→    setError(null);
   458→    const body: MLTrainRequest = {
   459→      task,
   460→      samples,
   461→      noise,
   462→      model: task === "regression" ? regModel : "logistic",
   463→      testSize: testSizePct / 100,
   464→      cv: (Number.parseInt(cv, 10) || 0) as 0 | 3 | 5 | 10,
   465→      seed: Number.parseInt(seed, 10) || 0,
   466→    };
   467→    if (task === "regression" && regModel === "ridge") body.alpha = Number.parseFloat(alpha);
   468→    if (task === "classification") body.l2 = Number.parseFloat(l2);
   469→    if (source === "csv" && mlCsv) {
   470→      body.csv = mlCsv.text;
   471→      body.csvName = mlCsv.fileName;
   472→      if (task === "regression") {
   473→        if (featureColumns.length > 0) body.featureColumns = featureColumns;
   474→        if (targetColumn) body.targetColumn = targetColumn;
   475→      } else {
   476→        if (featureColumn) body.featureColumn = featureColumn;
   477→        if (featureColumn2) body.featureColumn2 = featureColumn2;
   478→        if (targetColumn) body.targetColumn = targetColumn;
   479→      }
   480→    }
   481→
   482→    try {
   483→      const res = await fetch("/api/ml/train", {
   484→        method: "POST",
   485→        headers: { "Content-Type": "application/json" },
   486→        body: JSON.stringify(body),
   487→      });
   488→      const data: unknown = await res.json();
   489→      if (!res.ok) {
   490→        const message =
   491→          typeof data === "object" && data !== null && "error" in data
   492→            ? `${String((data as { error: unknown }).error)}${
   493→                "detail" in data && Array.isArray((data as { detail: unknown }).detail)
   494→                  ? ` ${((data as { detail: unknown[] }).detail as string[]).join("; ")}`
   495→                  : ""
   496→              }`
   497→            : `HTTP ${res.status}`;
   498→        throw new Error(message);
   499→      }
   500→      const payload = data as MLTrainResponse;
   501→      setResult(payload);
   502→      setStatus("done");
   503→      /* Record the completed run in the history strip (restorable). */
   504→      const key = payload.task === "regression" ? "r2" : "accuracy";
   505→      const keyValue = payload.metrics[key];
   506→      if (typeof keyValue === "number" && Number.isFinite(keyValue)) {
   507→        pushMlRun({
   508→          id: Date.now(),
   509→          at: new Date().toISOString(),
   510→          task: payload.task,
   511→          model: payload.model,
   512→          keyMetric: payload.task === "regression" ? "R²" : "Accuracy",
   513→          keyValue,
   514→          config: { ...mlConfig },
   515→          result: payload,
   516→          ...(source === "csv" && mlCsv
   517→            ? {
   518→                csv: {
   519→                  fileName: mlCsv.fileName,
   520→                  featureColumns:
   521→                    task === "regression"
   522→                      ? featureColumns
   523→                      : [featureColumn, featureColumn2].filter(Boolean),
   524→                  targetColumn,
   525→                },
   526→              }
   527→            : {}),
   528→        });
   529→      }
   530→    } catch (err) {
   531→      setStatus("error");
   532→      const message = err instanceof Error ? err.message : "Training request failed.";
   533→      setError(message);
   534→    }
   535→  }
