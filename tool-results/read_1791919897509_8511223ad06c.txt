     4→import { Brain, Loader2, Play, TriangleAlert } from "lucide-react";
     5→import {
     6→  CartesianGrid,
     7→  Legend,
     8→  ResponsiveContainer,
     9→  Scatter,
    10→  ScatterChart,
    11→  Tooltip,

    13→  YAxis,
    14→} from "recharts";
    15→import { Alert, AlertDescription, AlertTitle } from "@/components/ui/alert";
    16→import { Button } from "@/components/ui/button";
    17→import {
    18→  Card,
    19→  CardContent,
    20→  CardDescription,
    21→  CardHeader,
    22→  CardTitle,
    23→} from "@/components/ui/card";
    24→import { Input } from "@/components/ui/input";
    25→import { Label } from "@/components/ui/label";
    26→import {
    27→  Select,
    28→  SelectContent,
    29→  SelectItem,
    30→  SelectTrigger,
    31→  SelectValue,
    32→} from "@/components/ui/select";
    33→import { Slider } from "@/components/ui/slider";
    34→import {

    36→  TableBody,

    44→import type { MLTrainResponse, MLTrainRequest } from "@/lib/toolkit-types";
    45→
    46→type Task = "regression" | "classification";
    47→type RegModel = "linear" | "polynomial" | "ridge";
    48→type CvValue = "0" | "3" | "5" | "10";
    49→
    50→const REG_METRICS: Array<{ key: string; label: string }> = [
    51→  { key: "r2", label: "R²" },
    52→  { key: "rmse", label: "RMSE" },
    53→  { key: "mae", label: "MAE" },
    54→];
    55→
    56→const CLS_METRICS: Array<{ key: string; label: string }> = [
    57→  { key: "accuracy", label: "Accuracy" },
    58→  { key: "precision", label: "Precision" },
    59→  { key: "recall", label: "Recall" },
    60→  { key: "f1", label: "F1" },
    61→];
    62→
    63→function MetricCard({ label, value }: { label: string; value: number }) {
    64→  return (
    65→    <div className="rounded-lg border bg-card px-3 py-2.5 shadow-sm">
    66→      <p className="text-xs text-muted-foreground">{label}</p>
    67→      <p className="font-mono text-xl font-semibold tabular-nums">{fmtNum(value, 4)}</p>
    68→    </div>
    69→  );
    70→}
    71→
    72→/** Factory for tiny boundary dot renderers (60×60 grid → 3600 points). */
    73→function TinyDot(color: string) {
    74→  return function renderTinyDot(props: { cx?: number; cy?: number }) {
    75→    const { cx, cy } = props;
    76→    if (cx === undefined || cy === undefined) return null;
    77→    return <circle cx={cx} cy={cy} r={1.5} fill={color} fillOpacity={0.3} />;
    78→  };
    79→}
    80→
    81→/** Factory for the larger moon-class dots. */
    82→function MoonDot(color: string) {
    83→  return function renderMoonDot(props: { cx?: number; cy?: number }) {
    84→    const { cx, cy } = props;
    85→    if (cx === undefined || cy === undefined) return null;
    86→    return <circle cx={cx} cy={cy} r={3.2} fill={color} fillOpacity={0.9} stroke={color} strokeWidth={0.6} />;
    87→  };
    88→}
    89→
    90→export function MlPlayground() {
    91→  const [task, setTask] = useState<Task>("regression");
    92→  const [samples, setSamples] = useState(300);
    93→  const [noise, setNoise] = useState(0.4);
    94→  const [regModel, setRegModel] = useState<RegModel>("linear");
    95→  const [alpha, setAlpha] = useState("1.0");
    96→  const [l2, setL2] = useState("1.0");
    97→  const [testSizePct, setTestSizePct] = useState(20);
    98→  const [cv, setCv] = useState<CvValue>("5");
    99→  const [seed, setSeed] = useState("42");
   100→
   101→  const [status, setStatus] = useState<"idle" | "running" | "done" | "error">("idle");
   102→  const [error, setError] = useState<string | null>(null);
   103→  const [result, setResult] = useState<MLTrainResponse | null>(null);
   104→
   105→  const noiseMax = task === "regression" ? 2 : 1.5;
   106→  const metricSpecs = task === "regression" ? REG_METRICS : CLS_METRICS;
   107→
   108→  function switchTask(next: Task) {
   109→    setTask(next);
   110→    if (next === "classification" && noise > 1.5) setNoise(0.3);
   111→    if (next === "regression" && regModel !== "ridge" && regModel !== "linear" && regModel !== "polynomial") {
   112→      setRegModel("linear");
   113→    }
   114→    setStatus("idle");
   115→    setResult(null);
   116→    setError(null);
   117→  }
   118→
   119→  const seedValid = /^\d{1,10}$/.test(seed.trim());
   120→  const alphaValid = Number.isFinite(Number.parseFloat(alpha)) && Number.parseFloat(alpha) >= 0;
   121→  const l2Valid = Number.isFinite(Number.parseFloat(l2)) && Number.parseFloat(l2) >= 0;
   122→
   123→  async function run() {
   124→    if (!seedValid) {
   125→      setError("Seed must be a non-negative integer (0 – 2147483647).");
   126→      return;
   127→    }
   128→    if (task === "regression" && regModel === "ridge" && !alphaValid) {
   129→      setError("Ridge alpha must be a number ≥ 0.");
   130→      return;
   131→    }
   132→    if (task === "classification" && !l2Valid) {
   133→      setError("L2 strength must be a number ≥ 0.");
   134→      return;
   135→    }
   136→    setStatus("running");
   137→    setError(null);
   138→    const body: MLTrainRequest = {
   139→      task,
   140→      samples,
   141→      noise,
   142→      model: task === "regression" ? regModel : "logistic",
   143→      testSize: testSizePct / 100,
   144→      cv: (Number.parseInt(cv, 10) || 0) as 0 | 3 | 5 | 10,
   145→      seed: Number.parseInt(seed, 10) || 0,
   146→    };
   147→    if (task === "regression" && regModel === "ridge") body.alpha = Number.parseFloat(alpha);
   148→    if (task === "classification") body.l2 = Number.parseFloat(l2);
   149→
   150→    try {
   151→      const res = await fetch("/api/ml/train", {
   152→        method: "POST",
   153→        headers: { "Content-Type": "application/json" },
   154→        body: JSON.stringify(body),
   155→      });
   156→      const data: unknown = await res.json();
   157→      if (!res.ok) {
   158→        const message =
   159→          typeof data === "object" && data !== null && "error" in data
   160→            ? `${String((data as { error: unknown }).error)}${
   161→                "detail" in data && Array.isArray((data as { detail: unknown }).detail)
   162→                  ? ` ${((data as { detail: unknown[] }).detail as string[]).join("; ")}`
   163→                  : ""
   164→              }`
   165→            : `HTTP ${res.status}`;
   166→        throw new Error(message);
   167→      }
   168→      setResult(data as MLTrainResponse);
   169→      setStatus("done");
   170→    } catch (err) {
   171→      setStatus("error");
   172→      const message = err instanceof Error ? err.message : "Training request failed.";
   173→      setError(message);
   174→    }
   175→  }
   176→
   177→  const running = status === "running";
   178→
   179→  /* Chart series (memoized splits so recharts gets stable references). */
   180→  const clsSeries = useMemo(() => {
   181→    if (!result || result.task !== "classification") return null;
   182→    const boundary = result.series.boundary ?? [];
   183→    return {
   184→      grid0: boundary.filter((p) => p.pred === 0),
   185→      grid1: boundary.filter((p) => p.pred === 1),
   186→      moon0: result.series.test.filter((p) => p.label === 0),
   187→      moon1: result.series.test.filter((p) => p.label === 1),
   188→    };
   189→  }, [result]);
   190→
   191→  const coefficientsText = useMemo(() => {
   192→    if (!result) return null;
   193→    const { intercept, coefs } = result.coefficients;
   194→    if (result.task === "regression") {
   195→      const terms = coefs
   196→        .map((c, i) => `${fmtNum(c, 4)}·x̂${coefs.length > 1 ? `^${i + 1}` : ""}`)
   197→        .join(" + ");
   198→      return `ŷ = ${fmtNum(intercept, 4)} + ${terms}   (x̂ = standardized x)`;
   199→    }
   200→    return `σ(b₀ + w₁x̂₁ + w₂x̂₂)   b₀=${fmtNum(intercept, 4)}, w=[${coefs
   201→      .map((c) => fmtNum(c, 4))
   202→      .join(", ")}]`;
   203→  }, [result]);
   204→
   205→  return (
   206→    <div className="grid gap-4 xl:grid-cols-[minmax(300px,360px)_minmax(0,1fr)]">
   207→      {/* ------------------------------ Controls ------------------------------ */}
   208→      <Card className="h-fit overflow-hidden py-0 shadow-sm xl:sticky xl:top-24">
   209→        <CardHeader className="border-b bg-muted/30 py-3">
   210→          <div className="flex items-center gap-2.5">
   211→            <span className="flex size-9 items-center justify-center rounded-lg bg-primary/10 text-primary">
   212→              <Brain className="size-4.5" aria-hidden />
   213→            </span>
   214→            <div>
   215→              <CardTitle className="text-base">Experiment setup</CardTitle>
   216→              <CardDescription className="text-xs">
   217→                Deterministic — same seed, same result
   218→              </CardDescription>
   219→            </div>
   220→          </div>
   221→        </CardHeader>
   222→        <CardContent className="space-y-5 pb-4">
   223→          {/* Task */}
   224→          <div className="space-y-1.5">
   225→            <Label className="text-xs font-medium">Task</Label>
   226→            <div className="grid grid-cols-2 gap-1 rounded-lg border bg-muted/40 p-1" role="tablist" aria-label="ML task">
   227→              {(["regression", "classification"] as const).map((t) => (
   228→                <button
   229→                  key={t}
   230→                  type="button"
   231→                  role="tab"
   232→                  aria-selected={task === t}
   233→                  onClick={() => switchTask(t)}
   234→                  className={`h-9 rounded-md text-sm font-medium capitalize transition-colors focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring ${
   235→                    task === t ? "bg-background shadow-sm text-foreground" : "text-muted-foreground hover:text-foreground"
   236→                  }`}
   237→                >
   238→                  {t}
   239→                </button>
   240→              ))}
   241→            </div>
   242→          </div>
   243→
   244→          {/* Samples */}
   245→          <div className="space-y-2">
   246→            <div className="flex items-center justify-between">
   247→              <Label htmlFor="ml-samples" className="text-xs font-medium">
   248→                Samples
   249→              </Label>
   250→              <span className="font-mono text-xs tabular-nums text-muted-foreground">{samples}</span>
   251→            </div>
   252→            <Slider
   253→              id="ml-samples"
   254→              min={100}
   255→              max={1000}
   256→              step={50}
   257→              value={[samples]}
   258→              onValueChange={(v) => setSamples(v[0] ?? samples)}
   259→            />
   260→          </div>
   261→
   262→          {/* Noise */}
   263→          <div className="space-y-2">
   264→            <div className="flex items-center justify-between">
   265→              <Label htmlFor="ml-noise" className="text-xs font-medium">
   266→                Noise {task === "classification" ? "(moons)" : "σ"}
   267→              </Label>
   268→              <span className="font-mono text-xs tabular-nums text-muted-foreground">
   269→                {noise.toFixed(2)}
   270→              </span>
   271→            </div>
   272→            <Slider
   273→              id="ml-noise"
   274→              min={0}
   275→              max={noiseMax}
   276→              step={0.05}
   277→              value={[Math.min(noise, noiseMax)]}
   278→              onValueChange={(v) => setNoise(v[0] ?? noise)}
   279→            />
   280→          </div>
   281→
   282→          {/* Model */}
   283→          <div className="space-y-1.5">
   284→            <Label className="text-xs font-medium">Model</Label>
   285→            {task === "regression" ? (
   286→              <Select value={regModel} onValueChange={(v) => setRegModel(v as RegModel)}>
   287→                <SelectTrigger className="h-10" aria-label="Regression model">
   288→                  <SelectValue />
   289→                </SelectTrigger>
   290→                <SelectContent>
   291→                  <SelectItem value="linear">Linear regression</SelectItem>
   292→                  <SelectItem value="polynomial">Polynomial (degree 2)</SelectItem>
   293→                  <SelectItem value="ridge">Ridge (α regularized)</SelectItem>
   294→                </SelectContent>
   295→              </Select>
   296→            ) : (
   297→              <div className="space-y-2">
   298→                <div className="flex h-10 items-center rounded-md border bg-muted/30 px-3 text-sm text-muted-foreground">
   299→                  Logistic regression (batch GD, 800 epochs)
   300→                </div>
   301→                <div className="space-y-1.5">
   302→                  <Label htmlFor="ml-l2" className="text-xs font-medium">
   303→                    L2 strength λ
   304→                  </Label>
   305→                  <Input
   306→                    id="ml-l2"
   307→                    inputMode="decimal"
   308→                    value={l2}
   309→                    onChange={(e) => setL2(e.target.value)}
   310→                    aria-invalid={l2Valid ? undefined : true}
   311→                    className={`h-10 font-mono text-sm tabular-nums ${l2Valid ? "" : "border-destructive/60"}`}
   312→                  />
   313→                  {!l2Valid ? (
   314→                    <p className="text-[11px] text-destructive">Must be a number ≥ 0.</p>
   315→                  ) : null}
   316→                </div>
   317→              </div>
   318→            )}
   319→          </div>
   320→
   321→          {task === "regression" && regModel === "ridge" ? (
   322→            <div className="space-y-1.5">
   323→              <Label htmlFor="ml-alpha" className="text-xs font-medium">
   324→                Ridge α
   325→              </Label>
   326→              <Input
   327→                id="ml-alpha"
   328→                inputMode="decimal"
   329→                value={alpha}
   330→                onChange={(e) => setAlpha(e.target.value)}
   331→                aria-invalid={alphaValid ? undefined : true}
   332→                className={`h-10 font-mono text-sm tabular-nums ${alphaValid ? "" : "border-destructive/60"}`}
   333→              />
   334→              {!alphaValid ? <p className="text-[11px] text-destructive">Must be a number ≥ 0.</p> : null}
   335→            </div>
   336→          ) : null}
   337→
   338→          {/* Test size */}
   339→          <div className="space-y-2">
   340→            <div className="flex items-center justify-between">
   341→              <Label htmlFor="ml-test" className="text-xs font-medium">
   342→                Test split
   343→              </Label>
   344→              <span className="font-mono text-xs tabular-nums text-muted-foreground">
   345→                {testSizePct}%
   346→              </span>
   347→            </div>
   348→            <Slider
   349→              id="ml-test"
   350→              min={10}
   351→              max={50}
   352→              step={5}
   353→              value={[testSizePct]}
   354→              onValueChange={(v) => setTestSizePct(v[0] ?? testSizePct)}
   355→            />
   356→          </div>
   357→
   358→          {/* CV + seed */}
   359→          <div className="grid grid-cols-2 gap-3">
   360→            <div className="space-y-1.5">
   361→              <Label className="text-xs font-medium">k-fold CV</Label>
   362→              <Select value={cv} onValueChange={(v) => setCv(v as CvValue)}>
   363→                <SelectTrigger className="h-10" aria-label="Cross-validation folds">
   364→                  <SelectValue />
   365→                </SelectTrigger>
   366→                <SelectContent>
   367→                  <SelectItem value="0">Off</SelectItem>
   368→                  <SelectItem value="3">3-fold</SelectItem>
   369→                  <SelectItem value="5">5-fold</SelectItem>
   370→                  <SelectItem value="10">10-fold</SelectItem>
   371→                </SelectContent>
   372→              </Select>
   373→            </div>
   374→            <div className="space-y-1.5">
   375→              <Label htmlFor="ml-seed" className="text-xs font-medium">
   376→                Seed
   377→              </Label>
   378→              <Input
   379→                id="ml-seed"
   380→                inputMode="numeric"
   381→                value={seed}
   382→                onChange={(e) => setSeed(e.target.value)}
   383→                aria-invalid={seedValid ? undefined : true}
   384→                className={`h-10 font-mono text-sm tabular-nums ${seedValid ? "" : "border-destructive/60"}`}
   385→              />
   386→              {!seedValid ? (
   387→                <p className="text-[11px] text-destructive">Non-negative integer.</p>
   388→              ) : null}
   389→            </div>
   390→          </div>
   391→
   392→          <Button
   393→            size="lg"
   394→            className="h-11 w-full gap-2"
   395→            onClick={() => void run()}
   396→            disabled={running}
   397→          >
   398→            {running ? (
   399→              <>
   400→                <Loader2 className="size-4 animate-spin" aria-hidden />
   401→                Training…
   402→              </>
   403→            ) : (
   404→              <>
   405→                <Play className="size-4" aria-hidden />
   406→                Train model
   407→              </>
   408→            )}
   409→          </Button>
   410→
   411→          {error ? (
   412→            <Alert variant="destructive">
   413→              <TriangleAlert className="size-4" aria-hidden />
   414→              <AlertTitle>Request failed</AlertTitle>
   415→              <AlertDescription>{error}</AlertDescription>
   416→            </Alert>
   417→          ) : null}
   418→        </CardContent>
   419→      </Card>
   420→
   421→      {/* ------------------------------ Results ------------------------------ */}
   422→      <div className="min-w-0 space-y-4">
   423→        {!result && status !== "running" ? (
   424→          <Card className="py-0 shadow-sm">
   425→            <CardContent className="flex flex-col items-center justify-center gap-2 py-16 text-center">
   426→              <span className="flex size-12 items-center justify-center rounded-full bg-primary/10 text-primary">
   427→                <Brain className="size-6" aria-hidden />
   428→              </span>
   429→              <p className="text-sm font-medium">No results yet</p>
   430→              <p className="max-w-sm text-xs text-muted-foreground">
   431→                Configure the experiment and press{" "}
   432→                <strong className="font-semibold text-foreground">Train model</strong>. Data is
   433→                generated server-side from your seed — regression fits a closed-form least-squares
   434→                model; classification trains logistic regression on two moons.
   435→              </p>
   436→            </CardContent>
   437→          </Card>
   438→        ) : null}
   439→
   440→        {status === "running" ? (
   441→          <Card className="py-0 shadow-sm">
   442→            <CardContent className="flex flex-col items-center justify-center gap-3 py-16 text-center">
   443→              <Loader2 className="size-8 animate-spin text-primary" aria-hidden />
   444→              <p className="text-sm font-medium">Training…</p>
   445→              <p className="text-xs text-muted-foreground">
   446→                Generating {samples} samples, fitting the model, evaluating the held-out split
   447→                {cv !== "0" ? ` and running ${cv}-fold CV` : ""}.
   448→              </p>
   449→            </CardContent>
   450→          </Card>
   451→        ) : null}
   452→
   453→        {result && status !== "running" ? (
   454→          <>
   455→            {/* Metrics */}
   456→            <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
   457→              {metricSpecs.map((m) => (
   458→                <MetricCard key={m.key} label={m.label} value={result.metrics[m.key] ?? Number.NaN} />
   459→              ))}
   460→              {result.cv ? (
   461→                <div className="col-span-2 rounded-lg border border-primary/25 bg-primary/5 px-3 py-2.5 shadow-sm sm:col-span-4">
   462→                  <p className="text-xs text-muted-foreground">
   463→                    {result.cv.k}-fold CV {result.cv.metricName}
   464→                  </p>
   465→                  <p className="font-mono text-xl font-semibold tabular-nums text-primary">
   466→                    {fmtNum(result.cv.mean, 4)}{" "}
   467→                    <span className="text-sm font-normal">± {fmtNum(result.cv.std, 3)}</span>
   468→                  </p>
   469→                </div>
   470→              ) : null}
   471→            </div>
   472→
   473→            {/* Chart */}
   474→            <Card className="overflow-hidden py-0 shadow-sm">
   475→              <CardHeader className="border-b bg-muted/30 py-3">
   476→                <CardTitle className="text-sm">
   477→                  {result.task === "regression"
   478→                    ? "Test points & fitted curve"
   479→                    : "Two moons & decision boundary"}
   480→                </CardTitle>
   481→                <CardDescription className="text-xs">
   482→                  {result.datasetMeta.nTotal} samples · {result.datasetMeta.nTrain} train ·{" "}
   483→                  {result.datasetMeta.nTest} test · seed {result.datasetMeta.seed}
   484→                </CardDescription>
   485→              </CardHeader>
   486→              <CardContent className="p-3 sm:p-4">
   487→                <div
   488→                  className="h-[340px] w-full"
   489→                  role="img"
   490→                  aria-label={
   491→                    result.task === "regression"
   492→                      ? "Scatter plot of test points with the fitted regression curve"
   493→                      : "Scatter plot of the two moon classes with model decision boundary"
   494→                  }
   495→                >
   496→                  <ResponsiveContainer width="100%" height="100%">
   497→                    {result.task === "regression" ? (
   498→                      <ScatterChart margin={{ top: 8, right: 14, bottom: 4, left: 0 }}>
   499→                        <CartesianGrid stroke={CHART.gridStroke} strokeOpacity={CHART.gridStrokeOpacity} />
   500→                        <XAxis
   501→                          type="number"
   502→                          dataKey="x"
   503→                          domain={["auto", "auto"]}
   504→                          tick={CHART.axisTick}
   505→                          tickFormatter={fmtTick}
   506→                          tickLine={false}
   507→                          axisLine={{ stroke: CHART.gridStroke, strokeOpacity: 0.4 }}
   508→                          label={{ value: "x", position: "insideBottomRight", offset: -2, fill: "#78716c", fontSize: 11 }}
   509→                        />
   510→                        <YAxis
   511→                          type="number"
   512→                          dataKey="y"
   513→                          width={44}
   514→                          tick={CHART.axisTick}
   515→                          tickFormatter={fmtTick}
   516→                          tickLine={false}
   517→                          axisLine={false}
   518→                        />
   519→                        <Tooltip
   520→                          cursor={{ strokeDasharray: "3 3" }}
   521→                          contentStyle={CHART.tooltipStyle}
   522→                          formatter={(value) => fmtNum(Number(value), 4)}
   523→                        />
   524→                        <Legend wrapperStyle={{ fontSize: 12, color: "#78716c" }} />
   525→                        <Scatter
   526→                          name="fitted curve"
   527→                          data={result.series.curve ?? []}
   528→                          line={{ stroke: CHART.emerald, strokeWidth: 2.5 }}
   529→                          shape="square"
   530→                          fill={CHART.emerald}
   531→                          fillOpacity={0.5}
