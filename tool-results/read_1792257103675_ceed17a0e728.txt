     1→"""Objective library for molecule generation: SMILES -> score in [0, 1].
     2→
     3→Every optimiser in :mod:`pharmaml.generative` (genetic algorithm and
     4→reinforcement learning alike) maximises a *scoring objective* -- a callable
     5→mapping a SMILES string to a float in ``[0, 1]``.  This module ships the
     6→common pharmacology-flavoured objectives:
     7→
     8→* :class:`QedObjective` -- RDKit quantitative estimate of drug-likeness;
     9→* :class:`LogPBandObjective` / :class:`MolecularWeightBandObjective` /
    10→  :class:`TpsaBandObjective` -- Gaussian bands around a target property
    11→  window (the interval ``[lo, hi]`` is the ±1σ band: 1.0 at the centre,
    12→  ≈0.61 at the edges, smoothly decaying outside);
    13→* :class:`SimilarityObjective` -- Morgan-fingerprint Tanimoto similarity
    14→  to a seed molecule;
    15→* :class:`NoveltyObjective` -- structural novelty against a reference
    16→  corpus (``1 - max`` Tanimoto to any corpus member -- a corpus member
    17→  scores ``0.0``, a structurally distant molecule scores high);
    18→* :class:`SubstructureObjective` -- hard pharmacophore constraint (1.0 when
    19→  the SMARTS matches, 0.0 otherwise);
    20→* :class:`MultiObjective` -- weighted geometric mean of several objectives
    21→  with zero-score vetoing, so one hard constraint can reject a molecule
    22→  outright.
    23→
    24→Invalid SMILES never crash an objective: they score ``0.0`` and are
    25→counted (``n_invalid`` / ``n_scored`` / :meth:`ScoringObjective.reset_stats`)
    26→-- the accounting the optimisers use to report valid-generation fractions.
    27→
    28→Custom objectives subclass :class:`ScoringObjective` and implement
    29→:meth:`ScoringObjective.score`; using the :meth:`_parse` helper gives the
    30→invalid-input accounting for free::
    31→
    32→    from pharmaml.generative import ScoringObjective
    33→
    34→    class HeavyAtomObjective(ScoringObjective):
    35→        name = "heavy_atoms"
    36→        def score(self, smiles: str) -> float:
    37→            mol = self._parse(smiles)
    38→            if mol is None:
    39→                return 0.0
    40→            return min(mol.GetNumHeavyAtoms() / 30.0, 1.0)
    41→
    42→RDKit is imported lazily (the :mod:`pharmaml.chem` precedent): importing
    43→this module never requires it, and calling ``score`` without RDKit raises
    44→an actionable :class:`~pharmaml.exceptions.DependencyError`.
    45→"""
    46→
    47→from __future__ import annotations
    48→
    49→import math
    50→from collections.abc import Sequence
    51→from typing import Any
    52→
    53→from pharmaml.exceptions import DataValidationError
    54→from pharmaml.logger import get_logger
    55→
    56→from ._utils import require_rdkit
    57→from .corpus import DEFAULT_CORPUS
    58→
    59→__all__ = [
    60→    "LogPBandObjective",
    61→    "MolecularWeightBandObjective",
    62→    "MultiObjective",

    71→_LOG = get_logger(__name__)
    72→
    73→#: Smallest band half-width accepted by the band objectives (a zero-width
    74→#: band would make the Gaussian explode; this floor keeps it finite).
    75→_MIN_BAND_WIDTH = 1e-6
    76→
    77→
    78→def _band_score(value: float, lo: float, hi: float) -> float:
    79→    """Gaussian band score: 1.0 at the centre, ≈0.61 at the edges."""
    80→    centre = (lo + hi) / 2.0
    81→    sigma = max((hi - lo) / 2.0, _MIN_BAND_WIDTH)
    82→    z = (value - centre) / sigma
    83→    return math.exp(-0.5 * z * z)

    86→class ScoringObjective:
    87→    """Base class for generation objectives: ``score(smiles) -> [0, 1]``.
    88→
    89→    Subclasses set the ``name`` class attribute (used as the breakdown
    90→    column name inside :class:`MultiObjective`) and implement
    91→    :meth:`score`.  Parsing a SMILES through :meth:`_parse` records the
    92→    invalid-input accounting (``n_scored`` / ``n_invalid``) that the
    93→    optimisers surface in their history.
    94→
    95→    Example
    96→    -------
    97→    >>> from pharmaml.generative import ScoringObjective
    98→    >>> class ConstantObjective(ScoringObjective):
    99→    ...     name = "constant"
   100→    ...     def score(self, smiles):
   101→    ...         return 0.5
   102→    """
   103→
   104→    #: Short identifier used in breakdown tables and result columns.
   105→    name: str = "objective"
   106→
   107→    def __init__(self) -> None:
   108→        self._n_scored = 0
   109→        self._n_invalid = 0
   110→
   111→    # -- accounting ------------------------------------------------------
   112→    @property
   113→    def n_scored(self) -> int:
   114→        """Number of SMILES scored since construction / last reset."""
   115→        return self._n_scored
   116→
   117→    @property
   118→    def n_invalid(self) -> int:
   119→        """Number of unparseable SMILES seen (each scored 0.0)."""
   120→        return self._n_invalid
   121→
   122→    def reset_stats(self) -> None:
   123→        """Zero the ``n_scored`` / ``n_invalid`` counters."""
   124→        self._n_scored = 0
   125→        self._n_invalid = 0
   126→
   127→    # -- contract --------------------------------------------------------
   128→    def _parse(self, smiles: str) -> Any:
   129→        """Parse ``smiles`` with RDKit, counting invalid inputs.
   130→
   131→        Returns the RDKit ``Mol`` or ``None`` (invalid / blank / non-str).
   132→        """
   133→        Chem = require_rdkit()  # actionable DependencyError when RDKit is absent
   134→# rdkit-stubs does not re-export DisableLog/EnableLog on rdkit.RDLogger,
   135→        # but rdBase (their defining module) is fully typed.
   136→        from rdkit.rdBase import DisableLog, EnableLog
   137→
   138→        self._n_scored += 1
   139→        if not isinstance(smiles, str) or not smiles.strip():
   140→            self._n_invalid += 1
   141→            return None
   142→        # RDKit logs parse errors to stderr before returning None; the
   143→        # accounting below is friendlier than the raw parser trace, and the
   144→        # optimisers parse thousands of candidates per run.
   145→        DisableLog("rdApp.error")
   146→        try:
   147→            mol = Chem.MolFromSmiles(smiles)
   148→        except Exception:  # noqa: BLE001 - RDKit can raise on hostile input
   149→            mol = None
   150→        finally:
   151→            EnableLog("rdApp.error")
   152→        if mol is None:
   153→            self._n_invalid += 1
   154→        return mol
   155→
   156→    def score(self, smiles: str) -> float:
   157→        """Return the objective value of ``smiles`` in ``[0, 1]``.
   158→
   159→        Invalid SMILES score ``0.0`` (counted in ``n_invalid``) instead of
   160→        raising -- generation loops feed raw sampled/mutated strings here.
   161→        """
   162→        raise NotImplementedError
   163→
   164→    def score_many(self, smiles_list: Sequence[str]) -> list[float]:
   165→        """Score a batch of SMILES (convenience loop over :meth:`score`)."""
   166→        return [self.score(smiles) for smiles in smiles_list]
   167→
   168→    def __repr__(self) -> str:  # pragma: no cover - debug nicety
   169→        return f"{type(self).__name__}(name={self.name!r})"
   170→
   171→
   172→class QedObjective(ScoringObjective):
   173→    """RDKit QED (quantitative estimate of drug-likeness).
   174→
   175→    QED blends eight desirability functions (molecular weight, LogP, TPSA,
   176→    rotatable bonds, aromatic rings, alerts, ...) into a single ``[0, 1]``
   177→    score -- 1.0 is a maximally drug-like molecule.
   178→
   179→    Example
   180→    -------
   181→    >>> QedObjective().score("CC(=O)Oc1ccccc1C(=O)O")   # aspirin
   182→    0.55...
   183→    """
   184→
   185→    name = "qed"
   186→
   187→    def score(self, smiles: str) -> float:
   188→        mol = self._parse(smiles)
   189→        if mol is None:
   190→            return 0.0
   191→        from rdkit.Chem.QED import qed
   192→
   193→        try:
   194→            return float(min(max(qed(mol), 0.0), 1.0))
   195→        except Exception:  # noqa: BLE001 - exotic valences can upset QED
   196→            _LOG.warning("QED failed on %r -- scoring 0.0", smiles)
   197→            return 0.0
   198→

   200→class _BandObjective(ScoringObjective):
   201→    """Shared machinery for the property-band objectives."""
   202→
   203→    def __init__(self, lo: float, hi: float, what: str) -> None:
   204→        super().__init__()
   205→        for bound, label in ((lo, "lo"), (hi, "hi")):
   206→            if (
   207→                not isinstance(bound, (int, float))
   208→                or isinstance(bound, bool)
   209→                or not math.isfinite(bound)
   210→            ):
   211→                raise DataValidationError(
   212→                    f"{what} band bound {label}={bound!r} must be a finite number"
   213→                )
   214→        if lo > hi:
   215→            raise DataValidationError(
   216→                f"{what} band is empty: lo={lo} is greater than hi={hi}"
   217→            )
   218→        if hi - lo < 2 * _MIN_BAND_WIDTH:
   219→            raise DataValidationError(
   220→                f"{what} band is too narrow: hi - lo = {hi - lo:.3g} "
   221→                f"(need > {2 * _MIN_BAND_WIDTH:.3g})"
   222→            )
   223→        self.lo = float(lo)
   224→        self.hi = float(hi)
   225→
   226→    def _descriptor(self, mol: Any) -> float:
   227→        raise NotImplementedError
   228→
   229→    def score(self, smiles: str) -> float:
   230→        mol = self._parse(smiles)
   231→        if mol is None:
   232→            return 0.0
   233→        return _band_score(self._descriptor(mol), self.lo, self.hi)

   236→class LogPBandObjective(_BandObjective):
   237→    """Gaussian band on Crippen LogP (wild octanol/water behaviour excluded).
   238→
   239→    The interval ``[lo, hi]`` is the ±1σ band around the target: a molecule
   240→    exactly at the centre scores 1.0, at either edge ≈0.61, and the score
   241→    decays smoothly further out -- so molecules *near* the window are
   242→    still rewarded.
   243→
   244→    Example
   245→    -------
   246→    >>> round(LogPBandObjective(0.31, 2.31).score("CC(=O)Oc1ccccc1C(=O)O"), 6)
   247→    1.0
   248→    """
   249→
   250→    name = "logp_band"
   251→
   252→    def __init__(self, lo: float, hi: float) -> None:
   253→        super().__init__(lo, hi, what="LogP")
   254→
   255→    def _descriptor(self, mol: Any) -> float:
   256→        from rdkit.Chem import Descriptors
   257→
   258→        # rdkit-stubs omits the dynamically-generated descriptor functors
   259→        # on rdkit.Chem.Descriptors (MolLogP among them); the runtime API
   260→        # is stable and documented.
   261→        return float(Descriptors.MolLogP(mol))  # type: ignore[attr-defined]

   305→class SimilarityObjective(ScoringObjective):
   306→    """Tanimoto similarity (Morgan fingerprints) to one seed molecule.
   307→
   308→    Parameters
   309→    ----------
   310→    seed_smiles:
   311→        The reference molecule.  Must be a parseable SMILES.
   312→    radius:
   313→        Morgan fingerprint radius (default 2, the "ECFP4" convention).
   314→    n_bits:
   315→        Fingerprint bit count (default 2048).
   316→
   317→    Raises
   318→    ------
   319→    DataValidationError
   320→        If ``seed_smiles`` cannot be parsed.
   321→
   322→    Example
   323→    -------
   324→    >>> SimilarityObjective("CC(=O)Oc1ccccc1C(=O)O").score(
   325→    ...     "CC(=O)Oc1ccccc1C(=O)O")   # aspirin vs itself
   326→    1.0
   327→    """
   328→
   329→    name = "similarity"
   330→
   331→    def __init__(self, seed_smiles: str, radius: int = 2, n_bits: int = 2048) -> None:
   332→        super().__init__()
   333→        Chem = require_rdkit()
   334→        from rdkit.Chem import rdFingerprintGenerator
   335→        from rdkit.rdBase import DisableLog, EnableLog
   336→
   337→        if not isinstance(seed_smiles, str) or not seed_smiles.strip():
   338→            raise DataValidationError("seed_smiles must be a non-empty string")
   339→        # Silence the raw parser trace: the DataValidationError below is the
   340→        # actionable message (the docking/ligands._validate_smiles precedent).
   341→        DisableLog("rdApp.error")
   342→        try:
   343→            mol = Chem.MolFromSmiles(seed_smiles)
   344→        finally:
   345→            EnableLog("rdApp.error")
   346→        if mol is None:
   347→            raise DataValidationError(
   348→                f"seed_smiles {seed_smiles!r} is not a parseable SMILES"
   349→            )
   350→        if not isinstance(radius, int) or isinstance(radius, bool) or radius < 0:
   351→            raise DataValidationError(
   352→                f"radius must be a non-negative int, got {radius!r}"
   353→            )
   354→        if not isinstance(n_bits, int) or isinstance(n_bits, bool) or n_bits < 32:
   355→            raise DataValidationError(f"n_bits must be an int >= 32, got {n_bits!r}")
   356→        # The modern generator API: bit-identical to the classic
   357→        # ``GetMorganFingerprintAsBitVect`` but without its per-call
   358→        # deprecation warning (generation loops score thousands of molecules).
   359→        self._generator = rdFingerprintGenerator.GetMorganGenerator(
   360→            radius=radius, fpSize=n_bits
   361→        )
   362→        self.seed_smiles = Chem.MolToSmiles(mol)
   363→        self.radius = radius
   364→        self.n_bits = n_bits
   365→        self._seed_fp = self._generator.GetFingerprint(mol)
   366→
   367→    def score(self, smiles: str) -> float:
   368→        mol = self._parse(smiles)
   369→        if mol is None:
   370→            return 0.0
   371→        from rdkit import DataStructs
   372→
   373→        fp = self._generator.GetFingerprint(mol)
   374→        return float(DataStructs.TanimotoSimilarity(fp, self._seed_fp))
   375→

   377→class NoveltyObjective(ScoringObjective):
   378→    """Structural novelty against a reference corpus.
   379→
   380→    ``score = 1 - max(Tanimoto(candidate, m) for m in corpus)`` computed on
   381→    Morgan fingerprints: a molecule already in the corpus is by definition
   382→    *not* novel (score ``0.0``), while a molecule far from every corpus
   383→    member scores towards ``1.0``.  Used inside a :class:`MultiObjective`
   384→    the objective acts as a soft anti-rediscovery constraint: one corpus
   385→    look-alike vetoes the whole molecule.
   386→
   387→    Parameters
   388→    ----------
   389→    corpus:
   390→        Reference SMILES (default: :data:`~pharmaml.generative.DEFAULT_CORPUS`).
   391→        Entries are canonicalised and de-duplicated; invalid entries are
   392→        dropped with a warning.
   393→    radius:
   394→        Morgan fingerprint radius (default 2, the "ECFP4" convention).
   395→    n_bits:
   396→        Fingerprint bit count (default 2048).
   397→
   398→    Raises
   399→    ------
   400→    DataValidationError
   401→        If ``corpus`` is not a sequence of SMILES, is empty, holds no
   402→        parseable entry, or ``radius`` / ``n_bits`` are out of range.
   403→
   404→    Example
   405→    -------
   406→    >>> NoveltyObjective(corpus=["CCO"]).score("CCO")   # corpus member
   407→    0.0
   408→    >>> round(NoveltyObjective(corpus=["CC(=O)Oc1ccccc1C(=O)O"]).score(
   409→    ...     "CC(=O)Nc1ccc(O)cc1"), 3)   # paracetamol vs aspirin-only corpus
   410→    0.778
   411→    """
   412→
   413→    name = "novelty"
   414→
   415→    def __init__(
   416→        self,
   417→        corpus: Sequence[str] | None = None,
   418→        radius: int = 2,
   419→        n_bits: int = 2048,
   420→    ) -> None:
   421→        super().__init__()
   422→        Chem = require_rdkit()
   423→        from rdkit.Chem import rdFingerprintGenerator
   424→        from rdkit.rdBase import DisableLog, EnableLog
   425→
   426→        if not isinstance(radius, int) or isinstance(radius, bool) or radius < 0:
   427→            raise DataValidationError(
   428→                f"radius must be a non-negative int, got {radius!r}"
   429→            )
   430→        if not isinstance(n_bits, int) or isinstance(n_bits, bool) or n_bits < 32:
   431→            raise DataValidationError(f"n_bits must be an int >= 32, got {n_bits!r}")
   432→
   433→        if corpus is None:
   434→            entries: list[str] = list(DEFAULT_CORPUS)
   435→        elif isinstance(corpus, str):
   436→            # iterating a bare string would treat CHARACTERS as molecules
   437→            raise DataValidationError(
   438→                "corpus must be a sequence of SMILES strings, not a single "
   439→                f"string (got {corpus!r} of length {len(corpus)})"
   440→            )
   441→        else:
   442→            try:
   443→                entries = list(corpus)
   444→            except TypeError as exc:
   445→                raise DataValidationError(
   446→                    "corpus must be a sequence of SMILES strings "
   447→                    f"(got {type(corpus).__name__})"
   448→                ) from exc
   449→        n_non_string = sum(1 for entry in entries if not isinstance(entry, str))
   450→        entries = [entry for entry in entries if isinstance(entry, str)]
   451→        if n_non_string:
   452→            # the docstring contract: invalid entries are dropped WITH A
   453→            # WARNING (unparseable strings warn below; non-strings must not
   454→            # vanish silently either -- silent data loss is the failure mode
   455→            # this package refuses).
   456→            _LOG.warning(
   457→                "dropping %d non-string corpus entr(ies) -- corpus entries "
   458→                "must be SMILES strings",
   459→                n_non_string,
   460→            )
   461→        if not entries:
   462→            raise DataValidationError(
   463→                "corpus is empty (or holds no string entries) -- pass a "
   464→                "non-empty list of SMILES strings"
   465→                + (
   466→                    f" ({n_non_string} non-string entr(ies) dropped)"
   467→                    if n_non_string
   468→                    else ""
   469→                )
   470→            )
   471→
   472→        canonical: list[str] = []
   473→        # RDKit logs parse errors to stderr before returning None; invalid
   474→        # corpus entries are dropped with a warning instead (the RL prior's
   475→        # corpus-sanitisation precedent).
   476→        DisableLog("rdApp.error")
   477→        try:
   478→            for entry in entries:
   479→                if not entry.strip():
   480→                    _LOG.warning("dropping blank corpus SMILES %r", entry)
   481→                    continue
   482→                mol = Chem.MolFromSmiles(entry)
   483→                if mol is None:
   484→                    _LOG.warning("dropping invalid corpus SMILES %r", entry)
   485→                    continue
   486→                canonical.append(Chem.MolToSmiles(mol))
   487→        finally:
   488→            EnableLog("rdApp.error")
   489→        canonical = list(dict.fromkeys(canonical))
   490→        if not canonical:
   491→            raise DataValidationError(
   492→                "corpus holds no valid SMILES -- every entry failed to parse"
   493→            )
   494→
   495→        self.radius = radius
   496→        self.n_bits = n_bits
   497→        self.corpus = tuple(canonical)
   498→        self._generator = rdFingerprintGenerator.GetMorganGenerator(
   499→            radius=radius, fpSize=n_bits
   500→        )
   501→        self._corpus_fps = [
   502→            self._generator.GetFingerprint(Chem.MolFromSmiles(smiles))
   503→            for smiles in canonical
   504→        ]
   505→
   506→    def score(self, smiles: str) -> float:
   507→        mol = self._parse(smiles)
   508→        if mol is None:
   509→            return 0.0
   510→        from rdkit import DataStructs
   511→
   512→        fp = self._generator.GetFingerprint(mol)
   513→        return float(1.0 - max(DataStructs.BulkTanimotoSimilarity(fp, self._corpus_fps)))
   514→

   516→class SubstructureObjective(ScoringObjective):
   517→    """Hard pharmacophore constraint: 1.0 when ``smarts`` matches.
   518→
   519→    Raises
   520→    ------
   521→    DataValidationError
   522→        If ``smarts`` cannot be parsed as a SMARTS query.
   523→
   524→    Example
   525→    -------
   526→    >>> SubstructureObjective("c1ccccc1").score(
   527→    ...     "CC(=O)Oc1ccccc1C(=O)O")   # aspirin contains benzene
   528→    1.0
   529→    """
   530→
   531→    name = "substructure"
   532→
   533→    def __init__(self, smarts: str) -> None:
   534→        super().__init__()
   535→        Chem = require_rdkit()
   536→        from rdkit.rdBase import DisableLog, EnableLog
   537→
   538→        if not isinstance(smarts, str) or not smarts.strip():
   539→            raise DataValidationError("smarts must be a non-empty string")
   540→        DisableLog("rdApp.error")
   541→        try:
   542→            query = Chem.MolFromSmarts(smarts)
   543→        finally:
   544→            EnableLog("rdApp.error")
   545→        if query is None:
   546→            raise DataValidationError(f"smarts {smarts!r} is not a parseable SMARTS")
   547→        self.smarts = smarts
   548→        self._query = query
   549→
   550→    def score(self, smiles: str) -> float:
   551→        mol = self._parse(smiles)
   552→        if mol is None:
   553→            return 0.0
   554→        return 1.0 if mol.HasSubstructMatch(self._query) else 0.0

   557→class MultiObjective(ScoringObjective):
   558→    """Weighted geometric mean of several objectives, with veto semantics.
   559→
   560→    ``total = prod(score_i ** weight_i) ** (1 / sum(weight_i))`` -- a
   561→    product of per-objective scores, so **any zero score vetoes the whole
   562→    molecule** (one hard constraint can reject what the others love).
   563→    Geometric (not arithmetic) weighting keeps the total in ``[0, 1]`` and
   564→    rewards balanced solutions.
   565→
   566→    Parameters
   567→    ----------
   568→    objectives:
   569→        Two or more objectives (any :class:`ScoringObjective`, or any
   570→        object exposing ``name`` and a callable ``score``).
   571→    weights:
   572→        Positive weights, one per objective (default: all 1.0).
   573→
   574→    Raises
   575→    ------
   576→    DataValidationError
   577→        On empty / single-objective lists, mismatched weights, or
   578→        non-positive / non-finite weights.
   579→
   580→    Example
   581→    -------
   582→    >>> obj = MultiObjective(
   583→    ...     [QedObjective(), LogPBandObjective(1.0, 3.0)], weights=[1.0, 1.0])
   584→    >>> total, breakdown = obj.score_with_breakdown(
   585→    ...     "CC(=O)Oc1ccccc1C(=O)O")
   586→    >>> sorted(breakdown)
   587→    ['logp_band', 'qed']
   588→    """
   589→
   590→    name = "multi"
   591→
   592→    def __init__(
   593→        self,
   594→        objectives: Sequence[Any],
   595→        weights: Sequence[float] | None = None,
   596→    ) -> None:
   597→        super().__init__()
   598→        objectives = list(objectives)
   599→        if len(objectives) < 2:
   600→            raise DataValidationError(
   601→                "MultiObjective needs at least two objectives "
   602→                f"(got {len(objectives)}) -- a single objective needs no combinator"
   603→            )
   604→        for i, objective in enumerate(objectives):
   605→            if not hasattr(objective, "score") or not callable(objective.score):
   606→                raise DataValidationError(
   607→                    f"objective #{i} ({objective!r}) has no callable score() method"
   608→                )
   609→            if not hasattr(objective, "name"):
   610→                raise DataValidationError(
   611→                    f"objective #{i} ({objective!r}) has no name attribute"
   612→                )
   613→        if weights is None:
   614→            weights = [1.0] * len(objectives)
   615→        weights = list(weights)
   616→        if len(weights) != len(objectives):
   617→            raise DataValidationError(
   618→                f"{len(objectives)} objectives but {len(weights)} weight(s) "
   619→                "-- the two lists must match"
   620→            )
   621→        for weight in weights:
   622→            if not isinstance(weight, (int, float)) or not math.isfinite(weight):
   623→                raise DataValidationError(
   624→                    f"weights must be finite numbers, got {weight!r}"
   625→                )
   626→            if weight <= 0:
   627→                raise DataValidationError(
   628→                    f"weights must be strictly positive, got {weight!r} "
   629→                    "(a zero-weight objective should simply be removed)"
   630→                )
   631→        self.objectives = objectives
   632→        self.weights = [float(weight) for weight in weights]
   633→
   634→    @property
   635→    def n_invalid(self) -> int:
   636→        """Sum of the children's invalid-SMILES counters."""
   637→        return int(
   638→            sum(
   639→                getattr(objective, "n_invalid", 0)
   640→                for objective in self.objectives
   641→            )
   642→        )
   643→
   644→    def reset_stats(self) -> None:
   645→        """Zero the counters of the combinator AND its children.
   646→
   647→        A ``reset_stats`` that left the children counting made the summed
   648→        ``n_invalid`` outlive the reset (4 while ``n_scored`` went 0).
   649→        Duck-typed children without the method are skipped, not crashed
   650→        on.
   651→        """
   652→        super().reset_stats()
   653→        for objective in self.objectives:
   654→            reset = getattr(objective, "reset_stats", None)
   655→            if callable(reset):
   656→                reset()
   657→
   658→    def score_with_breakdown(self, smiles: str) -> tuple[float, dict[str, float]]:
   659→        """Return ``(total, {objective_name: score})``.
   660→
   661→        Duplicate objective names are de-duplicated as ``name_2``,
   662→        ``name_3`` ... so the breakdown stays addressable.  A child score
   663→        that is NaN or infinite (a misbehaving custom objective) counts
   664→        as ``0.0`` -- i.e. it vetoes the molecule, keeping the documented
   665→        ``[0, 1]`` total contract true.
   666→        """
   667→        self._n_scored += 1
   668→        used: set[str] = set()
   669→        breakdown: dict[str, float] = {}
   670→        for objective in self.objectives:
   671→            base = str(getattr(objective, "name", "objective"))
   672→            key, suffix = base, 1
   673→            while key in used:
   674→                suffix += 1
   675→                key = f"{base}_{suffix}"
   676→            used.add(key)
   677→            try:
   678→                value = float(objective.score(smiles))
   679→            except (TypeError, ValueError) as exc:
   680→                raise DataValidationError(
   681→                    f"objective {key!r} returned a non-numeric score for "
   682→                    f"{smiles!r} -- objectives must map SMILES to floats"
   683→                ) from exc
   684→            if not math.isfinite(value):
   685→                value = 0.0
   686→            breakdown[key] = value
   687→        if any(value <= 0.0 for value in breakdown.values()):
   688→            return 0.0, breakdown
   689→        # Weighted geometric mean in LOG space: the linear-space product
   690→        # ``prod(v ** w)`` underflows to 0.0 long before the weights get
   691→        # extreme (weights [2000, 2000] scored 0.5/0.5 as 0.0) and the
   692→        # exponent sum overflows to inf ([1e308, 1e308] fabricated a
   693→        # perfect 1.0 via ``0.0 ** 0.0``).  Only weight RATIOS matter, so
   694→        # rescale by the max weight; math.fsum keeps the log-sum exact.
   695→        max_weight = max(self.weights)
   696→        log_total = math.fsum(
   697→            math.log(value) * (weight / max_weight)
   698→            for value, weight in zip(breakdown.values(), self.weights)
   699→        )
   700→        weight_sum = math.fsum(weight / max_weight for weight in self.weights)
   701→        total = math.exp(log_total / weight_sum)
   702→        return float(min(max(total, 0.0), 1.0)), breakdown
   703→
   704→    def score(self, smiles: str) -> float:
   705→        """Total weighted score (see :meth:`score_with_breakdown`)."""
   706→        return self.score_with_breakdown(smiles)[0]
