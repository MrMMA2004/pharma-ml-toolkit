     1→"""BREED-style scaffold-hop crossover for the genetic algorithm.
     2→
     3→The structural recombination operator for
     4→:class:`~pharmaml.generative.GeneticOptimizer`: two parent molecules are
     5→cut and recombined so an offspring is a hybrid that keeps parts of each
     6→parent (Nicolaou et al.'s BREED algorithm tradition, in the pragmatic
     7→spirit of this package: RDKit graph surgery + ``Chem.molzip``, no
     8→learned model).  Three cut geometries cover the two things a scaffold
     9→hop actually has to recombine -- fragment *arrangement* and ring
    10→*topology*:
    11→
    12→* **bridge** (the classic): both parents are cut at a pair of
    13→  *homologous* single acyclic bonds and the fragments recombined
    14→  cross-wise; every fragment arrives intact, so ring systems survive by
    15→  construction, and whenever the parents' frameworks span different
    16→  fragments (biphenyls, diaryl ethers, sulfonyl bridges) the child's
    17→  framework differs from BOTH parents.
    18→* **merge**: both parents are cut at homologous single *ring* bonds and
    19→  the two opened rings are zipped into one combined ring holding atoms
    20→  from both parents (ring expansion / macrocycle formation).
    21→* **transplant**: one parent is cut at a single ring bond and a
    22→  two-ended linker from the other parent -- a single atom (an ether O,
    23→  a methylene, a secondary amine N, ...) or a short contiguous acyclic
    24→  *path* of up to four atoms (an ethylene bridge ``-CH2CH2-``, an
    25→  oxyethylene ``-OCH2-`` segment, a diethylene ``-OCH2CH2O-``) -- is
    26→  excised and zipped into the cut: ring homologation by one atom or a
    27→  whole chain (piperidine + an ether oxygen -> an oxazepane, + an
    28→  ethylene bridge -> an azocane).
    29→
    30→All cuts are *single* bonds (aromatic ring bonds are never cut -- the
    31→re-zip would have to invent bond orders), and the join points must be
    32→**homologous**: the atoms joined by each new bond match on (atomic
    33→number, degree, aromaticity, formal charge), so recombination happens
    34→at chemically analogous positions (an aryl-CH2 bond crosses with
    35→another aryl-CH2 bond, not with a sulfonamide S-N).
    36→
    37→Deterministic given the caller's ``random.Random``; ``None`` whenever no
    38→viable cut exists (aromatic-only parents, no homologous pair, children
    39→unsanitisable or over the length cap) -- the caller falls back, never
    40→crashes.
    41→
    42→Example
    43→-------
    44→>>> from pharmaml.generative import scaffold_hop_crossover
    45→>>> import random
    46→>>> child = scaffold_hop_crossover("c1ccccc1CC(=O)O", "c1ccncc1CO",
    47→...                                random.Random(11))
    48→>>> child is not None   # a valid hybrid of both parents
    49→True
    50→>>> # ring expansion by a whole linker path: piperidine x diethylene
    51→>>> # glycol, the -CH2-O- path transplanted into a cut C-C bond -> 8-ring
    52→>>> scaffold_hop_crossover("C1CCNCC1", "OCCOCCO", random.Random(0))
    53→'C1CNCCCOC1'
    54→>>> # benzene has no acyclic single bond to cut: no crossover possible
    55→>>> scaffold_hop_crossover("c1ccccc1", "c1ccncc1", random.Random(0)) is None
    56→True
    57→"""
    58→
    59→from __future__ import annotations
    60→
    61→import itertools
    62→import random
    63→
    64→from ._utils import require_rdkit
    65→
    66→#: Minimum ``max_smiles_length`` the operator accepts (matches the
    67→#: :class:`~pharmaml.generative.GeneticOptimizer` floor, so the standalone
    68→#: use cannot be configured weaker than the GA use).
    69→_MIN_LENGTH_CAP = 20
    70→
    71→#: Maximum atoms in an excised multi-atom linker *path*.  Drug-like
    72→#: linkers are short (ethylene, oxyethylene, diethyleneoxy); longer
    73→#: chains would grow the candidate pool quadratically and the offspring
    74→#: past drug-like size -- the ``max_smiles_length`` cap polices the
    75→#: output, this cap keeps the search honest.
    76→_MAX_LINKER_PATH_ATOMS = 4
    77→
    78→
    79→def _atom_signature(atom: object) -> tuple[int, int, bool, int]:
    80→    """Homology key: two atoms match when these four fields all agree.
    81→
    82→    Atomic number and formal charge fix the element/charge class, degree
    83→    fixes the connectivity class (a methyl carbon never crosses with a
    84→    quaternary one), aromaticity separates aryl from alkyl carbons --
    85→    the pragmatic BREED-style environment descriptor, cheap and strict
    86→    enough that recombination happens at analogous positions.
    87→    """
    88→    return (
    89→        atom.GetAtomicNum(),  # type: ignore[attr-defined]
    90→        atom.GetDegree(),  # type: ignore[attr-defined]
    91→        atom.GetIsAromatic(),  # type: ignore[attr-defined]
    92→        atom.GetFormalCharge(),  # type: ignore[attr-defined]
    93→    )

    96→def _cut_into_two(
    97→    mol: object,
    98→    bond_idx: int,
    99→) -> tuple[object, object] | None:
   100→    """Cut ``mol`` at single bond ``bond_idx``; return ``(begin, end)`` fragments.
   101→
   102→    Both returned fragments carry exactly one ``[*:1]`` dummy at the cut
   103→    (map number 1), so ``Chem.molzip`` on any two of them forms exactly
   104→    one new bond.  ``None`` on any RDKit failure -- robustness first.
   105→    """
   106→    Chem = require_rdkit()
   107→    try:
   108→        rw = Chem.RWMol(mol)
   109→        bond = rw.GetBondWithIdx(bond_idx)
   110→        begin, end = bond.GetBeginAtomIdx(), bond.GetEndAtomIdx()
   111→        rw.RemoveBond(begin, end)
   112→        # Valence bookkeeping is exact for single bonds: each endpoint
   113→        # loses one single-bonded neighbour and gains one single-bonded
   114→        # dummy, so no implicit hydrogens appear or vanish -- the cut
   115→        # fragments sanitise exactly like the parent.
   116→        d_begin = rw.AddAtom(Chem.Atom(0))
   117→        d_end = rw.AddAtom(Chem.Atom(0))
   118→        rw.AddBond(begin, d_begin, Chem.BondType.SINGLE)
   119→        rw.AddBond(end, d_end, Chem.BondType.SINGLE)
   120→        cut = rw.GetMol()
   121→        idx_frags = Chem.GetMolFrags(cut)
   122→        mol_frags = Chem.GetMolFrags(cut, asMols=True)
   123→        if len(idx_frags) != 2 or len(mol_frags) != 2:  # pragma: no cover
   124→            return None  # a bridge bond always splits a connected mol in two
   125→        begin_frag = end_frag = None
   126→        for indices, fragment in zip(idx_frags, mol_frags):
   127→            if d_begin in indices:
   128→                begin_frag = fragment
   129→            if d_end in indices:
   130→                end_frag = fragment
   131→        if begin_frag is None or end_frag is None:  # pragma: no cover
   132→            return None
   133→        for fragment in (begin_frag, end_frag):
   134→            Chem.SanitizeMol(fragment)
   135→            dummies = [
   136→                atom for atom in fragment.GetAtoms() if atom.GetAtomicNum() == 0
   137→            ]
   138→            if len(dummies) != 1:  # pragma: no cover
   139→                return None
   140→            dummy = dummies[0]
   141→            dummy.SetAtomMapNum(1)
   142→            dummy.SetIsotope(0)
   143→        return begin_frag, end_frag
   144→    except Exception:  # noqa: BLE001 - robustness-first fallback contract
   145→        return None

   148→def _open_ring(mol: object, bond_idx: int) -> object | None:
   149→    """Cut one single *ring* bond; return the ring-opened molecule.
   150→
   151→    A ring cut never disconnects a connected molecule: the result is ONE
   152→    fragment carrying two dummies -- ``[*:1]`` on the former begin atom
   153→    of the cut bond, ``[*:2]`` on the former end atom -- so
   154→    ``Chem.molzip`` against another two-ended piece forms two new bonds
   155→    (a ring merge or a linker transplant).  ``None`` on any RDKit
   156→    failure.
   157→    """
   158→    Chem = require_rdkit()
   159→    try:
   160→        rw = Chem.RWMol(mol)
   161→        bond = rw.GetBondWithIdx(bond_idx)
   162→        begin, end = bond.GetBeginAtomIdx(), bond.GetEndAtomIdx()
   163→        rw.RemoveBond(begin, end)
   164→        # Same exact valence bookkeeping as _cut_into_two: each endpoint
   165→        # trades one single-bonded neighbour for one single-bonded dummy.
   166→        d_begin = rw.AddAtom(Chem.Atom(0))
   167→        d_end = rw.AddAtom(Chem.Atom(0))
   168→        rw.AddBond(begin, d_begin, Chem.BondType.SINGLE)
   169→        rw.AddBond(end, d_end, Chem.BondType.SINGLE)
   170→        opened = rw.GetMol()
   171→        if len(Chem.GetMolFrags(opened)) != 1:  # pragma: no cover
   172→            return None  # a ring cut keeps a connected mol connected
   173→        Chem.SanitizeMol(opened)
   174→        # RWMol.GetMol() preserves atom indices, so the begin/end atoms
   175→        # are still addressable -- label each dummy by its host's side.
   176→        dummies = [
   177→            atom for atom in opened.GetAtoms() if atom.GetAtomicNum() == 0
   178→        ]
   179→        if len(dummies) != 2:  # pragma: no cover
   180→            return None
   181→        for dummy in dummies:
   182→            host = dummy.GetNeighbors()[0].GetIdx()
   183→            if host == begin:
   184→                dummy.SetAtomMapNum(1)
   185→            elif host == end:
   186→                dummy.SetAtomMapNum(2)
   187→            else:  # pragma: no cover
   188→                return None
   189→            dummy.SetIsotope(0)
   190→        return opened
   191→    except Exception:  # noqa: BLE001 - robustness-first fallback contract
   192→        return None

   195→def _excise_linker_path(
   196→    mol: object,
   197→    path: tuple[int, ...],
   198→) -> tuple[object, tuple] | None:
   199→    """Cut a contiguous acyclic linker path loose; return ``(donor, host_sigs)``.
   200→
   201→    A linker path is either a single atom (the classic two-ended linker:
   202→    an ether O, a methylene, a secondary amine N, a thioether S ...) or
   203→    a contiguous chain of up to ``_MAX_LINKER_PATH_ATOMS`` such atoms
   204→    (an ethylene bridge, an oxyethylene segment).  Every path atom must
   205→    have exactly two neighbours, both reached via single acyclic bonds,
   206→    and be outside every ring; attachment dummies (atomic number 0) are
   207→    never excisable -- a parent's dummy must not be transplanted into a
   208→    child.  The donor is a fresh molecule holding the path atoms and two
   209→    dummies -- ``[*:1]`` replacing the bond to the first path atom's
   210→    outside host, ``[*:2]`` the last's -- ready to be zipped into a ring
   211→    cut of the other parent.  ``host_sigs`` are the two former outside
   212→    neighbours' signatures, in the path's own orientation, for the
   213→    homology check.  ``None`` when any validation fails (a chain that
   214→    runs through a double-bonded atom -- e.g. an aldehyde carbon --
   215→    never excises).
   216→
   217→    Valence bookkeeping is exact for every seam: each path end atom
   218→    trades one single-bonded host for one single-bonded dummy, interior
   219→    atoms keep both neighbours, so the donor sanitises exactly like the
   220→    corresponding parent fragment.
   221→    """
   222→    Chem = require_rdkit()
   223→    try:
   224→        if not path or len(path) > _MAX_LINKER_PATH_ATOMS:
   225→            return None
   226→        if len(set(path)) != len(path):
   227→            return None  # a repeating index is not a path
   228→        for idx in path:
   229→            atom = mol.GetAtomWithIdx(idx)  # type: ignore[attr-defined]
   230→            if atom.GetDegree() != 2 or atom.IsInRing():
   231→                return None
   232→            if atom.GetAtomicNum() == 0:  # attachment dummy: never excised
   233→                return None
   234→            # degree 2 -> exactly these two bonds, so one loop covers both
   235→            # the interior path bonds and the two host bonds
   236→            for nbr in atom.GetNeighbors():
   237→                bond = mol.GetBondBetweenAtoms(  # type: ignore[attr-defined]
   238→                    idx, nbr.GetIdx()
   239→                )
   240→                if (
   241→                    bond is None
   242→                    or bond.IsInRing()
   243→                    or bond.GetBondType() != Chem.BondType.SINGLE
   244→                ):
   245→                    return None
   246→        # consecutive path atoms must actually be bonded (the enumeration
   247→        # emits true chains; a hand-made tuple may not be one)
   248→        for left, right in itertools.pairwise(path):
   249→            if mol.GetBondBetweenAtoms(left, right) is None:  # type: ignore[attr-defined]
   250→                return None
   251→        # hosts: n == 1 -> the atom's two neighbours in GetNeighbors
   252→        # order (the classic linker semantics); n >= 2 -> each end's
   253→        # unique outside neighbour (the other neighbour is the next path
   254→        # atom, so the lookups are unambiguous)
   255→        if len(path) == 1:
   256→            nbrs = list(mol.GetAtomWithIdx(path[0]).GetNeighbors())  # type: ignore[attr-defined]
   257→            host_first, host_second = nbrs[0], nbrs[1]
   258→        else:
   259→            first_outside = [
   260→                nbr
   261→                for nbr in mol.GetAtomWithIdx(path[0]).GetNeighbors()  # type: ignore[attr-defined]
   262→                if nbr.GetIdx() != path[1]
   263→            ]
   264→            last_outside = [
   265→                nbr
   266→                for nbr in mol.GetAtomWithIdx(path[-1]).GetNeighbors()  # type: ignore[attr-defined]
   267→                if nbr.GetIdx() != path[-2]
   268→            ]
   269→            if len(first_outside) != 1 or len(last_outside) != 1:
   270→                return None
   271→            host_first, host_second = first_outside[0], last_outside[0]
   272→        rw = Chem.RWMol()
   273→        for idx in path:
   274→            anchor = Chem.Atom(mol.GetAtomWithIdx(idx))  # type: ignore[attr-defined]
   275→            anchor.SetAtomMapNum(0)
   276→            anchor.SetIsotope(0)
   277→            anchor.SetChiralTag(Chem.ChiralType.CHI_UNSPECIFIED)
   278→            rw.AddAtom(anchor)
   279→        d_first = rw.AddAtom(Chem.Atom(0))
   280→        d_second = rw.AddAtom(Chem.Atom(0))
   281→        for i in range(len(path) - 1):
   282→            rw.AddBond(i, i + 1, Chem.BondType.SINGLE)
   283→        rw.AddBond(0, d_first, Chem.BondType.SINGLE)
   284→        rw.AddBond(len(path) - 1, d_second, Chem.BondType.SINGLE)
   285→        donor = rw.GetMol()
   286→        Chem.SanitizeMol(donor)
   287→        # The donor was hand-built (path atoms in order, then the two
   288→        # dummies in exactly that atom order), so the dummy indices are
   289→        # fixed: label the first-host-side dummy [*:1], the second [*:2].
   290→        donor.GetAtomWithIdx(d_first).SetAtomMapNum(1)
   291→        donor.GetAtomWithIdx(d_second).SetAtomMapNum(2)
   292→        return donor, (_atom_signature(host_first), _atom_signature(host_second))
   293→    except Exception:  # noqa: BLE001 - robustness-first fallback contract
   294→        return None

   297→def _linker_path_units(mol: object) -> list[tuple[tuple[int, ...], tuple]]:
   298→    """Enumerate multi-atom linker units: ``[(path, host_sigs)]``.
   299→
   300→    Every maximal run ("chain") of linker-eligible atoms -- degree two,
   301→    acyclic, non-dummy -- contributes its contiguous sublists of two up
   302→    to ``_MAX_LINKER_PATH_ATOMS`` atoms as candidate excisions; the
   303→    single atoms keep their dedicated scan in
   304→    :func:`scaffold_hop_crossover` so that geometry's candidate pool is
   305→    unchanged.  A sublist is admitted only when
   306→    :func:`_excise_linker_path` validates it -- a chain may run through
   307→    a double-bonded atom (an aldehyde carbon, a carbonyl), and any unit
   308→    containing it never excises, so it is silently dropped here.
   309→
   310→    Determinism: chains are discovered in order of their smallest atom
   311→    index (the first unvisited eligible atom in index order is its
   312→    chain's minimum), each chain oriented smallest-index end first, and
   313→    sublists emitted left-to-right, shorter before longer within the
   314→    same start -- a pure function of the molecule, so the crossover's
   315→    candidate pool stays a pure function of the parents.
   316→    """
   317→    assigned: set[int] = set()
   318→    units: list[tuple[tuple[int, ...], tuple]] = []
   319→
   320→    def eligible(atom: object) -> bool:
   321→        return (
   322→            atom.GetDegree() == 2  # type: ignore[attr-defined]
   323→            and not atom.IsInRing()  # type: ignore[attr-defined]
   324→            and atom.GetAtomicNum() != 0  # type: ignore[attr-defined]
   325→        )
   326→
   327→    for atom in mol.GetAtoms():  # type: ignore[attr-defined]
   328→        start = atom.GetIdx()  # type: ignore[attr-defined]
   329→        if start in assigned or not eligible(atom):
   330→            continue
   331→        # Walk the maximal chain through `start` in both directions:
   332→        # both of its neighbours are eligible-or-host, each eligible
   333→        # neighbour extends the run until a host (degree != 2, ring
   334→        # atom, dummy) terminates it.
   335→        chain = [start]
   336→        neighbours = list(atom.GetNeighbors())  # type: ignore[attr-defined]
   337→        for is_first_side, seed in (
   338→            (True, neighbours[1]),
   339→            (False, neighbours[0]),
   340→        ):
   341→            prev, cur = start, seed.GetIdx()
   342→            while eligible(mol.GetAtomWithIdx(cur)):  # type: ignore[attr-defined]
   343→                chain.append(cur)
   344→                nxt = [
   345→                    nbr.GetIdx()
   346→                    for nbr in mol.GetAtomWithIdx(cur).GetNeighbors()  # type: ignore[attr-defined]
   347→                    if nbr.GetIdx() != prev
   348→                ]
   349→                prev, cur = cur, nxt[0]
   350→            if is_first_side:
   351→                chain.reverse()  # the far side becomes the front
   352→        assigned.update(chain)
   353→        if chain[0] > chain[-1]:
   354→            chain.reverse()  # canonical orientation: smaller end first
   355→        for begin in range(len(chain)):
   356→            for length in range(
   357→                2, min(_MAX_LINKER_PATH_ATOMS, len(chain) - begin) + 1
   358→            ):
   359→                path = tuple(chain[begin:begin + length])
   360→                excision = _excise_linker_path(mol, path)
   361→                if excision is None:
   362→                    continue
   363→                units.append((path, excision[1]))
   364→    return units

   367→def _zip_child(
   368→    frag_x: object,
   369→    frag_y: object,
   370→    max_smiles_length: int,
   371→) -> str | None:
   372→    """Zip two prepared fragments into one child SMILES, or ``None``.
   373→
   374→    Belt-and-braces: a child that still carries an attachment dummy
   375→    (atomic number 0) is rejected -- no operator of this module may
   376→    ever leak a ``*`` into an offspring (round 100x-28 review finding 1).
   377→    """
   378→    Chem = require_rdkit()
   379→    try:
   380→        child = Chem.molzip(frag_x, frag_y)
   381→        Chem.SanitizeMol(child)
   382→        if any(atom.GetAtomicNum() == 0 for atom in child.GetAtoms()):
   383→            return None
   384→        smiles = Chem.MolToSmiles(child)
   385→    except Exception:  # noqa: BLE001 - no viable child
   386→        return None
   387→    if smiles and len(smiles) <= max_smiles_length:
   388→        return smiles
   389→    return None

   392→def scaffold_hop_crossover(
   393→    parent_a: str,
   394→    parent_b: str,
   395→    rng: random.Random,
   396→    *,
   397→    max_smiles_length: int = 120,
   398→) -> str | None:
   399→    """BREED-style crossover: homologous cuts, hybrid child.
   400→
   401→    Three cut geometries compete for the rng draw -- a **bridge** cut
   402→    per parent (fragment swap, the classic), a **ring merge** (one
   403→    single ring bond cut per parent, the opened rings zipped into one
   404→    combined ring) and a **linker transplant** (a single ring bond cut
   405→    in one parent, a two-ended linker -- a single atom or a contiguous
   406→    acyclic path of up to four atoms -- excised from the other and
   407→    zipped into the cut: ring homologation by one atom or a whole
   408→    chain).  Aromatic ring bonds are never cut (the re-zip would have
   409→    to invent bond orders).
   410→
   411→    Parameters
   412→    ----------
   413→    parent_a, parent_b:
   414→        Canonical (or any parseable) SMILES of the two parents.  Both must
   415→        be single connected molecules -- a dot-disconnected parent has no
   416→        meaningful fragment recombination and yields ``None``.
   417→    rng:
   418→        The caller's dedicated ``random.Random`` instance (the GA passes
   419→        its own; standalone callers pass ``random.Random(seed)``).  The
   420→        returned child is a pure function of the parents, the cap and the
   421→        rng state -- same state, same child.
   422→    max_smiles_length:
   423→        Hard cap on the child's canonical SMILES length (default 120, the
   424→        :class:`~pharmaml.generative.GeneticOptimizer` default).  All
   425→        viable pairings of a drawn bridge candidate are tried before
   426→        giving up.
   427→
   428→    Returns
   429→    -------
   430→    str | None
   431→        Canonical SMILES of the hybrid child, or ``None`` when no viable
   432→        crossover exists (unparseable parent, no homologous cut pair,
   433→        children unsanitisable or over the cap).  ``None`` is a
   434→        "try the other operator" signal, never an error.
   435→
   436→    Example
   437→    -------
   438→    >>> from pharmaml.generative import scaffold_hop_crossover
   439→    >>> import random
   440→    >>> # aspirin x benzamide: homologous bridge cuts, hybrids come out
   441→    >>> child = scaffold_hop_crossover("CC(=O)Oc1ccccc1C(=O)O",
   442→    ...                                 "NC(=O)c1ccccc1", random.Random(0))
   443→    >>> child is not None
   444→    True
   445→    >>> # ring recombination: piperidine x diethyl ether -- the ether O
   446→    >>> # is transplanted into a cut C-C ring bond -> a 7-ring oxazepane
   447→    >>> child = scaffold_hop_crossover("C1CCNCC1", "CCOCC", random.Random(0))
   448→    >>> child is not None
   449→    True
   450→    >>> # benzene has no acyclic single bond and no cuttable ring bond
   451→    >>> scaffold_hop_crossover("c1ccccc1", "c1ccncc1", random.Random(0)) is None
   452→    True
   453→    """
   454→    Chem = require_rdkit()
   455→    from rdkit.rdBase import DisableLog, EnableLog
   456→
   457→    if max_smiles_length < _MIN_LENGTH_CAP:
   458→        raise ValueError(
   459→            f"max_smiles_length must be >= {_MIN_LENGTH_CAP}, "
   460→            f"got {max_smiles_length}"
   461→        )
   462→
   463→    # Unsanitisable recombinations are EXPECTED (the caller falls back);
   464→    # RDKit logs every sanitisation failure to stderr -- keep the GA loop
   465→    # quiet (the canonical_smiles / _scaffold_mutate precedent).
   466→    DisableLog("rdApp.error")
   467→    try:
   468→        mol_a = Chem.MolFromSmiles(parent_a)
   469→        mol_b = Chem.MolFromSmiles(parent_b)
   470→        if mol_a is None or mol_b is None:
   471→            return None
   472→        if len(Chem.GetMolFrags(mol_a)) != 1 or len(Chem.GetMolFrags(mol_b)) != 1:
   473→            return None  # dot-disconnected parent: no recombination semantics
   474→
   475→        sig_a = {atom.GetIdx(): _atom_signature(atom) for atom in mol_a.GetAtoms()}
   476→        sig_b = {atom.GetIdx(): _atom_signature(atom) for atom in mol_b.GetAtoms()}
   477→
   478→        # Homologous cut candidates, three geometries in a fixed scan
   479→        # order (determinism: the pool is a pure function of the parents;
   480→        # the rng draw then picks one uniformly).
   481→        #
   482→        # bridge: single acyclic bonds whose endpoints match in some
   483→        # orientation (the classic fragment swap -- ring bonds excluded
   484→        # here because a bridge-style cut must split the molecule).
   485→        candidates: list[tuple] = []
   486→        for bond_a in mol_a.GetBonds():
   487→            if bond_a.IsInRing() or bond_a.GetBondType() != Chem.BondType.SINGLE:
   488→                continue
   489→            a1, a2 = bond_a.GetBeginAtomIdx(), bond_a.GetEndAtomIdx()
   490→            for bond_b in mol_b.GetBonds():
   491→                if bond_b.IsInRing() or bond_b.GetBondType() != Chem.BondType.SINGLE:
   492→                    continue
   493→                b1, b2 = bond_b.GetBeginAtomIdx(), bond_b.GetEndAtomIdx()
   494→                direct = sig_a[a1] == sig_b[b1] and sig_a[a2] == sig_b[b2]
   495→                flipped = sig_a[a1] == sig_b[b2] and sig_a[a2] == sig_b[b1]
   496→                if direct or flipped:
   497→                    candidates.append(("bridge", bond_a.GetIdx(), bond_b.GetIdx(), direct, flipped))
   498→
   499→        # merge: single RING bonds whose endpoints match -- the opened
   500→        # rings zip into one combined ring (atom count = both rings').
   501→        # The orientation matters: a direct match zips begin-begin and
   502→        # end-end; a flipped-only match must zip begin-end so each seam
   503→        # bond joins signature-matched atoms (the module's invariant).
   504→        ring_bonds_a = [
   505→            bond for bond in mol_a.GetBonds()
   506→            if bond.IsInRing() and bond.GetBondType() == Chem.BondType.SINGLE
   507→        ]
   508→        ring_bonds_b = [
   509→            bond for bond in mol_b.GetBonds()
   510→            if bond.IsInRing() and bond.GetBondType() == Chem.BondType.SINGLE
   511→        ]
   512→        for bond_a in ring_bonds_a:
   513→            a1, a2 = bond_a.GetBeginAtomIdx(), bond_a.GetEndAtomIdx()
   514→            for bond_b in ring_bonds_b:
   515→                b1, b2 = bond_b.GetBeginAtomIdx(), bond_b.GetEndAtomIdx()
   516→                direct = sig_a[a1] == sig_b[b1] and sig_a[a2] == sig_b[b2]
   517→                flipped = sig_a[a1] == sig_b[b2] and sig_a[a2] == sig_b[b1]
   518→                if direct or flipped:
   519→                    candidates.append(("merge", bond_a.GetIdx(), bond_b.GetIdx(), direct))
   520→
   521→        # transplant: a single ring bond in one parent + a two-ended
   522→        # linker in the other -- a single atom or a multi-atom path
   523→        # (homologous flanks: the linker's former outside hosts must
   524→        # match the cut bond's endpoints, so the linker lands in a
   525→        # chemically analogous environment).  Both argument orders are
   526→        # scanned.  For single atoms the orientation is admission-only
   527→        # (the anchor bonds to BOTH ring endpoints, so the child's edge
   528→        # set is orientation-independent); a *path* is asymmetric, so
   529→        # flipped-only matches must excise the reversed path at
   530→        # execution time (handled below).
   531→        for ring_side, other, ring_bonds, sig_ring in (
   532→            ("a", mol_b, ring_bonds_a, sig_a),
   533→            ("b", mol_a, ring_bonds_b, sig_b),
   534→        ):
   535→            for bond in ring_bonds:
   536→                r1, r2 = bond.GetBeginAtomIdx(), bond.GetEndAtomIdx()
   537→                for atom in other.GetAtoms():
   538→                    if atom.GetDegree() != 2 or atom.IsInRing():
   539→                        continue
   540→                    excision = _excise_linker_path(other, (atom.GetIdx(),))
   541→                    if excision is None:
   542→                        continue
   543→                    _, host_sigs = excision
   544→                    direct = host_sigs[0] == sig_ring[r1] and host_sigs[1] == sig_ring[r2]
   545→                    flipped = host_sigs[0] == sig_ring[r2] and host_sigs[1] == sig_ring[r1]
   546→                    if direct or flipped:
   547→                        candidates.append(
   548→                            ("transplant", ring_side, bond.GetIdx(), atom.GetIdx())
   549→                        )
   550→            # the same ring bonds against multi-atom linker paths of the
   551→            # donor (round 100x-29): ring expansion by 2-4 atoms.  The
   552→            # guard skips the enumeration for acyclic donors (no ring
   553→            # bond to transplant into -- the units would be dead weight)
   554→            if ring_bonds:
   555→                for path, host_sigs in _linker_path_units(other):
   556→                    for bond in ring_bonds:
   557→                        r1, r2 = bond.GetBeginAtomIdx(), bond.GetEndAtomIdx()
   558→                        direct = host_sigs[0] == sig_ring[r1] and host_sigs[1] == sig_ring[r2]
   559→                        flipped = host_sigs[0] == sig_ring[r2] and host_sigs[1] == sig_ring[r1]
   560→                        if direct or flipped:
   561→                            candidates.append(
   562→                                ("transplant_path", ring_side, bond.GetIdx(), path, direct)
   563→                            )
   564→        if not candidates:
   565→            return None
   566→
   567→        choice = candidates[rng.randrange(len(candidates))]
   568→
   569→        if choice[0] == "bridge":
   570→            _, bond_a_idx, bond_b_idx, direct, flipped = choice
   571→            cut_a = _cut_into_two(mol_a, bond_a_idx)
   572→            cut_b = _cut_into_two(mol_b, bond_b_idx)
   573→            if cut_a is None or cut_b is None:
   574→                return None
   575→            frag_a_begin, frag_a_end = cut_a
   576→            frag_b_begin, frag_b_end = cut_b
   577→            # Eligible pairings join signature-matched atoms: begin-begin
   578→            # and end-end when the direct orientation matched, begin-end
   579→            # and end-begin when only the flipped one did (all four when
   580→            # both).
   581→            pairings: list[tuple[object, object]] = []
   582→            if direct:
   583→                pairings.append((frag_a_begin, frag_b_begin))
   584→                pairings.append((frag_a_end, frag_b_end))
   585→            if flipped:
   586→                pairings.append((frag_a_begin, frag_b_end))
   587→                pairings.append((frag_a_end, frag_b_begin))
   588→            rng.shuffle(pairings)
   589→            for frag_x, frag_y in pairings:
   590→                child = _zip_child(frag_x, frag_y, max_smiles_length)
   591→                if child is not None:
   592→                    return child
   593→            return None
   594→
   595→        if choice[0] == "merge":
   596→            _, bond_a_idx, bond_b_idx, direct = choice
   597→            open_a = _open_ring(mol_a, bond_a_idx)
   598→            open_b = _open_ring(mol_b, bond_b_idx)
   599→            if open_a is None or open_b is None:
   600→                return None
   601→            if not direct:
   602→                # flipped-only match: swap open_b's dummy labels so the
   603→                # seams join begin_a-end_b / end_a-begin_b -- the
   604→                # signature-matched orientation
   605→                for dummy in open_b.GetAtoms():  # type: ignore[attr-defined]
   606→                    if dummy.GetAtomicNum() == 0:
   607→                        dummy.SetAtomMapNum(3 - dummy.GetAtomMapNum())
   608→            return _zip_child(open_a, open_b, max_smiles_length)
   609→
   610→        # transplant (single-atom linker: orientation-independent)
   611→        if choice[0] == "transplant":
   612→            _, ring_side, ring_bond_idx, donor_atom_idx = choice
   613→            if ring_side == "a":
   614→                ring_mol, donor_mol = mol_a, mol_b
   615→            else:
   616→                ring_mol, donor_mol = mol_b, mol_a
   617→            opened = _open_ring(ring_mol, ring_bond_idx)
   618→            excision = _excise_linker_path(donor_mol, (donor_atom_idx,))
   619→            if opened is None or excision is None:
   620→                return None
   621→            donor, _ = excision
   622→            return _zip_child(opened, donor, max_smiles_length)
   623→
   624→        # transplant_path (multi-atom linker: orientation-aware)
   625→        _, ring_side, ring_bond_idx, path, direct = choice
   626→        if ring_side == "a":
   627→            ring_mol, donor_mol = mol_a, mol_b
   628→        else:
   629→            ring_mol, donor_mol = mol_b, mol_a
   630→        opened = _open_ring(ring_mol, ring_bond_idx)
   631→        # a flipped-only match must excise the REVERSED path so each seam
   632→        # bonds a path atom to a signature-matched ring endpoint (the
   633→        # single-atom transplant is symmetric -- reversing a one-atom
   634→        # path is the identity -- which is why orientation could be
   635→        # admission-only before paths existed)
   636→        excision = _excise_linker_path(
   637→            donor_mol, path if direct else tuple(reversed(path))
   638→        )
   639→        if opened is None or excision is None:
   640→            return None
   641→        donor, _ = excision
   642→        return _zip_child(opened, donor, max_smiles_length)
   643→    except Exception:  # noqa: BLE001 - robustness-first fallback contract
   644→        return None
   645→    finally:
   646→        EnableLog("rdApp.error")
