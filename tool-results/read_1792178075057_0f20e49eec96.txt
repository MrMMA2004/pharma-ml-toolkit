   108→def _hide_torch(monkeypatch: pytest.MonkeyPatch) -> None:
   109→    """Simulate a torch-less machine (import of torch raises ImportError)."""
   110→    monkeypatch.setitem(sys.modules, "torch", None)

   113→def _hide_torch_spec(monkeypatch: pytest.MonkeyPatch) -> None:
   114→    """Simulate torch being undetectable by the registry availability probe."""
   115→    real = models_module._module_available
   116→
   117→    def fake(name: str) -> bool:
   118→        return False if name == "torch" else real(name)
   119→
   120→    monkeypatch.setattr(models_module, "_module_available", fake)

   126→class TestImportSafety:
   127→    def test_module_imports_without_torch(self, monkeypatch: pytest.MonkeyPatch) -> None:
   128→        # Executing the module source with torch "missing" must succeed:
   129→        # all torch imports live inside functions (or TYPE_CHECKING).
   130→        _hide_torch(monkeypatch)
   131→        spec = importlib.util.find_spec("pharmaml.ml.neural")
   132→        assert spec is not None and spec.loader is not None
   133→        module = importlib.util.module_from_spec(spec)
   134→        spec.loader.exec_module(module)
   135→        for name in (
   136→            "DeepMLPClassifier",
   137→            "DeepMLPRegressor",
   138→            "ResidualMLPClassifier",
   139→            "ResidualMLPRegressor",
   140→        ):
   141→            assert hasattr(module, name)

   679→    def test_best_epoch_weights_restored_not_last(self, binary_data: tuple) -> None:
   680→        # A deliberately hot learning rate makes the validation loss dip
   681→        # early and then WORSEN; the stored weights must be the ones from
   682→        # the best epoch, not the final one.  A second estimator trained
   683→        # for exactly best_epoch epochs (same seed -> identical epoch
   684→        # sequence) ends up holding those best-epoch weights.

   691→            "random_state": 7,
   692→            "learning_rate": 0.05,
   693→        }
   694→        full = DeepMLPClassifier(**common).fit(X_train, y_train)
   695→        val_losses = [entry["val_loss"] for entry in full.history_]
   696→        best_epoch = int(np.argmin(val_losses)) + 1
   697→        # meaningfulness precondition: the curve really does worsen
   698→        assert val_losses[-1] > val_losses[best_epoch - 1]
   699→        truncated = DeepMLPClassifier(
   700→            **{**common, "epochs": best_epoch, "patience": best_epoch}
   701→        ).fit(X_train, y_train)
   702→        assert full._weights.keys() == truncated._weights.keys()

   704→            np.allclose(full._weights[key], truncated._weights[key])
   705→            for key in full._weights

   904→    long before the intercept is learned.  Pre-fix repro (this exact
   905→    test's data, 60 epochs): rmse 4.39 -> post-fix 0.36.
   906→    """

   908→    def test_pic50_scale_target_is_learnable(self) -> None:
   909→        rng = np.random.default_rng(11)
   910→        X = rng.normal(size=(120, 6))
   911→        # pIC50-style target: informative signal + noise, centred ~5.3.
   912→        y = 5.3 + 0.4 * X[:, 0] - 0.3 * X[:, 1] + 0.05 * rng.normal(size=120)
   913→        X_train, X_test, y_train, y_test = train_test_split(
   914→            X, y, test_size=0.25, random_state=11
   915→        )
   916→        model = fast_regressor(epochs=60, patience=10).fit(X_train, y_train)
   917→        rmse = float(np.sqrt(np.mean((model.predict(X_test) - y_test) ** 2)))
   918→        # A linear baseline reaches ~0.05; the pre-fix network sat at
   919→        # ~4.4 (never learned the intercept).  1.0 is generous but
   920→        # still fails the pre-fix code by 4x.
   921→        assert rmse < 1.0, f"off-centre target not learned: rmse={rmse:.3f}"
