     7→     7→                               ``rng.random() < crossover_rate`` Bernoulli)
     8→     8→  Level 2  (per recombination): the seven scaffold-hop geometries

    80→    80→def _scan_candidates_unlocked(parent_a: str, parent_b: str):
    81→    81→    """Replicate the deterministic candidate scan of scaffold_hop_crossover.
    82→    82→
    83→    83→    Returns the candidates list (same order as the original), or None when
    84→    84→    either parent does not parse / is dot-disconnected (the original's early
    85→    85→    None path).
    86→    86→    """
    87→    87→    Chem = sh.require_rdkit()
    88→    88→    mol_a = Chem.MolFromSmiles(parent_a)
    89→    89→    mol_b = Chem.MolFromSmiles(parent_b)
    90→    90→    if mol_a is None or mol_b is None:
    91→    91→        return None
    92→    92→    if len(Chem.GetMolFrags(mol_a)) != 1 or len(Chem.GetMolFrags(mol_b)) != 1:
    93→    93→        return None
    94→    94→
    95→    95→    sig_a = {atom.GetIdx(): sh._atom_signature(atom) for atom in mol_a.GetAtoms()}
    96→    96→    sig_b = {atom.GetIdx(): sh._atom_signature(atom) for atom in mol_b.GetAtoms()}
    97→    97→
    98→    98→    candidates: list[tuple] = []
    99→    99→    for bond_a in mol_a.GetBonds():
   100→   100→        if bond_a.IsInRing() or bond_a.GetBondType() != Chem.BondType.SINGLE:
   101→   101→            continue
   102→   102→        a1, a2 = bond_a.GetBeginAtomIdx(), bond_a.GetEndAtomIdx()
   103→   103→        for bond_b in mol_b.GetBonds():
   104→   104→            if bond_b.IsInRing() or bond_b.GetBondType() != Chem.BondType.SINGLE:
   105→   105→                continue
   106→   106→            b1, b2 = bond_b.GetBeginAtomIdx(), bond_b.GetEndAtomIdx()
   107→   107→            direct = sig_a[a1] == sig_b[b1] and sig_a[a2] == sig_b[b2]
   108→   108→            flipped = sig_a[a1] == sig_b[b2] and sig_a[a2] == sig_b[b1]
   109→   109→            if direct or flipped:
   110→   110→                candidates.append(
   111→   111→                    ("bridge", bond_a.GetIdx(), bond_b.GetIdx(), direct, flipped)
   112→   112→                )
   113→   113→
   114→   114→    ring_bonds_a = [
   115→   115→        bond for bond in mol_a.GetBonds()
   116→   116→        if bond.IsInRing() and bond.GetBondType() == Chem.BondType.SINGLE
   117→   117→    ]
   118→   118→    ring_bonds_b = [
   119→   119→        bond for bond in mol_b.GetBonds()
   120→   120→        if bond.IsInRing() and bond.GetBondType() == Chem.BondType.SINGLE
   121→   121→    ]
   122→   122→    for bond_a in ring_bonds_a:
   123→   123→        a1, a2 = bond_a.GetBeginAtomIdx(), bond_a.GetEndAtomIdx()
   124→   124→        for bond_b in ring_bonds_b:
   125→   125→            b1, b2 = bond_b.GetBeginAtomIdx(), bond_b.GetEndAtomIdx()
   126→   126→            direct = sig_a[a1] == sig_b[b1] and sig_a[a2] == sig_b[b2]
   127→   127→            flipped = sig_a[a1] == sig_b[b2] and sig_a[a2] == sig_b[b1]
   128→   128→            if direct or flipped:
   129→   129→                candidates.append(("merge", bond_a.GetIdx(), bond_b.GetIdx(), direct))
   130→   130→
   131→   131→    for ring_side, other, ring_bonds, sig_ring in (
   132→   132→        ("a", mol_b, ring_bonds_a, sig_a),
   133→   133→        ("b", mol_a, ring_bonds_b, sig_b),
   134→   134→    ):
   135→   135→        for bond in ring_bonds:
   136→   136→            r1, r2 = bond.GetBeginAtomIdx(), bond.GetEndAtomIdx()
   137→   137→            for atom in other.GetAtoms():
   138→   138→                if atom.GetDegree() != 2 or atom.IsInRing():
   139→   139→                    continue
   140→   140→                excision = sh._excise_linker_path(other, (atom.GetIdx(),))
   141→   141→                if excision is None:
   142→   142→                    continue
   143→   143→                _, host_sigs = excision
   144→   144→                direct = host_sigs[0] == sig_ring[r1] and host_sigs[1] == sig_ring[r2]
   145→   145→                flipped = host_sigs[0] == sig_ring[r2] and host_sigs[1] == sig_ring[r1]
   146→   146→                if direct or flipped:
   147→   147→                    candidates.append(
   148→   148→                        ("transplant", ring_side, bond.GetIdx(), atom.GetIdx())
   149→   149→                    )
   150→   150→        if ring_bonds:
   151→   151→            for path, host_sigs in sh._linker_path_units(other):
   152→   152→                for bond in ring_bonds:
   153→   153→                    r1, r2 = bond.GetBeginAtomIdx(), bond.GetEndAtomIdx()
   154→   154→                    direct = host_sigs[0] == sig_ring[r1] and host_sigs[1] == sig_ring[r2]
   155→   155→                    flipped = host_sigs[0] == sig_ring[r2] and host_sigs[1] == sig_ring[r1]
   156→   156→                    if direct or flipped:
   157→   157→                        candidates.append(
   158→   158→                            ("transplant_path", ring_side, bond.GetIdx(), path, direct)
   159→   159→                        )
   160→   160→            for ring, a1, a2, host_sigs in sh._ring_units(other):
   161→   161→                for bond in ring_bonds:
   162→   162→                    r1, r2 = bond.GetBeginAtomIdx(), bond.GetEndAtomIdx()
   163→   163→                    direct = host_sigs[0] == sig_ring[r1] and host_sigs[1] == sig_ring[r2]
   164→   164→                    flipped = host_sigs[0] == sig_ring[r2] and host_sigs[1] == sig_ring[r1]
   165→   165→                    if direct or flipped:
   166→   166→                        candidates.append(
   167→   167→                            ("fuse", ring_side, bond.GetIdx(), ring, a1, a2, direct)
   168→   168→                        )
   169→   169→
   170→   170→    for donor_side, donor, recipient, sig_recipient in (
   171→   171→        ("a", mol_a, mol_b, sig_b),
   172→   172→        ("b", mol_b, mol_a, sig_a),
   173→   173→    ):
   174→   174→        acyclic_bonds = [
   175→   175→            bond for bond in recipient.GetBonds()
   176→   176→            if not bond.IsInRing() and bond.GetBondType() == Chem.BondType.SINGLE
   177→   177→        ]
   178→   178→        if not acyclic_bonds:
   179→   179→            continue
   180→   180→        for ring, a1, a2, host_sigs in sh._ring_units(donor):
   181→   181→            for bond in acyclic_bonds:
   182→   182→                b1, b2 = bond.GetBeginAtomIdx(), bond.GetEndAtomIdx()
   183→   183→                direct = (
   184→   184→                    host_sigs[0] == sig_recipient[b1]
   185→   185→                    and host_sigs[1] == sig_recipient[b2]
   186→   186→                )
   187→   187→                flipped = (
   188→   188→                    host_sigs[0] == sig_recipient[b2]
   189→   189→                    and host_sigs[1] == sig_recipient[b1]
   190→   190→                )
   191→   191→                if direct or flipped:
   192→   192→                    candidates.append(
   193→   193→                        ("insert", donor_side, bond.GetIdx(), ring, a1, a2, direct)
   194→   194→                    )
   195→   195→
   196→   196→    middle_units_a = sh._middle_units(mol_a)
   197→   197→    middle_units_b = sh._middle_units(mol_b)
   198→   198→    for flank_side, flank_units, piece_units in (
   199→   199→        ("a", middle_units_a, middle_units_b),
   200→   200→        ("b", middle_units_b, middle_units_a),
   201→   201→    ):
   202→   202→        for flank_path, flank_sigs in flank_units:
   203→   203→            for piece_path, piece_sigs in piece_units:
   204→   204→                direct = flank_sigs == piece_sigs
   205→   205→                flipped = (
   206→   206→                    flank_sigs[0] == piece_sigs[1]
   207→   207→                    and flank_sigs[1] == piece_sigs[0]
   208→   208→                )
   209→   209→                if direct or flipped:
   210→   210→                    candidates.append(
   211→   211→                        ("swap", flank_side, flank_path, piece_path, direct)
   212→   212→                    )
   213→   213→    return candidates

   216→   216→class RecordingRng:
   217→   217→    """Delegate rng that logs randrange/shuffle calls (scan validation)."""
   218→   218→
   219→   219→    def __init__(self, base: random.Random) -> None:
   220→   220→        self.base = base
   221→   221→        self.randrange_calls: list[tuple] = []
   222→   222→        self.shuffle_calls = 0
   223→   223→
   224→   224→    def randrange(self, *args):
   225→   225→        self.randrange_calls.append(args)
   226→   226→        return self.base.randrange(*args)
   227→   227→
   228→   228→    def shuffle(self, seq):
   229→   229→        self.shuffle_calls += 1
   230→   230→        return self.base.shuffle(seq)

   233→   233→class RiggedRng:
   234→   234→    """Delegate rng forcing ``randrange(n) -> forced`` when n == target.
   235→   235→
   236→   236→    The original scaffold_hop_crossover consumes exactly one randrange (the
   237→   237→    candidate draw) plus an optional shuffle (bridge geometry); the shuffle is
   238→   238→    delegated to the wrapped GA rng so the TOTAL rng consumption matches the
   239→   239→    production call (``noop_shuffle=True`` freezes the bridge pairing order
   240→   240→    for the scan-consistency validation below).
   241→   241→    """
   242→   242→
   243→   243→    def __init__(self, base: random.Random, forced: int, target: int,
   244→   244→                 *, noop_shuffle: bool = False) -> None:
   245→   245→        self.base = base
   246→   246→        self.forced = forced
   247→   247→        self.target = target
   248→   248→        self.noop_shuffle = noop_shuffle
   249→   249→
   250→   250→    def randrange(self, *args):
   251→   251→        if len(args) == 1 and args[0] == self.target:
   252→   252→            return self.forced
   253→   253→        return self.base.randrange(*args)
   254→   254→
   255→   255→    def shuffle(self, seq):
   256→   256→        if self.noop_shuffle:
   257→   257→            return
   258→   258→        return self.base.shuffle(seq)
   259→   259→

   262→   262→    """Recording rng with a no-op shuffle (validation only).
   263→   263→
   264→   264→    The candidate draw is the function's only 1-arg randrange, so its return
   265→   265→    value IS the drawn candidate index.

   267→   267→
   268→   268→    def __init__(self, base: random.Random) -> None:
   269→   269→        self.base = base

   271→   271→
   272→   272→    def randrange(self, *args):
   273→   273→        value = self.base.randrange(*args)
   274→   274→        if len(args) == 1:
   275→   275→            self.drawn_index = value
   276→   276→        return value
   277→   277→
   278→   278→    def shuffle(self, seq):
   279→   279→        return  # freeze the bridge pairing order (construction order)

   282→   282→def validate_scan_copy(pairs: list[tuple[str, str]], log) -> None:
   283→   283→    """Prove the scan replication matches the original function's pool.
   284→   284→
   285→   285→    Two checks per pair: (1) a recording rng reveals the original's candidate
   286→   286→    pool size -- it must equal ``len(_scan_candidates)``; (2) with the bridge
   287→   287→    shuffle frozen, forcing candidate index i through the ORIGINAL function
   288→   288→    must reproduce exactly the child a plain (shuffle-frozen) draw of index i
   289→   289→    produces -- same index, same child, so the replication is
   290→   290→    order-consistent with the production pool.
   291→   291→    """
   292→   292→    for a, b in pairs:
   293→   293→        mine = _scan_candidates(a, b)
   294→   294→        rec = RecordingRng(random.Random(0))
   295→   295→        child = sh.scaffold_hop_crossover(a, b, rec, max_smiles_length=120)
   296→   296→        if mine is None or not mine:
   297→   297→            assert child is None or rec.randrange_calls == [], (a, b, child)
   298→   298→            log(f"  scan OK: {a[:18]:>18} x {b[:18]:<18} -> no candidates "
   299→   299→                f"(original returns None: {child is None})")
   300→   300→            continue
   301→   301→        assert rec.randrange_calls and rec.randrange_calls[0] == (len(mine),), (
   302→   302→            a, b, rec.randrange_calls, len(mine)
   303→   303→        )
   304→   304→        for seed in range(12):
   305→   305→            plain_rng = NoShuffleRng(random.Random(seed))
   306→   306→            plain = sh.scaffold_hop_crossover(a, b, plain_rng,
   307→   307→                                              max_smiles_length=120)
   308→   308→            forced = sh.scaffold_hop_crossover(
   309→   309→                a, b,
   310→   310→                RiggedRng(random.Random(seed), plain_rng.drawn_index, len(mine),
   311→   311→                          noop_shuffle=True),
   312→   312→                max_smiles_length=120,
   313→   313→            )
   314→   314→            assert plain == forced, (a, b, seed, plain_rng.drawn_index, plain,
   315→   315→                                     forced)
   316→   316→        log(f"  scan OK: {a[:18]:>18} x {b[:18]:<18} -> {len(mine):3d} candidates, "
   317→   317→            f"{len({c[0] for c in mine})} geometries, index-consistent x12")

   321→   321→# 2. The AOS machinery (the production design, verbatim semantics)
   322→   322→# ---------------------------------------------------------------------------
   323→   323→class OperatorBook:
   324→   324→    """Sliding-window credit + softmax-with-floor selection for one level."""
   325→   325→
   326→   326→    def __init__(self, operators: list[str], window: int, temperature: float,
   327→   327→                 label: str) -> None:
   328→   328→        self.operators = list(operators)
   329→   329→        self.window = window
   330→   330→        self.temperature = temperature
   331→   331→        self.label = label
   332→   332→        self.rewards: dict[str, deque] = {op: deque(maxlen=window) for op in operators}
   333→   333→        self.draws: dict[str, int] = {op: 0 for op in operators}
   334→   334→        self.successes: dict[str, int] = {op: 0 for op in operators}
   335→   335→        self.best: dict[str, tuple[float, str]] = {}
   336→   336→
   337→   337→    def credit(self, op: str) -> float:
   338→   338→        window = self.rewards[op]
   339→   339→        if not window:
   340→   340→            return 0.0  # empty window -> neutral prior (never NaN)
   341→   341→        return sum(window) / len(window)
   342→   342→
   343→   343→    def probabilities(self, present: list[str] | None = None) -> dict[str, float]:
   344→   344→        ops = [op for op in self.operators if present is None or op in present]
   345→   345→        if not ops:
   346→   346→            return {}
   347→   347→        if len(ops) == 1:
   348→   348→            return {ops[0]: 1.0}
   349→   349→        k = len(ops)
   350→   350→        floor = 1.0 / (k * FLOOR_MARGIN)
   351→   351→        credits = [self.credit(op) for op in ops]
   352→   352→        top = max(credits)
   353→   353→        exps = [math.exp((c - top) / self.temperature) for c in credits]
   354→   354→        total = sum(exps)
   355→   355→        return {
   356→   356→            op: floor + (1.0 - k * floor) * (e / total)
   357→   357→            for op, e in zip(ops, exps)
   358→   358→        }
   359→   359→
   360→   360→    def draw(self, rng: random.Random, present: list[str] | None = None) -> str:
   361→   361→        probs = self.probabilities(present)
   362→   362→        ops = list(probs)
   363→   363→        pick = rng.random()
   364→   364→        cumulative = 0.0
   365→   365→        for op in ops:
   366→   366→            cumulative += probs[op]
   367→   367→            if pick < cumulative:
   368→   368→                return op
   369→   369→        return ops[-1]  # float-tail guard (deterministic)
   370→   370→
   371→   371→    def push(self, op: str, reward: float, success: bool,
   372→   372→             child: str | None, child_score: float | None) -> None:
   373→   373→        if op not in self.rewards:  # lazily discovered operator
   374→   374→            self.rewards[op] = deque(maxlen=self.window)
   375→   375→            self.draws[op] = 0
   376→   376→            self.successes[op] = 0
   377→   377→            self.operators.append(op)
   378→   378→        self.rewards[op].append(reward)
   379→   379→        self.draws[op] += 1
   380→   380→        if success:
   381→   381→            self.successes[op] += 1
   382→   382→            if child is not None and (
   383→   383→                op not in self.best or child_score > self.best[op][0]
   384→   384→            ):
   385→   385→                self.best[op] = (child_score, child)
   386→   386→
   387→   387→    def table(self) -> list[dict]:
   388→   388→        probs = self.probabilities(None)
   389→   389→        rows = []
   390→   390→        for op in self.operators:
   391→   391→            credit = self.credit(op)
   392→   392→            assert math.isfinite(credit)  # the NaN guard, asserted per row
   393→   393→            rows.append({
   394→   394→                "level": self.label,
   395→   395→                "operator": op,
   396→   396→                "draws": self.draws[op],
   397→   397→                "successes": self.successes[op],
   398→   398→                "credit": round(credit, 4),
   399→   399→                "final_probability": round(probs.get(op, 0.0), 4),
   400→   400→                "best_offspring": self.best.get(op, ("", None))[1],
   401→   401→                "best_offspring_score": round(self.best.get(op, (0.0, None))[0], 4),
   402→   402→            })
   403→   403→        return rows

   422→   422→    crossover_rate: float = 0.7,
   423→   423→    mutation_rate: float = 0.08,
   424→   424→    elitism: int = 4,
   425→   425→    tournament_k: int = 3,
   426→   426→    max_smiles_length: int = 120,
   427→   427→    fail_geometry: str | None = None,
   428→   428→):
   429→   429→    adaptive = operator_selection == "adaptive"
   430→   430→    opt = GeneticOptimizer(
   431→   431→        objective,
   432→   432→        population_size=population_size,
   433→   433→        generations=generations,
   434→   434→        mutation_rate=mutation_rate,
   435→   435→        crossover_rate=crossover_rate,
   436→   436→        elitism=elitism,
   437→   437→        tournament_k=tournament_k,
   438→   438→        max_smiles_length=max_smiles_length,
   439→   439→        mutation=mutation,
   440→   440→        crossover=crossover,
   441→   441→        seed=seed,

   443→   443→    rng = random.Random(seed)
   444→   444→

   451→   451→            canonical_pool.append(canonical)

   453→   453→    assert canonical_pool
   454→   454→    alphabet = opt._mutation_alphabet(canonical_pool)
   455→   455→    population, _kept = opt._initial_population(None, canonical_pool, rng)
   456→   456→    decoration_pool: list[str] = []
   457→   457→    if mutation == "scaffold":
   458→   458→        decoration_pool = _build_decoration_pool(canonical_pool)
   459→   459→
   460→   460→    # -- AOS state ----------------------------------------------------------
   461→   461→    level1 = OperatorBook(["crossover", "clone"], aos_window, aos_temperature,
   462→   462→                          "level1")
   463→   463→    geometry = OperatorBook(list(GEOMETRY_ORDER), aos_window, aos_temperature,
   464→   464→                            "geometry")
   465→   465→    if mutation == "topology":
   466→   466→        family = OperatorBook(["topology", "character"], aos_window,
   467→   467→                              aos_temperature, "family")
   468→   468→    elif mutation == "scaffold":
   469→   469→        family = OperatorBook(["scaffold", "character"], aos_window,
   470→   470→                              aos_temperature, "family")
   471→   471→    else:
   472→   472→        family = OperatorBook(["character"], aos_window, aos_temperature, "family")
   473→   473→
   474→   474→    produced: dict[str, float] = {}
   475→   475→    history: list[dict] = []
   476→   476→    last_fraction = 1.0
   477→   477→
   478→   478→    def character_mutate(smiles: str) -> str:
   479→   479→        # verbatim copy of the _mutate character block (rng-identical)
   480→   480→        chars = list(smiles)
   481→   481→        n_mutations = max(1, round(len(chars) * opt.mutation_rate))
   482→   482→        for _ in range(n_mutations):
   483→   483→            if not chars:
   484→   484→                break
   485→   485→            position = rng.randrange(len(chars))
   486→   486→            draw = rng.random()
   487→   487→            if draw < 0.45:  # substitute
   488→   488→                chars[position] = alphabet[rng.randrange(len(alphabet))]
   489→   489→            elif draw < 0.8:  # insert
   490→   490→                chars.insert(position, alphabet[rng.randrange(len(alphabet))])
   491→   491→            else:  # delete
   492→   492→                del chars[position]
   493→   493→        return "".join(chars)
   494→   494→
   495→   495→    for generation in range(generations):
   496→   496→        fitness = {s: _validated_score(objective, s) for s in population}
   497→   497→        for smiles, score in fitness.items():
   498→   498→            produced[smiles] = max(produced.get(smiles, 0.0), score)
   499→   499→        ranked = sorted(population, key=lambda s: (-fitness[s], s))
   500→   500→        history.append({
   501→   501→            "generation": generation,
   502→   502→            "best_score": fitness[ranked[0]],
   503→   503→            "mean_score": sum(fitness.values()) / len(fitness),
   504→   504→            "best_smiles": ranked[0],
   505→   505→            "n_unique": len(set(population)),
   506→   506→            "valid_offspring_fraction": last_fraction,
   507→   507→        })

   509→   509→            break

   512→   512→        attempts = 0

   514→   514→            attempts += 1
   515→   515→            if attempts > 50 * population_size:
   516→   516→                while len(next_population) < population_size:
   517→   517→                    next_population.append(
   518→   518→                        canonical_pool[rng.randrange(len(canonical_pool))]
   519→   519→                    )
   520→   520→                break

   522→   522→            drawn_l1: str | None = None
   523→   523→            drawn_geometry: str | None = None
   524→   524→            drawn_family: str | None = None
   525→   525→            recombined = False  # level-1 crossover product survived?
   526→   526→            geometry_survived = False
   527→   527→            family_survived = False
   528→   528→            mid_parent = 0.0

   531→   531→            else:
   532→   532→                parent_b = opt._tournament(population, fitness, rng)
   533→   533→                mid_parent = (fitness[parent_a] + fitness[parent_b]) / 2.0
   534→   534→                if adaptive:
   535→   535→                    drawn_l1 = level1.draw(rng)
   536→   536→                    use_cross = drawn_l1 == "crossover"
   537→   537→                else:
   538→   538→                    use_cross = rng.random() < opt.crossover_rate
   539→   539→                if use_cross:
   540→   540→                    if crossover == "scaffold_hop":
   541→   541→                        if adaptive:
   542→   542→                            candidates = _scan_candidates(parent_a, parent_b)
   543→   543→                            if not candidates:
   544→   544→                                child = None  # no viable cut: fallback below
   545→   545→                            else:
   546→   546→                                buckets: dict[str, list[int]] = {}
   547→   547→                                for idx, cand in enumerate(candidates):
   548→   548→                                    buckets.setdefault(cand[0], []).append(idx)

   550→   550→                                           if g in buckets]

   552→   552→                                if drawn_geometry == fail_geometry:
   553→   553→                                    child = None  # injected always-fail operator
   554→   554→                                else:
   555→   555→                                    bucket = buckets[drawn_geometry]
   556→   556→                                    idx = bucket[rng.randrange(len(bucket))]
   557→   557→                                    child = sh.scaffold_hop_crossover(
   558→   558→                                        parent_a, parent_b,
   559→   559→                                        RiggedRng(rng, idx, len(candidates)),
   560→   560→                                        max_smiles_length=max_smiles_length,
   561→   561→                                    )
   562→   562→                        else:
   563→   563→                            child = opt._crossover(parent_a, parent_b, rng)
   564→   564→                        if child is not None:
   565→   565→                            recombined = True
   566→   566→                            geometry_survived = True
   567→   567→                        child = child if child is not None else parent_a
   568→   568→                    else:  # splice: inline replica (rng-identical to _crossover)
   569→   569→                        if len(parent_a) < 2 or len(parent_b) < 2:
   570→   570→                            child = parent_a
   571→   571→                        else:
   572→   572→                            cut_a = rng.randrange(1, len(parent_a))
   573→   573→                            cut_b = rng.randrange(1, len(parent_b))
   574→   574→                            spliced = parent_a[:cut_a] + parent_b[cut_b:]
   575→   575→                            if len(spliced) > max_smiles_length:
   576→   576→                                child = parent_a
   577→   577→                            else:
   578→   578→                                child = spliced
   579→   579→                                recombined = True
   580→   580→                                geometry_survived = True
   581→   581→                        if adaptive:
   582→   582→                            drawn_geometry = "splice"
   583→   583→                else:
   584→   584→                    child = parent_a
   585→   585→            # -- mutation stage ------------------------------------------------
   586→   586→            if not adaptive:
   587→   587→                child = opt._mutate(child, alphabet, rng, decoration_pool)
   588→   588→            elif mutation == "topology":
   589→   589→                drawn_family = family.draw(rng)
   590→   590→                if drawn_family == "topology":
   591→   591→                    mutant = ring_topology_mutate(
   592→   592→                        child, rng, max_smiles_length=max_smiles_length
   593→   593→                    )
   594→   594→                    if mutant is not None:
   595→   595→                        child = mutant
   596→   596→                        family_survived = True
   597→   597→                    else:
   598→   598→                        child = character_mutate(child)
   599→   599→                else:
   600→   600→                    child = character_mutate(child)
   601→   601→                    family_survived = True
   602→   602→            else:
   603→   603→                child = character_mutate(child)
   604→   604→                drawn_family = "character"
   605→   605→                family_survived = True
   606→   606→

   609→   609→            score_child: float | None = None
   610→   610→            canonical: str | None = None

   613→   613→                if canonical is not None and len(canonical) <= max_smiles_length:
   614→   614→                    valid = True

   616→   616→                    if canonical not in next_population:
   617→   617→                        next_population.append(canonical)
   618→   618→

   621→   621→                end_to_end = (score_child if valid else 0.0) - mid_parent
   622→   622→                null_reward = 0.0 - mid_parent

   624→   624→                            end_to_end if (drawn_l1 == "clone" or recombined)
   625→   625→                            else null_reward,
   626→   626→                            valid, canonical if valid else None, score_child)
   627→   627→                if drawn_geometry is not None:
   628→   628→                    geometry.push(drawn_geometry,
   629→   629→                                  end_to_end if geometry_survived else null_reward,
   630→   630→                                  valid and geometry_survived,
   631→   631→                                  canonical if valid else None, score_child)
   632→   632→                if drawn_family is not None:
   633→   633→                    family.push(drawn_family,
   634→   634→                                end_to_end if family_survived else null_reward,
   635→   635→                                valid and family_survived,
   636→   636→                                canonical if valid else None, score_child)
   637→   637→

   639→   639→        if attempts:
   640→   640→            last_fraction = accepted / attempts
   641→   641→        history[-1]["valid_offspring_fraction"] = last_fraction
   642→   642→        population = next_population
   643→   643→

   648→   648→        "family": family.table(),
   649→   649→    }

   663→   663→from rdkit import Chem  # noqa: E402

   666→   666→def sat_ring_sizes(mol):
   667→   667→    sizes = []
   668→   668→    for ring in mol.GetRingInfo().AtomRings():
   669→   669→        bonds = [
   670→   670→            mol.GetBondBetweenAtoms(ring[i], ring[(i + 1) % len(ring)])
   671→   671→            for i in range(len(ring))
   672→   672→        ]
   673→   673→        if all(
   674→   674→            b.GetBondType() == Chem.BondType.SINGLE and not b.GetIsAromatic()
   675→   675→            for b in bonds
   676→   676→        ):
   677→   677→            sizes.append(len(ring))
   678→   678→    return sizes
   679→   679→
   680→   680→
   681→   681→class RingSizeBandObjective(ScoringObjective):
   682→   682→    name = "ring_size_band"
   683→   683→
   684→   684→    def __init__(self, lo, hi):
   685→   685→        super().__init__()
   686→   686→        self.centre = (lo + hi) / 2.0
   687→   687→        self.sigma = max((hi - lo) / 2.0, 1e-6)
   688→   688→
   689→   689→    def score(self, smiles):
   690→   690→        mol = self._parse(smiles)
   691→   691→        if mol is None:
   692→   692→            return 0.0
   693→   693→        sizes = sat_ring_sizes(mol)
   694→   694→        if not sizes:
   695→   695→            return 0.0
   696→   696→        z = (sum(sizes) / len(sizes) - self.centre) / self.sigma
   697→   697→        return math.exp(-0.5 * z * z)
   698→   698→
