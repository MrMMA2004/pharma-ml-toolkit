   106→def _hide_torch(monkeypatch: pytest.MonkeyPatch) -> None:
   107→    """Simulate a torch-less machine (import of torch raises ImportError)."""
   108→    monkeypatch.setitem(sys.modules, "torch", None)

   111→def _hide_torch_spec(monkeypatch: pytest.MonkeyPatch) -> None:
   112→    """Simulate torch being undetectable by the registry availability probe."""
   113→    real = models_module._module_available
   114→
   115→    def fake(name: str) -> bool:
   116→        return False if name == "torch" else real(name)
   117→
   118→    monkeypatch.setattr(models_module, "_module_available", fake)

   126→        # Executing the module source with torch "missing" must succeed:

   128→        _hide_torch(monkeypatch)

   130→        assert spec is not None and spec.loader is not None
   131→        module = importlib.util.module_from_spec(spec)

   133→        for name in (
   134→            "DeepMLPClassifier",
   135→            "DeepMLPRegressor",
   136→            "ResidualMLPClassifier",
   137→            "ResidualMLPRegressor",
   138→        ):
   139→            assert hasattr(module, name)

   314→        )
   315→        X_train, X_test, y_train, y_test = train_test_split(
   316→            X, y, test_size=0.25, random_state=5, stratify=y
   317→        )
   318→        model = fast_classifier(epochs=40, patience=8).fit(X_train, y_train)
   319→        assert accuracy_score(y_test, model.predict(X_test)) > 0.85
