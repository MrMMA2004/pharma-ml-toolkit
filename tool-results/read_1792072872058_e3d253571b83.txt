     1→"""Deep-learning estimators for QSAR (PyTorch, scikit-learn compatible).
     2→
     3→The legacy codebase shipped one shallow ``MLPClassifier`` recipe under
     4→the name "ANN"; this module brings *real* deep learning to the
     5→toolkit: feed-forward networks (:class:`DeepMLPClassifier` /
     6→:class:`DeepMLPRegressor`) and residual networks
     7→(:class:`ResidualMLPClassifier` / :class:`ResidualMLPRegressor`) built
     8→on PyTorch, with minibatch Adam training (weight decay), early
     9→stopping on a held-out validation split and per-feature
    10→standardisation fitted on the training data.
    11→
    12→The estimators follow the scikit-learn estimator contract (``fit`` /
    13→``predict`` / ``predict_proba``, ``get_params`` / ``set_params``,
    14→clonable by :class:`~sklearn.model_selection.GridSearchCV`), so they
    15→drop into :func:`~pharmaml.ml.training.train_model`,
    16→:func:`~pharmaml.ml.training.compare_models` and
    17→:func:`~pharmaml.ml.evaluation.evaluate_model` like any registry
    18→model — the registry keys are ``'dnn'`` and ``'dnn_residual'``.
    19→
    20→PyTorch is an *optional* dependency imported lazily, mirroring how
    21→:mod:`pharmaml.ml.models` treats XGBoost/LightGBM/CatBoost: importing
    22→this module (or ``pharmaml`` itself) never requires torch; the import
    23→happens at :meth:`fit` time and raises an actionable
    24→:class:`~pharmaml.exceptions.DependencyError` when missing::
    25→
    26→    pip install 'pharmaml[deep]'
    27→
    28→Determinism: ``fit`` seeds both the PyTorch and NumPy random number
    29→generators from ``random_state``, so two estimators constructed with
    30→the same seed and trained on the same data produce identical
    31→predictions (CPU).  Fitted weights are always stored as plain NumPy
    32→arrays, so pickled estimators stay portable across torch versions —
    33→and even across machines without torch installed (torch is only
    34→needed again to *predict*).
    35→"""

    89→def _make_activation(activation: str) -> Any:
    90→    """Return a fresh ``torch.nn`` activation module for a token."""
    91→    from torch import nn  # noqa: PLC0415 - lazy import
    92→
    93→    factory = {
    94→        "relu": nn.ReLU,
    95→        "tanh": nn.Tanh,
    96→        "gelu": nn.GELU,
    97→        "sigmoid": nn.Sigmoid,
    98→    }
    99→    return factory[activation]()

   102→def _build_network(
   103→    residual: bool,
   104→    n_in: int,
   105→    hidden: tuple[int, ...],
   106→    n_out: int,
   107→    activation: str,
   108→    dropout: float,
   109→) -> Any:
   110→    """Construct an *untrained* torch network for the given architecture.
   111→
   112→    ``residual=False`` builds a plain feed-forward stack (Linear ->
   113→    activation -> dropout between hidden layers); ``residual=True``
   114→    builds :class:`_ResidualNet` blocks with skip connections.  An
   115→    empty ``hidden`` tuple yields a single linear layer (a generalised
   116→    linear model) in both cases.
   117→    """
   118→    if residual:
   119→        return _torch_networks()[2](n_in, hidden, n_out, activation, dropout)
   120→
   121→    from torch import nn  # noqa: PLC0415 - lazy import
   122→
   123→    layers: list[Any] = []
   124→    widths = [n_in, *hidden, n_out]
   125→    for i in range(len(widths) - 1):
   126→        layers.append(nn.Linear(widths[i], widths[i + 1]))
   127→        if i < len(widths) - 2:  # hidden layer -> activation + dropout
   128→            layers.append(_make_activation(activation))
   129→            if dropout > 0:
   130→                layers.append(nn.Dropout(dropout))
   131→    return nn.Sequential(*layers)
   132→
   133→
   134→#: Lazily-created torch network classes, cached so state-dict keys are
   135→#: stable across every build within a process.
   136→_TORCH_NETS: tuple[Any, type, type] | None = None
   137→
   138→
   139→def _torch_networks() -> tuple[Any, type, type]:
   140→    """Define (once) and return ``(torch, ResidualBlock, ResidualNet)``.
   141→
   142→    The ``torch.nn.Module`` subclasses are defined lazily — never at
   143→    module import — so ``import pharmaml.ml.neural`` works on machines
   144→    without torch; caching them keeps ``state_dict`` keys identical
   145→    between the training build and the rebuild-on-predict.
   146→    """
   147→    global _TORCH_NETS
   148→    if _TORCH_NETS is None:
   149→        import torch
   150→        from torch import nn
   151→
   152→        class ResidualBlock(nn.Module):
   153→            """Linear -> activation -> dropout -> Linear with an input skip.
   154→
   155→            The skip is an identity when input and output widths match,
   156→            else a learned linear projection (the dense analogue of a
   157→            1x1 convolution).
   158→            """
   159→
   160→            def __init__(
   161→                self, width_in: int, width_out: int, activation: str, dropout: float
   162→            ) -> None:
   163→                super().__init__()
   164→                self.fc1 = nn.Linear(width_in, width_out)
   165→                self.fc2 = nn.Linear(width_out, width_out)
   166→                self.act = _make_activation(activation)
   167→                self.drop = nn.Dropout(dropout) if dropout > 0 else nn.Identity()
   168→                self.skip = (
   169→                    nn.Identity() if width_in == width_out else nn.Linear(width_in, width_out)
   170→                )
   171→
   172→            def forward(self, x: "torch.Tensor") -> "torch.Tensor":
   173→                out = self.fc2(self.drop(self.act(self.fc1(x))))
   174→                return self.act(out + self.skip(x))
   175→
   176→        class ResidualNet(nn.Module):
   177→            """A stack of :class:`ResidualBlock` layers plus a linear head."""
   178→
   179→            def __init__(
   180→                self,
   181→                n_in: int,
   182→                hidden: tuple[int, ...],
   183→                n_out: int,
   184→                activation: str,
   185→                dropout: float,
   186→            ) -> None:
   187→                super().__init__()
   188→                widths = [n_in, *hidden]
   189→                self.blocks = nn.ModuleList(
   190→                    ResidualBlock(widths[i], widths[i + 1], activation, dropout)
   191→                    for i in range(len(widths) - 1)
   192→                )
   193→                self.head = nn.Linear(widths[-1], n_out)
   194→
   195→            def forward(self, x: "torch.Tensor") -> "torch.Tensor":
   196→                for block in self.blocks:
   197→                    x = block(x)
   198→                return self.head(x)
   199→
   200→        _TORCH_NETS = (torch, ResidualBlock, ResidualNet)
   201→    return _TORCH_NETS

   204→# ---------------------------------------------------------------------------
   205→# Shared estimator machinery
   206→# ---------------------------------------------------------------------------
   207→class _BaseDeepEstimator(BaseEstimator):
   208→    """Private base with the fit/predict machinery shared by the four
   209→    public deep estimators (never instantiated directly).
   210→
   211→    Subclasses set two class flags and define the sklearn-visible
   212→    ``__init__`` (all constructor parameters verbatim, so
   213→    ``get_params``/``set_params``/``clone`` work); everything else —
   214→    validation, standardisation, the torch training loop, early
   215→    stopping, weight (de)serialisation — lives here.
   216→    """
   217→
   218→    _is_classifier: bool = False
   219→    _residual: bool = False
   220→
   221→    # -- validation -----------------------------------------------------
   222→    def _validate_hyperparameters(self) -> None:
   223→        """Raise :class:`ModelError` early for bad constructor values.
   224→
   225→        Raises
   226→        ------
   227→        ModelError
   228→            When ``hidden_layers`` holds a non-positive/non-integer
   229→            width, ``dropout``/``val_split`` fall outside ``[0, 1)``,
   230→            ``activation``/``device`` are unknown tokens, ``epochs``/
   231→            ``batch_size``/``patience`` are below 1, ``learning_rate``
   232→            is not positive-finite, ``weight_decay`` is negative, or
   233→            ``random_state`` is not an integer.
   234→        """
   235→        hidden = self.hidden_layers
   236→        if isinstance(hidden, (str, bytes)) or not hasattr(hidden, "__iter__"):
   237→            raise ModelError(
   238→                f"hidden_layers must be a sequence of positive ints, got {hidden!r}"
   239→            )
   240→        for width in hidden:
   241→            if isinstance(width, bool) or not isinstance(width, (int, np.integer)) or width < 1:
   242→                raise ModelError(
   243→                    "hidden_layers entries must be positive ints, got "
   244→                    f"{width!r} in {tuple(hidden)!r}"
   245→                )
   246→        if isinstance(self.dropout, bool) or not np.isfinite(self.dropout) or not (
   247→            0.0 <= float(self.dropout) < 1.0
   248→        ):
   249→            raise ModelError(f"dropout must be in [0, 1), got {self.dropout!r}")
   250→        if self.activation not in _ACTIVATIONS:
   251→            raise ModelError(
   252→                f"activation must be one of {_ACTIVATIONS}, got {self.activation!r}"
   253→            )
   254→        for name in ("epochs", "batch_size", "patience"):
   255→            value = getattr(self, name)
   256→            if isinstance(value, bool) or not isinstance(value, (int, np.integer)) or value < 1:
   257→                raise ModelError(f"{name} must be an integer >= 1, got {value!r}")
   258→        if not np.isfinite(self.learning_rate) or float(self.learning_rate) <= 0:
   259→            raise ModelError(f"learning_rate must be > 0, got {self.learning_rate!r}")
   260→        if not np.isfinite(self.weight_decay) or float(self.weight_decay) < 0:
   261→            raise ModelError(f"weight_decay must be >= 0, got {self.weight_decay!r}")
   262→        if not np.isfinite(self.val_split) or not 0.0 <= float(self.val_split) < 1.0:
   263→            raise ModelError(f"val_split must be in [0, 1), got {self.val_split!r}")
   264→        if self.device not in _DEVICES:
   265→            raise ModelError(f"device must be one of {_DEVICES}, got {self.device!r}")
   266→        if isinstance(self.random_state, bool) or not isinstance(
   267→            self.random_state, (int, np.integer)
   268→        ):
   269→            raise ModelError(
   270→                f"random_state must be an integer seed (determinism contract), "
   271→                f"got {self.random_state!r}"
   272→            )
   273→
   274→    def _check_X(self, X: Any) -> np.ndarray:
   275→        """Validate and convert ``X`` to a finite float64 matrix.
   276→
   277→        Raises
   278→        ------
   279→        DataValidationError
   280→            When ``X`` is not 2-dimensional, non-numeric, empty, has no
   281→            feature columns, or contains NaN/infinite values.
   282→        """
   283→        try:
   284→            X_arr = np.asarray(X, dtype=np.float64)
   285→        except (TypeError, ValueError) as exc:
   286→            raise DataValidationError(
   287→                f"features must be numeric — conversion failed: {exc}"
   288→            ) from exc
   289→        if X_arr.ndim != 2:
   290→            raise DataValidationError(
   291→                "X must be 2-dimensional (n_samples, n_features), got "
   292→                f"{X_arr.ndim} dimension(s)"
   293→            )
   294→        if X_arr.shape[0] == 0:
   295→            raise DataValidationError(
   296→                f"X has 0 rows — {type(self).__name__} needs at least one sample"
   297→            )
   298→        if X_arr.shape[1] == 0:
   299→            raise DataValidationError(
   300→                f"X has 0 feature columns — {type(self).__name__} needs at least "
   301→                "one input feature"
   302→            )
   303→        if not np.isfinite(X_arr).all():
   304→            raise DataValidationError(
   305→                "X contains NaN or infinite values — impute or drop them first "
   306→                "(pharmaml.ml.build_preprocessor does this inside a Pipeline)"
   307→            )
   308→        return X_arr
   309→
   310→    def _check_fitted(self) -> None:
   311→        """Raise :class:`ModelError` when the estimator is used before fit."""
   312→        if getattr(self, "_weights", None) is None:
   313→            raise ModelError(
   314→                f"{type(self).__name__} has not been fitted yet — call fit(X, y) "
   315→                "first"
   316→            )
   317→
   318→    # -- fit ------------------------------------------------------------
   319→    def fit(self, X: Any, y: Any) -> "_BaseDeepEstimator":
   320→        """Train the network on ``(X, y)`` with early stopping.
   321→
   322→        Parameters
   323→        ----------
   324→        X:
   325→            Feature matrix — DataFrame or 2-D array-like; must be
   326→            numeric and finite (the standard ``pharmaml`` pipeline
   327→            imputes/scales upstream).
   328→        y:
   329→            Targets — class labels (any hashable, e.g. strings or ints)
   330→            for the classifiers, numeric values for the regressors.
   331→
   332→        Returns
   333→        -------
   334→        self
   335→            Fitted estimator (``classes_``, ``n_features_in_``,
   336→            ``history_``, ``n_iter_`` and the standardisation
   337→            attributes are now available).
   338→
   339→        Raises
   340→        ------
   341→        DependencyError
   342→            When torch is not installed (see ``pip install
   343→            'pharmaml[deep]'``).
   344→        ModelError
   345→            For invalid hyperparameters (validated before any work) or
   346→            when predicting before fitting.
   347→        DataValidationError
   348→            For empty/mismatched/non-numeric ``X``/``y``, NaN targets,
   349→            or a single-class classification target.
   350→
   351→        Notes
   352→        -----
   353→        Determinism: PyTorch and NumPy are seeded from
   354→        ``random_state`` at fit start, so the same seed + data always
   355→        yields the same fitted network on CPU.
   356→        """
   357→        self._validate_hyperparameters()
   358→        X_arr = self._check_X(X)
   359→
   360→        y_arr = np.asarray(y)
   361→        if y_arr.ndim != 1:
   362→            raise DataValidationError(
   363→                f"y must be 1-dimensional, got {y_arr.ndim} dimension(s)"
   364→            )
   365→        if y_arr.size == 0:
   366→            raise DataValidationError(
   367→                f"y is empty — {type(self).__name__} needs at least one target value"
   368→            )
   369→        if y_arr.shape[0] != X_arr.shape[0]:
   370→            raise DataValidationError(
   371→                f"X and y have inconsistent lengths: {X_arr.shape[0]} rows vs "
   372→                f"{y_arr.shape[0]} targets"
   373→            )
   374→
   375→        if self._is_classifier:
   376→            classes, y_enc = np.unique(y_arr, return_inverse=True)
   377→            if classes.size < 2:
   378→                raise DataValidationError(
   379→                    "classification needs at least 2 distinct classes, got "
   380→                    f"{classes.size}: {classes.tolist()[:5]}"
   381→                )
   382→            if classes.dtype.kind in "fc" and np.isnan(classes.astype(float)).any():
   383→                raise DataValidationError(
   384→                    "classification targets contain NaN — drop those rows first"
   385→                )
   386→            self.classes_ = classes
   387→            n_out = int(classes.size)
   388→        else:
   389→            try:
   390→                y_enc = np.asarray(y_arr, dtype=np.float64)
   391→            except (TypeError, ValueError) as exc:
   392→                raise DataValidationError(
   393→                    f"regression targets must be numeric — conversion failed: {exc}"
   394→                ) from exc
   395→            if not np.isfinite(y_enc).all():
   396→                raise DataValidationError(
   397→                    "regression targets contain NaN or infinite values — drop or "
   398→                    "impute those rows first"
   399→                )
   400→            n_out = 1
   401→
   402→        self.n_features_in_ = int(X_arr.shape[1])
   403→        if isinstance(X, pd.DataFrame):
   404→            self.feature_names_in_ = np.asarray(X.columns, dtype=object)
   405→
   406→        # Per-feature standardisation, fitted on the training block only
   407→        # (constant columns get std 1 to avoid division by zero).
   408→        mean = X_arr.mean(axis=0)
   409→        std = X_arr.std(axis=0)
   410→        std = np.where(std > 0, std, 1.0)
   411→        self._mean_ = mean
   412→        self._std_ = std
   413→        X_scaled = (X_arr - mean) / std
   414→
   415→        torch = _require_torch()
   416→        device = self._resolve_device(torch)
   417→        self._train_torch(X_scaled, y_enc, n_out, device, torch)
   418→        return self
   419→
   420→    def _resolve_device(self, torch: "torch") -> str:
   421→        """Map the ``device`` token to a concrete torch device string.
   422→
   423→        Raises
   424→        ------
   425→        ModelError
   426→            When ``device='cuda'`` is requested but no CUDA device is
   427→            visible to torch.
   428→        """
   429→        if self.device == "auto":
   430→            return "cuda" if torch.cuda.is_available() else "cpu"
   431→        if self.device == "cuda" and not torch.cuda.is_available():
   432→            raise ModelError(
   433→                "device='cuda' requested but no CUDA device is visible to torch"
   434→            )
   435→        return self.device
   436→
   437→    def _train_val_split(
   438→        self, y_enc: np.ndarray, n: int, rng: np.random.Generator
   439→    ) -> tuple[np.ndarray, np.ndarray | None]:
   440→        """Split row indices into (train, validation) index arrays.
   441→
   442→        Classification splits are *stratified* (every class keeps at
   443→        least one training row); regression uses a plain random split.
   444→        Returns ``(all_rows, None)`` when ``val_split`` is 0, the data
   445→        is a single row, or the split would be degenerate — in that
   446→        case training runs for all ``epochs`` with no early stopping.
   447→        """
   448→        val_split = float(self.val_split)
   449→        if val_split <= 0 or n < 2:
   450→            return np.arange(n), None
   451→        if self._is_classifier:
   452→            parts: list[np.ndarray] = []
   453→            for cls in np.unique(y_enc):
   454→                members = np.flatnonzero(y_enc == cls)
   455→                members = members[rng.permutation(members.size)]
   456→                k = int(round(val_split * members.size))
   457→                if members.size > 1:
   458→                    k = min(k, members.size - 1)  # keep >= 1 training row per class
   459→                parts.append(members[:k])
   460→            val_idx = np.concatenate(parts).astype(int) if parts else np.array([], dtype=int)
   461→        else:
   462→            perm = rng.permutation(n)
   463→            val_idx = perm[: max(1, int(round(val_split * n)))]
   464→        if val_idx.size == 0 or val_idx.size >= n:
   465→            return np.arange(n), None
   466→        mask = np.zeros(n, dtype=bool)
   467→        mask[val_idx] = True
   468→        return np.flatnonzero(~mask), np.flatnonzero(mask)
   469→
   470→    def _train_torch(
   471→        self,
   472→        X_scaled: np.ndarray,
   473→        y_enc: np.ndarray,
   474→        n_out: int,
   475→        device: str,
   476→        torch: "torch",
   477→    ) -> None:
   478→        """Run the minibatch Adam training loop with early stopping."""
   479→        from torch import nn  # noqa: PLC0415 - lazy import
   480→
   481→        seed = int(self.random_state)
   482→        torch.manual_seed(seed)
   483→        rng = np.random.default_rng(seed)
   484→
   485→        hidden = tuple(int(w) for w in self.hidden_layers)
   486→        dropout = float(self.dropout)
   487→        # Snapshot of the *fitted* architecture — later set_params calls
   488→        # cannot desynchronise the weight rebuild.
   489→        self._net_spec_: dict[str, Any] = {
   490→            "residual": self._residual,
   491→            "n_in": int(X_scaled.shape[1]),
   492→            "hidden": hidden,
   493→            "n_out": int(n_out),
   494→            "activation": self.activation,
   495→            "dropout": dropout,
   496→        }
   497→        network = _build_network(
   498→            self._residual, self._net_spec_["n_in"], hidden, n_out, self.activation, dropout
   499→        )
   500→        network.to(device)
   501→
   502→        X_t = torch.as_tensor(X_scaled, dtype=torch.float32, device=device)
   503→        if self._is_classifier:
   504→            y_t = torch.as_tensor(y_enc, dtype=torch.long, device=device)
   505→            criterion: Any = nn.CrossEntropyLoss()
   506→        else:
   507→            y_t = torch.as_tensor(y_enc, dtype=torch.float32, device=device).reshape(-1, 1)
   508→            criterion = nn.MSELoss()
   509→        optimizer = torch.optim.Adam(
   510→            network.parameters(),
   511→            lr=float(self.learning_rate),
   512→            weight_decay=float(self.weight_decay),
   513→        )
   514→
   515→        n = X_scaled.shape[0]
   516→        train_idx, val_idx = self._train_val_split(y_enc, n, rng)
   517→        train_t = torch.as_tensor(train_idx, dtype=torch.long, device=device)
   518→        X_tr, y_tr = X_t[train_t], y_t[train_t]
   519→        X_val = y_val = None
   520→        if val_idx is not None:
   521→            val_t = torch.as_tensor(val_idx, dtype=torch.long, device=device)
   522→            X_val, y_val = X_t[val_t], y_t[val_t]
   523→
   524→        epochs = int(self.epochs)
   525→        batch_size = int(self.batch_size)
   526→        patience = int(self.patience)
   527→
   528→        history: list[dict[str, Any]] = []
   529→        best_val = float("inf")
   530→        best_state: dict[str, Any] | None = None
   531→        stale = 0
   532→
   533→        for epoch in range(1, epochs + 1):
   534→            network.train()
   535→            perm = torch.as_tensor(rng.permutation(train_idx.size), device=device)
   536→            batch_losses: list[float] = []
   537→            for start in range(0, perm.numel(), batch_size):
   538→                batch = perm[start : start + batch_size]
   539→                optimizer.zero_grad()
   540→                loss = criterion(network(X_tr[batch]), y_tr[batch])
   541→                loss.backward()
   542→                optimizer.step()
   543→                batch_losses.append(float(loss.item()))
   544→            train_loss = float(np.mean(batch_losses)) if batch_losses else float("nan")
   545→
   546→            val_loss: float | None = None
   547→            if X_val is not None:
   548→                network.eval()
   549→                with torch.no_grad():
   550→                    val_loss = float(criterion(network(X_val), y_val).item())
   551→                if val_loss < best_val:
   552→                    best_val = val_loss
   553→                    best_state = {
   554→                        key: value.detach().cpu().clone()
   555→                        for key, value in network.state_dict().items()
   556→                    }
   557→                    stale = 0
   558→                else:
   559→                    stale += 1
   560→
   561→            history.append({"epoch": epoch, "train_loss": train_loss, "val_loss": val_loss})
   562→            _LOG.debug(
   563→                "epoch %03d train_loss %.3f val_loss %s",
   564→                epoch,
   565→                train_loss,
   566→                "n/a" if val_loss is None else f"{val_loss:.3f}",
   567→            )
   568→            if val_loss is not None and stale >= patience:
   569→                _LOG.debug(
   570→                    "early stopping at epoch %03d (no val improvement for %d epochs)",
   571→                    epoch,
   572→                    patience,
   573→                )
   574→                break
   575→
   576→        if best_state is not None:  # restore the best (lowest val loss) weights
   577→            network.load_state_dict(best_state)
   578→        network.eval()
   579→
   580→        self.history_ = history
   581→        self.n_iter_ = len(history)
   582→        # Portable CPU weights — pickles never embed torch classes.
   583→        self._weights = {
   584→            key: value.detach().cpu().numpy().copy()
   585→            for key, value in network.state_dict().items()
   586→        }
   587→        self._network = network
   588→
   589→        last = history[-1]
   590→        val_text = "n/a" if last["val_loss"] is None else f"{last['val_loss']:.4f}"
   591→        n_params = sum(int(p.numel()) for p in network.parameters())
   592→        _LOG.info(
   593→            "%s fitted on %d samples: epoch %03d train_loss %.4f val_loss %s "
   594→            "(%d epochs run, %d parameters, device=%s)",
   595→            type(self).__name__,
   596→            n,
   597→            last["epoch"],
   598→            last["train_loss"],
   599→            val_text,
   600→            self.n_iter_,
   601→            n_params,
   602→            device,
   603→        )
   604→
   605→    # -- predict ----------------------------------------------------------
   606→    def _ensure_network(self) -> Any:
   607→        """Return the live torch network, rebuilding it when needed.
   608→
   609→        After unpickling (or a ``set_params`` mutation) the live module
   610→        is ``None``; it is reconstructed from the fitted architecture
   611→        snapshot and the stored NumPy weights — deterministic, because
   612→        ``load_state_dict`` overwrites every parameter.
   613→        """
   614→        network = getattr(self, "_network", None)
   615→        if network is None:
   616→            torch = _require_torch()
   617→            spec = self._net_spec_
   618→            network = _build_network(
   619→                spec["residual"], spec["n_in"], spec["hidden"], spec["n_out"],
   620→                spec["activation"], spec["dropout"],
   621→            )
   622→            network.load_state_dict(
   623→                {key: torch.from_numpy(value) for key, value in self._weights.items()}
   624→            )
   625→            self._network = network
   626→        return network
   627→
   628→    def _forward(self, X: Any) -> np.ndarray:
   629→        """Standardise ``X`` and forward it through the network (eval mode)."""
   630→        self._check_fitted()
   631→        X_arr = self._check_X(X)
   632→        torch = _require_torch()
   633→        network = self._ensure_network()
   634→        network.eval()  # disable dropout at inference time
   635→        X_scaled = (X_arr - self._mean_) / self._std_
   636→        with torch.no_grad():
   637→            out = network(torch.as_tensor(X_scaled, dtype=torch.float32))
   638→        return out.numpy()
   639→
   640→    # -- pickle support ----------------------------------------------------
   641→    def __getstate__(self) -> dict[str, Any]:
   642→        """Drop the live torch network from pickles (rebuilt on demand).
   643→
   644→        Only the NumPy weight dict is serialised, so pickled estimators
   645→        stay portable across torch versions — and loadable even where
   646→        torch is absent (torch is required again only to predict).
   647→        """
   648→        state = self.__dict__.copy()
   649→        state["_network"] = None
   650→        return state

   656→class DeepMLPClassifier(ClassifierMixin, _BaseDeepEstimator):
   657→    """Deep feed-forward neural-network classifier (PyTorch).
   658→
   659→    A fully connected network trained with minibatch Adam and
   660→    weight-decay regularisation, early-stopped on a stratified
   661→    validation split.  Multiclass (K > 2) is fully supported: the
   662→    network has K outputs and ``predict_proba`` returns a full softmax
   663→    matrix aligned with ``classes_``.
   664→
   665→    Parameters
   666→    ----------
   667→    hidden_layers:
   668→        Widths of the hidden layers, e.g. ``(64, 32)``.  An empty tuple
   669→        builds a single linear layer — i.e. a (regularised) multinomial
   670→        logistic regression.
   671→    dropout:
   672→        Dropout probability applied after every hidden activation
   673→        (``0`` disables it).
   674→    activation:
   675→        Hidden activation token: ``'relu'``, ``'tanh'``, ``'gelu'`` or
   676→        ``'sigmoid'``.
   677→    epochs:
   678→        Maximum number of training epochs (>= 1).
   679→    batch_size:
   680→        Minibatch size (>= 1).
   681→    learning_rate:
   682→        Adam step size (> 0).
   683→    weight_decay:
   684→        Adam L2 penalty (>= 0).
   685→    patience:
   686→        Early-stopping patience: stop when the validation loss has not
   687→        improved for this many epochs (>= 1).
   688→    val_split:
   689→        Fraction of the training rows held out for early stopping
   690→        (stratified).  ``0`` trains on everything and disables early
   691→        stopping.
   692→    random_state:
   693→        Integer seed — same seed + same data ⇒ identical fitted
   694→        predictions on CPU.
   695→    device:
   696→        ``'auto'`` (cuda if available else cpu), ``'cpu'`` or
   697→        ``'cuda'``.  Training only: fitted weights are always stored
   698→        on CPU.
   699→
   700→    Attributes
   701→    ----------
   702→    classes_:
   703→        Sorted class labels (strings label-encoded internally).
   704→    n_features_in_:
   705→        Number of input features seen at fit.
   706→    feature_names_in_:
   707→        Column names when ``X`` was a DataFrame.
   708→    history_:
   709→        List of per-epoch dicts ``{'epoch', 'train_loss', 'val_loss'}``
   710→        (``val_loss`` is ``None`` without a validation split).
   711→    n_iter_:
   712→        Number of epochs actually run (early stopping included).
   713→
   714→    Raises
   715→    ------
   716→    DependencyError
   717→        At fit/predict time when torch is missing.
   718→    ModelError
   719→        For invalid hyperparameters or use before ``fit``.
   720→    DataValidationError
   721→        For empty/non-numeric/mismatched data or single-class targets.
   722→
   723→    See Also
   724→    --------
   725→    DeepMLPRegressor, ResidualMLPClassifier : siblings in this module.
   726→
   727→    Notes
   728→    -----
   729→    Registry alias: ``build_model('dnn', 'classification')``.  Requires
   730→    the optional torch dependency — ``pip install 'pharmaml[deep]'``.
   731→    """
   732→
   733→    _is_classifier = True
   734→    _residual = False
   735→
   736→    def __init__(
   737→        self,
   738→        hidden_layers=(64, 32),
   739→        dropout=0.1,
   740→        activation="relu",
   741→        epochs=120,
   742→        batch_size=32,
   743→        learning_rate=1e-3,
   744→        weight_decay=1e-4,
   745→        patience=15,
   746→        val_split=0.15,
   747→        random_state=42,
   748→        device="auto",
   749→    ) -> None:
   750→        self.hidden_layers = hidden_layers
   751→        self.dropout = dropout
   752→        self.activation = activation
   753→        self.epochs = epochs
   754→        self.batch_size = batch_size
   755→        self.learning_rate = learning_rate
   756→        self.weight_decay = weight_decay
   757→        self.patience = patience
   758→        self.val_split = val_split
   759→        self.random_state = random_state
   760→        self.device = device
   761→
   762→    def predict(self, X: Any) -> np.ndarray:
   763→        """Predict class labels for ``X`` (argmax of the softmax).
   764→
   765→        Parameters
   766→        ----------
   767→        X:
   768→            Feature matrix — DataFrame or 2-D array-like, numeric.
   769→
   770→        Returns
   771→        -------
   772→        numpy.ndarray
   773→            Labels in the original ``classes_`` dtype (strings stay
   774→            strings).
   775→        """
   776→        logits = self._forward(X)
   777→        return self.classes_[np.argmax(logits, axis=1)]
   778→
   779→    def predict_proba(self, X: Any) -> np.ndarray:
   780→        """Class probabilities for ``X`` (softmax, columns == classes_).
   781→
   782→        Parameters
   783→        ----------
   784→        X:
   785→            Feature matrix — DataFrame or 2-D array-like, numeric.
   786→
   787→        Returns
   788→        -------
   789→        numpy.ndarray
   790→            Shape ``(n_samples, len(classes_))``; rows sum to 1.
   791→        """
   792→        logits = self._forward(X)
   793→        shifted = logits - logits.max(axis=1, keepdims=True)
   794→        exp = np.exp(shifted)
   795→        return exp / exp.sum(axis=1, keepdims=True)
