     1→"""Drug-response screens: GDSC/CCLE-style pharmacology tables.
     2→
     3→A *response screen* is a long-format table with one row per
     4→(cell line, compound) pair and one or more numeric metric columns --
     5→the layout popularised by the GDSC and CCLE sensitivity datasets
     6→(``LN_IC50``, ``AUC``, ``RMSE`` ...).  This module loads, audits and
     7→reshapes such tables:
     8→
     9→* :func:`load_response_table` accepts a DataFrame or a csv/xlsx path,
    10→  validates the identifier columns, auto-detects the metric columns
    11→  (every remaining numeric column) and coerces stray non-numeric
    12→  entries to NaN *with a warning* (round-11/16 semantics: warn, keep
    13→  NaN -- never drop rows silently),
    14→* :class:`ResponseScreen` is the immutable container carrying the
    15→  frame plus the column configuration, with a ``summary()`` in the
    16→  package's audit style,
    17→* :func:`audit_response_table` reports the classic screen hygiene
    18→  issues (missing identifiers, missing metrics, duplicate pairs, cell
    19→  lines claimed by two tissues, under-tested compounds),
    20→* :func:`collapse_replicates` resolves duplicated pairs by the median
    21→  per metric (rows with a missing identifier are kept as their own
    22→  NaN-keyed group, never silently dropped),
    23→* :func:`ln_ic50_to_pic50` / :func:`pic50_to_ln_ic50` convert between
    24→  GDSC's ``LN_IC50`` (natural log of the IC50 in micromolar) and the
    25→  chemist's pIC50 scale,
    26→* :func:`response_matrix` pivots one metric into the classic
    27→  cell-line x compound sensitivity matrix (untested pairs stay NaN),
    28→* :func:`summarize_compounds` gives per-compound potency statistics
    29→  across the panel, and
    30→* :func:`call_sensitive_resistant` thresholds one compound's response
    31→  vector into sensitive / intermediate / resistant calls.
    32→
    33→Direction conventions matter everywhere here: for ``LN_IC50``-style
    34→metrics *lower* means more sensitive, for pIC50-style metrics *higher*
    35→means more sensitive.  Every direction-sensitive function documents
    36→its assumption and exposes a switch.
    37→"""
    38→
    39→from __future__ import annotations
    40→
    41→import math
    42→from dataclasses import dataclass, field
    43→from pathlib import Path
    44→from typing import TypeAlias
    45→
    46→import numpy as np
    47→import pandas as pd
    48→
    49→from pharmaml.exceptions import DataValidationError
    50→from pharmaml.logger import get_logger
    51→from pharmaml.utils.io import read_table
    52→from pharmaml.utils.validation import require_columns
    53→
    54→_LOG = get_logger(__name__)
    55→
    56→__all__ = [
    57→    "ResponseScreen",
    58→    "audit_response_table",
    59→    "call_sensitive_resistant",
    60→    "collapse_replicates",
    61→    "ln_ic50_to_pic50",
    62→    "load_response_table",
    63→    "pic50_to_ln_ic50",
    64→    "response_matrix",
    65→    "summarize_compounds",
    66→]
    67→
    68→#: What a table argument may be: an in-memory frame or a csv/xlsx path.
    69→ResponseTableSource: TypeAlias = pd.DataFrame | str | Path
    70→
    71→#: ``ln(10)`` -- the natural-log/decade conversion factor.
    72→_LN10 = math.log(10.0)

    74→
    75→def _detect_metric_columns(
    76→    frame: pd.DataFrame, exclude: list[str]
    77→) -> list[str]:
    78→    """Return the remaining columns that hold numeric measurements.
    79→
    80→    A column qualifies when its dtype is numeric or when at least one
    81→    value parses as a number (``pd.to_numeric`` with coercion) -- the
    82→    latter keeps a ``LN_IC50`` column that went ``object`` dtype
    83→    because of a few stray strings in an otherwise numeric export.
    84→
    85→    Boolean columns never qualify: ``pandas`` counts ``bool`` as a
    86→    numeric dtype, but an ``msi_status``-style flag is an annotation,
    87→    not a measurement.  The rule covers both ``bool`` dtype and an
    88→    ``object`` column holding only ``True``/``False`` (with missing
    89→    entries -- what a csv round-trip of a flag column produces).
    90→    Datetime columns are excluded in the same spirit: a
    91→    ``screened_on`` screening date is an annotation, and
    92→    ``pd.to_numeric`` would otherwise coerce it to ns-epoch integers
    93→    that masquerade as measurements (the ``datetime64`` dtypes pandas'
    94→    Excel/parquet readers emit).
    95→    Pass such a column explicitly via ``metrics=[...]`` to treat its
    96→    ``True``/``False`` values as 1/0.
    97→    """
    98→    metrics: list[str] = []
    99→    for column in frame.columns:
   100→        if column in exclude:
   101→            continue
   102→        series = frame[column]
   103→        if pd.api.types.is_bool_dtype(series):
   104→            continue
   105→        if pd.api.types.is_datetime64_any_dtype(series):
   106→            continue
   107→        if pd.api.types.is_numeric_dtype(series):
   108→            metrics.append(column)
   109→            continue
   110→        non_missing = series.dropna()
   111→        if not non_missing.empty and non_missing.map(
   112→            lambda value: isinstance(value, (bool, np.bool_))
   113→        ).all():
   114→            continue
   115→        coerced = pd.to_numeric(series, errors="coerce")
   116→        if coerced.notna().any():
   117→            metrics.append(column)
   118→    return metrics
   119→
   120→
   121→def _coerce_metrics(frame: pd.DataFrame, metrics: list[str]) -> pd.DataFrame:
   122→    """Coerce metric columns to numeric dtype, warning per column.
   123→
   124→    Values that fail to parse become NaN *after* a warning names the
   125→    column and the affected count (round-11/16 semantics: warn, keep
   126→    NaN).  Already-numeric columns pass through unchanged.  Boolean
   127→    metric columns (only reachable through the explicit ``metrics=``
   128→    path -- auto-detection skips them) are converted to 0/1 floats:
   129→    ``pd.to_numeric`` passes ``bool`` dtype through unchanged, but a
   130→    metric column must support arithmetic.
   131→    """
   132→    for metric in metrics:
   133→        series = frame[metric]
   134→        if pd.api.types.is_bool_dtype(series):
   135→            frame[metric] = series.astype("float64")
   136→            continue
   137→        if pd.api.types.is_numeric_dtype(series):
   138→            continue
   139→        coerced = pd.to_numeric(series, errors="coerce")
   140→        lost = int((series.notna() & coerced.isna()).sum())
   141→        if lost:
   142→            _LOG.warning(
   143→                "%d non-numeric value(s) in metric column %r coerced to NaN",
   144→                lost,
   145→                metric,
   146→            )
   147→        frame[metric] = coerced
   148→    return frame

   151→@dataclass(frozen=True, slots=True)
   152→class ResponseScreen:
   153→    """One loaded drug-response screen.
   154→
   155→    Attributes
   156→    ----------
   157→    frame:
   158→        The long-format table: one row per (cell line, compound) pair
   159→        with the metric columns already coerced to numeric dtype.
   160→    cell_col:
   161→        Name of the cell-line identifier column.
   162→    compound_col:
   163→        Name of the compound identifier column.
   164→    tissue_col:
   165→        Name of the optional tissue/lineage annotation column.
   166→    metrics:
   167→        Names of the numeric response-metric columns (``LN_IC50``,
   168→        ``AUC`` ...).  When empty at construction time the numeric
   169→        non-identifier columns are auto-detected.
   170→
   171→    Raises
   172→    ------
   173→    DataValidationError
   174→        If no metric column can be auto-detected.
   175→    """
   176→
   177→    frame: pd.DataFrame
   178→    cell_col: str
   179→    compound_col: str
   180→    tissue_col: str | None = None
   181→    metrics: list[str] = field(default_factory=list)
   182→
   183→    def __post_init__(self) -> None:
   184→        if not self.metrics:
   185→            detected = _detect_metric_columns(
   186→                self.frame,
   187→                exclude=[
   188→                    column
   189→                    for column in (self.cell_col, self.compound_col, self.tissue_col)
   190→                    if column is not None
   191→                ],
   192→            )
   193→            if not detected:
   194→                raise DataValidationError(
   195→                    "no metric columns detected in the response table -- "
   196→                    "pass metrics=[...] explicitly. Available columns: "
   197→                    f"{list(self.frame.columns)}"
   198→                )
   199→            object.__setattr__(self, "metrics", detected)
   200→
   201→    @property
   202→    def cell_lines(self) -> list[str]:
   203→        """Sorted unique cell-line identifiers (missing ones excluded)."""
   204→        return sorted(self.frame[self.cell_col].dropna().astype(str).unique().tolist())
   205→
   206→    @property
   207→    def compounds(self) -> list[str]:
   208→        """Sorted unique compound identifiers (missing ones excluded)."""
   209→        return sorted(
   210→            self.frame[self.compound_col].dropna().astype(str).unique().tolist()
   211→        )
   212→
   213→    @property
   214→    def n_pairs(self) -> int:
   215→        """Number of unique (cell line, compound) pairs."""
   216→        return len(self.frame[[self.cell_col, self.compound_col]].drop_duplicates())
   217→
   218→    def _duplicate_pairs(self) -> list[tuple[str, str]]:
   219→        """(cell, compound) pairs that occupy more than one row.
   220→
   221→        Uses ``dropna=False`` so pairs keyed by a missing identifier
   222→        are counted too -- silently dropping them would hide
   223→        duplicated unusable rows from every audit.
   224→        """
   225→        counts = self.frame.groupby(
   226→            [self.cell_col, self.compound_col], sort=True, dropna=False
   227→        ).size()
   228→        return [pair for pair, count in counts.items() if count > 1]
   229→
   230→    def summary(self) -> str:
   231→        """Multi-line audit summary (the package's ``inspect`` style)."""
   232→        lines = [
   233→            (f"screen        : {len(self.cell_lines)} cell line(s) x "
   234→            f"{len(self.compounds)} compound(s) -> {self.n_pairs} (cell, compound) pair(s)")
   235→        ]
   236→        if len(self.frame) != self.n_pairs:
   237→            lines.append(
   238→                f"rows          : {len(self.frame)} "
   239→                f"({len(self.frame) - self.n_pairs} duplicate row(s))"
   240→            )
   241→        if self.tissue_col is not None:
   242→            tissues = sorted(
   243→                self.frame[self.tissue_col].dropna().astype(str).unique().tolist()
   244→            )
   245→            shown = ", ".join(tissues[:8])
   246→            extra = len(tissues) - 8
   247→            more = f" (+{extra} more)" if extra > 0 else ""
   248→            lines.append(
   249→                f"tissue        : {len(tissues)} value(s) in {self.tissue_col!r}: "
   250→                f"{shown}{more}"
   251→            )
   252→        lines.append(f"metrics       : {', '.join(self.metrics) or '(none)'}")
   253→        for metric in self.metrics:
   254→            present = int(self.frame[metric].notna().sum())
   255→            total = len(self.frame)
   256→            missing = total - present
   257→            note = f" ({missing} missing)" if missing else ""
   258→            lines.append(f"  {metric:<13s}: {present}/{total} value(s) present{note}")
   259→        duplicates = self._duplicate_pairs()
   260→        if duplicates:
   261→            shown = ", ".join(f"({cell} + {compound})" for cell, compound in duplicates[:3])
   262→            extra = len(duplicates) - 3
   263→            more = f" (+{extra} more)" if extra > 0 else ""
   264→            lines.append(f"dup. pairs    : {len(duplicates)}: {shown}{more}")
   265→        else:
   266→            lines.append("dup. pairs    : 0")
   267→        return "\n".join(lines)
   268→

   270→def load_response_table(
   271→    data: ResponseTableSource,
   272→    cell_column: str = "cell_line",
   273→    compound_column: str = "compound",
   274→    tissue_column: str | None = None,
   275→    metrics: list[str] | None = None,
   276→) -> ResponseScreen:
   277→    """Load a drug-response screen from a DataFrame or csv/xlsx path.
   278→
   279→    Parameters
   280→    ----------
   281→    data:
   282→        Long-format table (one row per cell line x compound pair) as a
   283→        DataFrame or a path read via
   284→        :func:`pharmaml.utils.io.read_table`.
   285→    cell_column, compound_column:
   286→        Names of the identifier columns.
   287→    tissue_column:
   288→        Optional tissue annotation column (kept out of the metrics).
   289→    metrics:
   290→        Response-metric column names.  When ``None`` (the default) all
   291→        remaining numeric columns are used -- including ``object``
   292→        columns that hold at least one parseable number, so a stray
   293→        string in an otherwise numeric export degrades to NaN instead
   294→        of hiding the whole metric.  Boolean columns (flags like
   295→        ``msi_status``) and datetime columns (screening dates) are
   296→        *never* auto-detected; pass one explicitly to have its
   297→        ``True``/``False`` values treated as 1/0 (or its datetimes
   298→        coerced, which is the caller's responsibility).
   299→
   300→    Returns
   301→    -------
   302→    ResponseScreen
   303→        The validated screen.  Metric columns are coerced to numeric
   304→        dtype on a *copy* of the frame (the caller's frame is never
   305→        mutated); non-numeric values become NaN after a warning that
   306→        names the column and the count.
   307→
   308→    Raises
   309→    ------
   310→    DataValidationError
   311→        If an identifier column is missing (all offenders listed at
   312→        once), an explicit metric column is missing, or no metric
   313→        column can be detected at all.
   314→    """
   315→    if isinstance(data, pd.DataFrame):
   316→        frame = data.copy()
   317→    elif isinstance(data, (str, Path)):
   318→        frame = read_table(data)
   319→    else:
   320→        raise DataValidationError(
   321→            "data must be a pandas DataFrame or a path to a csv/xlsx file, "
   322→            f"got {type(data).__name__}"
   323→        )
   324→
   325→    required = [cell_column, compound_column]
   326→    if tissue_column is not None:
   327→        required.append(tissue_column)
   328→    require_columns(frame, required)
   329→
   330→    if metrics is None:
   331→        detected = _detect_metric_columns(frame, exclude=required)
   332→        if not detected:
   333→            raise DataValidationError(
   334→                "no metric columns detected in the response table -- pass "
   335→                "metrics=[...] explicitly. Available columns: "
   336→                f"{list(frame.columns)}"
   337→            )
   338→        metrics = detected
   339→        _LOG.debug("Auto-detected metric column(s): %s", metrics)
   340→    else:
   341→        require_columns(frame, list(metrics))
   342→
   343→    frame = _coerce_metrics(frame, list(metrics))
   344→    _LOG.info(
   345→        "Loaded response screen: %d row(s), %d cell line(s), %d compound(s), "
   346→        "metric(s) %s",
   347→        len(frame),
   348→        frame[cell_column].nunique(),
   349→        frame[compound_column].nunique(),
   350→        metrics,
   351→    )
   352→    return ResponseScreen(
   353→        frame=frame,
   354→        cell_col=cell_column,
   355→        compound_col=compound_column,
   356→        tissue_col=tissue_column,
   357→        metrics=list(metrics),
   358→    )

   361→def audit_response_table(screen: ResponseScreen, n_min: int = 3) -> pd.DataFrame:
   362→    """Audit a response screen for the classic hygiene issues.
   363→
   364→    One row per concrete issue with columns ``issue``, ``detail`` and
   365→    ``count``:
   366→
   367→    * ``missing identifier`` -- per identifier column (cell line /
   368→      compound) holding NaN entries; ``count`` is the number of rows
   369→      without the identifier,
   370→    * ``missing metric values`` -- per metric column with missing
   371→      entries; ``count`` is the number of missing values,
   372→    * ``duplicate pairs`` -- per (cell line, compound) pair occupying
   373→      more than one row (pairs keyed by a missing identifier included);
   374→      ``count`` is the number of rows for the pair,
   375→    * ``ambiguous tissue`` -- per cell line mapped to more than one
   376→      tissue; ``count`` is the number of distinct tissues,
   377→    * ``under-tested compound`` -- per compound measured on fewer than
   378→      ``n_min`` cell lines; ``count`` is the number of cell lines.
   379→
   380→    A clean screen returns the single row ``("ok", "no issues found",
   381→    0)``.  ``to_string()`` on the result is print-friendly.
   382→
   383→    Parameters
   384→    ----------
   385→    screen:
   386→        The screen to audit.
   387→    n_min:
   388→        Minimum number of cell lines a compound should be measured on
   389→        (GDSC-style panels use 3 as the floor for any per-compound
   390→        statistic).
   391→    """
   392→    rows: list[dict[str, object]] = []
   393→
   394→    for column in (screen.cell_col, screen.compound_col):
   395→        missing_ids = int(screen.frame[column].isna().sum())
   396→        if missing_ids:
   397→            rows.append(
   398→                {
   399→                    "issue": "missing identifier",
   400→                    "detail": f"column {column!r}",
   401→                    "count": missing_ids,
   402→                }
   403→            )
   404→
   405→    for metric in screen.metrics:
   406→        missing = int(screen.frame[metric].isna().sum())
   407→        if missing:
   408→            rows.append(
   409→                {
   410→                    "issue": "missing metric values",
   411→                    "detail": f"metric {metric!r}",
   412→                    "count": missing,
   413→                }
   414→            )
   415→
   416→    pair_counts = screen.frame.groupby(
   417→        [screen.cell_col, screen.compound_col], sort=True, dropna=False
   418→    ).size()
   419→    for (cell, compound), count in pair_counts.items():
   420→        if count > 1:
   421→            rows.append(
   422→                {
   423→                    "issue": "duplicate pairs",
   424→                    "detail": f"{cell} / {compound}",
   425→                    "count": int(count),
   426→                }
   427→            )
   428→
   429→    if screen.tissue_col is not None:
   430→        tissues = (
   431→            screen.frame[[screen.cell_col, screen.tissue_col]]
   432→            .dropna(subset=[screen.tissue_col])
   433→            .groupby(screen.cell_col, sort=True, dropna=False)[screen.tissue_col]
   434→            .agg(lambda values: sorted(set(values.astype(str))))
   435→        )
   436→        for cell, tissue_list in tissues.items():
   437→            if len(tissue_list) > 1:
   438→                rows.append(
   439→                    {
   440→                        "issue": "ambiguous tissue",
   441→                        "detail": f"{cell}: {', '.join(tissue_list)}",
   442→                        "count": len(tissue_list),
   443→                    }
   444→                )
   445→
   446→    n_cells = screen.frame.groupby(screen.compound_col, sort=True, dropna=False)[
   447→        screen.cell_col
   448→    ].nunique()
   449→    for compound, count in n_cells.items():
   450→        if count < n_min:
   451→            rows.append(
   452→                {
   453→                    "issue": "under-tested compound",
   454→                    "detail": f"{compound}",
   455→                    "count": int(count),
   456→                }
   457→            )
   458→
   459→    if not rows:
   460→        rows.append({"issue": "ok", "detail": "no issues found", "count": 0})
   461→    report = pd.DataFrame(rows, columns=["issue", "detail", "count"])
   462→    _LOG.info("Response-table audit: %d issue row(s)", len(report))
   463→    return report

   466→def collapse_replicates(screen: ResponseScreen) -> ResponseScreen:
   467→    """Collapse duplicated (cell line, compound) pairs by the median.
   468→
   469→    Every pair is reduced to a single row: metric columns take the
   470→    median over the replicates, annotation columns (tissue, study ...)
   471→    keep the first value.  Pairs without duplicates pass through
   472→    untouched.  Rows whose cell-line or compound identifier is missing
   473→    are kept as their own (NaN-keyed) group instead of being dropped
   474→    (pandas' ``groupby`` default would silently discard them); a
   475→    warning names the count of such rows.  The collapsed pairs are
   476→    logged at DEBUG level.
   477→
   478→    Returns
   479→    -------
   480→    ResponseScreen
   481→        A new screen (the input is never mutated) with at most one row
   482→        per pair, preserving first-appearance row order.
   483→    """
   484→    frame = screen.frame
   485→    missing_ids = frame[screen.cell_col].isna() | frame[screen.compound_col].isna()
   486→    if missing_ids.any():
   487→        _LOG.warning(
   488→            "%d row(s) have a missing cell-line/compound identifier -- kept "
   489→            "as their own (NaN) group instead of being dropped "
   490→            "(audit_response_table reports them)",
   491→            int(missing_ids.sum()),
   492→        )
   493→    aggregations: dict[str, str] = {}
   494→    for column in frame.columns:
   495→        aggregations[column] = (
   496→            "median" if column in screen.metrics else "first"
   497→        )
   498→    collapsed = frame.groupby(
   499→        [screen.cell_col, screen.compound_col],
   500→        sort=False,
   501→        as_index=False,
   502→        dropna=False,
   503→    ).agg(aggregations)
   504→
   505→    duplicate_pairs = screen._duplicate_pairs()
   506→    if duplicate_pairs:
   507→        shown = ", ".join(f"({cell} + {compound})" for cell, compound in duplicate_pairs[:10])
   508→        extra = len(duplicate_pairs) - 10
   509→        more = f" (+{extra} more)" if extra > 0 else ""
   510→        _LOG.debug(
   511→            "Collapsed %d duplicate pair(s) by median: %s%s",
   512→            len(duplicate_pairs),
   513→            shown,
   514→            more,
   515→        )
   516→    _LOG.info(
   517→        "collapse_replicates: %d row(s) -> %d row(s)", len(frame), len(collapsed)
   518→    )
   519→    return ResponseScreen(
   520→        frame=collapsed,
   521→        cell_col=screen.cell_col,
   522→        compound_col=screen.compound_col,
   523→        tissue_col=screen.tissue_col,
   524→        metrics=list(screen.metrics),
   525→    )

   528→def ln_ic50_to_pic50(ln_ic50):
   529→    """Convert GDSC ``LN_IC50`` values to pIC50.
   530→
   531→    GDSC reports the natural log of the IC50 in micromolar, so
   532→    ``LN_IC50 = 0`` (IC50 = 1 uM) maps to ``pIC50 = 6`` via
   533→    ``pIC50 = 6 - ln(IC50_uM) / ln(10)``.
   534→
   535→    Parameters
   536→    ----------
   537→    ln_ic50:
   538→        Scalar, list/array or :class:`pandas.Series` of ``LN_IC50``
   539→        values.  NaN passes through; a Series keeps its index.
   540→
   541→    Returns
   542→    -------
   543→    Same kind as the input (float for scalars, ndarray for arrays,
   544→    Series for Series) with the converted values.
   545→    """
   546→    if isinstance(ln_ic50, pd.Series):
   547→        numeric = pd.to_numeric(ln_ic50, errors="coerce")
   548→        return 6.0 - numeric / _LN10
   549→    values = np.asarray(ln_ic50, dtype=float)
   550→    result = 6.0 - values / _LN10
   551→    if result.ndim == 0:
   552→        return float(result)
   553→    return result
   554→
   555→
   556→def pic50_to_ln_ic50(pic50):
   557→    """Convert pIC50 values back to GDSC ``LN_IC50`` (the exact inverse).
   558→
   559→    ``ln(IC50_uM) = (6 - pIC50) * ln(10)``; a 1 uM IC50 (pIC50 6)
   560→    maps back to ``LN_IC50 = 0``.  Accepts the same input kinds as
   561→    :func:`ln_ic50_to_pic50` with NaN passthrough.
   562→    """
   563→    if isinstance(pic50, pd.Series):
   564→        numeric = pd.to_numeric(pic50, errors="coerce")
   565→        return (6.0 - numeric) * _LN10
   566→    values = np.asarray(pic50, dtype=float)
   567→    result = (6.0 - values) * _LN10
   568→    if result.ndim == 0:
   569→        return float(result)
   570→    return result

   573→def response_matrix(screen: ResponseScreen, metric: str) -> pd.DataFrame:
   574→    """Pivot one metric into a cell-line x compound sensitivity matrix.
   575→
   576→    Rows are cell lines, columns are compounds (both sorted); untested
   577→    pairs hold NaN.  Duplicated pairs are *not* silently averaged --
   578→    collapse them first with :func:`collapse_replicates`.
   579→
   580→    Parameters
   581→    ----------
   582→    screen:
   583→        The screen to pivot.
   584→    metric:
   585→        Metric column to place in the matrix.
   586→
   587→    Returns
   588→    -------
   589→    pandas.DataFrame
   590→        The pivoted matrix.
   591→
   592→    Raises
   593→    ------
   594→    DataValidationError
   595→        If ``metric`` is not one of the screen's metrics, or duplicate
   596→        (cell line, compound) pairs are present.
   597→    """
   598→    if metric not in screen.metrics:
   599→        raise DataValidationError(
   600→            f"metric {metric!r} is not part of the screen -- available "
   601→            f"metrics: {list(screen.metrics)}"
   602→        )
   603→    duplicates = screen._duplicate_pairs()
   604→    if duplicates:
   605→        shown = ", ".join(f"({cell} + {compound})" for cell, compound in duplicates[:5])
   606→        extra = len(duplicates) - 5
   607→        more = f" (+{extra} more)" if extra > 0 else ""
   608→        raise DataValidationError(
   609→            f"{len(duplicates)} duplicate (cell, compound) pair(s) prevent "
   610→            f"pivoting: {shown}{more} -- collapse_replicates() first"
   611→        )
   612→    matrix = screen.frame.pivot(
   613→        index=screen.cell_col, columns=screen.compound_col, values=metric
   614→    )
   615→    matrix.index.name = screen.cell_col
   616→    matrix.columns.name = screen.compound_col
   617→    return matrix.sort_index().sort_index(axis=1)
   618→

   620→def summarize_compounds(
   621→    screen: ResponseScreen,
   622→    metric: str,
   623→    higher_is_more_sensitive: bool = False,
   624→) -> pd.DataFrame:
   625→    """Per-compound potency statistics across the cell-line panel.
   626→
   627→    Rows are compounds (sorted) with columns ``n_cell_lines`` (number of
   628→    distinct cell lines measured, replicates counted once -- missing
   629→    metric values are the audit's business), ``median``, ``min``,
   630→    ``max``, ``std`` (sample standard deviation, ddof=1) and
   631→    ``sensitive_spread`` = ``max - min`` -- the potency range across
   632→    the panel, i.e. how selectively the compound hits the panel
   633→    regardless of direction.  Rows whose compound identifier is missing
   634→    form their own trailing ``NaN`` row instead of being dropped
   635→    (:func:`audit_response_table` reports them).
   636→
   637→    Parameters
   638→    ----------
   639→    screen:
   640→        The screen to summarise.
   641→    metric:
   642→        Metric column to summarise.
   643→    higher_is_more_sensitive:
   644→        Direction of ``metric``: ``False`` (default) for LN_IC50-style
   645→        metrics where *lower* means more potent; ``True`` for
   646→        pIC50-style metrics where *higher* means more potent.  It only
   647→        affects the ``most_sensitive_cell_line`` column -- the cell
   648→        line with the strongest response under that direction.
   649→
   650→    Raises
   651→    ------
   652→    DataValidationError
   653→        If ``metric`` is not one of the screen's metrics, or the frame's
   654→        row index holds duplicated labels -- the positional
   655→        ``idxmin``/``idxmax`` re-alignment would crash there; call
   656→        ``screen.frame.reset_index(drop=True)`` (or reload the table)
   657→        first.
   658→    """
   659→    if metric not in screen.metrics:
   660→        raise DataValidationError(
   661→            f"metric {metric!r} is not part of the screen -- available "
   662→            f"metrics: {list(screen.metrics)}"
   663→        )
   664→    if screen.frame.index.has_duplicates:
   665→        raise DataValidationError(
   666→            "the screen frame has duplicated row index labels -- "
   667→            "summarize_compounds aligns its picks by label; call "
   668→            "screen.frame.reset_index(drop=True) (or reload the table) "
   669→            "first"
   670→        )
   671→    grouped = screen.frame.groupby(
   672→        screen.compound_col, sort=True, dropna=False
   673→    )[metric]
   674→    summary = pd.DataFrame(
   675→        {
   676→            # distinct cell lines (replicates counted once); missing
   677→            # metric values stay the audit's job to report.
   678→            "n_cell_lines": screen.frame.groupby(
   679→                screen.compound_col, sort=True, dropna=False
   680→            )[screen.cell_col].nunique(),
   681→            "median": grouped.median(),
   682→            "min": grouped.min(),
   683→            "max": grouped.max(),
   684→            "std": grouped.std(),
   685→        }
   686→    )
   687→    summary["sensitive_spread"] = summary["max"] - summary["min"]
   688→
   689→    # Direction-aware pick of the strongest-responding cell line per
   690→    # compound (NaN metric rows cannot win by construction).
   691→    lookup = screen.frame[
   692→        [screen.compound_col, screen.cell_col, metric]
   693→    ].dropna(subset=[metric])
   694→    if higher_is_more_sensitive:
   695→        positions = lookup.groupby(
   696→            screen.compound_col, sort=True, dropna=False
   697→        )[metric].idxmax()
   698→    else:
   699→        positions = lookup.groupby(
   700→            screen.compound_col, sort=True, dropna=False
   701→        )[metric].idxmin()
   702→    # ``positions`` maps compound -> winning row label; re-align the picked
   703→    # cell lines onto the compound index before assigning into the summary.
   704→    summary["most_sensitive_cell_line"] = pd.Series(
   705→        lookup.loc[positions, screen.cell_col].to_numpy(), index=positions.index
   706→    )
   707→    summary = summary[
   708→        [
   709→            "n_cell_lines",
   710→            "median",
   711→            "min",
   712→            "max",
   713→            "std",
   714→            "sensitive_spread",
   715→            "most_sensitive_cell_line",
   716→        ]
   717→    ]
   718→    summary.index.name = screen.compound_col
   719→    return summary

   722→def call_sensitive_resistant(
   723→    response_series,
   724→    sensitive_quantile: float = 0.25,
   725→    resistant_quantile: float = 0.75,
   726→    lower_is_sensitive: bool = True,
   727→):
   728→    """Threshold a response vector into sensitivity calls.
   729→
   730→    The lowest ``sensitive_quantile`` fraction of cells is called
   731→    ``"sensitive"``, the highest ``resistant_quantile`` fraction
   732→    ``"resistant"`` and everything in between ``"intermediate"``
   733→    (thresholds are inclusive, so tied values stay on the boundary
   734→    side).  NaN entries are labelled ``"not tested"``.
   735→
   736→    Parameters
   737→    ----------
   738→    response_series:
   739→        Response values, typically one compound's column of a
   740→        :func:`response_matrix` pivot.
   741→    sensitive_quantile, resistant_quantile:
   742→        Quantile cuts; must satisfy
   743→        ``0 < sensitive_quantile < resistant_quantile < 1``.
   744→    lower_is_sensitive:
   745→        Direction of the metric: ``True`` (default) for LN_IC50-style
   746→        metrics where *lower* means more sensitive; ``False`` for
   747→        pIC50-style metrics where *higher* means more sensitive (the
   748→        quantile cuts are mirrored accordingly).
   749→
   750→    Returns
   751→    -------
   752→    pandas.Series
   753→        Labels ``"sensitive"`` / ``"resistant"`` / ``"intermediate"``
   754→        / ``"not tested"``, indexed like the input.
   755→
   756→    Raises
   757→    ------
   758→    DataValidationError
   759→        If the quantile ordering is violated.
   760→    """
   761→    if not (
   762→        0.0 < sensitive_quantile < resistant_quantile < 1.0
   763→    ):
   764→        raise DataValidationError(
   765→            "quantiles must satisfy 0 < sensitive_quantile < "
   766→            f"resistant_quantile < 1, got {sensitive_quantile} and "
   767→            f"{resistant_quantile}"
   768→        )
   769→    series = (
   770→        response_series
   771→        if isinstance(response_series, pd.Series)
   772→        else pd.Series(response_series)
   773→    )
   774→    numeric = pd.to_numeric(series, errors="coerce")
   775→    labels = pd.Series("not tested", index=numeric.index, dtype=object)
   776→
   777→    valid = numeric.dropna()
   778→    if valid.empty:
   779→        _LOG.warning(
   780→            "call_sensitive_resistant: no non-NaN values -- every cell line "
   781→            "labelled 'not tested'"
   782→        )
   783→        return labels
   784→
   785→    if lower_is_sensitive:
   786→        sensitive_cut = valid.quantile(sensitive_quantile)
   787→        resistant_cut = valid.quantile(resistant_quantile)
   788→        is_sensitive = numeric <= sensitive_cut
   789→        is_resistant = numeric >= resistant_cut
   790→    else:
   791→        # High values are the sensitive ones: mirror the quantile cuts.
   792→        sensitive_cut = valid.quantile(1.0 - sensitive_quantile)
   793→        resistant_cut = valid.quantile(1.0 - resistant_quantile)
   794→        is_sensitive = numeric >= sensitive_cut
   795→        is_resistant = numeric <= resistant_cut
   796→
   797→    labels[is_sensitive] = "sensitive"
   798→    labels[is_resistant & ~is_sensitive] = "resistant"
   799→    intermediate = numeric.notna() & ~is_sensitive & ~is_resistant
   800→    labels[intermediate] = "intermediate"
   801→    _LOG.debug(
   802→        "Sensitivity calls: %d sensitive, %d resistant, %d intermediate, "
   803→        "%d not tested",
   804→        int((labels == "sensitive").sum()),
   805→        int((labels == "resistant").sum()),
   806→        int((labels == "intermediate").sum()),
   807→        int((labels == "not tested").sum()),
   808→    )
   809→    return labels
