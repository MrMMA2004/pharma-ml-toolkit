    79→/** Factory for tiny boundary dot renderers (60×60 grid → 3600 points). */
    80→function TinyDot(color: string) {
    81→  return function renderTinyDot(props: unknown) {
    82→    const { cx, cy } = props as { cx?: number; cy?: number };
    83→    if (cx === undefined || cy === undefined) return <g />;
    84→    return <circle cx={cx} cy={cy} r={1.5} fill={color} fillOpacity={0.3} />;
    85→  };
    86→}
    87→
    88→/** Factory for the larger moon-class dots. */
    89→function MoonDot(color: string) {
    90→  return function renderMoonDot(props: unknown) {
    91→    const { cx, cy } = props as { cx?: number; cy?: number };
    92→    if (cx === undefined || cy === undefined) return <g />;
    93→    return <circle cx={cx} cy={cy} r={3.2} fill={color} fillOpacity={0.9} stroke={color} strokeWidth={0.6} />;
    94→  };
    95→}

    97→/** Qualitative label for a Pearson correlation coefficient. */
    98→function corrStrength(r: number): string {
    99→  const a = Math.abs(r);
   100→  if (a >= 0.9) return "very strong";
   101→  if (a >= 0.7) return "strong";
   102→  if (a >= 0.5) return "moderate";
   103→  return "weak";
   104→}

   106→export function MlPlayground() {
   107→  /* Experiment setup + results live in the store, so the whole playground
   108→   * (config, trained model, dataset preview) survives tab switches. */
   109→  const mlConfig = useToolkitStore((s) => s.mlConfig);
   110→  const setMlConfig = useToolkitStore((s) => s.setMlConfig);
   111→  const resetMlConfig = useToolkitStore((s) => s.resetMlConfig);
   112→  const result = useToolkitStore((s) => s.mlResult);
   113→  const setResult = useToolkitStore((s) => s.setMlResult);
   114→  const preview = useToolkitStore((s) => s.mlPreview);
   115→  const setPreview = useToolkitStore((s) => s.setMlPreview);
   116→
   117→  const { task, samples, noise, regModel, alpha, l2, testSizePct, cv, seed } = mlConfig;
   118→
   119→  /* Transient request state — not worth persisting across tab switches. */
   120→  const [status, setStatus] = useState<"idle" | "running" | "done" | "error">(
   121→    "idle"
   122→  );
   123→  const [error, setError] = useState<string | null>(null);
   124→  const [previewLoading, setPreviewLoading] = useState(false);
   125→  const [previewOpen, setPreviewOpen] = useState(true);
   126→
   127→  /* X-range zoom on the preview scatter — null = full range. */
   128→  const [xZoom, setXZoom] = useState<[number, number] | null>(null);
   129→
   130→  /** Classification preview: the sampled points split by class for the chart. */
   131→  const previewClasses = useMemo(() => {
   132→    if (!preview || preview.task !== "classification") return null;
   133→    return {
   134→      cls0: preview.points.filter((p) => p.label === 0),
   135→      cls1: preview.points.filter((p) => p.label === 1),
   136→    };
   137→  }, [preview]);
   138→
   139→  /** Windowed view of the preview dataset (a true brush): the x-window
   140→   *  filters the series data AND re-derives both axis domains, so zooming
   141→   *  rescales the plot instead of merely clipping it. */
   142→  const previewView = useMemo(() => {
   143→    if (!preview || preview.points.length === 0) return null;
   144→    const xs = preview.points.map((p) => p.x);
   145→    const fullX: [number, number] = [Math.min(...xs), Math.max(...xs)];
   146→    const yPad = preview.task === "regression" ? 0.4 : 0.3;
   147→    const zoomed = xZoom !== null && (xZoom[0] > fullX[0] || xZoom[1] < fullX[1]);
   148→    const win: [number, number] = xZoom ?? fullX;
   149→
   150→    const inside =
   151→      zoomed ? preview.points.filter((p) => p.x >= win[0] && p.x <= win[1]) : preview.points;
   152→    const ySource = inside.length > 0 ? inside : preview.points;
   153→    const ys = ySource.map((p) => p.y);
   154→    // Pad the window slightly so edge dots aren't cut by the plot boundary.
   155→    const xPad = zoomed ? Math.max((win[1] - win[0]) * 0.04, 0.05) : yPad;
   156→
   157→    const inWin = (p: { x: number }) => !zoomed || (p.x >= win[0] && p.x <= win[1]);
   158→    return {
   159→      fullX,
   160→      win,
   161→      zoomed,
   162→      xDomain: [win[0] - xPad, win[1] + xPad] as [number, number],
   163→      yDomain: [Math.min(...ys) - yPad, Math.max(...ys) + yPad] as [number, number],
   164→      points: inside,
   165→      cls0: previewClasses ? previewClasses.cls0.filter(inWin) : [],
   166→      cls1: previewClasses ? previewClasses.cls1.filter(inWin) : [],
   167→      trend: (preview.trend ?? []).filter(inWin),
   168→    };
   169→  }, [preview, previewClasses, xZoom]);
   170→
   171→  const noiseMax = task === "regression" ? 2 : 1.5;
   172→  const metricSpecs = task === "regression" ? REG_METRICS : CLS_METRICS;
   173→
   174→  function switchTask(next: Task) {
   175→    setMlConfig({ task: next });
   176→    if (next === "classification" && noise > 1.5) setMlConfig({ noise: 0.3 });
   177→    if (next === "regression" && regModel !== "ridge" && regModel !== "linear" && regModel !== "polynomial") {
   178→      setMlConfig({ regModel: "linear" });
   179→    }
   180→    setStatus("idle");
   181→    setResult(null);
   182→    setError(null);
   183→    setPreview(null);
   184→    setXZoom(null);
   185→  }
   186→
   187→  /** Restore the default experiment setup and drop any stored results. */
   188→  function resetSetup() {
   189→    resetMlConfig();
   190→    setStatus("idle");
   191→    setResult(null);
   192→    setPreview(null);
   193→    setError(null);
   194→    setXZoom(null);
   195→  }
   196→
   197→  const seedValid = /^\d{1,10}$/.test(seed.trim());
   198→  const alphaValid = Number.isFinite(Number.parseFloat(alpha)) && Number.parseFloat(alpha) >= 0;
   199→  const l2Valid = Number.isFinite(Number.parseFloat(l2)) && Number.parseFloat(l2) >= 0;
   200→
   201→  async function run() {
   202→    if (!seedValid) {
   203→      setError("Seed must be a non-negative integer (0 – 2147483647).");
   204→      return;
   205→    }
   206→    if (task === "regression" && regModel === "ridge" && !alphaValid) {
   207→      setError("Ridge alpha must be a number ≥ 0.");
   208→      return;
   209→    }
   210→    if (task === "classification" && !l2Valid) {
   211→      setError("L2 strength must be a number ≥ 0.");
   212→      return;
   213→    }
   214→    setStatus("running");
   215→    setError(null);
   216→    const body: MLTrainRequest = {
   217→      task,
   218→      samples,
   219→      noise,
   220→      model: task === "regression" ? regModel : "logistic",
   221→      testSize: testSizePct / 100,
   222→      cv: (Number.parseInt(cv, 10) || 0) as 0 | 3 | 5 | 10,
   223→      seed: Number.parseInt(seed, 10) || 0,
   224→    };
   225→    if (task === "regression" && regModel === "ridge") body.alpha = Number.parseFloat(alpha);
   226→    if (task === "classification") body.l2 = Number.parseFloat(l2);
   227→
   228→    try {
   229→      const res = await fetch("/api/ml/train", {
   230→        method: "POST",
   231→        headers: { "Content-Type": "application/json" },
   232→        body: JSON.stringify(body),
   233→      });
   234→      const data: unknown = await res.json();
   235→      if (!res.ok) {
   236→        const message =
   237→          typeof data === "object" && data !== null && "error" in data
   238→            ? `${String((data as { error: unknown }).error)}${
   239→                "detail" in data && Array.isArray((data as { detail: unknown }).detail)
   240→                  ? ` ${((data as { detail: unknown[] }).detail as string[]).join("; ")}`
   241→                  : ""
   242→              }`
   243→            : `HTTP ${res.status}`;
   244→        throw new Error(message);
   245→      }
   246→      setResult(data as MLTrainResponse);
   247→      setStatus("done");
   248→    } catch (err) {
   249→      setStatus("error");
   250→      const message = err instanceof Error ? err.message : "Training request failed.";
   251→      setError(message);
   252→    }
   253→  }  /** Fetch the first rows + summary stats of the (deterministic) dataset. */
   254→  async function previewDataset() {
   255→    if (!seedValid) {
   256→      setError("Seed must be a non-negative integer (0 – 2147483647).");
   257→      return;
   258→    }
   259→    setPreviewLoading(true);
   260→    setError(null);
   261→    try {
   262→      const res = await fetch("/api/ml/dataset", {
   263→        method: "POST",
   264→        headers: { "Content-Type": "application/json" },
   265→        body: JSON.stringify({
   266→          task,
   267→          samples,
   268→          noise,
   269→          seed: Number.parseInt(seed, 10) || 0,
   270→        }),
   271→      });
   272→      const data: unknown = await res.json();
   273→      if (!res.ok) {
   274→        const message =
   275→          typeof data === "object" && data !== null && "error" in data
   276→            ? `${String((data as { error: unknown }).error)}${
   277→                "detail" in data && Array.isArray((data as { detail: unknown }).detail)
   278→                  ? ` ${((data as { detail: unknown[] }).detail as string[]).join("; ")}`
   279→                  : ""
   280→              }`
   281→            : `HTTP ${res.status}`;
   282→        throw new Error(message);
   283→      }
   284→      setPreview(data as DatasetPreviewResponse);
   285→      setPreviewOpen(true);
   286→      setXZoom(null); // fresh dataset — reset the x-range zoom
   287→    } catch (err) {
   288→      setError(err instanceof Error ? err.message : "Preview request failed.");
   289→    } finally {
   290→      setPreviewLoading(false);
   291→    }
   292→  }
   293→
   294→  const running = status === "running";
   295→
   296→  /* Chart series (memoized splits so recharts gets stable references). */
   297→  const clsSeries = useMemo(() => {
   298→    if (!result || result.task !== "classification") return null;
   299→    const boundary = result.series.boundary ?? [];
   300→    return {
   301→      grid0: boundary.filter((p) => p.pred === 0),
   302→      grid1: boundary.filter((p) => p.pred === 1),
   303→      moon0: result.series.test.filter((p) => p.label === 0),
   304→      moon1: result.series.test.filter((p) => p.label === 1),
   305→    };
   306→  }, [result]);
   307→
   308→  const coefficientsText = useMemo(() => {
   309→    if (!result) return null;
   310→    const { intercept, coefs } = result.coefficients;
   311→    if (result.task === "regression") {
   312→      const terms = coefs
   313→        .map((c, i) => `${fmtNum(c, 4)}·x̂${coefs.length > 1 ? `^${i + 1}` : ""}`)
   314→        .join(" + ");
   315→      return `ŷ = ${fmtNum(intercept, 4)} + ${terms}   (x̂ = standardized x)`;
   316→    }
   317→    return `σ(b₀ + w₁x̂₁ + w₂x̂₂)   b₀=${fmtNum(intercept, 4)}, w=[${coefs
   318→      .map((c) => fmtNum(c, 4))
   319→      .join(", ")}]`;
   320→  }, [result]);
   321→
   322→  return (
   323→    <div className="grid gap-4 xl:grid-cols-[minmax(300px,360px)_minmax(0,1fr)]">
   324→      {/* ------------------------------ Controls ------------------------------ */}
   325→      <Card className="h-fit overflow-hidden py-0 shadow-sm xl:sticky xl:top-24">
   326→        <CardHeader className="border-b bg-muted/30 py-3">
   327→          <div className="flex items-center gap-2.5">
   328→            <span className="flex size-9 items-center justify-center rounded-lg bg-primary/10 text-primary">
   329→              <Brain className="size-4.5" aria-hidden />
   330→            </span>
   331→            <div>
   332→              <CardTitle className="text-base">Experiment setup</CardTitle>
   333→              <CardDescription className="text-xs">
   334→                Deterministic — same seed, same result · survives tab switches
   335→              </CardDescription>
   336→            </div>
   337→          </div>
   338→        </CardHeader>
   339→        <CardContent className="space-y-5 pb-4">
   340→          {/* Task */}
   341→          <div className="space-y-1.5">
   342→            <Label className="text-xs font-medium">Task</Label>
   343→            <div className="grid grid-cols-2 gap-1 rounded-lg border bg-muted/40 p-1" role="tablist" aria-label="ML task">
   344→              {(["regression", "classification"] as const).map((t) => (
   345→                <button
   346→                  key={t}
   347→                  type="button"
   348→                  role="tab"
   349→                  aria-selected={task === t}
   350→                  onClick={() => switchTask(t)}
   351→                  className={`h-9 rounded-md text-sm font-medium capitalize transition-colors focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring ${
   352→                    task === t ? "bg-background shadow-sm text-foreground" : "text-muted-foreground hover:text-foreground"
   353→                  }`}
   354→                >
   355→                  {t}
   356→                </button>
   357→              ))}
   358→            </div>
   359→          </div>
   360→
   361→          {/* Samples */}
   362→          <div className="space-y-2">
   363→            <div className="flex items-center justify-between">
   364→              <Label htmlFor="ml-samples" className="text-xs font-medium">
   365→                Samples
   366→              </Label>
   367→              <span className="font-mono text-xs tabular-nums text-muted-foreground">{samples}</span>
   368→            </div>
   369→            <Slider
   370→              id="ml-samples"
   371→              min={100}
   372→              max={1000}
   373→              step={50}
   374→              value={[samples]}
   375→              onValueChange={(v) => setMlConfig({ samples: v[0] ?? samples })}
   376→            />
   377→          </div>
   378→
   379→          {/* Noise */}
   380→          <div className="space-y-2">
   381→            <div className="flex items-center justify-between">
   382→              <Label htmlFor="ml-noise" className="text-xs font-medium">
   383→                Noise {task === "classification" ? "(moons)" : "σ"}
   384→              </Label>
   385→              <span className="font-mono text-xs tabular-nums text-muted-foreground">
   386→                {noise.toFixed(2)}

   388→            </div>

   393→              step={0.05}
   394→              value={[Math.min(noise, noiseMax)]}
   395→              onValueChange={(v) => setMlConfig({ noise: v[0] ?? noise })}
   396→            />
   397→          </div>
   398→
   399→          {/* Model */}
   400→          <div className="space-y-1.5">
   401→            <Label className="text-xs font-medium">Model</Label>
   402→            {task === "regression" ? (
