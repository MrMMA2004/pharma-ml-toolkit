     1→"""Drug-response screens: GDSC/CCLE-style pharmacology tables.
     2→
     3→A *response screen* is a long-format table with one row per
     4→(cell line, compound) pair and one or more numeric metric columns --
     5→the layout popularised by the GDSC and CCLE sensitivity datasets
     6→(``LN_IC50``, ``AUC``, ``RMSE`` ...).  This module loads, audits and
     7→reshapes such tables:
     8→
     9→* :func:`load_response_table` accepts a DataFrame or a csv/xlsx path,
    10→  validates the identifier columns, auto-detects the metric columns
    11→  (every remaining numeric column) and coerces stray non-numeric
    12→  entries to NaN *with a warning* (round-11/16 semantics: warn, keep
    13→  NaN -- never drop rows silently),
    14→* :class:`ResponseScreen` is the immutable container carrying the
    15→  frame plus the column configuration, with a ``summary()`` in the
    16→  package's audit style,
    17→* :func:`audit_response_table` reports the classic screen hygiene
    18→  issues (missing metrics, duplicate pairs, cell lines claimed by two
    19→  tissues, under-tested compounds),
    20→* :func:`collapse_replicates` resolves duplicated pairs by the median
    21→  per metric,
    22→* :func:`ln_ic50_to_pic50` / :func:`pic50_to_ln_ic50` convert between
    23→  GDSC's ``LN_IC50`` (natural log of the IC50 in micromolar) and the
    24→  chemist's pIC50 scale,
    25→* :func:`response_matrix` pivots one metric into the classic
    26→  cell-line x compound sensitivity matrix (untested pairs stay NaN),
    27→* :func:`summarize_compounds` gives per-compound potency statistics
    28→  across the panel, and
    29→* :func:`call_sensitive_resistant` thresholds one compound's response
    30→  vector into sensitive / intermediate / resistant calls.
    31→
    32→Direction conventions matter everywhere here: for ``LN_IC50``-style
    33→metrics *lower* means more sensitive, for pIC50-style metrics *higher*
    34→means more sensitive.  Every direction-sensitive function documents
    35→its assumption and exposes a switch.
    36→"""
    37→
    38→from __future__ import annotations
    39→
    40→import math
    41→from dataclasses import dataclass, field
    42→from pathlib import Path
    43→from typing import Union
    44→
    45→import numpy as np
    46→import pandas as pd
    47→
    48→from pharmaml.exceptions import DataValidationError
    49→from pharmaml.logger import get_logger
    50→from pharmaml.utils.io import read_table
    51→from pharmaml.utils.validation import require_columns
    52→
    53→_LOG = get_logger(__name__)
    54→
    55→__all__ = [
    56→    "ResponseScreen",
    57→    "load_response_table",
    58→    "audit_response_table",

    65→]
    66→
    67→#: What a table argument may be: an in-memory frame or a csv/xlsx path.
    68→ResponseTableSource = Union[pd.DataFrame, str, Path]
    69→
    70→#: ``ln(10)`` -- the natural-log/decade conversion factor.
    71→_LN10 = math.log(10.0)

    73→
    74→def _detect_metric_columns(
    75→    frame: pd.DataFrame, exclude: list[str]
    76→) -> list[str]:
    77→    """Return the remaining columns that hold numeric measurements.
    78→
    79→    A column qualifies when its dtype is numeric or when at least one
    80→    value parses as a number (``pd.to_numeric`` with coercion) -- the
    81→    latter keeps a ``LN_IC50`` column that went ``object`` dtype
    82→    because of a few stray strings in an otherwise numeric export.
    83→    """
    84→    metrics: list[str] = []
    85→    for column in frame.columns:
    86→        if column in exclude:
    87→            continue
    88→        series = frame[column]
    89→        if pd.api.types.is_numeric_dtype(series):
    90→            metrics.append(column)
    91→            continue
    92→        coerced = pd.to_numeric(series, errors="coerce")
    93→        if coerced.notna().any():
    94→            metrics.append(column)
    95→    return metrics
    96→
    97→
    98→def _coerce_metrics(frame: pd.DataFrame, metrics: list[str]) -> pd.DataFrame:
    99→    """Coerce metric columns to numeric dtype, warning per column.
   100→
   101→    Values that fail to parse become NaN *after* a warning names the
   102→    column and the affected count (round-11/16 semantics: warn, keep
   103→    NaN).  Already-numeric columns pass through unchanged.
   104→    """
   105→    for metric in metrics:
   106→        series = frame[metric]
   107→        if pd.api.types.is_numeric_dtype(series):
   108→            continue
   109→        coerced = pd.to_numeric(series, errors="coerce")
   110→        lost = int((series.notna() & coerced.isna()).sum())
   111→        if lost:
   112→            _LOG.warning(
   113→                "%d non-numeric value(s) in metric column %r coerced to NaN",
   114→                lost,
   115→                metric,
   116→            )
   117→        frame[metric] = coerced
   118→    return frame

   121→@dataclass(frozen=True, slots=True)
   122→class ResponseScreen:
   123→    """One loaded drug-response screen.
   124→
   125→    Attributes
   126→    ----------
   127→    frame:
   128→        The long-format table: one row per (cell line, compound) pair
   129→        with the metric columns already coerced to numeric dtype.
   130→    cell_col:
   131→        Name of the cell-line identifier column.
   132→    compound_col:
   133→        Name of the compound identifier column.
   134→    tissue_col:
   135→        Name of the optional tissue/lineage annotation column.
   136→    metrics:
   137→        Names of the numeric response-metric columns (``LN_IC50``,
   138→        ``AUC`` ...).  When empty at construction time the numeric
   139→        non-identifier columns are auto-detected.
   140→
   141→    Raises
   142→    ------
   143→    DataValidationError
   144→        If no metric column can be auto-detected.
   145→    """
   146→
   147→    frame: pd.DataFrame
   148→    cell_col: str
   149→    compound_col: str
   150→    tissue_col: str | None = None
   151→    metrics: list[str] = field(default_factory=list)
   152→
   153→    def __post_init__(self) -> None:
   154→        if not self.metrics:
   155→            detected = _detect_metric_columns(
   156→                self.frame,
   157→                exclude=[self.cell_col, self.compound_col, self.tissue_col],
   158→            )
   159→            if not detected:
   160→                raise DataValidationError(
   161→                    "no metric columns detected in the response table -- "
   162→                    "pass metrics=[...] explicitly. Available columns: "
   163→                    f"{list(self.frame.columns)}"
   164→                )
   165→            object.__setattr__(self, "metrics", detected)
   166→
   167→    @property
   168→    def cell_lines(self) -> list[str]:
   169→        """Sorted unique cell-line identifiers."""
   170→        return sorted(self.frame[self.cell_col].astype(str).unique().tolist())
   171→
   172→    @property
   173→    def compounds(self) -> list[str]:
   174→        """Sorted unique compound identifiers."""
   175→        return sorted(self.frame[self.compound_col].astype(str).unique().tolist())
   176→
   177→    @property
   178→    def n_pairs(self) -> int:
   179→        """Number of unique (cell line, compound) pairs."""
   180→        return int(
   181→            len(self.frame[[self.cell_col, self.compound_col]].drop_duplicates())
   182→        )
   183→
   184→    def _duplicate_pairs(self) -> list[tuple[str, str]]:
   185→        """(cell, compound) pairs that occupy more than one row."""
   186→        counts = self.frame.groupby(
   187→            [self.cell_col, self.compound_col], sort=True
   188→        ).size()
   189→        return [pair for pair, count in counts.items() if count > 1]
   190→
   191→    def summary(self) -> str:
   192→        """Multi-line audit summary (the package's ``inspect`` style)."""
   193→        lines = [
   194→            f"screen        : {len(self.cell_lines)} cell line(s) x "
   195→            f"{len(self.compounds)} compound(s) -> {self.n_pairs} (cell, compound) pair(s)"
   196→        ]
   197→        if len(self.frame) != self.n_pairs:
   198→            lines.append(
   199→                f"rows          : {len(self.frame)} "
   200→                f"({len(self.frame) - self.n_pairs} duplicate row(s))"
   201→            )
   202→        if self.tissue_col is not None:
   203→            tissues = sorted(
   204→                self.frame[self.tissue_col].dropna().astype(str).unique().tolist()
   205→            )
   206→            shown = ", ".join(tissues[:8])
   207→            extra = len(tissues) - 8
   208→            more = f" (+{extra} more)" if extra > 0 else ""
   209→            lines.append(
   210→                f"tissue        : {len(tissues)} value(s) in {self.tissue_col!r}: "
   211→                f"{shown}{more}"
   212→            )
   213→        lines.append(f"metrics       : {', '.join(self.metrics) or '(none)'}")
   214→        for metric in self.metrics:
   215→            present = int(self.frame[metric].notna().sum())
   216→            total = len(self.frame)
   217→            missing = total - present
   218→            note = f" ({missing} missing)" if missing else ""
   219→            lines.append(f"  {metric:<13s}: {present}/{total} value(s) present{note}")
   220→        duplicates = self._duplicate_pairs()
   221→        if duplicates:
   222→            shown = ", ".join(f"({cell} + {compound})" for cell, compound in duplicates[:3])
   223→            extra = len(duplicates) - 3
   224→            more = f" (+{extra} more)" if extra > 0 else ""
   225→            lines.append(f"dup. pairs    : {len(duplicates)}: {shown}{more}")
   226→        else:
   227→            lines.append("dup. pairs    : 0")
   228→        return "\n".join(lines)
   229→

   231→def load_response_table(
   232→    data: ResponseTableSource,
   233→    cell_column: str = "cell_line",
   234→    compound_column: str = "compound",
   235→    tissue_column: str | None = None,
   236→    metrics: list[str] | None = None,
   237→) -> ResponseScreen:
   238→    """Load a drug-response screen from a DataFrame or csv/xlsx path.
   239→
   240→    Parameters
   241→    ----------
   242→    data:
   243→        Long-format table (one row per cell line x compound pair) as a
   244→        DataFrame or a path read via
   245→        :func:`pharmaml.utils.io.read_table`.
   246→    cell_column, compound_column:
   247→        Names of the identifier columns.
   248→    tissue_column:
   249→        Optional tissue annotation column (kept out of the metrics).
   250→    metrics:
   251→        Response-metric column names.  When ``None`` (the default) all
   252→        remaining numeric columns are used -- including ``object``
   253→        columns that hold at least one parseable number, so a stray
   254→        string in an otherwise numeric export degrades to NaN instead
   255→        of hiding the whole metric.
   256→
   257→    Returns
   258→    -------
   259→    ResponseScreen
   260→        The validated screen.  Metric columns are coerced to numeric
   261→        dtype on a *copy* of the frame (the caller's frame is never
   262→        mutated); non-numeric values become NaN after a warning that
   263→        names the column and the count.
   264→
   265→    Raises
   266→    ------
   267→    DataValidationError
   268→        If an identifier column is missing (all offenders listed at
   269→        once), an explicit metric column is missing, or no metric
   270→        column can be detected at all.
   271→    """
   272→    if isinstance(data, pd.DataFrame):
   273→        frame = data.copy()
   274→    elif isinstance(data, (str, Path)):
   275→        frame = read_table(data)
   276→    else:
   277→        raise DataValidationError(
   278→            "data must be a pandas DataFrame or a path to a csv/xlsx file, "
   279→            f"got {type(data).__name__}"
   280→        )
   281→
   282→    required = [cell_column, compound_column]
   283→    if tissue_column is not None:
   284→        required.append(tissue_column)
   285→    require_columns(frame, required)
   286→
   287→    if metrics is None:
   288→        detected = _detect_metric_columns(frame, exclude=required)
   289→        if not detected:
   290→            raise DataValidationError(
   291→                "no metric columns detected in the response table -- pass "
   292→                "metrics=[...] explicitly. Available columns: "
   293→                f"{list(frame.columns)}"
   294→            )
   295→        metrics = detected
   296→        _LOG.debug("Auto-detected metric column(s): %s", metrics)
   297→    else:
   298→        require_columns(frame, list(metrics))
   299→
   300→    frame = _coerce_metrics(frame, list(metrics))
   301→    _LOG.info(
   302→        "Loaded response screen: %d row(s), %d cell line(s), %d compound(s), "
   303→        "metric(s) %s",
   304→        len(frame),
   305→        frame[cell_column].nunique(),
   306→        frame[compound_column].nunique(),
   307→        metrics,
   308→    )
   309→    return ResponseScreen(
   310→        frame=frame,
   311→        cell_col=cell_column,
   312→        compound_col=compound_column,
   313→        tissue_col=tissue_column,
   314→        metrics=list(metrics),
   315→    )

   318→def audit_response_table(screen: ResponseScreen, n_min: int = 3) -> pd.DataFrame:
   319→    """Audit a response screen for the classic hygiene issues.
   320→
   321→    One row per concrete issue with columns ``issue``, ``detail`` and
   322→    ``count``:
   323→
   324→    * ``missing metric values`` -- per metric column with missing
   325→      entries; ``count`` is the number of missing values,
   326→    * ``duplicate pairs`` -- per (cell line, compound) pair occupying
   327→      more than one row; ``count`` is the number of rows for the pair,
   328→    * ``ambiguous tissue`` -- per cell line mapped to more than one
   329→      tissue; ``count`` is the number of distinct tissues,
   330→    * ``under-tested compound`` -- per compound measured on fewer than
   331→      ``n_min`` cell lines; ``count`` is the number of cell lines.
   332→
   333→    A clean screen returns the single row ``("ok", "no issues found",
   334→    0)``.  ``to_string()`` on the result is print-friendly.
   335→
   336→    Parameters
   337→    ----------
   338→    screen:
   339→        The screen to audit.
   340→    n_min:
   341→        Minimum number of cell lines a compound should be measured on
   342→        (GDSC-style panels use 3 as the floor for any per-compound
   343→        statistic).
   344→    """
   345→    rows: list[dict[str, object]] = []
   346→
   347→    for metric in screen.metrics:
   348→        missing = int(screen.frame[metric].isna().sum())
   349→        if missing:
   350→            rows.append(
   351→                {
   352→                    "issue": "missing metric values",
   353→                    "detail": f"metric {metric!r}",
   354→                    "count": missing,
   355→                }
   356→            )
   357→
   358→    pair_counts = screen.frame.groupby(
   359→        [screen.cell_col, screen.compound_col], sort=True
   360→    ).size()
   361→    for (cell, compound), count in pair_counts.items():
   362→        if count > 1:
   363→            rows.append(
   364→                {
   365→                    "issue": "duplicate pairs",
   366→                    "detail": f"{cell} / {compound}",
   367→                    "count": int(count),
   368→                }
   369→            )
   370→
   371→    if screen.tissue_col is not None:
   372→        tissues = (
   373→            screen.frame[[screen.cell_col, screen.tissue_col]]
   374→            .dropna(subset=[screen.tissue_col])
   375→            .groupby(screen.cell_col, sort=True)[screen.tissue_col]
   376→            .agg(lambda values: sorted(set(values.astype(str))))
   377→        )
   378→        for cell, tissue_list in tissues.items():
   379→            if len(tissue_list) > 1:
   380→                rows.append(
   381→                    {
   382→                        "issue": "ambiguous tissue",
   383→                        "detail": f"{cell}: {', '.join(tissue_list)}",
   384→                        "count": len(tissue_list),
   385→                    }
   386→                )
   387→
   388→    n_cells = screen.frame.groupby(screen.compound_col, sort=True)[
   389→        screen.cell_col
   390→    ].nunique()
   391→    for compound, count in n_cells.items():
   392→        if count < n_min:
   393→            rows.append(
   394→                {
   395→                    "issue": "under-tested compound",
   396→                    "detail": f"{compound}",
   397→                    "count": int(count),
   398→                }
   399→            )
   400→
   401→    if not rows:
   402→        rows.append({"issue": "ok", "detail": "no issues found", "count": 0})
   403→    report = pd.DataFrame(rows, columns=["issue", "detail", "count"])
   404→    _LOG.info("Response-table audit: %d issue row(s)", len(report))
   405→    return report

   408→def collapse_replicates(screen: ResponseScreen) -> ResponseScreen:
   409→    """Collapse duplicated (cell line, compound) pairs by the median.
   410→
   411→    Every pair is reduced to a single row: metric columns take the
   412→    median over the replicates, annotation columns (tissue, study ...)
   413→    keep the first value.  Pairs without duplicates pass through
   414→    untouched.  The collapsed pairs are logged at DEBUG level.
   415→
   416→    Returns
   417→    -------
   418→    ResponseScreen
   419→        A new screen (the input is never mutated) with at most one row
   420→        per pair, preserving first-appearance row order.
   421→    """
   422→    frame = screen.frame
   423→    aggregations: dict[str, str] = {}
   424→    for column in frame.columns:
   425→        aggregations[column] = (
   426→            "median" if column in screen.metrics else "first"
   427→        )
   428→    collapsed = frame.groupby(
   429→        [screen.cell_col, screen.compound_col], sort=False, as_index=False
   430→    ).agg(aggregations)
   431→
   432→    duplicate_pairs = screen._duplicate_pairs()
   433→    if duplicate_pairs:
   434→        shown = ", ".join(f"({cell} + {compound})" for cell, compound in duplicate_pairs[:10])
   435→        extra = len(duplicate_pairs) - 10
   436→        more = f" (+{extra} more)" if extra > 0 else ""
   437→        _LOG.debug(
   438→            "Collapsed %d duplicate pair(s) by median: %s%s",
   439→            len(duplicate_pairs),
   440→            shown,
   441→            more,
   442→        )
   443→    _LOG.info(
   444→        "collapse_replicates: %d row(s) -> %d row(s)", len(frame), len(collapsed)
   445→    )
   446→    return ResponseScreen(
   447→        frame=collapsed,
   448→        cell_col=screen.cell_col,
   449→        compound_col=screen.compound_col,
   450→        tissue_col=screen.tissue_col,
   451→        metrics=list(screen.metrics),
   452→    )

   455→def ln_ic50_to_pic50(ln_ic50):
   456→    """Convert GDSC ``LN_IC50`` values to pIC50.
   457→
   458→    GDSC reports the natural log of the IC50 in micromolar, so
   459→    ``LN_IC50 = 0`` (IC50 = 1 uM) maps to ``pIC50 = 6`` via
   460→    ``pIC50 = 6 - ln(IC50_uM) / ln(10)``.
   461→
   462→    Parameters
   463→    ----------
   464→    ln_ic50:
   465→        Scalar, list/array or :class:`pandas.Series` of ``LN_IC50``
   466→        values.  NaN passes through; a Series keeps its index.
   467→
   468→    Returns
   469→    -------
   470→    Same kind as the input (float for scalars, ndarray for arrays,
   471→    Series for Series) with the converted values.
   472→    """
   473→    if isinstance(ln_ic50, pd.Series):
   474→        numeric = pd.to_numeric(ln_ic50, errors="coerce")
   475→        return 6.0 - numeric / _LN10
   476→    values = np.asarray(ln_ic50, dtype=float)
   477→    result = 6.0 - values / _LN10
   478→    if result.ndim == 0:
   479→        return float(result)
   480→    return result
   481→
   482→
   483→def pic50_to_ln_ic50(pic50):
   484→    """Convert pIC50 values back to GDSC ``LN_IC50`` (the exact inverse).
   485→
   486→    ``ln(IC50_uM) = (6 - pIC50) * ln(10)``; a 1 uM IC50 (pIC50 6)
   487→    maps back to ``LN_IC50 = 0``.  Accepts the same input kinds as
   488→    :func:`ln_ic50_to_pic50` with NaN passthrough.
   489→    """
   490→    if isinstance(pic50, pd.Series):
   491→        numeric = pd.to_numeric(pic50, errors="coerce")
   492→        return (6.0 - numeric) * _LN10
   493→    values = np.asarray(pic50, dtype=float)
   494→    result = (6.0 - values) * _LN10
   495→    if result.ndim == 0:
   496→        return float(result)
   497→    return result

   500→def response_matrix(screen: ResponseScreen, metric: str) -> pd.DataFrame:
   501→    """Pivot one metric into a cell-line x compound sensitivity matrix.
   502→
   503→    Rows are cell lines, columns are compounds (both sorted); untested
   504→    pairs hold NaN.  Duplicated pairs are *not* silently averaged --
   505→    collapse them first with :func:`collapse_replicates`.
   506→
   507→    Parameters
   508→    ----------
   509→    screen:
   510→        The screen to pivot.
   511→    metric:
   512→        Metric column to place in the matrix.
   513→
   514→    Returns
   515→    -------
   516→    pandas.DataFrame
   517→        The pivoted matrix.
   518→
   519→    Raises
   520→    ------
   521→    DataValidationError
   522→        If ``metric`` is not one of the screen's metrics, or duplicate
   523→        (cell line, compound) pairs are present.
   524→    """
   525→    if metric not in screen.metrics:
   526→        raise DataValidationError(
   527→            f"metric {metric!r} is not part of the screen -- available "
   528→            f"metrics: {list(screen.metrics)}"
   529→        )
   530→    duplicates = screen._duplicate_pairs()
   531→    if duplicates:
   532→        shown = ", ".join(f"({cell} + {compound})" for cell, compound in duplicates[:5])
   533→        extra = len(duplicates) - 5
   534→        more = f" (+{extra} more)" if extra > 0 else ""
   535→        raise DataValidationError(
   536→            f"{len(duplicates)} duplicate (cell, compound) pair(s) prevent "
   537→            f"pivoting: {shown}{more} -- collapse_replicates() first"
   538→        )
   539→    matrix = screen.frame.pivot(
   540→        index=screen.cell_col, columns=screen.compound_col, values=metric
   541→    )
   542→    matrix.index.name = screen.cell_col
   543→    matrix.columns.name = screen.compound_col
   544→    return matrix.sort_index().sort_index(axis=1)

   547→def summarize_compounds(
   548→    screen: ResponseScreen,
   549→    metric: str,
   550→    higher_is_more_sensitive: bool = False,
   551→) -> pd.DataFrame:
   552→    """Per-compound potency statistics across the cell-line panel.
   553→
   554→    Rows are compounds (sorted) with columns ``n_cell_lines`` (number of
   555→    distinct cell lines measured, replicates counted once -- missing
   556→    metric values are the audit's business), ``median``, ``min``,
   557→    ``max``, ``std`` (sample standard deviation, ddof=1) and
   558→    ``sensitive_spread`` = ``max - min`` -- the potency range across
   559→    the panel, i.e. how selectively the compound hits the panel
   560→    regardless of direction.
   561→
   562→    Parameters
   563→    ----------
   564→    screen:
   565→        The screen to summarise.
   566→    metric:
   567→        Metric column to summarise.
   568→    higher_is_more_sensitive:
   569→        Direction of ``metric``: ``False`` (default) for LN_IC50-style
   570→        metrics where *lower* means more potent; ``True`` for
   571→        pIC50-style metrics where *higher* means more potent.  It only
   572→        affects the ``most_sensitive_cell_line`` column -- the cell
   573→        line with the strongest response under that direction.
   574→
   575→    Raises
   576→    ------
   577→    DataValidationError
   578→        If ``metric`` is not one of the screen's metrics.
   579→    """
   580→    if metric not in screen.metrics:
   581→        raise DataValidationError(
   582→            f"metric {metric!r} is not part of the screen -- available "
   583→            f"metrics: {list(screen.metrics)}"
   584→        )
   585→    grouped = screen.frame.groupby(screen.compound_col, sort=True)[metric]
   586→    summary = pd.DataFrame(
   587→        {
   588→            # distinct cell lines (replicates counted once); missing
   589→            # metric values stay the audit's job to report.
   590→            "n_cell_lines": screen.frame.groupby(screen.compound_col, sort=True)[
   591→                screen.cell_col
   592→            ].nunique(),
   593→            "median": grouped.median(),
   594→            "min": grouped.min(),
   595→            "max": grouped.max(),
   596→            "std": grouped.std(),
   597→        }
   598→    )
   599→    summary["sensitive_spread"] = summary["max"] - summary["min"]
   600→
   601→    # Direction-aware pick of the strongest-responding cell line per
   602→    # compound (NaN metric rows cannot win by construction).
   603→    lookup = screen.frame[
   604→        [screen.compound_col, screen.cell_col, metric]
   605→    ].dropna(subset=[metric])
   606→    if higher_is_more_sensitive:
   607→        positions = lookup.groupby(screen.compound_col, sort=True)[metric].idxmax()
   608→    else:
   609→        positions = lookup.groupby(screen.compound_col, sort=True)[metric].idxmin()
   610→    # ``positions`` maps compound -> winning row label; re-align the picked
   611→    # cell lines onto the compound index before assigning into the summary.
   612→    summary["most_sensitive_cell_line"] = pd.Series(
   613→        lookup.loc[positions, screen.cell_col].to_numpy(), index=positions.index
   614→    )
   615→    summary = summary[
   616→        [
   617→            "n_cell_lines",
   618→            "median",
   619→            "min",
   620→            "max",
   621→            "std",
   622→            "sensitive_spread",
   623→            "most_sensitive_cell_line",
   624→        ]
   625→    ]
   626→    summary.index.name = screen.compound_col
   627→    return summary
   628→
   629→
   630→def call_sensitive_resistant(
   631→    response_series,
   632→    sensitive_quantile: float = 0.25,
   633→    resistant_quantile: float = 0.75,
   634→    lower_is_sensitive: bool = True,
   635→):
   636→    """Threshold a response vector into sensitivity calls.
   637→
   638→    The lowest ``sensitive_quantile`` fraction of cells is called
   639→    ``"sensitive"``, the highest ``resistant_quantile`` fraction
   640→    ``"resistant"`` and everything in between ``"intermediate"``
   641→    (thresholds are inclusive, so tied values stay on the boundary
   642→    side).  NaN entries are labelled ``"not tested"``.
   643→
   644→    Parameters
   645→    ----------
   646→    response_series:
   647→        Response values, typically one compound's column of a
   648→        :func:`response_matrix` pivot.
   649→    sensitive_quantile, resistant_quantile:
   650→        Quantile cuts; must satisfy
   651→        ``0 < sensitive_quantile < resistant_quantile < 1``.
   652→    lower_is_sensitive:
   653→        Direction of the metric: ``True`` (default) for LN_IC50-style
   654→        metrics where *lower* means more sensitive; ``False`` for
   655→        pIC50-style metrics where *higher* means more sensitive (the
   656→        quantile cuts are mirrored accordingly).
   657→
   658→    Returns
   659→    -------
   660→    pandas.Series
   661→        Labels ``"sensitive"`` / ``"resistant"`` / ``"intermediate"``
   662→        / ``"not tested"``, indexed like the input.
   663→
   664→    Raises
   665→    ------
   666→    DataValidationError
   667→        If the quantile ordering is violated.
   668→    """
   669→    if not (
   670→        0.0 < sensitive_quantile < resistant_quantile < 1.0
   671→    ):
   672→        raise DataValidationError(
   673→            "quantiles must satisfy 0 < sensitive_quantile < "
   674→            f"resistant_quantile < 1, got {sensitive_quantile} and "
   675→            f"{resistant_quantile}"
   676→        )
   677→    series = (
   678→        response_series
   679→        if isinstance(response_series, pd.Series)
   680→        else pd.Series(response_series)
   681→    )
   682→    numeric = pd.to_numeric(series, errors="coerce")
   683→    labels = pd.Series("not tested", index=numeric.index, dtype=object)
   684→
   685→    valid = numeric.dropna()
   686→    if valid.empty:
   687→        _LOG.warning(
   688→            "call_sensitive_resistant: no non-NaN values -- every cell line "
   689→            "labelled 'not tested'"
   690→        )
   691→        return labels
   692→
   693→    if lower_is_sensitive:
   694→        sensitive_cut = valid.quantile(sensitive_quantile)
   695→        resistant_cut = valid.quantile(resistant_quantile)
   696→        is_sensitive = numeric <= sensitive_cut
   697→        is_resistant = numeric >= resistant_cut
   698→    else:
   699→        # High values are the sensitive ones: mirror the quantile cuts.
   700→        sensitive_cut = valid.quantile(1.0 - sensitive_quantile)
   701→        resistant_cut = valid.quantile(1.0 - resistant_quantile)
   702→        is_sensitive = numeric >= sensitive_cut
   703→        is_resistant = numeric <= resistant_cut
   704→
   705→    labels[is_sensitive] = "sensitive"
   706→    labels[is_resistant & ~is_sensitive] = "resistant"
   707→    intermediate = numeric.notna() & ~is_sensitive & ~is_resistant
   708→    labels[intermediate] = "intermediate"
   709→    _LOG.debug(
   710→        "Sensitivity calls: %d sensitive, %d resistant, %d intermediate, "
   711→        "%d not tested",
   712→        int((labels == "sensitive").sum()),
   713→        int((labels == "resistant").sum()),
   714→        int((labels == "intermediate").sum()),
   715→        int((labels == "not tested").sum()),
   716→    )
   717→    return labels
