    41→@contextmanager
    42→def _pharmaml_logs_visible():
    43→    """Re-enable propagation on the ``pharmaml`` logger for caplog.
    44→
    45→    Same dance as ``tests/test_ml_pipeline.py`` /
    46→    ``tests/test_docking.py``: CLI tests set ``propagate=False`` on the
    47→    ``pharmaml`` root logger, and pytest's caplog handler lives on the
    48→    true root logger.
    49→    """
    50→    pharma_root = logging.getLogger("pharmaml")
    51→    prev_propagate = pharma_root.propagate
    52→    pharma_root.propagate = True
    53→    try:
    54→        yield
    55→    finally:
    56→        pharma_root.propagate = prev_propagate

    59→def _screen_frame() -> pd.DataFrame:
    60→    """Clean 4x3 screen (4 cell lines, 3 compounds, no issues)."""
    61→    rows = []
    62→    for cell in ("A549", "HCT116", "MCF7", "HL60"):
    63→        for compound in ("cisplatin", "doxorubicin", "erlotinib"):
    64→            rows.append(
    65→                {
    66→                    "cell_line": cell,
    67→                    "compound": compound,
    68→                    "tissue": "pancreas" if cell == "A549" else "colon",
    69→                    "LN_IC50": 1.0 + 0.5 * len(cell),
    70→                    "AUC": 0.5,
    71→                }
    72→            )
    73→    return pd.DataFrame(rows)

   234→class TestLnIc50Conversion:
   235→    def test_known_values(self) -> None:
   236→        assert ln_ic50_to_pic50(0.0) == pytest.approx(6.0)  # IC50 = 1 uM
   237→        assert ln_ic50_to_pic50(math.log(2.0)) == pytest.approx(
   238→            6.0 - math.log10(2.0)
   239→        )  # IC50 = 2 uM
   240→        assert ln_ic50_to_pic50(math.log(0.001)) == pytest.approx(9.0)  # 1 nM
   241→
   242→    def test_round_trip(self) -> None:
   243→        values = np.linspace(-8.0, 5.0, 27)
   244→        round_trip = pic50_to_ln_ic50(ln_ic50_to_pic50(values))
   245→        np.testing.assert_allclose(round_trip, values)
   246→
   247→    def test_series_in_series_out_with_nan_passthrough(self) -> None:
   248→        series = pd.Series([0.0, math.log(2.0), np.nan], index=["a", "b", "c"])
   249→        result = ln_ic50_to_pic50(series)
   250→        assert isinstance(result, pd.Series)
   251→        assert list(result.index) == ["a", "b", "c"]
   252→        assert result["a"] == pytest.approx(6.0)
   253→        assert result["b"] == pytest.approx(6.0 - math.log10(2.0))
   254→        assert pd.isna(result["c"])
   255→        # and the inverse keeps its index too
   256→        back = pic50_to_ln_ic50(result)
   257→        assert isinstance(back, pd.Series)
   258→        np.testing.assert_allclose(back[:2], series[:2])
   259→
   260→    def test_scalar_and_array_kinds(self) -> None:

   490→class TestCorrelateWithResponse:
   491→    @staticmethod
   492→    def _panel(n: int = 12) -> tuple[pd.DataFrame, pd.Series]:
   493→        rng = np.random.default_rng(9)
   494→        response = pd.Series(rng.normal(size=n), index=[f"c{i}" for i in range(n)])
   495→        frame = pd.DataFrame(
   496→            rng.normal(50, 10, size=(n, 3)),
   497→            index=response.index,
   498→            columns=["planted", "random", "constant"],
   499→        )
   500→        frame["planted"] = 10.0 + 3.0 * response  # noise-free association
   501→        frame["constant"] = 7.0
   502→        return frame, response
   503→
   504→    def test_planted_association_is_perfect(self) -> None:
   505→        frame, response = self._panel()
   506→        result = correlate_with_response(frame, response)
   507→        planted = result[result["gene"] == "planted"].iloc[0]
   508→        assert abs(planted["rho"]) == pytest.approx(1.0, abs=1e-12)
   509→        assert planted["p_value"] < 1e-8
   510→        assert list(result.columns) == ["gene", "rho", "p_value"]
   511→        assert set(result["gene"]) == {"planted", "random", "constant"}
   512→
   513→    def test_unassociated_gene_is_weak(self) -> None:
   514→        frame, response = self._panel()
   515→        result = correlate_with_response(frame, response)
   516→        random_row = result[result["gene"] == "random"].iloc[0]
   517→        assert abs(random_row["rho"]) < 0.7  # pinned with the fixed seed
   518→        assert random_row["p_value"] > 0.05

   556→        rng = np.random.default_rng(11)
   557→        response = pd.Series(rng.normal(size=n), index=[f"c{i}" for i in range(n)])
   558→        genes = ["BRCA1", "EGFR", "KRAS", "TP53", "MYC", "ALK", "PTEN", "RB1"]

   561→            index=response.index,
   562→            columns=genes,
   563→        )
   564→        frame["BRCA1"] = -2.0 * response + rng.normal(scale=0.05, size=n)
   565→        # one gene with only 2 joint values -> NaN p must survive the FDR step
   566→        frame["SHORTY"] = np.nan
   567→        frame.loc[frame.index[:2], "SHORTY"] = 1.0
   568→        return sensitivity_associations(frame, response), frame, response
