    63→const CLS_METRICS: Array<{ key: string; label: string }> = [
    64→  { key: "accuracy", label: "Accuracy" },
    65→  { key: "precision", label: "Precision" },
    66→  { key: "recall", label: "Recall" },
    67→  { key: "f1", label: "F1" },

    70→function MetricCard({ label, value }: { label: string; value: number }) {
    71→  return (
    72→    <div className="rounded-lg border bg-card px-3 py-2.5 shadow-sm">
    73→      <p className="text-xs text-muted-foreground">{label}</p>
    74→      <p className="font-mono text-xl font-semibold tabular-nums">{fmtNum(value, 4)}</p>
    75→    </div>
    76→  );
    77→}

    79→/** Factory for tiny boundary dot renderers (60×60 grid → 3600 points). */
    80→function TinyDot(color: string) {
    81→  return function renderTinyDot(props: { cx?: number; cy?: number }) {
    82→    const { cx, cy } = props;
    83→    if (cx === undefined || cy === undefined) return null;
    84→    return <circle cx={cx} cy={cy} r={1.5} fill={color} fillOpacity={0.3} />;
    85→  };
    86→}
    87→
    88→/** Factory for the larger moon-class dots. */
    89→function MoonDot(color: string) {
    90→  return function renderMoonDot(props: { cx?: number; cy?: number }) {
    91→    const { cx, cy } = props;
    92→    if (cx === undefined || cy === undefined) return null;
    93→    return <circle cx={cx} cy={cy} r={3.2} fill={color} fillOpacity={0.9} stroke={color} strokeWidth={0.6} />;
    94→  };
    95→}
    96→
    97→/** Qualitative label for a Pearson correlation coefficient. */
    98→function corrStrength(r: number): string {
    99→  const a = Math.abs(r);
   100→  if (a >= 0.9) return "very strong";
   101→  if (a >= 0.7) return "strong";
   102→  if (a >= 0.5) return "moderate";
   103→  return "weak";
   104→}
   105→
   106→export function MlPlayground() {
   107→  const [task, setTask] = useState<Task>("regression");
   108→  const [samples, setSamples] = useState(300);
   109→  const [noise, setNoise] = useState(0.4);
   110→  const [regModel, setRegModel] = useState<RegModel>("linear");
   111→  const [alpha, setAlpha] = useState("1.0");
   112→  const [l2, setL2] = useState("1.0");
   113→  const [testSizePct, setTestSizePct] = useState(20);
   114→  const [cv, setCv] = useState<CvValue>("5");
   115→  const [seed, setSeed] = useState("42");
   116→
   117→  const [status, setStatus] = useState<"idle" | "running" | "done" | "error">("idle");
   118→  const [error, setError] = useState<string | null>(null);
   119→  const [result, setResult] = useState<MLTrainResponse | null>(null);
   120→
   121→  const [preview, setPreview] = useState<DatasetPreviewResponse | null>(null);
   122→  const [previewLoading, setPreviewLoading] = useState(false);
   123→  const [previewOpen, setPreviewOpen] = useState(true);
   124→
   125→  /** Classification preview: the sampled points split by class for the chart. */
   126→  const previewClasses = useMemo(() => {
   127→    if (!preview || preview.task !== "classification") return null;
   128→    return {
   129→      cls0: preview.points.filter((p) => p.label === 0),
   130→      cls1: preview.points.filter((p) => p.label === 1),
   131→    };
   132→  }, [preview]);
   133→
   134→  const noiseMax = task === "regression" ? 2 : 1.5;
   135→  const metricSpecs = task === "regression" ? REG_METRICS : CLS_METRICS;
   136→
   137→  function switchTask(next: Task) {
   138→    setTask(next);
   139→    if (next === "classification" && noise > 1.5) setNoise(0.3);
   140→    if (next === "regression" && regModel !== "ridge" && regModel !== "linear" && regModel !== "polynomial") {
   141→      setRegModel("linear");
   142→    }
   143→    setStatus("idle");
   144→    setResult(null);
   145→    setError(null);
   146→    setPreview(null);
   147→  }
   148→
   149→  const seedValid = /^\d{1,10}$/.test(seed.trim());
   150→  const alphaValid = Number.isFinite(Number.parseFloat(alpha)) && Number.parseFloat(alpha) >= 0;
   151→  const l2Valid = Number.isFinite(Number.parseFloat(l2)) && Number.parseFloat(l2) >= 0;
   152→
   153→  async function run() {
   154→    if (!seedValid) {
   155→      setError("Seed must be a non-negative integer (0 – 2147483647).");
   156→      return;
   157→    }
   158→    if (task === "regression" && regModel === "ridge" && !alphaValid) {
   159→      setError("Ridge alpha must be a number ≥ 0.");
   160→      return;
   161→    }
   162→    if (task === "classification" && !l2Valid) {
   163→      setError("L2 strength must be a number ≥ 0.");
   164→      return;
   165→    }
   166→    setStatus("running");
   167→    setError(null);
   168→    const body: MLTrainRequest = {
   169→      task,
   170→      samples,
   171→      noise,
   172→      model: task === "regression" ? regModel : "logistic",
   173→      testSize: testSizePct / 100,
   174→      cv: (Number.parseInt(cv, 10) || 0) as 0 | 3 | 5 | 10,
   175→      seed: Number.parseInt(seed, 10) || 0,
   176→    };
   177→    if (task === "regression" && regModel === "ridge") body.alpha = Number.parseFloat(alpha);
   178→    if (task === "classification") body.l2 = Number.parseFloat(l2);
   179→
   180→    try {
   181→      const res = await fetch("/api/ml/train", {
   182→        method: "POST",
   183→        headers: { "Content-Type": "application/json" },
   184→        body: JSON.stringify(body),
   185→      });
   186→      const data: unknown = await res.json();
   187→      if (!res.ok) {
   188→        const message =
   189→          typeof data === "object" && data !== null && "error" in data
   190→            ? `${String((data as { error: unknown }).error)}${
   191→                "detail" in data && Array.isArray((data as { detail: unknown }).detail)
   192→                  ? ` ${((data as { detail: unknown[] }).detail as string[]).join("; ")}`
   193→                  : ""
   194→              }`
   195→            : `HTTP ${res.status}`;
   196→        throw new Error(message);
   197→      }
   198→      setResult(data as MLTrainResponse);
   199→      setStatus("done");
   200→    } catch (err) {
   201→      setStatus("error");
   202→      const message = err instanceof Error ? err.message : "Training request failed.";
   203→      setError(message);
   204→    }
   205→  }
   206→
   207→  /** Fetch the first rows + summary stats of the (deterministic) dataset. */
   208→  async function previewDataset() {
   209→    if (!seedValid) {
   210→      setError("Seed must be a non-negative integer (0 – 2147483647).");
   211→      return;
   212→    }
   213→    setPreviewLoading(true);
   214→    setError(null);
   215→    try {
   216→      const res = await fetch("/api/ml/dataset", {
   217→        method: "POST",
   218→        headers: { "Content-Type": "application/json" },
   219→        body: JSON.stringify({
   220→          task,
   221→          samples,
   222→          noise,
   223→          seed: Number.parseInt(seed, 10) || 0,
   224→        }),
   225→      });
   226→      const data: unknown = await res.json();
   227→      if (!res.ok) {
   228→        const message =
   229→          typeof data === "object" && data !== null && "error" in data
   230→            ? `${String((data as { error: unknown }).error)}${
   231→                "detail" in data && Array.isArray((data as { detail: unknown }).detail)
   232→                  ? ` ${((data as { detail: unknown[] }).detail as string[]).join("; ")}`
   233→                  : ""
   234→              }`
   235→            : `HTTP ${res.status}`;
   236→        throw new Error(message);
   237→      }
   238→      setPreview(data as DatasetPreviewResponse);
   239→      setPreviewOpen(true);
   240→    } catch (err) {
   241→      setError(err instanceof Error ? err.message : "Preview request failed.");
   242→    } finally {
   243→      setPreviewLoading(false);
   244→    }
   245→  }
   246→
   247→  const running = status === "running";
   248→
   249→  /* Chart series (memoized splits so recharts gets stable references). */
   250→  const clsSeries = useMemo(() => {
   251→    if (!result || result.task !== "classification") return null;
   252→    const boundary = result.series.boundary ?? [];
   253→    return {
   254→      grid0: boundary.filter((p) => p.pred === 0),
   255→      grid1: boundary.filter((p) => p.pred === 1),
   256→      moon0: result.series.test.filter((p) => p.label === 0),
   257→      moon1: result.series.test.filter((p) => p.label === 1),
   258→    };
   259→  }, [result]);
   260→
   261→  const coefficientsText = useMemo(() => {
   262→    if (!result) return null;
   263→    const { intercept, coefs } = result.coefficients;
   264→    if (result.task === "regression") {
   265→      const terms = coefs
   266→        .map((c, i) => `${fmtNum(c, 4)}·x̂${coefs.length > 1 ? `^${i + 1}` : ""}`)
   267→        .join(" + ");
   268→      return `ŷ = ${fmtNum(intercept, 4)} + ${terms}   (x̂ = standardized x)`;
   269→    }
   270→    return `σ(b₀ + w₁x̂₁ + w₂x̂₂)   b₀=${fmtNum(intercept, 4)}, w=[${coefs
   271→      .map((c) => fmtNum(c, 4))
   272→      .join(", ")}]`;
   273→  }, [result]);
   274→
   275→  return (
   276→    <div className="grid gap-4 xl:grid-cols-[minmax(300px,360px)_minmax(0,1fr)]">
   277→      {/* ------------------------------ Controls ------------------------------ */}
   278→      <Card className="h-fit overflow-hidden py-0 shadow-sm xl:sticky xl:top-24">
   279→        <CardHeader className="border-b bg-muted/30 py-3">
   280→          <div className="flex items-center gap-2.5">
   281→            <span className="flex size-9 items-center justify-center rounded-lg bg-primary/10 text-primary">
   282→              <Brain className="size-4.5" aria-hidden />
   283→            </span>
   284→            <div>
   285→              <CardTitle className="text-base">Experiment setup</CardTitle>
   286→              <CardDescription className="text-xs">
   287→                Deterministic — same seed, same result
   288→              </CardDescription>
   289→            </div>
   290→          </div>
   291→        </CardHeader>
   292→        <CardContent className="space-y-5 pb-4">
   293→          {/* Task */}
   294→          <div className="space-y-1.5">
   295→            <Label className="text-xs font-medium">Task</Label>
   296→            <div className="grid grid-cols-2 gap-1 rounded-lg border bg-muted/40 p-1" role="tablist" aria-label="ML task">
   297→              {(["regression", "classification"] as const).map((t) => (
   298→                <button
   299→                  key={t}
   300→                  type="button"
   301→                  role="tab"
   302→                  aria-selected={task === t}
   303→                  onClick={() => switchTask(t)}
   304→                  className={`h-9 rounded-md text-sm font-medium capitalize transition-colors focus-visible:outline-none focus-visible:ring-1 focus-visible:ring-ring ${
   305→                    task === t ? "bg-background shadow-sm text-foreground" : "text-muted-foreground hover:text-foreground"
   306→                  }`}
   307→                >
   308→                  {t}
   309→                </button>
   310→              ))}
   311→            </div>
   312→          </div>
   313→
   314→          {/* Samples */}
   315→          <div className="space-y-2">
   316→            <div className="flex items-center justify-between">
   317→              <Label htmlFor="ml-samples" className="text-xs font-medium">
   318→                Samples
   319→              </Label>
   320→              <span className="font-mono text-xs tabular-nums text-muted-foreground">{samples}</span>
   321→            </div>
   322→            <Slider
   323→              id="ml-samples"
   324→              min={100}
   325→              max={1000}
   326→              step={50}
   327→              value={[samples]}
   328→              onValueChange={(v) => setSamples(v[0] ?? samples)}
   329→            />
   330→          </div>
   331→
   332→          {/* Noise */}
   333→          <div className="space-y-2">
   334→            <div className="flex items-center justify-between">
   335→              <Label htmlFor="ml-noise" className="text-xs font-medium">
   336→                Noise {task === "classification" ? "(moons)" : "σ"}
   337→              </Label>
   338→              <span className="font-mono text-xs tabular-nums text-muted-foreground">
   339→                {noise.toFixed(2)}
   340→              </span>
   341→            </div>
   342→            <Slider
   343→              id="ml-noise"
   344→              min={0}
   345→              max={noiseMax}
   346→              step={0.05}
   347→              value={[Math.min(noise, noiseMax)]}
   348→              onValueChange={(v) => setNoise(v[0] ?? noise)}
   349→            />
   350→          </div>
   351→
   352→          {/* Model */}
   353→          <div className="space-y-1.5">
   354→            <Label className="text-xs font-medium">Model</Label>
   355→            {task === "regression" ? (
   356→              <Select value={regModel} onValueChange={(v) => setRegModel(v as RegModel)}>
   357→                <SelectTrigger className="h-10" aria-label="Regression model">
   358→                  <SelectValue />
   359→                </SelectTrigger>
   360→                <SelectContent>
   361→                  <SelectItem value="linear">Linear regression</SelectItem>
   362→                  <SelectItem value="polynomial">Polynomial (degree 2)</SelectItem>
   363→                  <SelectItem value="ridge">Ridge (α regularized)</SelectItem>
   364→                </SelectContent>
   365→              </Select>
   366→            ) : (
   367→              <div className="space-y-2">
   368→                <div className="flex h-10 items-center rounded-md border bg-muted/30 px-3 text-sm text-muted-foreground">
   369→                  Logistic regression (batch GD, 800 epochs)
   370→                </div>
   371→                <div className="space-y-1.5">
   372→                  <Label htmlFor="ml-l2" className="text-xs font-medium">
   373→                    L2 strength λ
   374→                  </Label>
   375→                  <Input
   376→                    id="ml-l2"
   377→                    inputMode="decimal"
   378→                    value={l2}
   379→                    onChange={(e) => setL2(e.target.value)}
   380→                    aria-invalid={l2Valid ? undefined : true}
   381→                    className={`h-10 font-mono text-sm tabular-nums ${l2Valid ? "" : "border-destructive/60"}`}
   382→                  />
   383→                  {!l2Valid ? (
   384→                    <p className="text-[11px] text-destructive">Must be a number ≥ 0.</p>
   385→                  ) : null}
   386→                </div>
   387→              </div>
   388→            )}
   389→          </div>
   390→
   391→          {task === "regression" && regModel === "ridge" ? (
   392→            <div className="space-y-1.5">
   393→              <Label htmlFor="ml-alpha" className="text-xs font-medium">
   394→                Ridge α
   395→              </Label>
   396→              <Input
   397→                id="ml-alpha"
   398→                inputMode="decimal"
   399→                value={alpha}
   400→                onChange={(e) => setAlpha(e.target.value)}
   401→                aria-invalid={alphaValid ? undefined : true}
   402→                className={`h-10 font-mono text-sm tabular-nums ${alphaValid ? "" : "border-destructive/60"}`}
   403→              />
   404→              {!alphaValid ? <p className="text-[11px] text-destructive">Must be a number ≥ 0.</p> : null}
   405→            </div>
   406→          ) : null}
   407→
   408→          {/* Test size */}
   409→          <div className="space-y-2">
   410→            <div className="flex items-center justify-between">
   411→              <Label htmlFor="ml-test" className="text-xs font-medium">
   412→                Test split
   413→              </Label>
   414→              <span className="font-mono text-xs tabular-nums text-muted-foreground">
   415→                {testSizePct}%
   416→              </span>
   417→            </div>
   418→            <Slider
   419→              id="ml-test"
   420→              min={10}
   421→              max={50}
   422→              step={5}
   423→              value={[testSizePct]}
   424→              onValueChange={(v) => setTestSizePct(v[0] ?? testSizePct)}
   425→            />
   426→          </div>
   427→
   428→          {/* CV + seed */}
   429→          <div className="grid grid-cols-2 gap-3">
   430→            <div className="space-y-1.5">
   431→              <Label className="text-xs font-medium">k-fold CV</Label>
   432→              <Select value={cv} onValueChange={(v) => setCv(v as CvValue)}>
   433→                <SelectTrigger className="h-10" aria-label="Cross-validation folds">
   434→                  <SelectValue />
   435→                </SelectTrigger>
   436→                <SelectContent>
   437→                  <SelectItem value="0">Off</SelectItem>
   438→                  <SelectItem value="3">3-fold</SelectItem>
   439→                  <SelectItem value="5">5-fold</SelectItem>
   440→                  <SelectItem value="10">10-fold</SelectItem>
   441→                </SelectContent>
   442→              </Select>
   443→            </div>
   444→            <div className="space-y-1.5">
   445→              <Label htmlFor="ml-seed" className="text-xs font-medium">
   446→                Seed
   447→              </Label>
   448→              <Input
   449→                id="ml-seed"
   450→                inputMode="numeric"
   451→                value={seed}
   452→                onChange={(e) => setSeed(e.target.value)}
   453→                aria-invalid={seedValid ? undefined : true}
   454→                className={`h-10 font-mono text-sm tabular-nums ${seedValid ? "" : "border-destructive/60"}`}
   455→              />
   456→              {!seedValid ? (
   457→                <p className="text-[11px] text-destructive">Non-negative integer.</p>
   458→              ) : null}
   459→            </div>
   460→          </div>
   461→
   462→          <div className="flex gap-2">
   463→            <Button
   464→              size="lg"
   465→              className="h-11 flex-1 gap-2"
   466→              onClick={() => void run()}
   467→              disabled={running}
   468→            >
   469→              {running ? (
   470→                <>
   471→                  <Loader2 className="size-4 animate-spin" aria-hidden />
   472→                  Training…
   473→                </>
   474→              ) : (
   475→                <>
   476→                  <Play className="size-4" aria-hidden />
   477→                  Train model
   478→                </>
   479→              )}
   480→            </Button>
   481→            <Button
   482→              size="lg"
   483→              variant="outline"
   484→              className="h-11 shrink-0 gap-2"
   485→              onClick={() => void previewDataset()}
   486→              disabled={previewLoading}
   487→              aria-label="Preview the dataset before training"
   488→            >
   489→              {previewLoading ? (
   490→                <Loader2 className="size-4 animate-spin" aria-hidden />
   491→              ) : (
   492→                <Table2 className="size-4" aria-hidden />
   493→              )}
   494→              <span className="hidden sm:inline">Preview data</span>
   495→              <span className="sm:hidden">Data</span>
   496→            </Button>
   497→          </div>
   498→
   499→          {error ? (
   500→            <Alert variant="destructive">
   501→              <TriangleAlert className="size-4" aria-hidden />
   502→              <AlertTitle>Request failed</AlertTitle>
   503→              <AlertDescription>{error}</AlertDescription>
   504→            </Alert>
   505→          ) : null}
   506→        </CardContent>
   507→      </Card>
   508→
   509→      {/* ------------------------------ Results ------------------------------ */}
   510→      <div className="min-w-0 space-y-4">
   511→        {preview ? (
   512→          <Card className="overflow-hidden py-0 shadow-sm">
   513→            <CardHeader className="border-b bg-muted/30 py-3">
   514→              <div className="flex items-center justify-between gap-2">
   515→                <div className="min-w-0">
   516→                  <CardTitle className="text-sm">Dataset preview</CardTitle>
   517→                  <CardDescription className="text-xs">
   518→                    First {preview.rows.length} of {preview.nTotal} rows · seed {preview.seed} ·
   519→                    noise σ {preview.noise.toFixed(2)} — same generator the trainer uses
   520→                  </CardDescription>
   521→                </div>
   522→                <Button
   523→                  variant="ghost"
   524→                  size="sm"
   525→                  className="h-8 w-8 shrink-0 p-0"
   526→                  onClick={() => setPreviewOpen((v) => !v)}
   527→                  aria-expanded={previewOpen}
   528→                  aria-label={previewOpen ? "Collapse dataset preview" : "Expand dataset preview"}
   529→                >
   530→                  <ChevronDown
   531→                    className={cn(
   532→                      "size-4 transition-transform duration-200",
   533→                      previewOpen ? "" : "-rotate-90"
   534→                    )}
   535→                    aria-hidden
   536→                  />
   537→                </Button>
   538→              </div>
   539→            </CardHeader>
   540→            {previewOpen ? (
   541→              <CardContent className="space-y-3 p-3 sm:p-4">
   542→                <div className="grid gap-3 lg:grid-cols-2">
   543→                  <div className="thin-scrollbar max-h-72 overflow-auto rounded-md border">
   544→                    <Table className="data-table">
   545→                      <TableHeader>
   546→                        <TableRow className="bg-muted/40 hover:bg-muted/40">
   547→                          {preview.columns.map((c) => (
   548→                            <TableHead key={c} className="h-8 font-mono text-[11px]">
   549→                              {c}
