     1→"""Bioactivity endpoint and file-format conversions.
     2→
     3→Upgrades the original flat scripts ``Convert IC50(nM) to pIC50.py``,
     4→``Split CSV file.py`` and friends into typed, reusable functions that:
     5→
     6→* handle censored values (``<`` / ``>`` qualifiers) explicitly instead
     7→  of silently stripping them,
     8→* never mutate the caller's DataFrame in place,
     9→* validate units and ranges with precise error messages.
    10→
    11→The molecular file-format converters pick up the remaining scripts of
    12→the same family: ``Convert SMI to CSV.py`` (memory-flat streaming, BOM
    13→CSV output), ``Convert CSV to SDF.py`` (ETKDG-embedded ligand libraries
    14→with per-row failure accounting instead of a bare ``print``) and the
    15→``Sdf reading.ipynb`` notebook (SDF -> table round-trip without
    16→notebook-only ``PandasTools`` helpers).  RDKit stays optional: only the
    17→three molecule-aware functions import it, lazily, with an actionable
    18→:class:`~pharmaml.exceptions.DependencyError` when it is missing.
    19→"""

    52→@dataclass(frozen=True, slots=True)
    53→class EndpointValue:
    54→    """A parsed bioactivity measurement, censored or exact.
    55→
    56→    Attributes
    57→    ----------
    58→    value:
    59→        Numeric magnitude in the input unit (e.g. nM).
    60→    qualifier:
    61→        ``''`` for exact measurements, ``'<'`` or ``'>'`` for censored
    62→        ones (reporter-style entries such as ``<0.5`` or ``>10000``).
    63→    """
    64→
    65→    value: float
    66→    qualifier: str = ""
    67→
    68→    @property
    69→    def is_censored(self) -> bool:
    70→        return self.qualifier != ""
    71→
    72→
    73→def parse_endpoint(raw: object) -> EndpointValue | None:
    74→    """Parse one endpoint cell into an :class:`EndpointValue`.
    75→
    76→    Accepts numbers or strings, tolerating ``'<'``/``'>'`` prefixes,
    77→    thousands separators, surrounding units (``nM``, ``uM``) and stray
    78→    whitespace.  Returns ``None`` for genuinely unusable entries
    79→    (blank, text, or non-positive magnitudes).
    80→    """
    81→    if raw is None or (isinstance(raw, float) and math.isnan(raw)):
    82→        return None
    83→    text = str(raw).strip()
    84→    if not text or text.lower() in {"nan", "none", "null", "-"}:
    85→        return None
    86→
    87→    qualifier = ""
    88→    if text[0] in "<>":
    89→        qualifier = text[0]
    90→        text = text[1:].strip()
    91→
    92→    # Strip a trailing unit token such as 'nM' / 'uM' / 'M'.
    93→    for unit in ("nM", "uM", "µM", "mM", "M"):
    94→        if text.endswith(unit):
    95→            text = text[: -len(unit)].strip()
    96→            break
    97→    text = text.replace(",", "")
    98→
    99→    try:
   100→        value = float(text)
   101→    except ValueError:
   102→        return None
   103→    if not math.isfinite(value) or value <= 0:
   104→        return None
   105→    return EndpointValue(value=value, qualifier=qualifier)
   106→

   108→def ic50_to_pic50(ic50: object, unit: str = "nM") -> float | None:
   109→    """Convert an IC50 measurement to pIC50 = -log10(IC50 [M]).
   110→
   111→    Parameters
   112→    ----------
   113→    ic50:
   114→        Numeric value or string (censored entries like ``"<100"`` are
   115→        accepted; the qualifier is stripped and only the magnitude is
   116→        converted).
   117→    unit:
   118→        Input concentration unit: one of ``'M', 'mM', 'uM', 'nM', 'pM'``.
   119→
   120→    Returns
   121→    -------
   122→    float or None
   123→        pIC50 value, or ``None`` when the input cannot be parsed.
   124→
   125→    Examples
   126→    --------
   127→    >>> ic50_to_pic50(1000)            # 1000 nM == 1 uM
   128→    6.0
   129→    >>> ic50_to_pic50("<10")           # censored: 10 nM
   130→    8.0
   131→    """
   132→    scale = {"M": 1.0, "mM": 1e-3, "uM": 1e-6, "µM": 1e-6, "nM": 1e-9, "pM": 1e-12}
   133→    unit_key = unit.strip()
   134→    if unit_key not in scale:
   135→        raise DataValidationError(
   136→            f"Unknown concentration unit {unit!r}; expected one of {sorted(scale)}"
   137→        )
   138→    parsed = parse_endpoint(ic50)
   139→    if parsed is None:
   140→        return None
   141→    molar = parsed.value * scale[unit_key]
   142→    return -math.log10(molar)
   143→
   144→
   145→def ki_to_pki(ki: object, unit: str = "nM") -> float | None:
   146→    """Convert an inhibition constant to pKi = -log10(Ki [M])."""
   147→    return ic50_to_pic50(ki, unit=unit)  # same math, different label

   150→def pic50_from_ic50_series(
   151→    series: pd.Series, unit: str = "nM"
   152→) -> pd.Series:
   153→    """Vectorised IC50 -> pIC50 conversion preserving NaN for bad rows."""
   154→    return series.map(lambda value: ic50_to_pic50(value, unit=unit))

   157→def add_pic50_column(
   158→    frame: pd.DataFrame,
   159→    source_col: str = "IC50 (nM)",
   160→    out_col: str = "pIC50",
   161→    unit: str = "nM",
   162→    drop_invalid: bool = True,
   163→) -> pd.DataFrame:
   164→    """Append a pIC50 column to a table of measurements.
   165→
   166→    This is the upgraded equivalent of ``Convert IC50(nM) to pIC50.py``:
   167→    the original mutated a global ``df`` read from a hardcoded Excel
   168→    path and wrote to another hardcoded path; this pure function
   169→    validates the column, converts, and returns a new frame.
   170→
   171→    Parameters
   172→    ----------
   173→    frame:
   174→        Table containing the raw activity column.
   175→    source_col:
   176→        Column holding IC50 values (numeric or censored strings).
   177→    out_col:
   178→        Name of the appended column.
   179→    unit:
   180→        Unit of the source values.
   181→    drop_invalid:
   182→        Drop rows whose activity could not be parsed.
   183→
   184→    Returns
   185→    -------
   186→    pandas.DataFrame
   187→        A copy with the new column; rows with unparseable activity are
   188→        removed when ``drop_invalid`` is True, otherwise they receive
   189→        ``NaN``.
   190→    """
   191→    require_columns(frame, [source_col])
   192→    result = frame.copy()
   193→    result[out_col] = pic50_from_ic50_series(result[source_col], unit=unit)
   194→    n_invalid = int(result[out_col].isna().sum())
   195→    if n_invalid:
   196→        _LOG.warning("%d rows had unparseable %s values", n_invalid, source_col)
   197→    if drop_invalid:
   198→        result = result[result[out_col].notna()].reset_index(drop=True)
   199→    return result

   202→def split_table_into_chunks(
   203→    input_path: str | Path,
   204→    output_dir: str | Path,
   205→    chunk_rows: int = 1_500_000,
   206→    prefix: str | None = None,
   207→) -> list[Path]:
   208→    """Stream a large CSV into fixed-size part files.
   209→
   210→    Upgrades ``Split CSV file.py``: instead of writing parts next to a
   211→    hardcoded input, chunks land in an explicit output directory with a
   212→    generated manifest-friendly naming scheme, and memory stays flat
   213→    because the file is read lazily.
   214→
   215→    Parameters
   216→    ----------
   217→    input_path:
   218→        CSV (or Excel) file to split.
   219→    output_dir:
   220→        Destination directory (created if missing).
   221→    chunk_rows:
   222→        Maximum rows per part.
   223→    prefix:
   224→        Base name for parts; defaults to the input stem.
   225→
   226→    Returns
   227→    -------
   228→    list of Path
   229→        Paths of the written part files, in order.
   230→    """
   231→    if chunk_rows < 1:
   232→        raise DataValidationError(f"chunk_rows must be >= 1, got {chunk_rows}")
   233→    input_file = Path(input_path)
   234→    out_dir = Path(output_dir)
   235→    out_dir.mkdir(parents=True, exist_ok=True)
   236→    base = prefix or input_file.stem
   237→
   238→    written: list[Path] = []
   239→    if input_file.suffix.lower() == ".csv":
   240→        iterator: Iterable[pd.DataFrame] = pd.read_csv(
   241→            input_file, chunksize=chunk_rows
   242→        )
   243→    else:
   244→        frame = read_table(input_file)
   245→        iterator = (
   246→            frame.iloc[start : start + chunk_rows]
   247→            for start in range(0, len(frame), chunk_rows)
   248→        )
   249→
   250→    for index, chunk in enumerate(iterator, start=1):
   251→        part_path = out_dir / f"{base}_part{index:03d}.csv"
   252→        chunk.to_csv(part_path, index=False)
   253→        written.append(part_path)
   254→        _LOG.info("Wrote %s (%d rows)", part_path, len(chunk))
   255→    return written
   256→

   263→def _require_rdkit():
   264→    """Import the RDKit ``Chem`` module or raise an actionable error.
   265→
   266→    Mirrors :func:`pharmaml.chem.descriptors.require_rdkit` but with a
   267→    message scoped to file-format conversion.
   268→    """
   269→    try:
   270→        from rdkit import Chem
   271→        from rdkit.Chem import AllChem
   272→    except ImportError as exc:
   273→        raise DependencyError(
   274→            "RDKit is required for molecular file-format conversion. "
   275→            "Install it with:  pip install 'pharmaml[chem]'  "
   276→            "(or pip install rdkit)"
   277→        ) from exc
   278→    return Chem, AllChem

   281→def _iter_smi_lines(smi_path: Path) -> Iterable[tuple[str, str]]:
   282→    """Stream ``(smiles, identifier)`` pairs from a ``.smi`` file.
   283→
   284→    Blank lines are skipped and the identifier is the whitespace-joined
   285→    remainder of the line (possibly empty) -- exactly the parsing of
   286→    ``Convert SMI to CSV.py``, minus its in-memory list.
   287→    """
   288→    with open(smi_path, "r", encoding="utf-8", errors="ignore") as handle:
   289→        for line in handle:
   290→            parts = line.strip().split()
   291→            if not parts:
   292→                continue
   293→            yield parts[0], " ".join(parts[1:])
   294→

   296→def smi_to_csv(smi_path: str | Path, out_csv: str | Path | None = None) -> Path:
   297→    """Stream a ``.smi`` file to CSV one line at a time.
   298→
   299→    Upgrades ``Convert SMI to CSV.py``: rows are written as they are
   300→    read (no full in-memory list, so million-ligand files stay flat on
   301→    memory), multi-word identifiers survive as a single ID cell, blank
   302→    lines are skipped, and the output carries a ``utf-8-sig`` BOM so
   303→    Excel opens it without mangling the header.
   304→
   305→    Parameters
   306→    ----------
   307→    smi_path:
   308→        Input ``.smi`` file -- one ``<SMILES> <ID>`` pair per line.
   309→    out_csv:
   310→        Output CSV path; defaults to ``<stem>.csv`` next to the input.
   311→
   312→    Returns
   313→    -------
   314→    Path
   315→        The written CSV file (columns ``SMILES`` and ``ID``).
   316→
   317→    Raises
   318→    ------
   319→    DataValidationError
   320→        If the input file does not exist.
   321→    """
   322→    src = Path(smi_path)
   323→    if not src.is_file():
   324→        raise DataValidationError(f"SMILES file not found: {src}")
   325→    out = Path(out_csv) if out_csv is not None else src.with_suffix(".csv")
   326→    out.parent.mkdir(parents=True, exist_ok=True)
   327→
   328→    n_rows = 0
   329→    with open(out, "w", encoding="utf-8-sig", newline="") as handle:
   330→        writer = csv.writer(handle)
   331→        writer.writerow(["SMILES", "ID"])
   332→        for smiles, identifier in _iter_smi_lines(src):
   333→            writer.writerow([smiles, identifier])
   334→            n_rows += 1
   335→    _LOG.info("Converted %d SMILES lines: %s -> %s", n_rows, src, out)
   336→    return out

   339→def _cell_text(raw: object) -> str | None:
   340→    """Stringify a name/tag cell; NaN/blank cells return ``None``.
   341→
   342→    Numeric identifiers lose their trailing ``.0`` (``1.0`` -> ``"1"``),
   343→    mirroring :func:`pharmaml.docking.ligands._row_name`.
   344→    """
   345→    if raw is None or (isinstance(raw, float) and pd.isna(raw)):
   346→        return None
   347→    text = str(raw).strip()
   348→    if not text or text.lower() == "nan":
   349→        return None
   350→    if text.endswith(".0"):
   351→        try:
   352→            return str(int(float(text)))
   353→        except ValueError:
   354→            return text
   355→    return text

   358→def _load_conversion_table(data: pd.DataFrame | str | Path) -> tuple[pd.DataFrame, Path]:
   359→    """Accept a DataFrame or a CSV/TSV/Excel path via :func:`read_table`."""
   360→    if isinstance(data, pd.DataFrame):
   361→        return data, Path("<dataframe>")
   362→    if not isinstance(data, (str, Path)):
   363→        raise DataValidationError(
   364→            f"data must be a DataFrame or a table path, got {type(data).__name__}"
   365→        )
   366→    source = Path(data)
   367→    if not source.is_file():
   368→        raise DataValidationError(f"input table not found: {source}")
   369→    return read_table(source), source

   372→@dataclass(slots=True)
   373→class SdfWriteReport:
   374→    """Audit of one :func:`table_to_sdf` conversion.
   375→
   376→    Attributes
   377→    ----------
   378→    source:
   379→        Table file (or the ``"<dataframe>"`` marker) the molecules came
   380→        from.
   381→    output_sdf:
   382→        SDF file that was written.
   383→    molecules:
   384→        Number of molecules written.
   385→    failures:
   386→        Rows whose SMILES failed to parse (counted, not written).
   387→    smiles_column:
   388→        Column the SMILES were read from.
   389→    embedded_3d:
   390→        Molecules that received ETKDG 3D coordinates.
   391→    notes:
   392→        Non-fatal remarks: embedding fallbacks, UFF convergence.
   393→    failed_smiles:
   394→        The raw SMILES strings that failed -- the same accounting the
   395→        :class:`~pharmaml.docking.ligands.LigandLibrary` keeps for its
   396→        ``failed_smiles.csv`` exports.
   397→    """
   398→
   399→    source: Path
   400→    output_sdf: Path
   401→    molecules: int
   402→    failures: int
   403→    smiles_column: str
   404→    embedded_3d: int
   405→    notes: list[str]
   406→    failed_smiles: list[str]
   407→
   408→    def __str__(self) -> str:
   409→        lines = [
   410→            "Table -> SDF conversion",
   411→            f"  source        : {self.source}",
   412→            f"  output sdf    : {self.output_sdf}",
   413→            f"  molecules     : {self.molecules}",
   414→            f"  3D embedded   : {self.embedded_3d}",
   415→            f"  failures      : {self.failures}"
   416→            + ("  (invalid SMILES, skipped)" if self.failures else ""),
   417→            f"  smiles column : {self.smiles_column}",
   418→        ]
   419→        lines.extend(f"  note          : {note}" for note in self.notes)
   420→        return "\n".join(lines)
   421→
   422→
   423→def table_to_sdf(
   424→    data: pd.DataFrame | str | Path,
   425→    out_sdf: str | Path,
   426→    *,
   427→    smiles_col: str = "SMILES",
   428→    name_col: str | None = None,
   429→    embed3d: bool = True,
   430→    uff: bool = True,
   431→    seed: int = 0xF10E,
   432→) -> SdfWriteReport:
   433→    """Write a SMILES table to an SD file with 3D structures.
   434→
   435→    Upgrades ``Convert CSV to SDF.py``: the original hard-coded
   436→    ``yyy.csv``/``ligands.sdf``, printed ``Invalid SMILES at index i``
   437→    and silently lost the row.  Here every non-SMILES column becomes an
   438→    SD tag, invalid SMILES are counted in the returned
   439→    :class:`SdfWriteReport` (and never abort the batch), and each
   440→    molecule is AddHs -> ETKDG-embedded with a fixed ``seed`` ->
   441→    UFF-optimised.  Embedding or UFF failures are non-fatal: the
   442→    molecule is kept without coordinates and a note is recorded.
   443→
   444→    Parameters
   445→    ----------
   446→    data:
   447→        CSV/TSV/Excel path or a DataFrame (read via
   448→        :func:`pharmaml.utils.io.read_table`).
   449→    out_sdf:
   450→        Output ``.sdf`` path.
   451→    smiles_col:
   452→        Column holding the SMILES.
   453→    name_col:
   454→        Column used for molecule titles; empty/NaN cells fall back to
   455→        the 1-based row number.  ``None`` reproduces the rar's
   456→        ``ligand_{i}`` numbering.
   457→    embed3d:
   458→        Add hydrogens and embed 3D coordinates (ETKDG, fixed seed).
   459→    uff:
   460→        UFF energy minimisation after embedding (with a convergence
   461→        check that only *notes* non-convergence).
   462→    seed:
   463→        Random seed for the embedding (reproducible geometries).
   464→
   465→    Returns
   466→    -------
   467→    SdfWriteReport
   468→
   469→    Raises
   470→    ------
   471→    DataValidationError
   472→        Missing columns, an empty table, or a table where no SMILES
   473→        parses (nothing would be written).
   474→    DependencyError
   475→        If RDKit is not installed.
   476→    """
   477→    frame, source = _load_conversion_table(data)
   478→    require_columns(frame, [smiles_col])
   479→    if name_col is not None and name_col not in frame.columns:
   480→        raise DataValidationError(
   481→            f"name column {name_col!r} not found -- available columns: "
   482→            f"{list(frame.columns)}"
   483→        )
   484→    if frame.empty:
   485→        raise DataValidationError(f"table has no rows: {source}")
   486→
   487→    Chem, AllChem = _require_rdkit()
   488→# rdkit-stubs does not re-export DisableLog/EnableLog on rdkit.RDLogger,
   489→    # but rdBase (their defining module) is fully typed.
   490→    from rdkit.Chem import SDWriter
   491→    from rdkit.rdBase import DisableLog, EnableLog
   492→
   493→    embed_params = None
   494→    if embed3d:
   495→        embed_params = AllChem.ETKDG()
   496→        embed_params.randomSeed = seed
   497→
   498→    tag_columns = [col for col in frame.columns if col != smiles_col]
   499→    prepared: list = []
   500→    failures: list[str] = []
   501→    notes: list[str] = []
   502→    embedded = 0
   503→    columns = list(frame.columns)
   504→
   505→    # RDKit logs parse errors to stderr before returning None; the
   506→    # report below is friendlier than the raw parser trace.
   507→    DisableLog("rdApp.error")
   508→    try:
   509→        for i, record in enumerate(frame.itertuples(index=False, name=None), start=1):
   510→            values = dict(zip(columns, record))
   511→            raw_smiles = values[smiles_col]
   512→            # Blank cells must count as parse failures: RDKit parses "" as
   513→            # a *valid* 0-atom molecule, which then crashes the 3D embedder
   514→            # ("molecule has no atoms") or silently writes an empty record.
   515→            # Same guard family as chem/descriptors._smiles_to_molecules.
   516→            mol = (
   517→                Chem.MolFromSmiles(raw_smiles)
   518→                if isinstance(raw_smiles, str) and raw_smiles.strip()
   519→                else None
   520→            )
   521→            if mol is None:
   522→                failures.append(str(raw_smiles))
   523→                _LOG.warning("Invalid SMILES at row %d: %r", i, raw_smiles)
   524→                continue
   525→
   526→            if name_col is not None:
   527→                name = _cell_text(values.get(name_col)) or str(i)
   528→            else:
   529→                name = f"ligand_{i}"  # rar parity: ligand_1, ligand_2, ...
   530→            mol.SetProp("_Name", name)
   531→            for col in tag_columns:
   532→                cell = values[col]
   533→                if cell is None or (isinstance(cell, float) and pd.isna(cell)):
   534→                    continue  # missing tag values are simply not written
   535→                mol.SetProp(col, str(cell))
   536→
   537→            mol = Chem.AddHs(mol)
   538→            if embed_params is not None:
   539→                if AllChem.EmbedMolecule(mol, embed_params) != 0:
   540→                    notes.append(
   541→                        f"3D embedding failed for row {i} ({raw_smiles}) "
   542→                        "-- kept without coordinates"
   543→                    )
   544→                else:
   545→                    embedded += 1
   546→                    if uff and AllChem.UFFOptimizeMolecule(mol) != 0:
   547→                        notes.append(f"UFF optimisation did not converge for row {i}")
   548→            prepared.append(mol)
   549→    finally:
   550→        EnableLog("rdApp.error")
   551→
   552→    if not prepared:
   553→        raise DataValidationError(
   554→            f"no valid molecules to write -- {len(failures)} of {len(frame)} "
   555→            "SMILES failed to parse"
   556→        )
   557→
   558→    out_path = Path(out_sdf)
   559→    out_path.parent.mkdir(parents=True, exist_ok=True)
   560→    writer = SDWriter(str(out_path))
   561→    try:
   562→        for mol in prepared:
   563→            writer.write(mol)
   564→    finally:
   565→        writer.close()
   566→
   567→    report = SdfWriteReport(
   568→        source=source,
   569→        output_sdf=out_path,
   570→        molecules=len(prepared),
   571→        failures=len(failures),
   572→        smiles_column=smiles_col,
   573→        embedded_3d=embedded,
   574→        notes=notes,
   575→        failed_smiles=failures,
   576→    )
   577→    _LOG.info(
   578→        "Wrote %d molecules to %s (%d embedded, %d failures)",
   579→        len(prepared),
   580→        out_path,
   581→        embedded,
   582→        len(failures),
   583→    )
   584→    return report

   587→def sdf_to_table(
   588→    sdf_path: str | Path,
   589→    out: str | Path | None = None,
   590→    smiles_col: str = "SMILES",
   591→) -> pd.DataFrame:
   592→    """Read an SD file back into a table (SMILES, tags and titles).
   593→
   594→    Upgrades the ``Sdf reading.ipynb`` notebook: instead of
   595→    notebook-only ``PandasTools.LoadSDF`` plus ad-hoc
   596→    ``df['SMILES'] = df.ROMol.apply(Chem.MolToSmiles)``, every molecule
   597→    becomes one row with its canonical SMILES, its title in a ``name``
   598→    column and every SD property as a typed column (numbers come back
   599→    as numbers via ``GetPropsAsDict``).
   600→
   601→    Records that fail to parse are never dropped *silently*: the count
   602→    is reported with a warning (``"N of M SDF records unparseable and
   603→    dropped"``) and recorded on ``frame.attrs`` as
   604→    ``sdf_dropped_records`` / ``sdf_total_records`` so CLI callers can
   605→    exit non-zero -- the mirror of :func:`table_to_sdf`'s failure
   606→    accounting (``data convert`` exits 1 on any drop).  A file whose
   607→    records are *all* unparseable still raises.
   608→
   609→    Parameters
   610→    ----------
   611→    sdf_path:
   612→        Input ``.sdf`` file.
   613→    out:
   614→        Optional output table path (CSV/TSV/Excel via
   615→        :func:`pharmaml.utils.io.write_table`).
   616→    smiles_col:
   617→        Name of the SMILES column in the returned frame.
   618→
   619→    Returns
   620→    -------
   621→    pandas.DataFrame
   622→
   623→    Raises
   624→    ------
   625→    DataValidationError
   626→        Missing file, or an empty/unparseable SDF.
   627→    DependencyError
   628→        If RDKit is not installed.
   629→    """
   630→    src = Path(sdf_path)
   631→    if not src.is_file():
   632→        raise DataValidationError(f"SDF file not found: {src}")
   633→    Chem, _AllChem = _require_rdkit()
   634→    # rdkit-stubs does not re-export DisableLog/EnableLog on rdkit.RDLogger,
   635→    # but rdBase (their defining module) is fully typed (cf. table_to_sdf).
   636→    from rdkit.rdBase import DisableLog, EnableLog
   637→
   638→    rows: list[dict] = []
   639→    total = 0
   640→    # RDKit logs parse errors to stderr before returning None; the
   641→    # warning below is the single typed report (same guard as
   642→    # table_to_sdf).
   643→    DisableLog("rdApp.error")
   644→    try:
   645→        supplier = Chem.SDMolSupplier(str(src))
   646→        total = len(supplier)  # record count, dropped records included
   647→        for mol in supplier:
   648→            if mol is None:
   649→                continue  # unparseable record: dropped, counted below
   650→            row: dict[str, object] = dict(
   651→                mol.GetPropsAsDict(includePrivate=False, includeComputed=False)
   652→            )
   653→            row[smiles_col] = Chem.MolToSmiles(mol)
   654→            row["name"] = mol.GetProp("_Name") if mol.HasProp("_Name") else ""
   655→            rows.append(row)
   656→    except OSError as exc:
   657→        raise DataValidationError(f"empty or unreadable SDF file: {src}") from exc
   658→    finally:
   659→        EnableLog("rdApp.error")
   660→
   661→    if not rows:
   662→        raise DataValidationError(f"no parseable molecules found in {src}")
   663→
   664→    dropped = total - len(rows)
   665→    if dropped:
   666→        _LOG.warning("%d of %d SDF records unparseable and dropped", dropped, total)
   667→
   668→    frame = pd.DataFrame(rows)
   669→    ordered = [smiles_col, "name"] + [
   670→        col for col in frame.columns if col not in (smiles_col, "name")
   671→    ]
   672→    frame = frame[ordered]
   673→    # Drop accounting for downstream callers (the CLI exits 1 on any drop).
   674→    frame.attrs["sdf_dropped_records"] = dropped
   675→    frame.attrs["sdf_total_records"] = total
   676→    _LOG.info("Read %d molecules from %s", len(frame), src)
   677→    if out is not None:
   678→        written = write_table(frame, out)
   679→        _LOG.info("Wrote %d rows to %s", len(frame), written)
   680→    return frame

   683→def table_to_smi(
   684→    data: pd.DataFrame | str | Path,
   685→    out_smi: str | Path,
   686→    *,
   687→    smiles_col: str = "SMILES",
   688→    name_col: str | None = None,
   689→) -> Path:
   690→    """Write a SMILES table to a ``.smi`` list file.
   691→
   692→    The inverse of :func:`smi_to_csv`: each row becomes one
   693→    ``<smiles> <name>`` line (the name is omitted when ``name_col`` is
   694→    not given or the cell is empty), blank SMILES cells are skipped the
   695→    way blank ``.smi`` lines are, and validation mirrors
   696→    :func:`smi_to_csv` (missing file -> :class:`DataValidationError`).
   697→
   698→    Parameters
   699→    ----------
   700→    data:
   701→        CSV/TSV/Excel path or a DataFrame.
   702→    out_smi:
   703→        Output ``.smi`` path.
   704→    smiles_col:
   705→        Column holding the SMILES.
   706→    name_col:
   707→        Column written after the SMILES; ``None`` writes bare SMILES.
   708→
   709→    Returns
   710→    -------
   711→    Path
   712→        The written ``.smi`` file.
   713→
   714→    Raises
   715→    ------
   716→    DataValidationError
   717→        Missing file, missing columns, or an empty table.
   718→    """
   719→    frame, source = _load_conversion_table(data)
   720→    require_columns(frame, [smiles_col])
   721→    if name_col is not None and name_col not in frame.columns:
   722→        raise DataValidationError(
   723→            f"name column {name_col!r} not found -- available columns: "
   724→            f"{list(frame.columns)}"
   725→        )
   726→    if frame.empty:
   727→        raise DataValidationError(f"table has no rows: {source}")
   728→
   729→    out_path = Path(out_smi)
   730→    out_path.parent.mkdir(parents=True, exist_ok=True)
   731→    columns = list(frame.columns)
   732→    n_rows = 0
   733→    with open(out_path, "w", encoding="utf-8") as handle:
   734→        for record in frame.itertuples(index=False, name=None):
   735→            values = dict(zip(columns, record))
   736→            smiles = _cell_text(values[smiles_col])
   737→            if smiles is None:
   738→                continue  # blank SMILES cell: skipped like blank .smi lines
   739→            name = _cell_text(values.get(name_col)) if name_col is not None else None
   740→            handle.write(smiles + (f" {name}" if name else "") + "\n")
   741→            n_rows += 1
   742→    _LOG.info("Wrote %d SMILES lines to %s", n_rows, out_path)
   743→    return out_path
