  1081→    def _convert_affinity(self) -> None:
  1082→        if not self._require_pharmaml() or not self._require("pandas", "conversion"):
  1083→            return
  1084→        from pharmaml.chem import ic50_to_pic50, ki_to_pki
  1085→
  1086→        mode = self._aff_mode.get()
  1087→        func = ic50_to_pic50 if mode.startswith("IC50") else ki_to_pki
  1088→        try:
  1089→            value = func(self._aff_value.get().strip(), unit=self._aff_unit.get())
  1090→        except Exception as exc:  # noqa: BLE001
  1091→            self._show_error("Conversion failed", exc)
  1092→            return
  1093→        if value is None:
  1094→            self._aff_result.configure(text="n/a (unparseable entry)")
  1095→            return
  1096→        self._aff_result.configure(text=f"{value:.3f}")

  1098→    def _molar_mass(self) -> None:
  1099→        if not self._require_pharmaml():

  1110→    def _analyse_smiles(self) -> None:
  1111→        if not self._require_pharmaml() or not self._require("rdkit", "descriptors"):
  1112→            return
  1113→        smiles = self._smiles_entry.get().strip()
  1114→        if not smiles:
  1115→            messagebox.showwarning("Empty SMILES", "Type a SMILES string first.")
  1116→            return
  1117→
  1118→        def work() -> Any:
  1119→            from pharmaml.chem import compute_descriptors
  1120→
  1121→            return compute_descriptors([smiles], progress=False)
  1122→
  1123→        self._start_job(f"descriptors for {smiles[:28]}…", work,
  1124→                        self._render_descriptors)

  1126→    def _render_descriptors(self, frame: Any) -> None:
  1127→        self._desc_frame = frame
  1128→        self._desc_rows = []
  1129→        self._desc_tree.delete(*self._desc_tree.get_children())
  1130→        items = sorted(zip(frame.columns, frame.iloc[0].tolist()))
  1131→        for name, value in items:
  1132→            text = _cell_text(value, limit=40)
  1133→            item = self._desc_tree.insert("", "end", values=(name, text))
  1134→            self._desc_rows.append((item, str(name), text))
  1135→        highlights = {
  1136→            "MolWt": "MW", "MolLogP": "LogP", "TPSA": "TPSA",
  1137→            "NumHBD": "H-bond donors", "NumHBA": "H-bond acceptors",
  1138→            "NumRotatableBonds": "rotatable bonds", "RingCount": "rings",
  1139→        }
  1140→        found = [f"{short} {format_number(frame.iloc[0][key])}"
  1141→                 for key, short in highlights.items() if key in frame.columns]
  1142→        self._smiles_summary.configure(
  1143→            text=("  ·  ".join(found) if found else f"{len(items)} descriptors")
  1144→        )
  1145→        self._filter_descriptors()

  1147→    def _filter_descriptors(self, _event: Any = None) -> None:
  1148→        query = self._desc_search.get().strip().lower()
  1149→        for item, name, _value in self._desc_rows:
  1150→            if query and query not in name.lower():
  1151→                self._desc_tree.detach(item)
  1152→            else:
  1153→                self._desc_tree.move(item, "", "end")

  1155→    def _save_descriptors(self) -> None:
  1156→        if self._desc_frame is None:
  1157→            messagebox.showinfo("Nothing to save", "Analyse a SMILES string first.")
  1158→            return
  1159→        if not self._require_pharmaml():
  1160→            return
  1161→        path = filedialog.asksaveasfilename(
  1162→            title="Save descriptors", defaultextension=".csv",
  1163→            filetypes=[("CSV table", "*.csv")],
  1164→        )

  1166→            return
  1167→        try:
  1168→            from pharmaml.utils.io import write_table
  1169→
  1170→            written = write_table(self._desc_frame, path)
  1171→            messagebox.showinfo("Saved", f"Descriptor table written to\n{written}")

  1175→    def _compare_similarity(self) -> None:
  1176→        if not self._require_pharmaml() or not self._require("rdkit", "fingerprints"):
  1177→            return
  1178→        from pharmaml.chem import compute_fingerprints, tanimoto_from_bitarrays
  1179→
  1180→        first = self._tani_a.get().strip()
  1181→        second = self._tani_b.get().strip()
  1182→        if not first or not second:
  1183→            messagebox.showwarning("Missing SMILES", "Both molecules are needed.")
  1184→            return
  1185→        try:
  1186→            prints, valid = compute_fingerprints([first, second])
  1187→        except Exception as exc:  # noqa: BLE001
  1188→            self._show_error("Fingerprint failed", exc)
  1189→            return
  1190→        if len(valid) < 2:
  1191→            self._tani_result.configure(text="invalid SMILES")
  1192→            self._tani_verdict.configure(
  1193→                text="One of the two strings is not valid RDKit input."
  1194→            )
  1195→            return
  1196→        score = float(tanimoto_from_bitarrays(prints[0], prints[1]))
  1197→        if score >= 0.85:
  1198→            verdict = "near-identical fingerprints"
  1199→        elif score >= 0.6:
  1200→            verdict = "clearly related scaffolds"
  1201→        elif score >= 0.35:
  1202→            verdict = "loosely related"
  1203→        else:
  1204→            verdict = "essentially unrelated"
  1205→        self._tani_result.configure(text=f"{score:.3f}")
  1206→        self._tani_verdict.configure(text=f"{verdict} (1.0 = identical, 0.0 = nothing shared)")
