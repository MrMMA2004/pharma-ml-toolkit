    77→/** Factory for tiny boundary dot renderers (60×60 grid → 3600 points). */
    78→function TinyDot(color: string) {
    79→  return function renderTinyDot(props: { cx?: number; cy?: number }) {
    80→    const { cx, cy } = props;
    81→    if (cx === undefined || cy === undefined) return null;
    82→    return <circle cx={cx} cy={cy} r={1.5} fill={color} fillOpacity={0.3} />;
    83→  };
    84→}
    85→
    86→/** Factory for the larger moon-class dots. */
    87→function MoonDot(color: string) {
    88→  return function renderMoonDot(props: { cx?: number; cy?: number }) {
    89→    const { cx, cy } = props;
    90→    if (cx === undefined || cy === undefined) return null;
    91→    return <circle cx={cx} cy={cy} r={3.2} fill={color} fillOpacity={0.9} stroke={color} strokeWidth={0.6} />;
    92→  };
    93→}

    95→/** Qualitative label for a Pearson correlation coefficient. */
    96→function corrStrength(r: number): string {
    97→  const a = Math.abs(r);
    98→  if (a >= 0.9) return "very strong";
    99→  if (a >= 0.7) return "strong";
   100→  if (a >= 0.5) return "moderate";
   101→  return "weak";
   102→}

   104→export function MlPlayground() {
   105→  const [task, setTask] = useState<Task>("regression");
   106→  const [samples, setSamples] = useState(300);
   107→  const [noise, setNoise] = useState(0.4);
   108→  const [regModel, setRegModel] = useState<RegModel>("linear");
   109→  const [alpha, setAlpha] = useState("1.0");
   110→  const [l2, setL2] = useState("1.0");
   111→  const [testSizePct, setTestSizePct] = useState(20);
   112→  const [cv, setCv] = useState<CvValue>("5");
   113→  const [seed, setSeed] = useState("42");
   114→
   115→  const [status, setStatus] = useState<"idle" | "running" | "done" | "error">("idle");
   116→  const [error, setError] = useState<string | null>(null);
   117→  const [result, setResult] = useState<MLTrainResponse | null>(null);
   118→

   120→  const [previewLoading, setPreviewLoading] = useState(false);
   121→  const [previewOpen, setPreviewOpen] = useState(true);
   122→
   123→  const noiseMax = task === "regression" ? 2 : 1.5;
   124→  const metricSpecs = task === "regression" ? REG_METRICS : CLS_METRICS;

   126→  function switchTask(next: Task) {
   127→    setTask(next);
   128→    if (next === "classification" && noise > 1.5) setNoise(0.3);
   129→    if (next === "regression" && regModel !== "ridge" && regModel !== "linear" && regModel !== "polynomial") {
   130→      setRegModel("linear");
   131→    }
   132→    setStatus("idle");
   133→    setResult(null);
   134→    setError(null);
   135→    setPreview(null);
   136→  }
   137→
   138→  const seedValid = /^\d{1,10}$/.test(seed.trim());
   139→  const alphaValid = Number.isFinite(Number.parseFloat(alpha)) && Number.parseFloat(alpha) >= 0;
   140→  const l2Valid = Number.isFinite(Number.parseFloat(l2)) && Number.parseFloat(l2) >= 0;
   141→
   142→  async function run() {
   143→    if (!seedValid) {
   144→      setError("Seed must be a non-negative integer (0 – 2147483647).");
   145→      return;
   146→    }
   147→    if (task === "regression" && regModel === "ridge" && !alphaValid) {
   148→      setError("Ridge alpha must be a number ≥ 0.");
   149→      return;
   150→    }
   151→    if (task === "classification" && !l2Valid) {
   152→      setError("L2 strength must be a number ≥ 0.");
   153→      return;
   154→    }
   155→    setStatus("running");
   156→    setError(null);
   157→    const body: MLTrainRequest = {
   158→      task,
   159→      samples,
   160→      noise,
   161→      model: task === "regression" ? regModel : "logistic",
   162→      testSize: testSizePct / 100,
   163→      cv: (Number.parseInt(cv, 10) || 0) as 0 | 3 | 5 | 10,
   164→      seed: Number.parseInt(seed, 10) || 0,
   165→    };
   166→    if (task === "regression" && regModel === "ridge") body.alpha = Number.parseFloat(alpha);
   167→    if (task === "classification") body.l2 = Number.parseFloat(l2);
   168→
   169→    try {
   170→      const res = await fetch("/api/ml/train", {
   171→        method: "POST",
   172→        headers: { "Content-Type": "application/json" },
   173→        body: JSON.stringify(body),
   174→      });
   175→      const data: unknown = await res.json();
   176→      if (!res.ok) {
   177→        const message =
   178→          typeof data === "object" && data !== null && "error" in data
   179→            ? `${String((data as { error: unknown }).error)}${
   180→                "detail" in data && Array.isArray((data as { detail: unknown }).detail)
   181→                  ? ` ${((data as { detail: unknown[] }).detail as string[]).join("; ")}`
   182→                  : ""
   183→              }`
   184→            : `HTTP ${res.status}`;
   185→        throw new Error(message);
   186→      }
   187→      setResult(data as MLTrainResponse);
   188→      setStatus("done");
   189→    } catch (err) {
   190→      setStatus("error");
   191→      const message = err instanceof Error ? err.message : "Training request failed.";
   192→      setError(message);
   193→    }
   194→  }
   195→
   196→  /** Fetch the first rows + summary stats of the (deterministic) dataset. */

   203→    setError(null);
   204→    try {
   205→      const res = await fetch("/api/ml/dataset", {

   207→        headers: { "Content-Type": "application/json" },
   208→        body: JSON.stringify({
   209→          task,
   210→          samples,
   211→          noise,
   212→          seed: Number.parseInt(seed, 10) || 0,
   213→        }),

   218→          typeof data === "object" && data !== null && "error" in data

   220→                "detail" in data && Array.isArray((data as { detail: unknown }).detail)
   221→                  ? ` ${((data as { detail: unknown[] }).detail as string[]).join("; ")}`
   222→                  : ""
   223→              }`
   224→            : `HTTP ${res.status}`;
   225→        throw new Error(message);
   226→      }

   228→      setPreviewOpen(true);
   229→    } catch (err) {

   232→      setPreviewLoading(false);
   233→    }
   234→  }

   236→  const running = status === "running";

   264→  return (

   266→      {/* ------------------------------ Controls ------------------------------ */}

   268→        <CardHeader className="border-b bg-muted/30 py-3">
   269→          <div className="flex items-center gap-2.5">
   270→            <span className="flex size-9 items-center justify-center rounded-lg bg-primary/10 text-primary">
   271→              <Brain className="size-4.5" aria-hidden />
   272→            </span>
   273→            <div>
   274→              <CardTitle className="text-base">Experiment setup</CardTitle>
   275→              <CardDescription className="text-xs">
   276→                Deterministic — same seed, same result
   277→              </CardDescription>
   278→            </div>
   279→          </div>
   280→        </CardHeader>
   281→        <CardContent className="space-y-5 pb-4">

   294→                    task === t ? "bg-background shadow-sm text-foreground" : "text-muted-foreground hover:text-foreground"
   295→                  }`}
   296→                >
   297→                  {t}
   298→                </button>

   300→            </div>
   301→          </div>

   328→                {noise.toFixed(2)}
   329→              </span>
   330→            </div>

   336→              value={[Math.min(noise, noiseMax)]}

   338→            />
   339→          </div>
   340→
   341→          {/* Model */}

   349→                <SelectContent>
   350→                  <SelectItem value="linear">Linear regression</SelectItem>

   352→                  <SelectItem value="ridge">Ridge (α regularized)</SelectItem>
   353→                </SelectContent>
   354→              </Select>
   355→            ) : (
   356→              <div className="space-y-2">

   361→                  <Label htmlFor="ml-l2" className="text-xs font-medium">
   362→                    L2 strength λ
   363→                  </Label>
   364→                  <Input
   365→                    id="ml-l2"
   366→                    inputMode="decimal"
   367→                    value={l2}

   369→                    aria-invalid={l2Valid ? undefined : true}
   370→                    className={`h-10 font-mono text-sm tabular-nums ${l2Valid ? "" : "border-destructive/60"}`}
   371→                  />
   372→                  {!l2Valid ? (
   373→                    <p className="text-[11px] text-destructive">Must be a number ≥ 0.</p>
   374→                  ) : null}
   375→                </div>
   376→              </div>
   377→            )}
   378→          </div>
   379→
   380→          {task === "regression" && regModel === "ridge" ? (
   381→            <div className="space-y-1.5">
   382→              <Label htmlFor="ml-alpha" className="text-xs font-medium">
   383→                Ridge α
   384→              </Label>
   385→              <Input
   386→                id="ml-alpha"
   387→                inputMode="decimal"
   388→                value={alpha}
   389→                onChange={(e) => setAlpha(e.target.value)}
   390→                aria-invalid={alphaValid ? undefined : true}
   391→                className={`h-10 font-mono text-sm tabular-nums ${alphaValid ? "" : "border-destructive/60"}`}

   394→            </div>
   395→          ) : null}
   396→
   397→          {/* Test size */}
   398→          <div className="space-y-2">
   399→            <div className="flex items-center justify-between">
   400→              <Label htmlFor="ml-test" className="text-xs font-medium">

   402→              </Label>
   403→              <span className="font-mono text-xs tabular-nums text-muted-foreground">
   404→                {testSizePct}%
   405→              </span>
   406→            </div>
   407→            <Slider
   408→              id="ml-test"
   409→              min={10}
   410→              max={50}
   411→              step={5}
   412→              value={[testSizePct]}
   413→              onValueChange={(v) => setTestSizePct(v[0] ?? testSizePct)}

   416→

   424→                </SelectTrigger>

   450→
   451→          <div className="flex gap-2">

   457→            >

   459→                <>
   460→                  <Loader2 className="size-4 animate-spin" aria-hidden />

   462→                </>
   463→              ) : (
   464→                <>
   465→                  <Play className="size-4" aria-hidden />

   485→            </Button>
   486→          </div>
   487→
   488→          {error ? (

   490→              <TriangleAlert className="size-4" aria-hidden />

   492→              <AlertDescription>{error}</AlertDescription>
   493→            </Alert>
   494→          ) : null}
   495→        </CardContent>
   496→      </Card>
   497→
   498→      {/* ------------------------------ Results ------------------------------ */}
   499→      <div className="min-w-0 space-y-4">
   500→        {preview ? (

   502→            <CardHeader className="border-b bg-muted/30 py-3">

   504→                <div className="min-w-0">
   505→                  <CardTitle className="text-sm">Dataset preview</CardTitle>
   506→                  <CardDescription className="text-xs">
   507→                    First {preview.rows.length} of {preview.nTotal} rows · seed {preview.seed} ·

   518→                >
   519→                  <ChevronDown
   520→                    className={cn(
   521→                      "size-4 transition-transform duration-200",
   522→                      previewOpen ? "" : "-rotate-90"
   523→                    )}
   524→                    aria-hidden
   525→                  />
   526→                </Button>
   527→              </div>
   528→            </CardHeader>
   529→            {previewOpen ? (
   530→              <CardContent className="space-y-3 p-3 sm:p-4">
   531→                <div className="thin-scrollbar max-h-72 overflow-auto rounded-md border">

   533→                    <TableHeader>

   535→                        {preview.columns.map((c) => (
   536→                          <TableHead key={c} className="h-8 font-mono text-[11px]">

   539→                        ))}
   540→                      </TableRow>
   541→                    </TableHeader>
   542→                    <TableBody>
   543→                      {preview.rows.map((row) => (
   544→                        <TableRow key={row.i}>
   545→                          {preview.columns.map((c) => (
   546→                            <TableCell key={c} className="py-1.5 font-mono text-xs tabular-nums">
