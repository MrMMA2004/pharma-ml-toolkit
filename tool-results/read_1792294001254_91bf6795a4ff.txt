    73→/**
    74→ * Compact terminal command strip: a header row naming the command context
    75→ * ("terminal", "terminal · macOS", …) with an icon-only copy button, then the
    76→ * command itself behind a `$` prompt in monospace. Copy feedback swaps the
    77→ * icon to a check, flips aria-pressed and announces itself via a live region.
    78→ */
    79→function CommandStrip({
    80→  command,
    81→  context = "terminal",
    82→  hint,
    83→  className,
    84→}: {
    85→  command: string;
    86→  context?: string;
    87→  hint?: string;
    88→  className?: string;
    89→}) {
    90→  const [copied, setCopied] = useState(false);
    91→
    92→  async function copy() {
    93→    const ok = await copyToClipboard(command);
    94→    if (ok) {
    95→      setCopied(true);
    96→      toast.success("Command copied", { description: command });
    97→      window.setTimeout(() => setCopied(false), 1600);
    98→    } else {
    99→      toast.error("Could not copy — your browser blocked clipboard access");
   100→    }
   101→  }
   102→
   103→  return (
   104→    <div className={cn("overflow-hidden rounded-lg border bg-card text-left", className)}>
   105→      <div className="flex items-center justify-between gap-2 border-b bg-muted/40 py-1 pl-2.5 pr-1.5">
   106→        <span className="flex min-w-0 items-center gap-1.5 font-mono text-[10px] uppercase tracking-wide text-muted-foreground">

   108→          <span className="truncate">{context}</span>
   109→        </span>
   110→        <button
   111→          type="button"
   112→          onClick={() => void copy()}
   113→          aria-label={`Copy command to clipboard: ${command}`}
   114→          aria-pressed={copied}
   115→          className="flex size-6 shrink-0 items-center justify-center rounded-md text-muted-foreground transition-colors hover:bg-muted hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"
   116→        >
   117→          {copied ? (
   118→            <Check className="size-3.5 text-primary" aria-hidden />
   119→          ) : (
   120→            <Copy className="size-3.5" aria-hidden />
   121→          )}

   127→      <div className="flex items-baseline gap-2 px-2.5 py-2">

   186→function countFiles(node: TreeNode): number {
   187→  if (node.type === "file") return 1;
   188→  return (node.children ?? []).reduce((acc, child) => acc + countFiles(child), 0);
   189→}

   191→/** Look up a node in the tree by its toolkit-relative path. */
   192→function findNode(node: TreeNode, path: string): TreeNode | null {
   193→  if (node.path === path) return node;
   194→  for (const child of node.children ?? []) {
   195→    const hit = findNode(child, path);
   196→    if (hit) return hit;
   197→  }
   198→  return null;
   199→}

   260→function collectDirSummary(
   261→  root: TreeNode
   262→): Array<{ name: string; files: number; size: number; note: string }> {
   263→  const notes: Record<string, string> = {
   264→    src: "the pharmaml package (data, chem, ml, pharma, docking, viz, utils + CLI)",
   265→    tests: "1478 pytest cases across all domains",
   266→    examples: "six executed notebooks + five no-output twins + runnable demos",
   267→    ".github": "GitHub Actions CI matrix (Python 3.10–3.12)",
   268→    docs: "module index — auto-generated API reference",
   269→    scratch: "idempotent build & patch scripts (notebooks, docs index)",
   270→  };
   271→  const out: Array<{ name: string; files: number; size: number; note: string }> = [];
   272→  let rootFiles = 0;
   273→  let rootBytes = 0;
   274→  for (const child of root.children ?? []) {
   275→    if (child.type === "dir") {
   276→      out.push({
   277→        name: child.name,
   278→        files: countFiles(child),
   279→        size: child.size,
   280→        note: notes[child.name] ?? "",
   281→      });
   282→    } else {
   283→      rootFiles++;
   284→      rootBytes += child.size;
   285→    }
   286→  }
   287→  if (rootFiles > 0) {
   288→    out.push({
   289→      name: "(root)",
   290→      files: rootFiles,
   291→      size: rootBytes,
   292→      note: "README, pyproject.toml, Dockerfile, requirements.txt, .gitignore",
   293→    });
   294→  }
   295→  return out;
   296→}

   305→
   306→  const [downloading, setDownloading] = useState(false);
   307→  /** Path of the featured file currently being fetched (spinner state). */
   308→  const [fileBusy, setFileBusy] = useState<string | null>(null);
   309→
   310→  useEffect(() => {
   311→    void ensureTree();

   313→
   314→  const dirSummary = useMemo(() => (tree ? collectDirSummary(tree) : []), [tree]);
   315→
   316→  /** Total files across the top-level folders — denominator for the share bars. */
   317→  const totalFiles = useMemo(
   318→    () => dirSummary.reduce((acc, d) => acc + d.files, 0),
   319→    [dirSummary]
   320→  );
   321→
   322→  /** Live metadata (size / lines) for each featured file, from the tree. */
   323→  const featuredMeta = useMemo(() => {
   324→    const map = new Map<string, TreeNode | null>();
   325→    for (const f of FEATURED_FILES) {

   331→  async function download() {
   332→    setDownloading(true);
   333→    try {
   334→      const res = await fetch("/api/toolkit/download");
   335→      if (!res.ok) {
   336→        let message = `HTTP ${res.status}`;
   337→        try {
   338→          const data: unknown = await res.json();
   339→          if (typeof data === "object" && data !== null && "error" in data) {
   340→            message = String((data as { error: unknown }).error);
   341→          }
   342→        } catch {
   343→          /* body wasn't JSON — keep the status message */
   344→        }
   345→        throw new Error(message);
   346→      }
   347→      const blob = await res.blob();
   348→      if (blob.size < 1024) {
   349→        throw new Error("The archive came back suspiciously small — try again.");
   350→      }
   351→      const url = URL.createObjectURL(blob);
   352→      const anchor = document.createElement("a");
   353→      anchor.href = url;
   354→      anchor.download = "pharma-ml-toolkit.zip";
   355→      document.body.appendChild(anchor);
   356→      anchor.click();
   357→      document.body.removeChild(anchor);
   358→      URL.revokeObjectURL(url);
   359→      toast.success("Download started", {
   360→        description: `pharma-ml-toolkit.zip · ${fmtBytes(blob.size)}`,
   361→      });
   362→    } catch (err) {
   363→      toast.error("Download failed", {
   364→        description: err instanceof Error ? err.message : "Unknown error.",
   365→      });
   366→    } finally {
   367→      setDownloading(false);
   368→    }
   369→  }

   390→      }

   392→      const blob = new Blob([payload.content], { type: "text/plain;charset=utf-8" });
