    92→    92→def _write_if_changed(path: Path, data: bytes) -> None:
    93→    93→    """Write bytes only when they differ from what is on disk (idempotency)."""
    94→    94→    path.parent.mkdir(parents=True, exist_ok=True)

    96→    96→        STATS["unchanged"] += 1

   128→   128→def build_qsar_classification() -> None:
   129→   129→    rng = np.random.default_rng(11)
   130→   130→    n = 420
   131→   131→    signal = rng.normal(size=(n, 3))
   132→   132→    noise = rng.normal(scale=0.4, size=(n, 8))
   133→   133→    descriptors = noise.copy()
   134→   134→    descriptors[:, :3] += 0.6 * signal      # first 3 descriptors are informative
   135→   135→    frame_demo = pd.DataFrame(descriptors, columns=[f"descr_{i:02d}" for i in range(8)])
   136→   136→    score = frame_demo[["descr_00", "descr_01", "descr_02"]].sum(axis=1)
   137→   137→    frame_demo["activity"] = np.where(
   138→   138→        score > np.quantile(score, 0.72), "active",
   139→   139→        np.where(score > np.quantile(score, 0.35), "weak", "inactive"),
   140→   140→    )
   141→   141→    frame_demo["SMILES"] = rng.choice(["CCO", "c1ccccc1O", "CC(=O)Oc1ccccc1C(=O)O"], size=n)
   142→   142→    frame_demo["compound_id"] = [f"CMPD-{i:04d}" for i in range(n)]
   143→   143→    _write_frame(DATA_DIR / "qsar_classification.csv", frame_demo, index=False)

   318→   318→def build_cellline_data() -> None:
   319→   319→    # --- cell 3: the screen (one rng continues into cell 17's matrix) ---
   320→   320→    rng = np.random.default_rng(42)
   321→   321→    egfr_level = {
   322→   322→        name: rng.normal(9.0, 0.8) if tissue == "lung" else rng.normal(5.5, 0.9)
   323→   323→        for name, tissue in CELL_LINES.items()
   324→   324→    }
   325→   325→    egfr_z = (pd.Series(egfr_level) - pd.Series(egfr_level).mean()) / pd.Series(egfr_level).std()
   326→   326→
   327→   327→    rows = []
   328→   328→    for cell, tissue in CELL_LINES.items():
   329→   329→        for drug, (base, tissue_map, egfr_coef) in DRUGS.items():
   330→   330→            ln_ic50 = base + tissue_map.get(tissue, 0.0) + egfr_coef * egfr_z[cell]
   331→   331→            ln_ic50 += rng.normal(0.0, 0.18)
   332→   332→            rows.append((cell, tissue, drug, ln_ic50))
   333→   333→    screen_frame = pd.DataFrame(rows, columns=["cell_line", "tissue", "compound", "LN_IC50"])
   334→   334→
   335→   335→    # One duplicate pair (assay repeat) and one missing value, planted for §2.
   336→   336→    screen_frame = pd.concat(
   337→   337→        [screen_frame, screen_frame.iloc[[5]]], ignore_index=True
   338→   338→    )  # duplicate of row 5
   339→   339→    screen_frame.loc[100, "LN_IC50"] = np.nan
   340→   340→    _write_frame(DATA_DIR / "cellline_screen.csv", screen_frame, index=False)
   341→   341→
   342→   342→    # --- cell 17: the expression matrix (same rng object continues) ---
   343→   343→    genes = [f"GENE_{i:03d}" for i in range(1, 299)] + ["EGFR", "KRT8"]
   344→   344→    base = rng.normal(6.0, 1.0, size=(len(CELL_LINES), len(genes) - 2))
   345→   345→    expression = pd.DataFrame(base, index=list(CELL_LINES), columns=genes[:-2])
   346→   346→    expression["EGFR"] = [egfr_level[c] for c in CELL_LINES]        # planted biomarker
   347→   347→    expression["KRT8"] = [                                        # planted breast marker
   348→   348→        rng.normal(9.0, 0.7) if t == "breast" else rng.normal(5.0, 0.8)
   349→   349→        for t in CELL_LINES.values()
   350→   350→    ]
   351→   351→    expression.index.name = "cell_line"
   352→   352→    _write_frame(DATA_DIR / "cellline_expression.csv", expression, index=True)

   356→   356→# complete_pk_workflow.ipynb -- cell 4 (oral concentration-time profile)
   357→   357→# ---------------------------------------------------------------------------
   358→   358→def make_pk_profile(ka: float = 1.2, ke: float = 0.35, f: float = 0.9, dose: float = 250.0,
   359→   359→                    vd: float = 35.0, duration_h: float = 24.0, step_h: float = 0.5,
   360→   360→                    error_pct: float = 0.05, seed: int = 7) -> pd.DataFrame:
   361→   361→    """One-compartment oral profile with proportional assay noise (cell verbatim)."""
   362→   362→    rng = np.random.default_rng(seed)
   363→   363→    times = np.arange(0.0, duration_h + step_h, step_h)
   364→   364→    coefficient = f * dose * ka / (vd * (ka - ke))
   365→   365→    conc = coefficient * (np.exp(-ke * times) - np.exp(-ka * times))
   366→   366→    conc *= 1.0 + rng.normal(scale=error_pct, size=times.size)
   367→   367→    conc = np.clip(conc, 0.0, None)
   368→   368→    return pd.DataFrame({"time_h": times, "conc_ng_ml": conc})

   374→   374→    profile = make_pk_profile()
   375→   375→    _write_frame(DATA_DIR / "pk_profile.csv", profile, index=False)

   474→   474→def make_dlg(seed: int, n_clusters: int = 5, total_runs: int = 50) -> str:
   475→   475→    """Synthetic AutoDock4 DLG ending in the 'Successful Completion' sentinel."""
   476→   476→    rng = np.random.default_rng(seed)
   477→   477→    lowest = np.sort(rng.uniform(-11.8, -4.5, size=n_clusters))[::-1]
   478→   478→    sizes = rng.multinomial(total_runs, [1 / n_clusters] * n_clusters) + 1
   479→   479→    mean = lowest + rng.uniform(0.05, 0.6, size=n_clusters)   # mean sits above the lowest
   480→   480→    lines = [
   481→   481→        "AutoDock4 log file (demo)",
   482→   482→        "",
   483→   483→        "________",
   484→   484→        "CLUSTERING HISTOGRAM",
   485→   485→        "__________________",
   486→   486→        "______________________________________________________________________",
   487→   487→        "Cluster Rank | Lowest Energy | Run | Mean Energy | # in Cluster",
   488→   488→        "_____________|_______________|_____|_____________|________________",
   489→   489→    ]
   490→   490→    run = int(rng.integers(1, total_runs))
   491→   491→    for rank in range(n_clusters):
   492→   492→        lines.append(
   493→   493→            f"     {rank + 1}       |     {lowest[rank]:7.2f}    |  {run:3d} |"
   494→   494→            f"    {mean[rank]:7.2f}    |   {sizes[rank]:3d} |"
   495→   495→        )
   496→   496→    lines += [
   497→   497→        "____________|_______________|_____|_____________|________________",
   498→   498→        "",
   499→   499→        "Successful Completion",
   500→   500→    ]
   501→   501→    return "\n".join(lines) + "\n"
