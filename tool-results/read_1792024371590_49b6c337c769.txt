    94→/** Read the persisted recent-files list (client only, failure-tolerant). */
    95→function loadRecentFiles(): string[] {
    96→  if (typeof window === "undefined") return [];
    97→  try {
    98→    const raw = window.sessionStorage.getItem(RECENT_STORAGE_KEY);
    99→    if (!raw) return [];
   100→    const parsed: unknown = JSON.parse(raw);
   101→    if (!Array.isArray(parsed)) return [];
   102→    return parsed
   103→      .filter((p): p is string => typeof p === "string")
   104→      .slice(0, RECENT_LIMIT);
   105→  } catch {
   106→    return []; // private mode / quota / corrupted JSON — start fresh
   107→  }
   108→}

   116→    /* quota exceeded or private mode — persistence is optional */
   117→  }

   132→/** Validate a persisted PK regimen field-by-field, falling back to defaults. */
   133→function validatePk(raw: unknown): PkParams {
   134→  const base = DEFAULT_PK_PARAMS;
   135→  if (typeof raw !== "object" || raw === null) return base;
   136→  const p = raw as Partial<PkParams>;
   137→  const num = (v: unknown, def: number, lo: number, hi: number) =>
   138→    typeof v === "number" && Number.isFinite(v) && v >= lo && v <= hi ? v : def;
   139→  return {
   140→    route: p.route === "oral" ? "oral" : "iv",
   141→    dose: num(p.dose, base.dose, 0.1, 10_000),
   142→    vd: num(p.vd, base.vd, 0.1, 1_000),
   143→    cl: num(p.cl, base.cl, 0.01, 100),
   144→    tau: num(p.tau, base.tau, 0.25, 168),
   145→    doses: Math.round(num(p.doses, base.doses, 1, 48)),
   146→    f: num(p.f, base.f, 0.05, 1),
   147→    ka: num(p.ka, base.ka, 0.01, 12),
   148→  };
   149→}

   151→/** Validate a persisted ML config field-by-field, falling back to defaults. */
   152→function validateMl(raw: unknown): MlConfig {
   153→  const base = DEFAULT_ML_CONFIG;
   154→  if (typeof raw !== "object" || raw === null) return base;
   155→  const p = raw as Partial<MlConfig>;
   156→  const str = (v: unknown, def: string) =>
   157→    typeof v === "string" && v.length <= 12 ? v : def;
   158→  const str128 = (v: unknown, def: string) =>
   159→    typeof v === "string" && v.length <= 128 ? v : def;
   160→  const strArr = (v: unknown): string[] =>
   161→    Array.isArray(v)
   162→      ? v.filter((c): c is string => typeof c === "string" && c.length > 0 && c.length <= 128).slice(0, 8)
   163→      : [];
   164→  return {
   165→    task: p.task === "classification" ? "classification" : "regression",
   166→    samples:
   167→      typeof p.samples === "number" && Number.isFinite(p.samples) && p.samples >= 100 && p.samples <= 1000
   168→        ? Math.round(p.samples)
   169→        : base.samples,
   170→    noise:
   171→      typeof p.noise === "number" && Number.isFinite(p.noise) && p.noise >= 0 && p.noise <= 2
   172→        ? p.noise
   173→        : base.noise,
   174→    regModel:
   175→      p.regModel === "polynomial" || p.regModel === "ridge" ? p.regModel : "linear",
   176→    alpha: str(p.alpha, base.alpha),
   177→    l2: str(p.l2, base.l2),
   178→    testSizePct:
   179→      typeof p.testSizePct === "number" && Number.isFinite(p.testSizePct) && p.testSizePct >= 5 && p.testSizePct <= 50
   180→        ? p.testSizePct
   181→        : base.testSizePct,
   182→    cv: str(p.cv, base.cv),
   183→    seed: str(p.seed, base.seed),
   184→    source: p.source === "csv" ? "csv" : "synthetic",
   185→    featureColumn: str128(p.featureColumn, ""),
   186→    featureColumn2: str128(p.featureColumn2, ""),
   187→    targetColumn: str128(p.targetColumn, ""),
   188→    featureColumns: strArr(p.featureColumns),
   189→  };
   190→}

   192→/* ---- ML result persistence (last training run: metrics + chart series) ----
   193→ * Restored field-by-field with the same defensive validation as the rest of
   194→ * the persisted state; anything malformed falls back to "no result". */
   195→
   196→function finiteArray(raw: unknown, cap: number, validate: (v: unknown) => boolean): unknown[] | null {
   197→  if (!Array.isArray(raw) || raw.length === 0 || raw.length > cap) return null;
   198→  return raw.every(validate) ? raw : null;
   199→}
   200→
   201→const isFiniteNumber = (v: unknown): v is number =>
   202→  typeof v === "number" && Number.isFinite(v);
   203→
   204→const isPoint = (v: unknown): boolean => {
   205→  if (typeof v !== "object" || v === null) return false;
   206→  const p = v as Record<string, unknown>;
   207→  return isFiniteNumber(p.x) && isFiniteNumber(p.y);
   208→};
   209→
   210→const isCurvePoint = (v: unknown): boolean => {
   211→  if (typeof v !== "object" || v === null) return false;
   212→  const p = v as Record<string, unknown>;
   213→  return isFiniteNumber(p.x) && isFiniteNumber(p.yhat);
   214→};
   215→
   216→const isBoundaryPoint = (v: unknown): boolean => {
   217→  if (typeof v !== "object" || v === null) return false;
   218→  const p = v as Record<string, unknown>;
   219→  return isFiniteNumber(p.x) && isFiniteNumber(p.y) && isFiniteNumber(p.pred);
   220→};

   222→/** Validate a persisted ML train response; null when unusable. */
   223→function validateMlResult(raw: unknown): MLTrainResponse | null {
   224→  if (typeof raw !== "object" || raw === null) return null;
   225→  const r = raw as Record<string, unknown>;
   226→  if (r.task !== "regression" && r.task !== "classification") return null;
   227→  if (typeof r.model !== "string" || r.model.length > 40) return null;
   228→
   229→  const metricsRaw = r.metrics;
   230→  if (typeof metricsRaw !== "object" || metricsRaw === null) return null;
   231→  const metrics: Record<string, number> = {};
   232→  for (const [k, v] of Object.entries(metricsRaw as Record<string, unknown>)) {
   233→    if (isFiniteNumber(v) && k.length <= 24) metrics[k] = v;
   234→  }
   235→  if (Object.keys(metrics).length === 0) return null;
   236→
   237→  const coefsRaw = r.coefficients;
   238→  if (typeof coefsRaw !== "object" || coefsRaw === null) return null;
   239→  const c = coefsRaw as Record<string, unknown>;
   240→  const coefsArr = finiteArray(c.coefs, 512, isFiniteNumber);
   241→  if (!isFiniteNumber(c.intercept) || !coefsArr || typeof c.standardized !== "boolean") {
   242→    return null;
   243→  }
   244→
   245→  const seriesRaw = r.series;
   246→  if (typeof seriesRaw !== "object" || seriesRaw === null) return null;
   247→  const s = seriesRaw as Record<string, unknown>;
   248→  const test = finiteArray(s.test, 1200, isPoint);
   249→  if (!test) return null;
   250→  const curve = finiteArray(s.curve, 400, isCurvePoint);
   251→  const boundary = finiteArray(s.boundary, 400, isBoundaryPoint);
   252→
   253→  const metaRaw = r.datasetMeta;
   254→  if (typeof metaRaw !== "object" || metaRaw === null) return null;
   255→  const m = metaRaw as Record<string, unknown>;
   256→  const features = Array.isArray(m.features)
   257→    ? m.features.filter((f): f is string => typeof f === "string" && f.length <= 24).slice(0, 12)
   258→    : [];
   259→  if (
   260→    !isFiniteNumber(m.nTotal) ||
   261→    !isFiniteNumber(m.nTrain) ||
   262→    !isFiniteNumber(m.nTest) ||
   263→    !isFiniteNumber(m.noise) ||
   264→    !isFiniteNumber(m.seed)
   265→  ) {
   266→    return null;
   267→  }
   268→
   269→  // Optional CSV provenance — preserved when well-formed, dropped otherwise.
   270→  let source: CsvSourceMeta | undefined;
   271→  if (typeof m.source === "object" && m.source !== null) {
   272→    const s = m.source as Record<string, unknown>;
   273→    const colList = (v: unknown, cap: number): string[] =>
   274→      Array.isArray(v)
   275→        ? v.filter((c): c is string => typeof c === "string" && c.length <= 128).slice(0, cap)
   276→        : [];
   277→    const columns = colList(s.columns, 64);
   278→    const featureColumns = colList(s.featureColumns, 10);
   279→    const targetColumn = typeof s.targetColumn === "string" ? s.targetColumn.slice(0, 128) : "";
   280→    if (
   281→      s.kind === "csv" &&
   282→      columns.length > 0 &&
   283→      featureColumns.length > 0 &&
   284→      targetColumn !== "" &&
   285→      isFiniteNumber(s.nRows) &&
   286→      isFiniteNumber(s.droppedRows)
   287→    ) {
   288→      const labelMap =
   289→        Array.isArray(s.labelMap) &&
   290→        s.labelMap.length === 2 &&
   291→        s.labelMap.every((v) => typeof v === "string" && v.length <= 64)
   292→          ? (s.labelMap as [string, string])
   293→          : undefined;
   294→      source = {
   295→        kind: "csv",
   296→        fileName: typeof s.fileName === "string" ? s.fileName.slice(0, 128) : undefined,
   297→        columns,
   298→        featureColumns,
   299→        targetColumn,
   300→        nRows: s.nRows,
   301→        droppedRows: s.droppedRows,
   302→        ...(labelMap ? { labelMap } : {}),
   303→      };
   304→    }
   305→  }
   306→
   307→  // Optional cross-validation summary — dropped when malformed.
   308→  let cv: MLTrainResponse["cv"];
   309→  if (typeof r.cv === "object" && r.cv !== null) {
   310→    const cvRaw = r.cv as Record<string, unknown>;
   311→    const folds = Array.isArray(cvRaw.folds)
   312→      ? cvRaw.folds.filter(
   313→          (f): f is { fold: number; value: number } =>
   314→            typeof f === "object" &&
   315→            f !== null &&
   316→            isFiniteNumber((f as Record<string, unknown>).fold) &&
   317→            isFiniteNumber((f as Record<string, unknown>).value)
   318→        )
   319→      : [];
   320→    if (
   321→      (cvRaw.k === 3 || cvRaw.k === 5 || cvRaw.k === 10) &&
   322→      typeof cvRaw.metricName === "string" &&
   323→      folds.length === cvRaw.k &&
   324→      isFiniteNumber(cvRaw.mean) &&
   325→      isFiniteNumber(cvRaw.std)
   326→    ) {
   327→      cv = {
   328→        k: cvRaw.k,
   329→        metricName: cvRaw.metricName,
   330→        folds,
   331→        mean: cvRaw.mean,
   332→        std: cvRaw.std,
   333→      };
   334→    }
   335→  }
   336→
   337→  return {
   338→    task: r.task,
   339→    model: r.model,
   340→    metrics,
   341→    cv,
   342→    coefficients: {
   343→      intercept: c.intercept as number,
   344→      coefs: coefsArr as number[],
   345→      standardized: c.standardized,
   346→    },
   347→    series: {
   348→      test: test as { x: number; y: number; label?: number }[],
   349→      ...(curve ? { curve: curve as { x: number; yhat: number }[] } : {}),
   350→      ...(boundary ? { boundary: boundary as { x: number; y: number; pred: number }[] } : {}),
   351→    },
   352→    datasetMeta: {
   353→      nTotal: m.nTotal,
   354→      nTrain: m.nTrain,
   355→      nTest: m.nTest,
   356→      noise: m.noise,
   357→      seed: m.seed,
   358→      features,
   359→      ...(source ? { source } : {}),
   360→    },
   361→  };
   362→}

   376→/** Validate a persisted run-history payload; malformed entries are skipped.
   377→ * Accepts both the current `{ currentId, runs }` shape and the legacy bare
   378→ * array (current run = newest). */
   379→function validateMlHistoryPayload(raw: unknown): {
   380→  runs: MlRunRecord[];
   381→  currentId: number | null;
   382→} {
   383→  const body = Array.isArray(raw) ? { runs: raw, currentId: undefined } : raw;
   384→  if (typeof body !== "object" || body === null || !Array.isArray((body as { runs?: unknown }).runs)) {
   385→    return { runs: [], currentId: null };
   386→  }
   387→  const runs: MlRunRecord[] = [];
   388→  for (const item of ((body as { runs: unknown[] }).runs).slice(0, ML_HISTORY_LIMIT)) {
   389→    if (typeof item !== "object" || item === null) continue;
   390→    const r = item as Record<string, unknown>;
   391→    const result = validateMlResult(r.result);
   392→    if (
   393→      !result ||
   394→      typeof r.id !== "number" ||
   395→      !Number.isFinite(r.id) ||
   396→      typeof r.at !== "string" ||
   397→      r.at.length > 40 ||
   398→      (r.task !== "regression" && r.task !== "classification") ||
   399→      typeof r.model !== "string" ||
   400→      r.model.length > 40 ||
   401→      typeof r.keyMetric !== "string" ||
   402→      r.keyMetric.length > 16 ||
   403→      typeof r.keyValue !== "number" ||
   404→      !Number.isFinite(r.keyValue)
   405→    ) {
   406→      continue;
   407→    }
   408→    runs.push({
   409→      id: r.id,
   410→      at: r.at,
   411→      task: r.task,
   412→      model: r.model,
   413→      keyMetric: r.keyMetric,
   414→      keyValue: r.keyValue,
   415→      config: validateMl(r.config),
   416→      result,
   417→      ...(typeof r.csv === "object" && r.csv !== null
   418→        ? {
   419→            csv: {
   420→              fileName:
   421→                typeof (r.csv as Record<string, unknown>).fileName === "string"
   422→                  ? (r.csv as Record<string, unknown>).fileName.slice(0, 128) || "pasted.csv"
   423→                  : "pasted.csv",
   424→              featureColumns: Array.isArray((r.csv as Record<string, unknown>).featureColumns)
   425→                ? ((r.csv as Record<string, unknown>).featureColumns as unknown[])
   426→                    .filter((c): c is string => typeof c === "string" && c.length <= 128)
   427→                    .slice(0, 8)
   428→                : [],
   429→              targetColumn:
   430→                typeof (r.csv as Record<string, unknown>).targetColumn === "string"
   431→                  ? (r.csv as Record<string, unknown>).targetColumn.slice(0, 128)
   432→                  : "",
   433→            },
   434→          }
   435→        : {}),
   436→    });
   437→  }
   438→  const rawId = (body as { currentId?: unknown }).currentId;
   439→  const currentId =
   440→    typeof rawId === "number" && Number.isFinite(rawId) && runs.some((h) => h.id === rawId)
   441→      ? rawId
   442→      : (runs[0]?.id ?? null);
   443→  return { runs, currentId };
   444→}

   446→/** Read the persisted run history (client only, failure-tolerant). */
   447→function loadMlHistory(): { runs: MlRunRecord[]; currentId: number | null } {
   448→  if (typeof window === "undefined") return { runs: [], currentId: null };
   449→  try {
   450→    const raw = window.sessionStorage.getItem(ML_HISTORY_STORAGE_KEY);
   451→    if (!raw) return { runs: [], currentId: null };
   452→    return validateMlHistoryPayload(JSON.parse(raw));
   453→  } catch {
   454→    return { runs: [], currentId: null }; // corrupted JSON / private mode — start fresh
   455→  }
   456→}

   459→function saveMlHistory(history: MlRunRecord[], currentId: number | null): void {

   464→    window.sessionStorage.setItem(ML_HISTORY_STORAGE_KEY, json);
   465→  } catch {
   466→    /* quota exceeded or private mode — persistence is optional */
   467→  }

   470→/** Persist the ML training result (client only, size-capped, best-effort). */
   471→function saveMlResult(result: MLTrainResponse | null): void {
   472→  if (typeof window === "undefined") return;
   473→  try {
   474→    if (result === null) {
   475→      window.sessionStorage.removeItem(ML_RESULT_STORAGE_KEY);
   476→      return;
   477→    }
   478→    const json = JSON.stringify(result);
   479→    if (json.length > ML_RESULT_MAX_BYTES) return; // oversized — persistence is optional
   480→    window.sessionStorage.setItem(ML_RESULT_STORAGE_KEY, json);
   481→  } catch {
   482→    /* quota exceeded or private mode — persistence is optional */
   483→  }
   484→}

   487→function loadPlayground(): Partial<PlaygroundPersisted> {
   488→  if (typeof window === "undefined") return {};

   521→/** Validate a persisted uploaded-CSV payload; null when unusable. */
   522→function validMlCsv(raw: unknown): MlCsvState | null {
   523→  if (typeof raw !== "object" || raw === null) return null;
   524→  const c = raw as Partial<MlCsvState>;
   525→  if (
   526→    typeof c.text !== "string" ||
   527→    c.text.length === 0 ||
   528→    c.text.length > MAX_CSV_TEXT ||
   529→    typeof c.fileName !== "string" ||
   530→    c.fileName.length > 128
   531→  ) {
   532→    return null;
   533→  }
   534→  return { fileName: c.fileName, text: c.text };

   651→/** Newest-first, deduplicated, capped recent list (pure helper). */
   652→function pushRecent(recent: string[], path: string): string[] {
   653→  const next = [path, ...recent.filter((p) => p !== path)];
   654→  return next.slice(0, RECENT_LIMIT);

   782→  tree: null,
   783→  stats: null,
   784→  treeStatus: "idle",
   785→  treeError: null,
   786→  ensureTree: async () => {
   787→    const { treeStatus, tree } = get();
   788→    if (treeStatus === "loading" || tree) return;
   789→    set({ treeStatus: "loading", treeError: null });
   790→    try {
   791→      const res = await fetch("/api/toolkit/tree");
   792→      if (!res.ok) {
   793→        throw new Error(`Tree API responded with HTTP ${res.status}`);
   794→      }
   795→      const data = (await res.json()) as TreeResponse;
   796→      set({ tree: data.root, stats: data.stats, treeStatus: "ready" });
   797→    } catch (err) {
   798→      set({
   799→        treeStatus: "error",
   800→        treeError:
   801→          err instanceof Error ? err.message : "Unknown error while loading the file tree.",
   802→      });
   803→    }
   804→  },
   805→}));
