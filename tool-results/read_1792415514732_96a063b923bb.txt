     1→"""Deep-learning estimators for QSAR (PyTorch, scikit-learn compatible).
     2→
     3→The legacy codebase shipped one shallow ``MLPClassifier`` recipe under
     4→the name "ANN"; this module brings *real* deep learning to the
     5→toolkit: feed-forward networks (:class:`DeepMLPClassifier` /
     6→:class:`DeepMLPRegressor`) and residual networks
     7→(:class:`ResidualMLPClassifier` / :class:`ResidualMLPRegressor`) built
     8→on PyTorch, with minibatch Adam training (weight decay), early
     9→stopping on a held-out validation split and per-feature
    10→standardisation fitted on the training data.
    11→
    12→The estimators follow the scikit-learn estimator contract (``fit`` /
    13→``predict`` / ``predict_proba``, ``get_params`` / ``set_params``,
    14→clonable by :class:`~sklearn.model_selection.GridSearchCV`), so they
    15→drop into :func:`~pharmaml.ml.training.train_model`,
    16→:func:`~pharmaml.ml.training.compare_models` and
    17→:func:`~pharmaml.ml.evaluation.evaluate_model` like any registry
    18→model — the registry keys are ``'dnn'`` and ``'dnn_residual'``.
    19→
    20→PyTorch is an *optional* dependency imported lazily, mirroring how
    21→:mod:`pharmaml.ml.models` treats XGBoost/LightGBM/CatBoost: importing
    22→this module (or ``pharmaml`` itself) never requires torch; the import
    23→happens at :meth:`fit` time and raises an actionable
    24→:class:`~pharmaml.exceptions.DependencyError` when missing::
    25→
    26→    pip install 'pharmaml[deep]'
    27→
    28→Determinism: ``fit`` seeds both the PyTorch and NumPy random number
    29→generators from ``random_state``, so two estimators constructed with
    30→the same seed and trained on the same data produce identical
    31→predictions (CPU).  The caller's *global* torch RNG is
    32→snapshot-and-restored around the training loop, so ``fit`` leaves the
    33→global stream exactly as it found it.  Fitted weights are always
    34→stored as plain NumPy arrays, so pickled estimators stay portable
    35→across torch versions — and even across machines without torch
    36→installed (torch is only needed again to *predict*).
    37→
    38→Examples
    39→--------
    40→>>> from pharmaml.ml.models import MODEL_REGISTRY
    41→>>> spec = MODEL_REGISTRY["dnn"]    # registry key of the DeepMLP estimators
    42→>>> spec.task, spec.requires
    43→('both', 'torch')
    44→>>> sorted(spec.param_grid)
    45→['dropout', 'hidden_layers', 'learning_rate']
    46→
    47→>>> from pharmaml.ml.neural import DeepMLPRegressor
    48→>>> DeepMLPRegressor().predict([[0.0]])  # sklearn contract: fit before predict
    49→Traceback (most recent call last):
    50→...
    51→pharmaml.exceptions.ModelError: DeepMLPRegressor has not been fitted yet — call fit(X, y) first
    52→
    53→>>> DeepMLPRegressor(dropout=1.5).fit([[0.0], [1.0]], [0.0, 1.0])
    54→Traceback (most recent call last):
    55→...
    56→pharmaml.exceptions.ModelError: dropout must be in [0, 1), got 1.5
    57→
    58→>>> import numpy as np
    59→>>> X = np.linspace(0.0, 1.0, 40).reshape(-1, 1)
    60→>>> y = 2.0 * X[:, 0] + 1.0
    61→>>> net = DeepMLPRegressor(hidden_layers=(32,), epochs=300, random_state=0).fit(X, y)
    62→>>> net.score(X, y) > 0.99   # R^2 on a 40-point linear function (seed 0)
    63→True
    64→>>> twin = DeepMLPRegressor(hidden_layers=(32,), epochs=300, random_state=0).fit(X, y)
    65→>>> np.array_equal(net.predict(X), twin.predict(X))   # same seed, identical net
    66→True
    67→"""

   106→def _require_torch() -> ModuleType:
   107→    """Import and return torch, or raise the actionable error.
   108→
   109→    Returns the module object for callers that only need its presence;
   110→    functions that touch torch APIs should do a plain local ``import
   111→    torch`` after calling this guard so mypy sees the fully-typed module
   112→    (a ``ModuleType`` annotation alone does not expose attributes).
   113→
   114→    Raises
   115→    ------
   116→    DependencyError
   117→        When torch is not installed (message names the ``pharmaml[deep]``
   118→        extra that installs it).
   119→    """
   120→    try:
   121→        import torch
   122→    except ImportError as exc:
   123→        raise DependencyError(_TORCH_HINT) from exc
   124→    return torch

   127→def _make_activation(activation: str) -> Any:
   128→    """Return a fresh ``torch.nn`` activation module for a token."""
   129→    from torch import nn
   130→
   131→    factory = {
   132→        "relu": nn.ReLU,
   133→        "tanh": nn.Tanh,
   134→        "gelu": nn.GELU,
   135→        "sigmoid": nn.Sigmoid,
   136→    }
   137→    return factory[activation]()

   140→def _build_network(
   141→    residual: bool,
   142→    n_in: int,
   143→    hidden: tuple[int, ...],
   144→    n_out: int,
   145→    activation: str,
   146→    dropout: float,
   147→) -> Any:
   148→    """Construct an *untrained* torch network for the given architecture.
   149→
   150→    ``residual=False`` builds a plain feed-forward stack (Linear ->
   151→    activation -> dropout between hidden layers); ``residual=True``
   152→    builds :class:`_ResidualNet` blocks with skip connections.  An
   153→    empty ``hidden`` tuple yields a single linear layer (a generalised
   154→    linear model) in both cases.
   155→    """
   156→    if residual:
   157→        return _torch_networks()[2](n_in, hidden, n_out, activation, dropout)
   158→
   159→    from torch import nn
   160→
   161→    layers: list[Any] = []
   162→    widths = [n_in, *hidden, n_out]
   163→    for i in range(len(widths) - 1):
   164→        layers.append(nn.Linear(widths[i], widths[i + 1]))
   165→        if i < len(widths) - 2:  # hidden layer -> activation + dropout
   166→            layers.append(_make_activation(activation))
   167→            if dropout > 0:
   168→                layers.append(nn.Dropout(dropout))
   169→    return nn.Sequential(*layers)
   170→
   171→
   172→#: Lazily-created torch network classes, cached so state-dict keys are
   173→#: stable across every build within a process.
   174→_TORCH_NETS: tuple[Any, type, type] | None = None
   175→
   176→
   177→def _torch_networks() -> tuple[Any, type, type]:
   178→    """Define (once) and return ``(torch, ResidualBlock, ResidualNet)``.
   179→
   180→    The ``torch.nn.Module`` subclasses are defined lazily — never at
   181→    module import — so ``import pharmaml.ml.neural`` works on machines
   182→    without torch; caching them keeps ``state_dict`` keys identical
   183→    between the training build and the rebuild-on-predict.
   184→    """
   185→    global _TORCH_NETS
   186→    if _TORCH_NETS is None:
   187→        import torch
   188→        from torch import nn
   189→
   190→        class ResidualBlock(nn.Module):
   191→            """Linear -> activation -> dropout -> Linear with an input skip.
   192→
   193→            The skip is an identity when input and output widths match,
   194→            else a learned linear projection (the dense analogue of a
   195→            1x1 convolution).
   196→            """
   197→
   198→            def __init__(
   199→                self, width_in: int, width_out: int, activation: str, dropout: float
   200→            ) -> None:
   201→                super().__init__()
   202→                self.fc1 = nn.Linear(width_in, width_out)
   203→                self.fc2 = nn.Linear(width_out, width_out)
   204→                self.act = _make_activation(activation)
   205→                self.drop = nn.Dropout(dropout) if dropout > 0 else nn.Identity()
   206→                self.skip = (
   207→                    nn.Identity() if width_in == width_out else nn.Linear(width_in, width_out)
   208→                )
   209→
   210→            def forward(self, x: torch.Tensor) -> torch.Tensor:
   211→                out = self.fc2(self.drop(self.act(self.fc1(x))))
   212→                return self.act(out + self.skip(x))
   213→
   214→        class ResidualNet(nn.Module):
   215→            """A stack of :class:`ResidualBlock` layers plus a linear head."""
   216→
   217→            def __init__(
   218→                self,
   219→                n_in: int,
   220→                hidden: tuple[int, ...],
   221→                n_out: int,
   222→                activation: str,
   223→                dropout: float,
   224→            ) -> None:
   225→                super().__init__()
   226→                widths = [n_in, *hidden]
   227→                self.blocks = nn.ModuleList(
   228→                    ResidualBlock(widths[i], widths[i + 1], activation, dropout)
   229→                    for i in range(len(widths) - 1)
   230→                )
   231→                self.head = nn.Linear(widths[-1], n_out)
   232→
   233→            def forward(self, x: torch.Tensor) -> torch.Tensor:
   234→                for block in self.blocks:
   235→                    x = block(x)
   236→                return self.head(x)
   237→
   238→        _TORCH_NETS = (torch, ResidualBlock, ResidualNet)
   239→    return _TORCH_NETS

   242→# ---------------------------------------------------------------------------
   243→# Shared estimator machinery
   244→# ---------------------------------------------------------------------------
   245→class _BaseDeepEstimator(BaseEstimator):
   246→    """Private base with the fit/predict machinery shared by the four
   247→    public deep estimators (never instantiated directly).
   248→
   249→    Subclasses set two class flags and define the sklearn-visible
   250→    ``__init__`` (all constructor parameters verbatim, so
   251→    ``get_params``/``set_params``/``clone`` work); everything else —
   252→    validation, standardisation, the torch training loop, early
   253→    stopping, weight (de)serialisation — lives here.
   254→    """
   255→
   256→    _is_classifier: bool = False
   257→    _residual: bool = False
   258→
   259→    # -- validation -----------------------------------------------------
   260→    def _validate_hyperparameters(self) -> tuple[int, ...]:
   261→        """Validate constructor values and return the hidden-layer widths.
   262→
   263→        Returns
   264→        -------
   265→        tuple of int
   266→            ``hidden_layers`` materialised as a tuple — generators are
   267→            consumed exactly once here, so the training loop never sees
   268→            an exhausted iterable (which would silently build a single
   269→            linear layer).
   270→
   271→        Raises
   272→        ------
   273→        ModelError
   274→            When ``hidden_layers`` holds a non-positive/non-integer
   275→            width, ``dropout``/``val_split`` fall outside ``[0, 1)``,
   276→            ``activation``/``device`` are unknown tokens, ``epochs``/
   277→            ``batch_size``/``patience`` are below 1, ``learning_rate``
   278→            is not positive-finite, ``weight_decay`` is negative, or
   279→            ``random_state`` is not an integer.
   280→        """
   281→        hidden = self.hidden_layers
   282→        if isinstance(hidden, (str, bytes)) or not hasattr(hidden, "__iter__"):
   283→            raise ModelError(
   284→                f"hidden_layers must be a sequence of positive ints, got {hidden!r}"
   285→            )
   286→        hidden = tuple(hidden)
   287→        for width in hidden:
   288→            if isinstance(width, bool) or not isinstance(width, (int, np.integer)) or width < 1:
   289→                raise ModelError(
   290→                    "hidden_layers entries must be positive ints, got "
   291→                    f"{width!r} in {tuple(hidden)!r}"
   292→                )
   293→        if isinstance(self.dropout, bool) or not np.isfinite(self.dropout) or not (
   294→            0.0 <= float(self.dropout) < 1.0
   295→        ):
   296→            raise ModelError(f"dropout must be in [0, 1), got {self.dropout!r}")
   297→        if self.activation not in _ACTIVATIONS:
   298→            raise ModelError(
   299→                f"activation must be one of {_ACTIVATIONS}, got {self.activation!r}"
   300→            )
   301→        for name in ("epochs", "batch_size", "patience"):
   302→            value = getattr(self, name)
   303→            if isinstance(value, bool) or not isinstance(value, (int, np.integer)) or value < 1:
   304→                raise ModelError(f"{name} must be an integer >= 1, got {value!r}")
   305→        if not np.isfinite(self.learning_rate) or float(self.learning_rate) <= 0:
   306→            raise ModelError(f"learning_rate must be > 0, got {self.learning_rate!r}")
   307→        if not np.isfinite(self.weight_decay) or float(self.weight_decay) < 0:
   308→            raise ModelError(f"weight_decay must be >= 0, got {self.weight_decay!r}")
   309→        if not np.isfinite(self.val_split) or not 0.0 <= float(self.val_split) < 1.0:
   310→            raise ModelError(f"val_split must be in [0, 1), got {self.val_split!r}")
   311→        if self.device not in _DEVICES:
   312→            raise ModelError(f"device must be one of {_DEVICES}, got {self.device!r}")
   313→        if isinstance(self.random_state, bool) or not isinstance(
   314→            self.random_state, (int, np.integer)
   315→        ):
   316→            raise ModelError(
   317→                f"random_state must be an integer seed (determinism contract), "
   318→                f"got {self.random_state!r}"
   319→            )
   320→        return hidden
   321→
   322→    def _check_X(self, X: Any) -> np.ndarray:
   323→        """Validate and convert ``X`` to a finite float64 matrix.
   324→
   325→        Raises
   326→        ------
   327→        DataValidationError
   328→            When ``X`` is not 2-dimensional, non-numeric, empty, has no
   329→            feature columns, or contains NaN/infinite values.
   330→        """
   331→        try:
   332→            X_arr = np.asarray(X, dtype=np.float64)
   333→        except (TypeError, ValueError) as exc:
   334→            raise DataValidationError(
   335→                f"features must be numeric — conversion failed: {exc}"
   336→            ) from exc
   337→        if X_arr.ndim != 2:
   338→            raise DataValidationError(
   339→                "X must be 2-dimensional (n_samples, n_features), got "
   340→                f"{X_arr.ndim} dimension(s)"
   341→            )
   342→        if X_arr.shape[0] == 0:
   343→            raise DataValidationError(
   344→                f"X has 0 rows — {type(self).__name__} needs at least one sample"
   345→            )
   346→        if X_arr.shape[1] == 0:
   347→            raise DataValidationError(
   348→                f"X has 0 feature columns — {type(self).__name__} needs at least "
   349→                "one input feature"
   350→            )
   351→        if not np.isfinite(X_arr).all():
   352→            raise DataValidationError(
   353→                "X contains NaN or infinite values — impute or drop them first "
   354→                "(pharmaml.ml.build_preprocessor does this inside a Pipeline)"
   355→            )
   356→        return X_arr
   357→
   358→    def _check_fitted(self) -> None:
   359→        """Raise :class:`ModelError` when the estimator is used before fit."""
   360→        if getattr(self, "_weights", None) is None:
   361→            raise ModelError(
   362→                f"{type(self).__name__} has not been fitted yet — call fit(X, y) "
   363→                "first"
   364→            )
   365→
   366→    def _check_feature_names(self, X: Any) -> None:
   367→        """Reject DataFrame columns that disagree with ``feature_names_in_``.
   368→
   369→        Mirrors sklearn's ``_check_feature_names``: an estimator fitted
   370→        on a DataFrame remembers its columns, and predicting on a
   371→        DataFrame whose columns differ in *names or order* is a silent
   372→        column-permutation bug (the values are consumed positionally),
   373→        so it raises a typed error.  Plain arrays stay positional — no
   374→        name checking — exactly like sklearn.
   375→
   376→        Raises
   377→        ------
   378→        DataValidationError
   379→            When ``X`` is a DataFrame whose column names/order differ
   380→            from the ones seen at fit time.
   381→        """
   382→        names = getattr(self, "feature_names_in_", None)
   383→        if names is None or not isinstance(X, pd.DataFrame):
   384→            return
   385→        columns = np.asarray(X.columns, dtype=object)
   386→        if not np.array_equal(columns, names):
   387→            raise DataValidationError(
   388→                f"X has feature names {columns.tolist()} that do not match the "
   389→                f"feature names seen during fit: {names.tolist()} — pass the same "
   390→                "columns in the fit order, or a plain (positional) array"
   391→            )
   392→
   393→    # -- fit ------------------------------------------------------------
   394→    def fit(self, X: Any, y: Any) -> _BaseDeepEstimator:
   395→        """Train the network on ``(X, y)`` with early stopping.
   396→
   397→        Parameters
   398→        ----------
   399→        X:
   400→            Feature matrix — DataFrame or 2-D array-like; must be
   401→            numeric and finite (the standard ``pharmaml`` pipeline
   402→            imputes/scales upstream).
   403→        y:
   404→            Targets — class labels (any hashable, e.g. strings or ints)
   405→            for the classifiers, numeric values for the regressors.
   406→
   407→        Returns
   408→        -------
   409→        self
   410→            Fitted estimator (``classes_``, ``n_features_in_``,
   411→            ``history_``, ``n_iter_`` and the standardisation
   412→            attributes are now available).
   413→
   414→        Raises
   415→        ------
   416→        DependencyError
   417→            When torch is not installed (see ``pip install
   418→            'pharmaml[deep]'``).
   419→        ModelError
   420→            For invalid hyperparameters (validated before any work),
   421→            when predicting before fitting, or when the training
   422→            diverged (weights became NaN/inf — typically an extreme
   423→            ``learning_rate``).
   424→        DataValidationError
   425→            For empty/mismatched/non-numeric ``X``/``y``, NaN targets,
   426→            a single-class classification target, unorderable labels,
   427→            or a feature-count (or, for DataFrames, feature-name)
   428→            mismatch at predict time.
   429→
   430→        Notes
   431→        -----
   432→        Determinism: PyTorch and NumPy are seeded from
   433→        ``random_state`` at fit start, so the same seed + data always
   434→        yields the same fitted network on CPU.  torch's *global* RNG is
   435→        snapshot-and-restored around the training loop: the caller's
   436→        global stream is left exactly as ``fit`` found it.
   437→        """
   438→        hidden = self._validate_hyperparameters()
   439→        X_arr = self._check_X(X)
   440→
   441→        y_arr = np.asarray(y)
   442→        if y_arr.ndim != 1:
   443→            raise DataValidationError(
   444→                f"y must be 1-dimensional, got {y_arr.ndim} dimension(s)"
   445→            )
   446→        if y_arr.size == 0:
   447→            raise DataValidationError(
   448→                f"y is empty — {type(self).__name__} needs at least one target value"
   449→            )
   450→        if y_arr.shape[0] != X_arr.shape[0]:
   451→            raise DataValidationError(
   452→                f"X and y have inconsistent lengths: {X_arr.shape[0]} rows vs "
   453→                f"{y_arr.shape[0]} targets"
   454→            )
   455→
   456→        if self._is_classifier:
   457→            try:
   458→                classes, y_enc = np.unique(y_arr, return_inverse=True)
   459→            except TypeError as exc:  # e.g. None mixed with labels
   460→                raise DataValidationError(
   461→                    "classification targets could not be label-encoded — they "
   462→                    f"contain unorderable values (e.g. None): {exc}"
   463→                ) from exc
   464→            if classes.size < 2:
   465→                raise DataValidationError(
   466→                    "classification needs at least 2 distinct classes, got "
   467→                    f"{classes.size}: {classes.tolist()[:5]}"
   468→                )
   469→            if classes.dtype.kind in "fc" and np.isnan(classes.astype(float)).any():
   470→                raise DataValidationError(
   471→                    "classification targets contain NaN — drop those rows first"
   472→                )
   473→            n_out = int(classes.size)
   474→        else:
   475→            try:
   476→                y_enc = np.asarray(y_arr, dtype=np.float64)
   477→            except (TypeError, ValueError) as exc:
   478→                raise DataValidationError(
   479→                    f"regression targets must be numeric — conversion failed: {exc}"
   480→                ) from exc
   481→            if not np.isfinite(y_enc).all():
   482→                raise DataValidationError(
   483→                    "regression targets contain NaN or infinite values — drop or "
   484→                    "impute those rows first"
   485→                )
   486→            n_out = 1
   487→
   488→        # torch is imported BEFORE any fitted attribute is written, so a
   489→        # failed fit (missing torch / unusable device) leaves the
   490→        # estimator unfitted for sklearn's check_is_fitted as well.
   491→        _require_torch()  # raises the actionable DependencyError
   492→
   493→        device = self._resolve_device()
   494→
   495→        if self._is_classifier:
   496→            self.classes_ = classes
   497→        self.n_features_in_ = int(X_arr.shape[1])
   498→        if isinstance(X, pd.DataFrame):
   499→            self.feature_names_in_ = np.asarray(X.columns, dtype=object)
   500→        elif hasattr(self, "feature_names_in_"):
   501→            # sklearn reset semantics: refitting on a positional array
   502→            # drops the stale names instead of enforcing the old column
   503→            # order at predict time.
   504→            del self.feature_names_in_
   505→
   506→        seed = int(self.random_state)
   507→        rng = np.random.default_rng(seed)
   508→        n = int(X_arr.shape[0])
   509→        train_idx, val_idx = self._train_val_split(y_enc, n, rng)
   510→
   511→        # Per-feature standardisation, fitted on the training rows only
   512→        # (the early-stopping validation rows never touch the scaler;
   513→        # constant columns get std 1 to avoid division by zero).
   514→        mean = X_arr[train_idx].mean(axis=0)
   515→        std = X_arr[train_idx].std(axis=0)
   516→        std = np.where(std > 0, std, 1.0)
   517→        self._mean_ = mean
   518→        self._std_ = std
   519→        X_scaled = (X_arr - mean) / std
   520→
   521→        # Regression targets are standardised too (training rows only):
   522→        # pharmaceutical endpoints (pIC50 ~ 5, LN_IC50 ~ 2-4, AUC in
   523→        # [0, 1]) live far from zero, and on a raw-scale MSE the output
   524→        # bias crawls toward the target mean at ~learning_rate per
   525→        # step -- early stopping fires long before the intercept is
   526→        # learned (a linear baseline then beats the network by orders
   527→        # of magnitude).  Training on z-scaled targets keeps the loss
   528→        # O(1) and the intercept one step away; ``predict`` inverts
   529→        # the scaling.  A constant target keeps std 1 (predict returns
   530→        # the constant).
   531→        if self._is_classifier:
   532→            self._y_mean_: float | None = None
   533→            self._y_std_: float | None = None
   534→        else:
   535→            y_train = y_enc[train_idx]
   536→            self._y_mean_ = float(y_train.mean())
   537→            y_std = float(y_train.std())
   538→            self._y_std_ = y_std if y_std > 0 else 1.0
   539→            y_enc = (y_enc - self._y_mean_) / self._y_std_
   540→
   541→        # The training loop seeds torch's *global* RNG from
   542→        # ``random_state`` (the same-seed determinism contract).  Snapshot
   543→        # and restore it around the call so the caller's global stream is
   544→        # left exactly as ``fit`` found it; ``manual_seed`` stays inside
   545→        # the protected window, so determinism is untouched.
   546→        import torch
   547→
   548→        rng_state = torch.get_rng_state()
   549→        try:
   550→            self._train_torch(
   551→                X_scaled, y_enc, n_out, device, hidden, train_idx, val_idx, rng
   552→            )
   553→        finally:
   554→            torch.set_rng_state(rng_state)
   555→        return self
   556→
   557→    def _resolve_device(self) -> str:
   558→        """Map the ``device`` token to a concrete torch device string.
   559→
   560→        Raises
   561→        ------
   562→        ModelError
   563→            When ``device='cuda'`` is requested but no CUDA device is
   564→            visible to torch.
   565→        """
   566→        import torch
   567→
   568→        if self.device == "auto":
   569→            return "cuda" if torch.cuda.is_available() else "cpu"
   570→        if self.device == "cuda" and not torch.cuda.is_available():
   571→            raise ModelError(
   572→                "device='cuda' requested but no CUDA device is visible to torch"
   573→            )
   574→        return self.device
   575→
   576→    def _train_val_split(
   577→        self, y_enc: np.ndarray, n: int, rng: np.random.Generator
   578→    ) -> tuple[np.ndarray, np.ndarray | None]:
   579→        """Split row indices into (train, validation) index arrays.
   580→
   581→        Classification splits are *stratified* — every class keeps at
   582→        least one training row (single-member classes stay in the
   583→        training fold entirely); regression uses a plain random split.
   584→        Returns ``(all_rows, None)`` when ``val_split`` is 0, the data
   585→        is a single row, or the split would be degenerate — in that
   586→        case training runs for all ``epochs`` with no early stopping.
   587→        """
   588→        val_split = float(self.val_split)
   589→        if val_split <= 0 or n < 2:
   590→            return np.arange(n), None
   591→        if self._is_classifier:
   592→            parts: list[np.ndarray] = []
   593→            for cls in np.unique(y_enc):
   594→                members = np.flatnonzero(y_enc == cls)
   595→                members = members[rng.permutation(members.size)]
   596→                k = round(val_split * members.size)
   597→                # every class keeps >= 1 training row, even when it has
   598→                # a single member (val_split can otherwise steal it).
   599→                k = min(k, max(members.size - 1, 0))
   600→                parts.append(members[:k])
   601→            val_idx = np.concatenate(parts).astype(int) if parts else np.array([], dtype=int)
   602→        else:
   603→            perm = rng.permutation(n)
   604→            val_idx = perm[: max(1, round(val_split * n))]
   605→        if val_idx.size == 0 or val_idx.size >= n:
   606→            return np.arange(n), None
   607→        mask = np.zeros(n, dtype=bool)
   608→        mask[val_idx] = True
   609→        return np.flatnonzero(~mask), np.flatnonzero(mask)
   610→
   611→    def _train_torch(
   612→        self,
   613→        X_scaled: np.ndarray,
   614→        y_enc: np.ndarray,
   615→        n_out: int,
   616→        device: str,
   617→        hidden: tuple[int, ...],
   618→        train_idx: np.ndarray,
   619→        val_idx: np.ndarray | None,
   620→        rng: np.random.Generator,
   621→    ) -> None:
   622→        """Run the minibatch Adam training loop with early stopping.
   623→
   624→        ``hidden`` is the materialised layer-width tuple from
   625→        :meth:`_validate_hyperparameters` (one-time consumption of the
   626→        ``hidden_layers`` iterable), and ``train_idx``/``val_idx`` (chosen
   627→        in :meth:`fit` before the standardisation was fitted) plus the
   628→        seeded ``rng`` keep the whole fit one deterministic random
   629→        stream.
   630→        """
   631→        import torch
   632→        from torch import nn
   633→
   634→        torch.manual_seed(int(self.random_state))
   635→
   636→        dropout = float(self.dropout)
   637→        # Snapshot of the *fitted* architecture — later set_params calls
   638→        # cannot desynchronise the weight rebuild.
   639→        self._net_spec_: dict[str, Any] = {
   640→            "residual": self._residual,
   641→            "n_in": int(X_scaled.shape[1]),
   642→            "hidden": hidden,
   643→            "n_out": int(n_out),
   644→            "activation": self.activation,
   645→            "dropout": dropout,
   646→        }
   647→        network = _build_network(
   648→            self._residual, self._net_spec_["n_in"], hidden, n_out, self.activation, dropout
   649→        )
   650→        network.to(device)
   651→
   652→        X_t = torch.as_tensor(X_scaled, dtype=torch.float32, device=device)
   653→        if self._is_classifier:
   654→            y_t = torch.as_tensor(y_enc, dtype=torch.long, device=device)
   655→            criterion: Any = nn.CrossEntropyLoss()
   656→        else:
   657→            y_t = torch.as_tensor(y_enc, dtype=torch.float32, device=device).reshape(-1, 1)
   658→            criterion = nn.MSELoss()
   659→        optimizer = torch.optim.Adam(
   660→            network.parameters(),
   661→            lr=float(self.learning_rate),
   662→            weight_decay=float(self.weight_decay),
   663→        )
   664→
   665→        n = X_scaled.shape[0]
   666→        train_t = torch.as_tensor(train_idx, dtype=torch.long, device=device)
   667→        X_tr, y_tr = X_t[train_t], y_t[train_t]
   668→        X_val = y_val = None
   669→        if val_idx is not None:
   670→            val_t = torch.as_tensor(val_idx, dtype=torch.long, device=device)
   671→            X_val, y_val = X_t[val_t], y_t[val_t]
   672→
   673→        epochs = int(self.epochs)
   674→        batch_size = int(self.batch_size)
   675→        patience = int(self.patience)
   676→
   677→        history: list[dict[str, Any]] = []
   678→        best_val = float("inf")
   679→        best_state: dict[str, Any] | None = None
   680→        stale = 0
   681→
   682→        for epoch in range(1, epochs + 1):
   683→            network.train()
   684→            perm = torch.as_tensor(rng.permutation(train_idx.size), device=device)
   685→            batch_losses: list[float] = []
   686→            for start in range(0, perm.numel(), batch_size):
   687→                batch = perm[start : start + batch_size]
   688→                optimizer.zero_grad()
   689→                loss = criterion(network(X_tr[batch]), y_tr[batch])
   690→                loss.backward()
   691→                optimizer.step()
   692→                batch_losses.append(float(loss.item()))
   693→            train_loss = float(np.mean(batch_losses)) if batch_losses else float("nan")
   694→
   695→            val_loss: float | None = None
   696→            if X_val is not None:
   697→                network.eval()
   698→                with torch.no_grad():
   699→                    val_loss = float(criterion(network(X_val), y_val).item())
   700→                if val_loss < best_val:
   701→                    best_val = val_loss
   702→                    best_state = {
   703→                        key: value.detach().cpu().clone()
   704→                        for key, value in network.state_dict().items()
   705→                    }
   706→                    stale = 0
   707→                else:
   708→                    stale += 1
   709→
   710→            history.append({"epoch": epoch, "train_loss": train_loss, "val_loss": val_loss})
   711→            _LOG.debug(
   712→                "epoch %03d train_loss %.3f val_loss %s",
   713→                epoch,
   714→                train_loss,
   715→                "n/a" if val_loss is None else f"{val_loss:.3f}",
   716→            )
   717→            if val_loss is not None and stale >= patience:
   718→                _LOG.debug(
   719→                    "early stopping at epoch %03d (no val improvement for %d epochs)",
   720→                    epoch,
   721→                    patience,
   722→                )
   723→                break
   724→
   725→        if best_state is not None:  # restore the best (lowest val loss) weights
   726→            network.load_state_dict(best_state)
   727→        network.eval()
   728→
   729→        # Portable CPU weights — pickles never embed torch classes.
   730→        weights = {
   731→            key: value.detach().cpu().numpy().copy()
   732→            for key, value in network.state_dict().items()
   733→        }
   734→        if not all(np.isfinite(value).all() for value in weights.values()):
   735→            # Diverged training (typically an extreme learning_rate):
   736→            # refuse to leave a silently-NaN estimator behind.
   737→            raise ModelError(
   738→                "training diverged — the fitted weights contain NaN/inf "
   739→                "values; lower learning_rate (e.g. 1e-3) and refit"
   740→            )
   741→        self.history_ = history
   742→        self.n_iter_ = len(history)
   743→        self._weights = weights
   744→        self._network = network
   745→
   746→        last = history[-1]
   747→        val_text = "n/a" if last["val_loss"] is None else f"{last['val_loss']:.4f}"
   748→        n_params = sum(int(p.numel()) for p in network.parameters())
   749→        _LOG.info(
   750→            "%s fitted on %d samples: epoch %03d train_loss %.4f val_loss %s "
   751→            "(%d epochs run, %d parameters, device=%s)",
   752→            type(self).__name__,
   753→            n,
   754→            last["epoch"],
   755→            last["train_loss"],
   756→            val_text,
   757→            self.n_iter_,
   758→            n_params,
   759→            device,
   760→        )
   761→
   762→    # -- predict ----------------------------------------------------------
   763→    def _ensure_network(self) -> Any:
   764→        """Return the live torch network, rebuilding it when needed.
   765→
   766→        After unpickling the live module is ``None``; it is
   767→        reconstructed from the fitted architecture snapshot and the
   768→        stored NumPy weights — deterministic, because
   769→        ``load_state_dict`` overwrites every parameter.  (A
   770→        ``set_params`` mutation keeps the *old* live network, so
   771→        predictions keep working until the next ``fit``.)
   772→        """
   773→        network = getattr(self, "_network", None)
   774→        if network is None:
   775→            _require_torch()  # raises the actionable DependencyError
   776→            import torch
   777→
   778→            spec = self._net_spec_
   779→            network = _build_network(
   780→                spec["residual"], spec["n_in"], spec["hidden"], spec["n_out"],
   781→                spec["activation"], spec["dropout"],
   782→            )
   783→            network.load_state_dict(
   784→                {key: torch.from_numpy(value) for key, value in self._weights.items()}
   785→            )
   786→            self._network = network
   787→        return network
   788→
   789→    def _forward(self, X: Any) -> np.ndarray:
   790→        """Standardise ``X`` and forward it through the network (eval mode)."""
   791→        self._check_fitted()
   792→        self._check_feature_names(X)
   793→        X_arr = self._check_X(X)
   794→        if X_arr.shape[1] != self.n_features_in_:
   795→            # Without this guard a wrong-width X silently *broadcasts*
   796→            # against the per-feature mean/std (NumPy pads the missing
   797→            # columns) and yields garbage predictions.
   798→            raise DataValidationError(
   799→                f"X has {X_arr.shape[1]} feature column(s), but "
   800→                f"{type(self).__name__} was fitted with "
   801→                f"{self.n_features_in_} — pass the same features as to fit()"
   802→            )
   803→        _require_torch()  # raises the actionable DependencyError
   804→        import torch
   805→
   806→        network = self._ensure_network()
   807→        network.eval()  # disable dropout at inference time
   808→        X_scaled = (X_arr - self._mean_) / self._std_
   809→        with torch.no_grad():
   810→            out = network(torch.as_tensor(X_scaled, dtype=torch.float32))
   811→        return out.numpy()
   812→
   813→    # -- pickle support ----------------------------------------------------
   814→    def __getstate__(self) -> dict[str, Any]:
   815→        """Drop the live torch network from pickles (rebuilt on demand).
   816→
   817→        Only the NumPy weight dict is serialised, so pickled estimators
   818→        stay portable across torch versions — and loadable even where
   819→        torch is absent (torch is required again only to predict).
   820→        """
   821→        state = self.__dict__.copy()
   822→        state["_network"] = None
   823→        return state

   829→class DeepMLPClassifier(ClassifierMixin, _BaseDeepEstimator):
   830→    """Deep feed-forward neural-network classifier (PyTorch).
   831→
   832→    A fully connected network trained with minibatch Adam and
   833→    weight-decay regularisation, early-stopped on a stratified
   834→    validation split.  Multiclass (K > 2) is fully supported: the
   835→    network has K outputs and ``predict_proba`` returns a full softmax
   836→    matrix aligned with ``classes_``.
   837→
   838→    Parameters
   839→    ----------
   840→    hidden_layers:
   841→        Widths of the hidden layers, e.g. ``(64, 32)``.  An empty tuple
   842→        builds a single linear layer — i.e. a (regularised) multinomial
   843→        logistic regression.
   844→    dropout:
   845→        Dropout probability applied after every hidden activation
   846→        (``0`` disables it).
   847→    activation:
   848→        Hidden activation token: ``'relu'``, ``'tanh'``, ``'gelu'`` or
   849→        ``'sigmoid'``.
   850→    epochs:
   851→        Maximum number of training epochs (>= 1).
   852→    batch_size:
   853→        Minibatch size (>= 1).
   854→    learning_rate:
   855→        Adam step size (> 0).
   856→    weight_decay:
   857→        Adam L2 penalty (>= 0).
   858→    patience:
   859→        Early-stopping patience: stop when the validation loss has not
   860→        improved for this many epochs (>= 1).
   861→    val_split:
   862→        Fraction of the training rows held out for early stopping
   863→        (stratified).  ``0`` trains on everything and disables early
   864→        stopping.
   865→    random_state:
   866→        Integer seed — same seed + same data ⇒ identical fitted
   867→        predictions on CPU.
   868→    device:
   869→        ``'auto'`` (cuda if available else cpu), ``'cpu'`` or
   870→        ``'cuda'``.  Training only: fitted weights are always stored
   871→        on CPU.
   872→
   873→    Attributes
   874→    ----------
   875→    classes_:
   876→        Sorted class labels (strings label-encoded internally).
   877→    n_features_in_:
   878→        Number of input features seen at fit.
   879→    feature_names_in_:
   880→        Column names when ``X`` was a DataFrame (checked against
   881→        predicting DataFrames in names *and* order; an ndarray refit
   882→        drops them, sklearn-style).
   883→    history_:
   884→        List of per-epoch dicts ``{'epoch', 'train_loss', 'val_loss'}``
   885→        (``val_loss`` is ``None`` without a validation split).
   886→    n_iter_:
   887→        Number of epochs actually run (early stopping included).
   888→
   889→    Raises
   890→    ------
   891→    DependencyError
   892→        At fit/predict time when torch is missing.
   893→    ModelError
   894→        For invalid hyperparameters or use before ``fit``.
   895→    DataValidationError
   896→        For empty/non-numeric/mismatched data or single-class targets.
   897→
   898→    See Also
   899→    --------
   900→    DeepMLPRegressor, ResidualMLPClassifier : siblings in this module.
   901→
   902→    Notes
   903→    -----
   904→    Registry alias: ``build_model('dnn', 'classification')``.  Requires
   905→    the optional torch dependency — ``pip install 'pharmaml[deep]'``.
   906→    """
   907→
   908→    _is_classifier = True
   909→    _residual = False
   910→
   911→    def __init__(
   912→        self,
   913→        hidden_layers=(64, 32),
   914→        dropout=0.1,
   915→        activation="relu",
   916→        epochs=120,
   917→        batch_size=32,
   918→        learning_rate=1e-3,
   919→        weight_decay=1e-4,
   920→        patience=15,
   921→        val_split=0.15,
   922→        random_state=42,
   923→        device="auto",
   924→    ) -> None:
   925→        self.hidden_layers = hidden_layers
   926→        self.dropout = dropout
   927→        self.activation = activation
   928→        self.epochs = epochs
   929→        self.batch_size = batch_size
   930→        self.learning_rate = learning_rate
   931→        self.weight_decay = weight_decay
   932→        self.patience = patience
   933→        self.val_split = val_split
   934→        self.random_state = random_state
   935→        self.device = device
   936→
   937→    def predict(self, X: Any) -> np.ndarray:
   938→        """Predict class labels for ``X`` (argmax of the softmax).
   939→
   940→        Parameters
   941→        ----------
   942→        X:
   943→            Feature matrix — DataFrame or 2-D array-like, numeric.
   944→
   945→        Returns
   946→        -------
   947→        numpy.ndarray
   948→            Labels in the original ``classes_`` dtype (strings stay
   949→            strings).
   950→        """
   951→        logits = self._forward(X)
   952→        return self.classes_[np.argmax(logits, axis=1)]
   953→
   954→    def predict_proba(self, X: Any) -> np.ndarray:
   955→        """Class probabilities for ``X`` (softmax, columns == classes_).
   956→
   957→        Parameters
   958→        ----------
   959→        X:
   960→            Feature matrix — DataFrame or 2-D array-like, numeric.
   961→
   962→        Returns
   963→        -------
   964→        numpy.ndarray
   965→            Shape ``(n_samples, len(classes_))``; rows sum to 1.
   966→        """
   967→        logits = self._forward(X)
   968→        shifted = logits - logits.max(axis=1, keepdims=True)
   969→        exp = np.exp(shifted)
   970→        return exp / exp.sum(axis=1, keepdims=True)

   973→class DeepMLPRegressor(RegressorMixin, _BaseDeepEstimator):
   974→    """Deep feed-forward neural-network regressor (PyTorch).
   975→
   976→    Same architecture and training protocol as
   977→    :class:`DeepMLPClassifier` (see its docstring for the full
   978→    parameter/attribute reference) with a single linear output and
   979→    mean-squared-error loss; ``score`` returns R².
   980→
   981→    Registry alias: ``build_model('dnn', 'regression')``.  Requires
   982→    the optional torch dependency — ``pip install 'pharmaml[deep]'``.
   983→    """
   984→
   985→    _is_classifier = False
   986→    _residual = False
   987→
   988→    def __init__(
   989→        self,
   990→        hidden_layers=(64, 32),
   991→        dropout=0.1,
   992→        activation="relu",
   993→        epochs=120,
   994→        batch_size=32,
   995→        learning_rate=1e-3,
   996→        weight_decay=1e-4,
   997→        patience=15,
   998→        val_split=0.15,
   999→        random_state=42,
  1000→        device="auto",
  1001→    ) -> None:
  1002→        self.hidden_layers = hidden_layers
  1003→        self.dropout = dropout
  1004→        self.activation = activation
  1005→        self.epochs = epochs
  1006→        self.batch_size = batch_size
  1007→        self.learning_rate = learning_rate
  1008→        self.weight_decay = weight_decay
  1009→        self.patience = patience
  1010→        self.val_split = val_split
  1011→        self.random_state = random_state
  1012→        self.device = device
  1013→
  1014→    def predict(self, X: Any) -> np.ndarray:
  1015→        """Predict continuous targets for ``X``.
  1016→
  1017→        The network's z-scaled output is mapped back to the target's
  1018→        original scale (``_y_mean_`` / ``_y_std_`` were fitted on the
  1019→        training rows).
  1020→
  1021→        Parameters
  1022→        ----------
  1023→        X:
  1024→            Feature matrix — DataFrame or 2-D array-like, numeric.
  1025→
  1026→        Returns
  1027→        -------
  1028→        numpy.ndarray
  1029→            Shape ``(n_samples,)``.
  1030→        """
  1031→        # _check_fitted first: an unfitted estimator must raise the
  1032→        # documented ModelError, not an AttributeError on the scaling
  1033→        # attributes that fit() has not written yet.
  1034→        self._check_fitted()
  1035→        y_mean, y_std = self._y_mean_, self._y_std_
  1036→        if y_mean is None or y_std is None:
  1037→            # fit() always writes both for regressors; a missing pair means
  1038→            # the estimator was never fitted in regressor mode.
  1039→            raise ModelError(
  1040→                f"{type(self).__name__} is missing its target scaling -- "
  1041→                "was it fitted as a regressor?"
  1042→            )
  1043→        return self._forward(X).reshape(-1) * y_std + y_mean

  1049→class ResidualMLPClassifier(ClassifierMixin, _BaseDeepEstimator):
  1050→    """Residual (skip-connection) deep-network classifier (PyTorch).
  1051→

  1053→    an additive skip connection: identity when the block's input and

  1055→    analogue of a 1x1 convolution).  This lets gradients flow through
  1056→    deeper stacks than a plain feed-forward network; the default
  1057→    ``(64, 64, 64)`` stack exercises the identity-skip path.
  1058→

  1060→    determinism guarantee are identical to :class:`DeepMLPClassifier`
  1061→    (see its docstring) — only the block architecture differs.
  1062→
  1063→    Registry alias: ``build_model('dnn_residual', 'classification')``.
  1064→    Requires the optional torch dependency — ``pip install
  1065→    'pharmaml[deep]'``.
  1066→    """
  1067→
  1068→    _is_classifier = True
  1069→    _residual = True
  1070→
  1071→    def __init__(
  1072→        self,
  1073→        hidden_layers=(64, 64, 64),
  1074→        dropout=0.1,
  1075→        activation="relu",
  1076→        epochs=120,
  1077→        batch_size=32,
  1078→        learning_rate=1e-3,
  1079→        weight_decay=1e-4,
  1080→        patience=15,
  1081→        val_split=0.15,
  1082→        random_state=42,
  1083→        device="auto",
  1084→    ) -> None:
  1085→        self.hidden_layers = hidden_layers
  1086→        self.dropout = dropout
  1087→        self.activation = activation
