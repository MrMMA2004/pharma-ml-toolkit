   158→/** Min/max via loop — safe for very large CSVs (spread would overflow). */
   159→function extent(values: number[]): [number, number] {
   160→  let lo = values[0] ?? 0;
   161→  let hi = lo;
   162→  for (const v of values) {
   163→    if (v < lo) lo = v;
   164→    if (v > hi) hi = v;
   165→  }
   166→  return [lo, hi];
   167→}

   169→/** k-fold CV helper: runs `fitFold` per fold with its own RNG slot and
   170→ *  reduces fold values to mean ± population std. */
   171→function runCv(
   172→  n: number,
   173→  k: number,
   174→  effSeed: number,
   175→  metricName: string,
   176→  fitFold: (fold: number[], trainIdx: number[], slot: number) => number
   177→): MLTrainResponse["cv"] {
   178→  if (k <= 1) return undefined;
   179→  const cvRand = mulberry32((effSeed + 7919) >>> 0);
   180→  const folds = kFoldSplit(n, k, cvRand);
   181→  const foldResults: CVFold[] = folds.map((fold, i) => {
   182→    const foldSet = new Set(fold);
   183→    const trainIdx = Array.from({ length: n }, (_, j) => j).filter((j) => !foldSet.has(j));
   184→    return { fold: i + 1, value: fitFold(fold, trainIdx, i + 1) };
   185→  });
   186→  const values = foldResults.map((f) => f.value);
   187→  return {
   188→    k,
   189→    metricName,
   190→    folds: foldResults,
   191→    mean: mean(values),
   192→    std: populationStd(values),
   193→  };
   194→}

   196→export async function POST(request: NextRequest) {
   197→  let body: unknown;
   198→  try {
   199→    body = await request.json();
   200→  } catch {
   201→    return NextResponse.json({ error: "Request body must be valid JSON." }, { status: 400 });
   202→  }
   203→
   204→  const parsed = bodySchema.safeParse(body);
   205→  if (!parsed.success) {
   206→    return NextResponse.json(
   207→      {
   208→        error: "Invalid request body.",
   209→        detail: parsed.error.issues.map((i) => `${i.path.join(".") || "body"}: ${i.message}`),
   210→      },
   211→      { status: 400 }
   212→    );
   213→  }
   214→  const req = parsed.data;
   215→
   216→  /* Uploaded-CSV mode: parse + resolve columns up front so bad files/picks
   217→   * surface as actionable 400s instead of opaque 500s. */
   218→  let csvData: CsvRegressionData | CsvClassificationData | CsvMultiRegressionData | undefined;
   219→  let csvName: string | undefined;
   220→  let splitSeed = req.seed;
   221→  if (req.csv !== undefined) {
   222→    csvName = req.csvName ?? "pasted.csv";
   223→    const table = parseCsv(req.csv);
   224→    if (table.rows.length === 0) {
   225→      return NextResponse.json(
   226→        { error: "The uploaded CSV has no data rows below the header." },
   227→        { status: 400 }
   228→      );
   229→    }
   230→    try {
   231→      if (req.task === "regression") {
   232→        /* Multivariate when multiple feature columns are requested (or auto
   233→         * with ≥2 candidate features); single-feature keeps the x↔y chart. */
   234→        const multi =
   235→          (req.featureColumns?.length ?? 0) >= 2 ||
   236→          (req.featureColumns === undefined &&
   237→            !req.featureColumn &&
   238→            numericColumnNames(table).length >= 3);
   239→        csvData = multi
   240→          ? resolveMultiRegression(table, req.featureColumns, req.targetColumn)
   241→          : resolveRegression(table, req.featureColumn ?? req.featureColumns?.[0], req.targetColumn);
   242→      } else {
   243→        csvData = resolveClassification(
   244→          table,
   245→          req.featureColumn,
   246→          req.featureColumn2,
   247→          req.targetColumn
   248→        );
   249→      }
   250→    } catch (err) {
   251→      return NextResponse.json(
   252→        {
   253→          error: "Could not use the uploaded CSV.",
   254→          detail: [err instanceof Error ? err.message : String(err)],
   255→        },
   256→        { status: 400 }
   257→      );
   258→    }
   259→    /* Same file → same split: derive the split seed from the file content. */
   260→    splitSeed = hashCsvSeed(req.csv);
   261→  }
   262→
   263→  try {
   264→    let result: MLTrainResponse;
   265→    if (req.task === "regression") {
   266→      result =
   267→        csvData && "X" in csvData
   268→          ? runMultiRegression(req, csvData, csvName, splitSeed)
   269→          : runRegression(req, csvData as CsvRegressionData | undefined, csvName, splitSeed);
   270→    } else {
   271→      result = runClassification(
   272→        req,
   273→        csvData as CsvClassificationData | undefined,
   274→        csvName,
   275→        splitSeed
   276→      );
   277→    }
   278→    return NextResponse.json(result);
   279→  } catch (err) {
   280→    return NextResponse.json(
   281→      {
   282→        error: "Training failed.",
   283→        detail: err instanceof Error ? err.message : String(err),
   284→      },
   285→      { status: 500 }
   286→    );
   287→  }
   288→}

   294→/** GP diagnostic summary: the kernel readout plus the honest held-out
   295→ *  uncertainty stats (coverage of the ±1.96σ band and the median σ). */
   296→function gpSummary(
   297→  fit: RegFit,
   298→  yTrue: number[],
   299→  yPred: number[],
   300→  sigmaAt: (i: number) => number
   301→): MLTrainResponse["gp"] {
   302→  if (!fit.gp || !fit.predictStd) return undefined;
   303→  const sigmas = yTrue.map((_, i) => sigmaAt(i));
   304→  let covered = 0;
   305→  for (let i = 0; i < yTrue.length; i++) {
   306→    if (Math.abs(yTrue[i] - yPred[i]) <= 1.96 * (sigmas[i] ?? 0)) covered++;
   307→  }
   308→  const sorted = [...sigmas].sort((a, b) => a - b);
   309→  const mid = Math.floor((sorted.length - 1) / 2);
   310→  const medianSigma = sorted.length
   311→    ? (sorted[mid]! + sorted[Math.ceil((sorted.length - 1) / 2)]!) / 2
   312→    : 0;
   313→  return {
   314→    ...fit.gp,
   315→    coverage95: yTrue.length ? covered / yTrue.length : 0,
   316→    medianSigma,
   317→  };
   318→}

   320→function runRegression(
   321→  req: TrainBody,
   322→  csv?: CsvRegressionData,
   323→  csvName?: string,
   324→  splitSeed?: number
   325→): MLTrainResponse {
   326→  let xs: number[];
   327→  let ys: number[];
   328→  let source: CsvSourceMeta | undefined;
   329→  let features: string[];
   330→  let featureNames: string[];
   331→  if (csv) {
   332→    /* Raw feature units — NO full-column standardization here. Every
   333→     * fitter (fitLinearRegression, fitRegressionModel, ...) standardizes
   334→     * internally on the statistics of the rows it is given, i.e. the
   335→     * TRAIN split only. Pre-scaling with mean/std of the whole column
   336→     * would push held-out test rows into the feature geometry (the
   337→     * train/test leakage this toolkit's audit found in the original
   338→     * rar scripts — scaler.fit_transform before train_test_split). */
   339→    xs = csv.x;
   340→    ys = csv.y; // target kept in original units
   341→    source = {
   342→      kind: "csv",
   343→      fileName: csvName,
   344→      columns: csv.columns,
   345→      featureColumns: [csv.feature],
   346→      targetColumn: csv.target,
   347→      nRows: csv.x.length,
   348→      droppedRows: csv.droppedRows,
   349→    };
   350→    features = [csv.feature];
   351→    featureNames = [csv.feature];
   352→  } else {
   353→    const rand = mulberry32(req.seed);
   354→    ({ xs, ys } = makeRegressionData(req.samples, req.noise, rand));
   355→    features = ["x"];
   356→    featureNames = ["x"];
   357→  }
   358→  const effSeed = splitSeed ?? req.seed;
   359→  const rand = mulberry32(effSeed);
   360→  const n = xs.length;
   361→  const { train, test } = shuffledSplit(n, req.testSize, rand);
   362→
   363→  const hp = resolveHyper(req);
   364→  /* GP: when the request leaves all three kernel hyperparameters unset,
   365→   *  the fit LEARNS them by the deterministic LML grid search (the web
   366→   *  mirror of the library's marginal-likelihood-tuned kernel). */
   367→  const directives =
   368→    req.lengthScale === undefined && req.signalStd === undefined && req.noiseStd === undefined
   369→      ? { learnGpKernel: true }
   370→      : {};
   371→  const isLegacy = req.model === "linear" || req.model === "polynomial" || req.model === "ridge";
   372→  /* Legacy single-feature models keep the exact original call (degree for
   373→   * polynomial, ridge alpha on the diagonal) so responses stay
   374→   * byte-compatible; every other model fits over the 1-column matrix. */
   375→  const degree = req.model === "polynomial" ? 2 : 1;
   376→  const effAlpha = req.model === "ridge" ? hp.alpha : 0;
   377→
   378→  const trainX = train.map((i) => [xs[i] ?? 0]);
   379→  const trainY = train.map((i) => ys[i] ?? 0);
   380→
   381→  const fit: RegFit = isLegacy
   382→    ? (() => {
   383→        const m = fitLinearRegression(trainX.map((r) => r[0] ?? 0), trainY, {
   384→          alpha: effAlpha,
   385→          degree,
   386→        });
   387→        return {
   388→          predict: (row) => m.predict(row[0] ?? 0),
   389→          coefficients: {
   390→            intercept: m.w[0] ?? 0,
   391→            coefs: m.w.slice(1),
   392→            standardized: true,
   393→          },
   394→        };
   395→      })()
   396→    : fitRegressionModel(
   397→        req.model,
   398→        trainX,
   399→        trainY,
   400→        hp,
   401→        modelRng(effSeed, 0),
   402→        featureNames,
   403→        directives
   404→      );
   405→
   406→  const yTrue = test.map((i) => ys[i] ?? 0);
   407→  const yPred = test.map((i) => fit.predict([xs[i] ?? 0]));
   408→  const metrics = regressionMetrics(yTrue, yPred);
   409→  const hasStd = typeof fit.predictStd === "function";
   410→  const gp = hasStd
   411→    ? gpSummary(fit, yTrue, yPred, (i) => fit.predictStd!([xs[i] ?? 0]))
   412→    : undefined;
   413→
   414→  // Smooth fitted curve across the full observed x range; GP runs carry
   415→  // the ±1.96σ band the uncertainty chart fills.
   416→  const [xMin, xMax] = extent(xs);
   417→  const curve = Array.from({ length: 100 }, (_, i) => {
   418→    const x = xMin + ((xMax - xMin) * i) / 99;
   419→    const yhat = fit.predict([x]);
   420→    if (!hasStd) return { x, yhat };
   421→    const s = fit.predictStd!([x]);
   422→    return { x, yhat, lo: yhat - 1.96 * s, hi: yhat + 1.96 * s };
   423→  });
   424→
   425→  // k-fold CV (R² per fold) with an independent, seeded shuffle.
   426→  const cv = runCv(n, req.cv, effSeed, "R²", (fold, trainIdx, slot) => {
   427→    if (isLegacy) {
   428→      const foldModel = fitLinearRegression(
   429→        trainIdx.map((i) => xs[i] ?? 0),
   430→        trainIdx.map((i) => ys[i] ?? 0),
   431→        { alpha: effAlpha, degree }
   432→      );
   433→      const fTrue = fold.map((i) => ys[i] ?? 0);
   434→      const fPred = fold.map((i) => foldModel.predict(xs[i] ?? 0));
   435→      return regressionMetrics(fTrue, fPred).r2;
   436→    }
   437→    const foldFit = fitRegressionModel(
   438→      req.model,
   439→      trainIdx.map((i) => [xs[i] ?? 0]),
   440→      trainIdx.map((i) => ys[i] ?? 0),
   441→      hp,
   442→      modelRng(effSeed, slot),
   443→      featureNames,
   444→      directives
   445→    );
   446→    const fTrue = fold.map((i) => ys[i] ?? 0);
   447→    const fPred = fold.map((i) => foldFit.predict([xs[i] ?? 0]));
   448→    return regressionMetrics(fTrue, fPred).r2;
   449→  });
   450→
   451→  return {
   452→    task: "regression",
   453→    model: req.model,
   454→    metrics: { r2: metrics.r2, rmse: metrics.rmse, mae: metrics.mae },
   455→    cv,
   456→    ...(fit.coefficients ? { coefficients: fit.coefficients } : {}),
   457→    ...(fit.featureImportance ? { featureImportance: fit.featureImportance } : {}),
   458→    ...(fit.modelParams ? { modelParams: fit.modelParams } : {}),
   459→    ...(fit.vip ? { vip: { ...fit.vip, threshold: 1 } } : {}),
   460→    ...(gp ? { gp } : {}),
   461→    ...(fit.hsvr ? { hsvr: fit.hsvr } : {}),
   462→    series: {
   463→      test: test.map((i) => {
   464→        const point: { x: number; y: number; err?: number } = {
   465→          x: xs[i] ?? 0,
   466→          y: ys[i] ?? 0,
   467→        };
   468→        if (hasStd) point.err = 1.96 * fit.predictStd!([xs[i] ?? 0]);
   469→        return point;
   470→      }),
   471→      curve,
   472→    },
   473→    datasetMeta: {
   474→      nTotal: n,
   475→      nTrain: train.length,
   476→      nTest: test.length,
   477→      noise: csv ? 0 : req.noise,
   478→      seed: csv ? 0 : req.seed,
   479→      features,
   480→      ...(source ? { source } : {}),
   481→    },
   482→  };
   483→}
   484→
   485→/* ------------------------------------------------------------------ */
   486→/* Multivariate regression (uploaded CSV, k ≥ 2 features)              */
   487→/* ------------------------------------------------------------------ */
   488→
   489→/**
   490→ * Multivariate CSV regression. The chart becomes predicted-vs-actual:
   491→ * series.test carries {x: ŷ, y: actual} and `curve` is the perfect-prediction
   492→ * identity line, so the existing chart renderer works unchanged.
   493→ */
   494→function runMultiRegression(
   495→  req: TrainBody,
   496→  csv: CsvMultiRegressionData,
   497→  csvName: string | undefined,
   498→  splitSeed: number
   499→): MLTrainResponse {
   500→  const effSeed = splitSeed ?? req.seed;
   501→  const rand = mulberry32(effSeed);
   502→  const n = csv.y.length;
   503→  const { train, test } = shuffledSplit(n, req.testSize, rand);
   504→
   505→  const hp = resolveHyper(req);
   506→  const directives =
   507→    req.lengthScale === undefined && req.signalStd === undefined && req.noiseStd === undefined
   508→      ? { learnGpKernel: true }
   509→      : {};
   510→  const trainX = train.map((i) => csv.X[i] ?? []);
   511→  const trainY = train.map((i) => csv.y[i] ?? 0);
   512→  const fit = fitRegressionModel(
   513→    req.model,
   514→    trainX,
   515→    trainY,
   516→    hp,
   517→    modelRng(effSeed, 0),
   518→    csv.features,
   519→    directives
   520→  );
   521→
   522→  const yTrue = test.map((i) => csv.y[i] ?? 0);
   523→  const yPred = test.map((i) => fit.predict(csv.X[i] ?? []));
   524→  const metrics = regressionMetrics(yTrue, yPred);
   525→  const hasStd = typeof fit.predictStd === "function";
   526→  const gp = hasStd
   527→    ? gpSummary(fit, yTrue, yPred, (i) => fit.predictStd!(csv.X[i] ?? []))
   528→    : undefined;
   529→
   530→  /* Perfect-prediction identity line over the combined ŷ/y extent. */
   531→  const all = [...yTrue, ...yPred];
   532→  const [lo, hi] = extent(all);
   533→  const curve = Array.from({ length: 100 }, (_, i) => {
   534→    const v = lo + ((hi - lo) * i) / 99;
   535→    return { x: v, yhat: v };
   536→  });
   537→
   538→  const source: CsvSourceMeta = {
   539→    kind: "csv",
   540→    fileName: csvName,
   541→    columns: csv.columns,
   542→    featureColumns: csv.features,
   543→    targetColumn: csv.target,
   544→    nRows: n,
   545→    droppedRows: csv.droppedRows,
   546→  };
   547→
   548→  // k-fold CV (R² per fold) with an independent, seeded shuffle.
   549→  const cv = runCv(n, req.cv, effSeed, "R²", (fold, trainIdx, slot) => {
   550→    const foldFit = fitRegressionModel(
   551→      req.model,
   552→      trainIdx.map((i) => csv.X[i] ?? []),
   553→      trainIdx.map((i) => csv.y[i] ?? 0),
   554→      hp,
   555→      modelRng(effSeed, slot),
   556→      csv.features,
   557→      directives
   558→    );
   559→    const fTrue = fold.map((i) => csv.y[i] ?? 0);
   560→    const fPred = fold.map((i) => foldFit.predict(csv.X[i] ?? []));
   561→    return regressionMetrics(fTrue, fPred).r2;
   562→  });
   563→
   564→  return {
   565→    task: "regression",
   566→    model: req.model,
   567→    metrics: { r2: metrics.r2, rmse: metrics.rmse, mae: metrics.mae },
   568→    cv,
   569→    ...(fit.coefficients ? { coefficients: fit.coefficients } : {}),
   570→    ...(fit.featureImportance ? { featureImportance: fit.featureImportance } : {}),
   571→    ...(fit.modelParams ? { modelParams: fit.modelParams } : {}),
   572→    ...(fit.vip ? { vip: { ...fit.vip, threshold: 1 } } : {}),
   573→    ...(gp ? { gp } : {}),
   574→    ...(fit.hsvr ? { hsvr: fit.hsvr } : {}),
   575→    series: {
   576→      /* ŷ ± 1.96σ rides on each test point as `err` — the whiskers the
   577→       * predicted-vs-actual chart draws on the ŷ axis. */
   578→      test: yPred.map((yhat, i) => {
   579→        const point: { x: number; y: number; err?: number } = {
   580→          x: yhat,
   581→          y: yTrue[i] ?? 0,
   582→        };
   583→        if (hasStd) point.err = 1.96 * fit.predictStd!(csv.X[i] ?? []);
   584→        return point;
   585→      }),
   586→      curve,
   587→    },
   588→    datasetMeta: {
   589→      nTotal: n,
   590→      nTrain: train.length,
   591→      nTest: test.length,
   592→      noise: 0,
   593→      seed: 0,
   594→      features: csv.features,
   595→      source,
   596→    },
   597→  };
   598→}

   600→/* ------------------------------------------------------------------ */
   601→/* Classification                                                      */
   602→/* ------------------------------------------------------------------ */
   603→
   604→function runClassification(
   605→  req: TrainBody,
   606→  csv?: CsvClassificationData,
   607→  csvName?: string,
   608→  splitSeed?: number
   609→): MLTrainResponse {
   610→  let pts: Array<{ x: number; y: number; label: number }>;
   611→  let source: CsvSourceMeta | undefined;
   612→  let features: string[];
   613→  let featureNames: string[];
   614→  if (csv) {
   615→    /* Raw feature units — the classification fitters standardize on
   616→     * train statistics internally (fitLogisticRegression,
   617→     * fitKnnClassifier, ...), so scaling the full column here would
   618→     * leak held-out rows into the kNN metric / RBF kernel geometry
   619→     * (the same train/test contamination audited out of the rar). */
   620→    const { x1, x2, label } = csv;
   621→    pts = x1.map((x, i) => ({ x, y: x2[i] ?? 0, label: label[i] ?? 0 }));
   622→    source = {
   623→      kind: "csv",
   624→      fileName: csvName,
   625→      columns: csv.columns,
   626→      featureColumns: [csv.feature1, csv.feature2],
   627→      targetColumn: csv.target,
   628→      nRows: csv.x1.length,
   629→      droppedRows: csv.droppedRows,
   630→      labelMap: csv.labelMap,
   631→    };
   632→    features = [csv.feature1, csv.feature2];
   633→    featureNames = [csv.feature1, csv.feature2];
   634→  } else {
   635→    const genRand = mulberry32(req.seed);
   636→    pts = makeMoons(req.samples, req.noise, genRand);
   637→    features = ["x1", "x2"];
   638→    featureNames = ["x1", "x2"];
   639→  }
   640→  const effSeed = splitSeed ?? req.seed;
   641→  const rand = mulberry32(effSeed);
   642→  const n = pts.length;
   643→  const { train, test } = shuffledSplit(n, req.testSize, rand);
   644→
   645→  const hp = resolveHyper(req);
   646→  /* GP classification: when the request leaves both kernel slots unset,
   647→   *  the fit LEARNS (ℓ, σf) by the Laplace-lml grid search (the
   648→   *  classification twin of the regression GP's learn directive). */
   649→  const directives: FitDirectives =
   650→    req.lengthScale === undefined && req.signalStd === undefined
   651→      ? { learnGpClsKernel: true }
   652→      : {};
   653→  const trainX = train.map((i) => [pts[i]?.x ?? 0, pts[i]?.y ?? 0]);
   654→  const trainLabels = train.map((i) => pts[i]?.label ?? 0);
   655→  const fit = fitClassificationModel(
   656→    req.model,
   657→    trainX,
   658→    trainLabels,
   659→    hp,
   660→    modelRng(effSeed, 0),
   661→    featureNames,
   662→    directives
   663→  );
   664→
   665→  const yTrue = test.map((i) => pts[i]?.label ?? 0);
   666→  const yPred = test.map((i) => fit.predict01(pts[i]?.x ?? 0, pts[i]?.y ?? 0));
   667→  const metrics = classificationMetrics(yTrue, yPred);
   668→
   669→  /* GP classification: the honest probabilistic readout — ROC AUC (rank
   670→   *  statistic over the predicted probabilities), log-loss and Brier on
   671→   *  the held-out rows, plus the mean latent σ the 5-erf integral saw. */
   672→  let gpCls: MLTrainResponse["gpCls"];
   673→  if (fit.gpCls && fit.predictProba) {
   674→    const probas = test.map((i) => fit.predictProba!(pts[i]?.x ?? 0, pts[i]?.y ?? 0));
   675→    gpCls = {
   676→      ...fit.gpCls,
   677→      rocAuc: rankAuc(yTrue, probas),
   678→      logLoss:
   679→        -yTrue.reduce(
   680→          (acc, y, i) =>
   681→            acc +
   682→            (y === 1 ? Math.log(Math.max(probas[i] ?? 0.5, 1e-12)) : Math.log(Math.max(1 - (probas[i] ?? 0.5), 1e-12))),
   683→          0
   684→        ) / Math.max(yTrue.length, 1),
   685→      brier: yTrue.reduce((acc, y, i) => acc + (y - (probas[i] ?? 0.5)) ** 2, 0) / Math.max(yTrue.length, 1),
   686→    };
   687→  }
   688→
   689→  // Decision boundary: dense grid classified by the trained model.
   690→  // GP classification also carries the calibrated probability so the
   691→  // chart can fade boundary dots by confidence (|p − 0.5|).
   692→  const xs = pts.map((p) => p.x);
   693→  const ys = pts.map((p) => p.y);
   694→  const [xMin, xMax] = extent(xs);
   695→  const [yMin, yMax] = extent(ys);
   696→  const G = 60;
   697→  const boundary: Array<{ x: number; y: number; pred: number; proba?: number }> = [];
   698→  for (let gi = 0; gi < G; gi++) {
   699→    for (let gj = 0; gj < G; gj++) {
   700→      const x = (xMin - 0.5) + ((xMax + 0.5 - (xMin - 0.5)) * gi) / (G - 1);
   701→      const y = (yMin - 0.5) + ((yMax + 0.5 - (yMin - 0.5)) * gj) / (G - 1);
   702→      boundary.push({ x, y, pred: fit.predict01(x, y) });
   703→    }
   704→  }
   705→  if (fit.predictProba) {
   706→    for (const p of boundary) p.proba = fit.predictProba(p.x, p.y);
   707→  }
   708→
   709→  // k-fold CV (accuracy per fold).
   710→  const cv = runCv(n, req.cv, effSeed, "Accuracy", (fold, trainIdx, slot) => {
   711→    const foldFit = fitClassificationModel(
   712→      req.model,
   713→      trainIdx.map((i) => [pts[i]?.x ?? 0, pts[i]?.y ?? 0]),
   714→      trainIdx.map((i) => pts[i]?.label ?? 0),
   715→      hp,
   716→      modelRng(effSeed, slot),
   717→      featureNames,
   718→      directives
   719→    );
   720→    let correct = 0;
   721→    for (const i of fold) {
   722→      const p = pts[i] ?? { x: 0, y: 0, label: 0 };
   723→      const pred = foldFit.predict01(p.x, p.y);
   724→      if (pred === p.label) correct++;
   725→    }
   726→    return correct / fold.length;
   727→  });
   728→
   729→  return {
   730→    task: "classification",
   731→    model: req.model,
   732→    metrics: {
   733→      accuracy: metrics.accuracy,
   734→      precision: metrics.precision,
   735→      recall: metrics.recall,
   736→      f1: metrics.f1,
   737→    },
   738→    cv,
   739→    ...(fit.coefficients ? { coefficients: fit.coefficients } : {}),
   740→    ...(fit.featureImportance ? { featureImportance: fit.featureImportance } : {}),
   741→    ...(fit.modelParams ? { modelParams: fit.modelParams } : {}),
   742→    ...(gpCls ? { gpCls } : {}),
   743→    series: {
   744→      test: test.map((i) => ({
   745→        x: pts[i]?.x ?? 0,
   746→        y: pts[i]?.y ?? 0,
   747→        label: pts[i]?.label ?? 0,
   748→      })),
   749→      boundary,
   750→    },
   751→    datasetMeta: {
   752→      nTotal: n,
   753→      nTrain: train.length,
   754→      nTest: test.length,
   755→      noise: csv ? 0 : req.noise,
   756→      seed: csv ? 0 : req.seed,
   757→      features,
   758→      ...(source ? { source } : {}),
   759→    },
   760→  };
   761→}
