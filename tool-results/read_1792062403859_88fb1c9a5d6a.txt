   531→   531→def _smi_bridge_frame(path: Path):
   532→   532→    """Read a ``.smi`` file into a SMILES/ID table (CLI-internal bridge).
   533→   533→
   534→   534→    Lets ``--data x.smi --to sdf|smi`` reuse the table converters

   540→   540→

   543→   543→        for line in handle:
   544→   544→            parts = line.strip().split()
   545→   545→            if not parts:
   546→   546→                continue
   547→   547→            rows.append((parts[0], " ".join(parts[1:])))

   549→   549→    if frame.empty:
   550→   550→        raise DataValidationError(f"no SMILES lines found in {path}")
   551→   551→    return frame

   555→   555→    from pharmaml.chem.converters import (

   596→   596→        name_col = args.name_col or ("name" if "name" in frame.columns else None)
   597→   597→        written = table_to_smi(
   598→   598→            frame, out, smiles_col=args.smiles_col, name_col=name_col
   599→   599→        )

   601→   601→        return 0
   602→   602→    else:  # CSV/TSV/Excel table input

   611→   611→            written = write_table(frame, out)

   614→   614→        report = table_to_sdf(
   615→   615→            src,
   616→   616→            out,
   617→   617→            smiles_col=args.smiles_col,
