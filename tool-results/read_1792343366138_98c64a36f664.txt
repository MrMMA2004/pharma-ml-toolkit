    91→function CommandStrip({
    92→  command,
    93→  context = "terminal",
    94→  hint,
    95→  className,
    96→}: {
    97→  command: string;
    98→  context?: string;
    99→  hint?: string;
   100→  className?: string;
   101→}) {
   102→  const [copied, setCopied] = useState(false);
   103→
   104→  async function copy() {
   105→    const ok = await copyToClipboard(command);
   106→    if (ok) {
   107→      setCopied(true);
   108→      toast.success("Command copied", { description: command });
   109→      window.setTimeout(() => setCopied(false), 1600);
   110→    } else {
   111→      toast.error("Could not copy — your browser blocked clipboard access");
   112→    }
   113→  }
   114→
   115→  return (
   116→    <div className={cn("overflow-hidden rounded-lg border bg-card text-left", className)}>
   117→      <div className="flex items-center justify-between gap-2 border-b bg-muted/40 py-1 pl-2.5 pr-1.5">

   121→        </span>
   122→        <button
   123→          type="button"
   124→          onClick={() => void copy()}
   125→          aria-label={`Copy command to clipboard: ${command}`}
   126→          aria-pressed={copied}
   127→          className="flex size-6 shrink-0 items-center justify-center rounded-md text-muted-foreground transition-colors hover:bg-muted hover:text-foreground focus-visible:outline-none focus-visible:ring-2 focus-visible:ring-ring"

   130→            <Check className="size-3.5 text-primary" aria-hidden />

   132→            <Copy className="size-3.5" aria-hidden />

   198→function countFiles(node: TreeNode): number {
   199→  if (node.type === "file") return 1;
   200→  return (node.children ?? []).reduce((acc, child) => acc + countFiles(child), 0);
   201→}

   203→/** Look up a node in the tree by its toolkit-relative path. */
   204→function findNode(node: TreeNode, path: string): TreeNode | null {
   205→  if (node.path === path) return node;
   206→  for (const child of node.children ?? []) {
   207→    const hit = findNode(child, path);
   208→    if (hit) return hit;
   209→  }
   210→  return null;
   211→}

   280→function collectDirSummary(
   281→  root: TreeNode
   282→): Array<{ name: string; files: number; size: number; note: string }> {
   283→  const notes: Record<string, string> = {
   284→    src: "the pharmaml package (data, chem, ml, pharma, docking, viz, utils + CLI)",
   285→    tests: "1745 pytest cases across all domains",
   286→    examples: "seven executed notebooks + six no-output twins + runnable demos",
   287→    ".github": "GitHub Actions CI matrix (Python 3.10–3.12)",
   288→    docs: "module index — auto-generated API reference",
   289→    scratch: "idempotent build & patch scripts (notebooks, docs index)",
   290→  };
   291→  const out: Array<{ name: string; files: number; size: number; note: string }> = [];
   292→  let rootFiles = 0;
   293→  let rootBytes = 0;
   294→  for (const child of root.children ?? []) {
   295→    if (child.type === "dir") {
   296→      out.push({
   297→        name: child.name,
   298→        files: countFiles(child),
   299→        size: child.size,
   300→        note: notes[child.name] ?? "",
   301→      });
   302→    } else {
   303→      rootFiles++;
   304→      rootBytes += child.size;
   305→    }
   306→  }
   307→  if (rootFiles > 0) {
   308→    out.push({
   309→      name: "(root)",
   310→      files: rootFiles,
   311→      size: rootBytes,
   312→      note: "README, CHANGELOG, VERSION, playground.py, pyproject.toml, Dockerfile, requirements.txt",
   313→    });
   314→  }
   315→  return out;
   316→}

   358→  async function download() {
   359→    setDownloading(true);
   360→    try {
   361→      const res = await fetch("/api/toolkit/download");
   362→      if (!res.ok) {
   363→        let message = `HTTP ${res.status}`;
   364→        try {
   365→          const data: unknown = await res.json();
   366→          if (typeof data === "object" && data !== null && "error" in data) {
   367→            message = String((data as { error: unknown }).error);
   368→          }
   369→        } catch {
   370→          /* body wasn't JSON — keep the status message */
   371→        }
   372→        throw new Error(message);
   373→      }
   374→      const blob = await res.blob();
   375→      if (blob.size < 1024) {
   376→        throw new Error("The archive came back suspiciously small — try again.");
   377→      }
   378→      const url = URL.createObjectURL(blob);
   379→      const anchor = document.createElement("a");
   380→      anchor.href = url;
   381→      anchor.download = zipName;
   382→      document.body.appendChild(anchor);
   383→      anchor.click();
   384→      document.body.removeChild(anchor);
   385→      URL.revokeObjectURL(url);
   386→      toast.success("Download started", {
   387→        description: `${zipName} · ${fmtBytes(blob.size)}`,
   388→      });
   389→    } catch (err) {
   390→      toast.error("Download failed", {
   391→        description: err instanceof Error ? err.message : "Unknown error.",
   392→      });
   393→    } finally {
   394→      setDownloading(false);
   395→    }
