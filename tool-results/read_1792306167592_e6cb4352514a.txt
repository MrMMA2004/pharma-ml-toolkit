   852→class TestNotebookArtifacts:
   853→    """The shipped notebooks must stay executed and error-free."""
   854→
   855→    @staticmethod
   856→    def _read(example_name: str):
   857→        import nbformat
   858→
   859→        path = Path(__file__).resolve().parents[1] / "examples" / example_name
   860→        assert path.is_file(), f"examples/{example_name} missing from the ZIP"
   861→        return nbformat.read(path, as_version=4)
   862→
   863→    @staticmethod
   864→    def _stats(nb):
   865→        code_cells = [c for c in nb.cells if c.cell_type == "code"]
   866→        errors = [
   867→            o for c in code_cells for o in c.get("outputs", [])
   868→            if o.output_type == "error"
   869→        ]
   870→        figures = sum(
   871→            1 for c in code_cells for o in c.get("outputs", [])
   872→            if o.get("data", {}).get("image/png")
   873→        )
   874→        executed = sum(1 for c in code_cells if c.get("execution_count"))
   875→        return code_cells, errors, figures, executed
   876→
   877→    def test_complete_ml_workflow_notebook(self):
   878→        nb = self._read("complete_ml_workflow.ipynb")
   879→        code_cells, errors, figures, executed = self._stats(nb)
   880→        assert len(code_cells) >= 30
   881→        assert executed == len(code_cells), "notebook has unexecuted code cells"
   882→        assert not errors, f"notebook has {len(errors)} committed error output(s)"
   883→        assert figures >= 14, "notebook lost its inline figures"
   884→
   885→        sources = "\n".join(c.source for c in code_cells)
   886→        # The feature-calculation section must stay shipped (§2).
   887→        for symbol in (
   888→            "compute_descriptors",
   889→            "compute_fingerprints",
   890→            "compute_maccs",
   891→            "tanimoto_matrix",
   892→            "plot_similarity_heatmap",
   893→        ):
   894→            assert symbol in sources, f"notebook lost its {symbol} section"
   895→        desc_cell = next(
   896→            (c for c in code_cells if "compute_descriptors" in c.source), None
   897→        )
   898→        assert desc_cell is not None
   899→        desc_text = "".join(
   900→            o.get("text", "") for o in desc_cell.get("outputs", [])
   901→            if o.get("output_type") == "stream"
   902→        )
   903→        assert "valid molecules" in desc_text, "descriptor cell lost its output"
   904→
   905→        # §7/§8 classification recipes: binary (2 classes) + multiclass
   906→        # with a selectable class count (N_CLASSES).  The demo tables load
   907→        # from examples/data/ (qsar_classification.csv / qsar_multiclass.csv)
   908→        # — never fabricated inline (see test_notebook_variants.py).
   909→        for symbol in (
   910→            'DATA_DIR / "qsar_classification.csv"',
   911→            'DATA_DIR / "qsar_multiclass.csv"',
