  1239→
  1240→        mode = self._aff_mode.get()

  1329→    def _compare_similarity(self) -> None:
  1330→        if not self._require_pharmaml() or not self._require("rdkit", "fingerprints"):
  1331→            return
  1332→        from pharmaml.chem import compute_fingerprints, tanimoto_from_bitarrays
  1333→
  1334→        first = self._tani_a.get().strip()
  1335→        second = self._tani_b.get().strip()
  1336→        if not first or not second:
  1337→            messagebox.showwarning("Missing SMILES", "Both molecules are needed.")
  1338→            return
  1339→        try:
  1340→            prints, valid = compute_fingerprints([first, second])
  1341→        except Exception as exc:  # noqa: BLE001
  1342→            self._show_error("Fingerprint failed", exc)
  1343→            return
  1344→        if len(valid) < 2:
  1345→            self._tani_result.configure(text="invalid SMILES")
  1346→            self._tani_verdict.configure(
  1347→                text="One of the two strings is not valid RDKit input."
  1348→            )
  1349→            return
  1350→        score = float(tanimoto_from_bitarrays(prints[0], prints[1]))
  1351→        if score >= 0.85:
  1352→            verdict = "near-identical fingerprints"
  1353→        elif score >= 0.6:
  1354→            verdict = "clearly related scaffolds"
  1355→        elif score >= 0.35:
  1356→            verdict = "loosely related"
  1357→        else:
  1358→            verdict = "essentially unrelated"
  1359→        self._tani_result.configure(text=f"{score:.3f}")
  1360→        self._tani_verdict.configure(text=f"{verdict} (1.0 = identical, 0.0 = nothing shared)")
