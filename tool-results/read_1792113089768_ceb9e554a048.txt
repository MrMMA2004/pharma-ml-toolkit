   340→    def test_multi_objective_non_finite_child_scores_veto(self) -> None:
   341→        # Regression: a NaN-returning child made the total NaN (``nan <= 0``
   342→        # is False, so the veto never fired) -- violating the documented
   343→        # [0, 1] total contract.  Non-finite children now count as 0.0,
   344→        # matching how the optimisers clamp misbehaving objectives.
   345→        class NonFiniteObjective:
   346→            def __init__(self, value: float) -> None:
   347→                self.value = value
   348→                self.name = f"child_{self.value}"
   349→
   350→            def score(self, smiles: str) -> float:
   351→                return self.value
   352→
   353→        nan_obj = MultiObjective(
   354→            [QedObjective(), NonFiniteObjective(float("nan"))]
   355→        )
   356→        total, breakdown = nan_obj.score_with_breakdown(ASPIRIN)
   357→        assert breakdown["child_nan"] == 0.0
   358→        assert total == 0.0
   359→
   360→        inf_obj = MultiObjective(
   361→            [QedObjective(), NonFiniteObjective(float("inf"))]
   362→        )
   363→        total, _ = inf_obj.score_with_breakdown(ASPIRIN)
   364→        assert total == 0.0
   365→
   366→        class NoneObjective:
   367→            name = "none_child"
   368→
   369→            def score(self, smiles: str) -> None:
   370→                return None
   371→
   372→        with pytest.raises(DataValidationError, match="non-numeric score"):
   373→            MultiObjective([QedObjective(), NoneObjective()]).score(ASPIRIN)
