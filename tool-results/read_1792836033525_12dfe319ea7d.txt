   134→/** Chart datum: recharts x/y plus the payload fields the tooltip shows. */
   135→interface ScatterDatum {
   136→  x: number;
   137→  y: number;
   138→  cellLine: string;
   139→  tissue: string;
   140→  value: number;
   141→  lnIc50: number;
   142→  auc: number;
   143→}
   144→
   145→function toDatum(p: BiomarkerPoint): ScatterDatum {
   146→  return {
   147→    x: p.value,
   148→    y: p.lnIc50,
   149→    cellLine: p.cellLine,
   150→    tissue: p.tissue,
   151→    value: p.value,
   152→    lnIc50: p.lnIc50,
   153→    auc: p.auc,
   154→  };
   155→}

   161→function StatCard({ label, value }: { label: string; value: string }) {
   162→  return (
   163→    <div className="rounded-lg border bg-card px-3 py-2.5 shadow-sm">
   164→      <p className="text-xs text-muted-foreground">{label}</p>
   165→      <p className="font-mono text-xl font-semibold tabular-nums">{value}</p>
   166→    </div>
   167→  );
   168→}
   169→
   170→/** Custom scatter tooltip: cell line + tissue + both values. */
   171→function ScatterTooltip({
   172→  active,
   173→  payload,
   174→}: {
   175→  active?: boolean;
   176→  payload?: Array<{ payload?: ScatterDatum }>;
   177→}) {
   178→  if (!active || !payload || payload.length === 0) return null;
   179→  const p = payload[0]?.payload;
   180→  if (!p) return null;
   181→  const color = TISSUE_COLORS[p.tissue] ?? CHART.stone;
   182→  return (
   183→    <div className="rounded-lg border bg-popover px-3 py-2 text-xs shadow-md">
   184→      <p className="font-mono font-semibold">{p.cellLine}</p>
   185→      <p className="mt-0.5 flex items-center gap-1.5 text-muted-foreground">
   186→        <span className="size-2 rounded-full" style={{ backgroundColor: color }} aria-hidden />
   187→        {p.tissue}
   188→      </p>
   189→      <p className="mt-1 font-mono tabular-nums">
   190→        expression {fmtNum(p.value, 3)} · LN_IC50 {fmtNum(p.lnIc50, 3)}
   191→      </p>
   192→      <p className="font-mono text-[11px] tabular-nums text-muted-foreground">
   193→        AUC {fmtNum(p.auc, 3)}
   194→      </p>
   195→    </div>
   196→  );
   197→}
   198→
   199→/** rho cell coloring: negative = sensitizing (emerald), positive = resisting (rose). */
   200→function RhoCell({ rho }: { rho: number }) {
   201→  const negative = rho < 0;
   202→  return (
   203→    <span className="inline-flex items-center gap-1.5 font-mono text-xs tabular-nums">
   204→      <span className={negative ? "text-emerald-600 dark:text-emerald-400" : "text-rose-600 dark:text-rose-400"}>
   205→        {negative ? <TrendingDown className="size-3" aria-hidden /> : <TrendingUp className="size-3" aria-hidden />}
   206→        {fmtNum(rho, 3)}
   207→      </span>
   208→      <Badge
   209→        variant="secondary"
   210→        className={cn(
   211→          "h-4 gap-0.5 px-1 text-[9px] font-medium",
   212→          negative
   213→            ? "bg-emerald-500/10 text-emerald-600 dark:text-emerald-400"
   214→            : "bg-rose-500/10 text-rose-600 dark:text-rose-400"
   215→        )}
   216→      >
   217→        {negative ? "sensitizing" : "resisting"}
   218→      </Badge>
   219→    </span>
   220→  );
   221→}

   227→export function BiologyPlayground() {
   228→  /* Transient panel state — same convention as the docking parser and the
   229→   * ligand reconciler (only the ML/PK setups persist in the store). */
   230→  const [compound, setCompound] = useState<BiologyCompound>(DEFAULTS.compound);
   231→  const [genes, setGenes] = useState(DEFAULTS.genes);
   232→  const [seed, setSeed] = useState(DEFAULTS.seed);
   233→  const [nCellLines, setNCellLines] = useState(DEFAULTS.nCellLines);
   234→  const [result, setResult] = useState<BiomarkersResponse | null>(null);
   235→  const [status, setStatus] = useState<"idle" | "running" | "done" | "error">("idle");
   236→  const [error, setError] = useState<string | null>(null);
   237→  /** Which gene the scatter shows: the top hit, or EGFR specifically. */
   238→  const [chartGene, setChartGene] = useState<"top" | "egfr">("top");
   239→
   240→  const running = status === "running";
   241→  const seedValid = /^\d{1,10}$/.test(seed.trim());
   242→
   243→  /** Tissue-grouped series (stable references for recharts). */
   244→  const series = useMemo(() => {
   245→    if (!result) return null;
   246→    const scatter: BiomarkerScatter =
   247→      chartGene === "egfr" && result.egfrScatter ? result.egfrScatter : result.topHit;
   248→    const byTissue = new Map<string, ScatterDatum[]>();
   249→    for (const p of scatter.points) {
   250→      const list = byTissue.get(p.tissue) ?? [];
   251→      list.push(toDatum(p));
   252→      byTissue.set(p.tissue, list);
   253→    }
   254→    return {
   255→      scatter,
   256→      groups: result.tissues
   257→        .filter((t) => byTissue.has(t))
   258→        .map((t) => ({ tissue: t, data: byTissue.get(t) ?? [] })),
   259→    };
   260→  }, [result, chartGene]);
   261→
   262→  function randomizeSeed() {
   263→    setSeed(String(Math.floor(Math.random() * 2_147_483_648)));
   264→  }
   265→
   266→  function resetSetup() {
   267→    setCompound(DEFAULTS.compound);
   268→    setGenes(DEFAULTS.genes);
   269→    setSeed(DEFAULTS.seed);
   270→    setNCellLines(DEFAULTS.nCellLines);
   271→    setStatus("idle");
   272→    setResult(null);
   273→    setError(null);
   274→    setChartGene("top");
   275→  }
   276→
   277→  async function discover() {
   278→    if (!seedValid) {
   279→      setError("Seed must be a non-negative integer (0 – 2147483647).");
   280→      return;
   281→    }
   282→    setStatus("running");
   283→    setError(null);
   284→    try {
   285→      const res = await fetch("/api/biology/biomarkers", {
   286→        method: "POST",
   287→        headers: { "Content-Type": "application/json" },
   288→        body: JSON.stringify({
   289→          compound,
   290→          genes,
   291→          seed: Number.parseInt(seed, 10) || 0,
   292→          nCellLines: Number.parseInt(nCellLines, 10) || 24,
   293→        }),
   294→      });
   295→      const data: unknown = await res.json();
   296→      if (!res.ok) {
   297→        const message =
   298→          typeof data === "object" && data !== null && "error" in data
   299→            ? `${String((data as { error: unknown }).error)}${
   300→                "detail" in data && Array.isArray((data as { detail: unknown }).detail)
   301→                  ? ` ${((data as { detail: unknown[] }).detail as string[]).join("; ")}`
   302→                  : ""
   303→              }`
   304→            : `HTTP ${res.status}`;
   305→        throw new Error(message);
   306→      }
   307→      const payload = data as BiomarkersResponse;
   308→      setResult(payload);
   309→      setStatus("done");
   310→      setChartGene("top");
   311→      toast.success("Biomarkers ranked", {
   312→        description:
   313→          payload.meta.plantedHit === "EGFR"
   314→            ? `${payload.meta.nGenes} genes screened — planted EGFR biomarker recovered at rank #1.`
   315→            : `${payload.meta.nGenes} genes screened — ${
   316→                payload.meta.nSignificant
   317→              } significant at FDR < 0.05.`,
   318→      });
   319→    } catch (err) {
   320→      setStatus("error");
   321→      const message = err instanceof Error ? err.message : "Discovery request failed.";
   322→      setError(message);
   323→      toast.error("Biomarker discovery failed", { description: message });
   324→    }
   325→  }
   326→
   327→  const panelLabel =
   328→    PANEL_SIZES.find((p) => p.value === nCellLines)?.label.split(" — ")[1] ?? `${nCellLines} lines`;
   329→
   330→  return (
   331→    <div className="grid gap-4 xl:grid-cols-[minmax(300px,360px)_minmax(0,1fr)]">
   332→      {/* ------------------------------ Controls ------------------------------ */}
   333→      <Card className="h-fit overflow-hidden py-0 shadow-sm xl:sticky xl:top-24">
   334→        <CardHeader className="border-b bg-muted/30 py-3">
   335→          <div className="flex items-center gap-2.5">
   336→            <span className="flex size-9 items-center justify-center rounded-lg bg-primary/10 text-primary">
   337→              <Dna className="size-4.5" aria-hidden />
   338→            </span>
   339→            <div>
   340→              <CardTitle className="text-base">Biomarker discovery</CardTitle>
   341→              <CardDescription className="text-xs">
   342→                Synthetic GDSC/CCLE panel — 24 cell lines × 10 tissues, planted EGFR signal
   343→              </CardDescription>
   344→            </div>
   345→          </div>
   346→        </CardHeader>
   347→        <CardContent className="space-y-5 pb-4">
   348→          {/* Compound */}
   349→          <div className="space-y-1.5">
   350→            <Label htmlFor="bio-compound" className="text-xs font-medium">
   351→              Compound
   352→            </Label>
   353→            <Select value={compound} onValueChange={(v) => setCompound(v as BiologyCompound)}>
   354→              <SelectTrigger id="bio-compound" className="h-10" aria-label="Screened compound">
   355→                {/* Short label in the trigger; the rich item (hint + badge)
   356→                 * only renders inside the dropdown itself. */}
   357→                <span className="truncate">
   358→                  {COMPOUNDS.find((c) => c.id === compound)?.label ?? compound}
   359→                </span>
   360→              </SelectTrigger>
   361→              <SelectContent>
   362→                {COMPOUNDS.map((c) => (
   363→                  <SelectItem key={c.id} value={c.id} textValue={c.label} className="items-start py-2">
   364→                    <span className="flex min-w-0 flex-col gap-0.5">
   365→                      <span className="text-sm">
   366→                        {c.label}
   367→                        {c.egfr ? (
   368→                          <span className="ml-1.5 rounded bg-emerald-500/10 px-1 py-0.5 text-[10px] font-medium text-emerald-600 dark:text-emerald-400">
   369→                            EGFR-linked
   370→                          </span>
   371→                        ) : null}
   372→                      </span>
   373→                      <span className="text-[11px] leading-snug text-muted-foreground">{c.hint}</span>
   374→                    </span>
   375→                  </SelectItem>
   376→                ))}
   377→              </SelectContent>
   378→            </Select>
   379→            <p className="text-[11px] leading-snug text-muted-foreground">
   380→              LN_IC50 = baseline + tissue effect + EGFR coefficient × EGFR z + noise σ 0.25
   381→            </p>
   382→          </div>
   383→
   384→          {/* Genes */}
   385→          <div className="space-y-2">
   386→            <div className="flex items-center justify-between">
   387→              <Label htmlFor="bio-genes" className="text-xs font-medium">
   388→                Genes screened
   389→              </Label>
   390→              <span className="font-mono text-xs tabular-nums text-muted-foreground">{genes}</span>
   391→            </div>
   392→            <Slider
   393→              id="bio-genes"
   394→              min={60}
   395→              max={600}
   396→              step={10}
   397→              value={[genes]}
   398→              onValueChange={(v) => setGenes(v[0] ?? genes)}
   399→            />
   400→            <p className="text-[11px] text-muted-foreground">
   401→              GENE_001… noise columns + EGFR + KRT8 (breast-hot)
   402→            </p>
   403→          </div>
   404→
   405→          {/* Seed + dice */}
   406→          <div className="space-y-1.5">
   407→            <Label htmlFor="bio-seed" className="text-xs font-medium">
   408→              Seed
   409→            </Label>
   410→            <div className="flex gap-2">
   411→              <Input
   412→                id="bio-seed"
   413→                inputMode="numeric"
   414→                value={seed}
   415→                onChange={(e) => setSeed(e.target.value)}
   416→                aria-invalid={seedValid ? undefined : true}
   417→                className={`h-10 font-mono text-sm tabular-nums ${seedValid ? "" : "border-destructive/60"}`}
   418→              />
   419→              <Button
   420→                type="button"
   421→                variant="outline"
   422→                size="icon"
   423→                className="size-10 shrink-0"
   424→                onClick={randomizeSeed}
   425→                disabled={running}
   426→                aria-label="Randomize the seed"
   427→                title="Randomize the seed"
   428→              >
   429→                <Dices className="size-4" aria-hidden />
   430→              </Button>
   431→            </div>
   432→            {!seedValid ? (
   433→              <p className="text-[11px] text-destructive">Non-negative integer.</p>
   434→            ) : null}
   435→          </div>
   436→
   437→          {/* Panel size */}
   438→          <div className="space-y-1.5">
   439→            <Label htmlFor="bio-panel" className="text-xs font-medium">
   440→              Cell line panel
   441→            </Label>
   442→            <Select value={nCellLines} onValueChange={setNCellLines}>
   443→              <SelectTrigger id="bio-panel" className="h-10" aria-label="Cell line panel size">
   444→                <SelectValue />
   445→              </SelectTrigger>
   446→              <SelectContent>
   447→                {PANEL_SIZES.map((p) => (
   448→                  <SelectItem key={p.value} value={p.value}>
   449→                    {p.label}
   450→                  </SelectItem>
   451→                ))}
   452→              </SelectContent>
   453→            </Select>
   454→          </div>
   455→
   456→          <div className="flex gap-2">
   457→            <Button
   458→              size="lg"
   459→              className="h-11 flex-1 gap-2"
   460→              onClick={() => void discover()}
   461→              disabled={running}
   462→            >
   463→              {running ? (
   464→                <>
   465→                  <Loader2 className="size-4 animate-spin" aria-hidden />
   466→                  Screening…
   467→                </>
   468→              ) : (
   469→                <>
   470→                  <Play className="size-4" aria-hidden />
   471→                  Discover biomarkers
   472→                </>
   473→              )}
   474→            </Button>
   475→            <Button
   476→              size="lg"
   477→              variant="outline"
   478→              className="h-11 shrink-0 gap-2"
   479→              onClick={resetSetup}
   480→              disabled={running}
   481→              aria-label="Reset the discovery setup to defaults"
   482→              title="Reset setup to defaults"
   483→            >
   484→              <RotateCcw className="size-4" aria-hidden />
   485→              <span className="hidden xl:inline">Reset</span>
   486→            </Button>
   487→          </div>
   488→
   489→          {error ? (
   490→            <Alert variant="destructive">
   491→              <TriangleAlert className="size-4" aria-hidden />
   492→              <AlertTitle>Request failed</AlertTitle>
   493→              <AlertDescription>{error}</AlertDescription>
   494→            </Alert>
   495→          ) : null}
   496→        </CardContent>
   497→      </Card>
   498→
   499→      {/* ------------------------------ Results ------------------------------ */}
   500→      <div className="min-w-0 space-y-4">
   501→        {!result && status !== "running" ? (
   502→          <Card className="py-0 shadow-sm">
   503→            <CardContent className="flex flex-col items-center justify-center gap-2 py-16 text-center">
   504→              <span className="ping-ring flex size-12 items-center justify-center rounded-full bg-primary/10 text-primary">
   505→                <Microscope className="size-6" aria-hidden />
   506→              </span>
   507→              <p className="text-sm font-medium">No discovery run yet</p>
   508→              <p className="max-w-sm text-xs text-muted-foreground">
   509→                Pick a compound and press{" "}
   510→                <strong className="font-semibold text-foreground">Discover biomarkers</strong>. Every
   511→                gene in the expression matrix is correlated (Spearman, BH-corrected) with the
   512→                compound&apos;s LN_IC50 across the panel — erlotinib should recover the planted EGFR
   513→                biomarker, cisplatin should not.
   514→              </p>
   515→            </CardContent>
   516→          </Card>
   517→        ) : null}
   518→
   519→        {status === "running" ? (
   520→          <Card className="py-0 shadow-sm">
   521→            <CardContent className="flex flex-col items-center justify-center gap-3 py-16 text-center">
   522→              <Loader2 className="size-8 animate-spin text-primary" aria-hidden />
   523→              <p className="text-sm font-medium">Screening…</p>
   524→              <p className="text-xs text-muted-foreground">
   525→                Generating the {panelLabel} screen + {genes}-gene expression matrix and ranking every
   526→                gene against {compound}.
   527→              </p>
   528→            </CardContent>
   529→          </Card>
   530→        ) : null}
   531→
   532→        {result && status !== "running" ? (
   533→          <>
   534→            {/* Stats strip */}
   535→            <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
   536→              <StatCard label="genes screened" value={String(result.meta.nGenes)} />
   537→              <StatCard
   538→                label="significant (FDR<0.05)"
   539→                value={String(result.meta.nSignificant)}
   540→              />
   541→              <StatCard label="strongest |ρ|" value={fmtNum(result.meta.strongestRho, 3)} />
   542→              <StatCard label="runtime" value={`${result.meta.runtimeMs} ms`} />
   543→            </div>
   544→
   545→            {/* Scatter chart */}
   546→            <Card className="overflow-hidden py-0 shadow-sm">
   547→              <CardHeader className="border-b bg-muted/30 py-3">
   548→                <div className="flex flex-wrap items-center justify-between gap-2">
   549→                  <div className="min-w-0">
   550→                    <CardTitle className="flex flex-wrap items-center gap-2 text-sm">
   551→                      <FlaskConical className="size-3.5 text-primary" aria-hidden />
   552→                      {series?.scatter.gene ?? result.topHit.gene} expression vs LN_IC50 —{" "}
   553→                      {result.meta.compound}
   554→                      {result.meta.plantedHit === "EGFR" ? (
