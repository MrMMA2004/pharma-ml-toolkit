     1→"""SMILES genetic algorithm: de novo and seeded molecule generation.
     2→
     3→A pragmatic population-based optimiser in the GuacaMol / LEADD tradition,
     4→CPU-friendly and deterministic given ``seed``:
     5→
     6→* **genome** = canonical SMILES string (character level);
     7→* **mutation** = random character substitution / insertion / deletion,
     8→  followed by RDKit sanitisation and re-canonicalisation (invalid mutants
     9→  are discarded) -- OR, in ``mutation="scaffold"`` mode, a
    10→  scaffold-preserving decoration resampling (see :func:`_scaffold_mutate`);
    11→* **crossover** = single-point splice of two parents, sanitised the same
    12→  way (not used in scaffold mode: a character splice routinely destroys
    13→  ring systems, defeating scaffold preservation);
    14→* **selection** = tournament of ``tournament_k`` molecules;
    15→* **elitism** = the top ``elitism`` molecules survive a generation
    16→  verbatim, so the best score is monotone across generations.
    17→
    18→Two initialisation modes:
    19→
    20→* **de novo** (``seed_smiles=None``): the population is sampled from
    21→  ``corpus`` (default: :data:`~pharmaml.generative.DEFAULT_CORPUS`);
    22→* **seeded** (``seed_smiles=[...]``): the population starts from the
    23→  user's molecules (duplicated to fill the population), so the search
    24→  explores their neighbourhood -- the "with some sample molecules" mode.
    25→
    26→Example
    27→-------
    28→>>> from pharmaml.generative import GeneticOptimizer, QedObjective
    29→>>> result = GeneticOptimizer(QedObjective(), population_size=32,
    30→...                           generations=5, seed=42).optimize()
    31→>>> result.best_score >= 0.5
    32→True
    33→"""
    34→
    35→from __future__ import annotations
    36→

    38→import random
    39→import time
    40→from collections.abc import Sequence
    41→from functools import lru_cache
    42→from typing import Any
    43→
    44→from pharmaml.exceptions import DataValidationError
    45→from pharmaml.logger import get_logger
    46→
    47→from ._utils import canonical_smiles, require_rdkit, smiles_alphabet
    48→from .corpus import DEFAULT_CORPUS
    49→from .result import GenerationResult
    50→

    55→#: Probability that a resampled decoration slot is dropped entirely (rather
    56→#: than replaced with a corpus decoration) by the scaffold mutation.
    57→_DROP_PROBABILITY = 0.25

    60→def _validated_score(objective: Any, smiles: str) -> float:
    61→    """Score ``smiles``, coercing stray values into ``[0, 1]``.
    62→
    63→    A non-numeric return value raises the same typed error as
    64→    :class:`~pharmaml.generative.MultiObjective` -- the objective contract
    65→    (``score(smiles) -> float``) is enforced consistently everywhere.
    66→    """
    67→    try:
    68→        value = float(objective.score(smiles))
    69→    except (TypeError, ValueError) as exc:
    70→        raise DataValidationError(
    71→            f"objective {getattr(objective, 'name', objective)!r} returned a "
    72→            f"non-numeric score for {smiles!r} -- objectives must map SMILES "
    73→            "to floats"
    74→        ) from exc
    75→    if not math.isfinite(value):
    76→        return 0.0
    77→    return min(max(value, 0.0), 1.0)

    81→# Scaffold-preserving mutation machinery
    82→# ---------------------------------------------------------------------------
    83→@lru_cache(maxsize=4096)
    84→def _scaffold_split(smiles: str) -> tuple[str, tuple[str, ...]] | None:
    85→    """Split ``smiles`` into its (labelled) Murcko core + decorations.
    86→
    87→    Returns ``(core_smiles, decoration_smiles)`` where the core carries
    88→    ``[*:k]`` dummy atoms at every attachment point (labelled 1..k in a
    89→    deterministic order) and each decoration fragment carries the matching
    90→    label -- so ``Chem.molzip(core, decorations)`` rebuilds exactly the
    91→    parent molecule.  ``None`` when there is nothing to split (acyclic
    92→    molecule, bare ring system without decorations, unparseable input, or
    93→    any RDKit failure -- robustness first).
    94→    """
    95→    Chem = require_rdkit()
    96→    from rdkit import RDLogger
    97→    from rdkit.Chem.Scaffolds import MurckoScaffold
    98→
    99→    if not isinstance(smiles, str) or not smiles.strip():
   100→        return None
   101→    RDLogger.DisableLog("rdApp.error")
   102→    try:
   103→        mol = Chem.MolFromSmiles(smiles)
   104→    except Exception:  # noqa: BLE001 - RDKit can raise on hostile input
   105→        return None
   106→    finally:
   107→        RDLogger.EnableLog("rdApp.error")
   108→    if mol is None:
   109→        return None
   110→
   111→    scaffold = MurckoScaffold.GetScaffoldForMol(mol)
   112→    if scaffold.GetNumAtoms() == 0:
   113→        return None  # acyclic parent: no framework to preserve
   114→    match = mol.GetSubstructMatch(scaffold)
   115→    if not match:
   116→        return None
   117→    in_scaffold = set(match)
   118→    # attachment pairs (scaffold atom position, outside neighbour), ranked
   119→    # deterministically: label k = k-th attachment in sorted pair order.
   120→    pairs = sorted(
   121→        (j, neighbour.GetIdx())
   122→        for j, atom_idx in enumerate(match)
   123→        for neighbour in mol.GetAtomWithIdx(atom_idx).GetNeighbors()
   124→        if neighbour.GetIdx() not in in_scaffold
   125→    )
   126→    if not pairs:
   127→        return None  # bare scaffold: no decorations to resample
   128→    ranks_by_j: dict[int, list[int]] = {}
   129→    for rank, (j, _neighbour) in enumerate(pairs, start=1):
   130→        ranks_by_j.setdefault(j, []).append(rank)
   131→
   132→    core = Chem.ReplaceSidechains(mol, scaffold)
   133→    if core is None:
   134→        return None
   135→    core_match = core.GetSubstructMatch(scaffold)
   136→    if not core_match:
   137→        return None
   138→    dummies_by_j: dict[int, list[int]] = {}
   139→    for atom in core.GetAtoms():
   140→        if atom.GetAtomicNum() != 0:
   141→            continue
   142→        neighbours = atom.GetNeighbors()
   143→        if len(neighbours) != 1:
   144→            return None
   145→        j = core_match.index(neighbours[0].GetIdx())
   146→        dummies_by_j.setdefault(j, []).append(atom.GetIdx())
   147→    rw_core = Chem.RWMol(core)
   148→    for j, dummy_ids in dummies_by_j.items():
   149→        ranks = ranks_by_j.get(j, [])
   150→        if len(dummy_ids) != len(ranks):
   151→            return None
   152→        for dummy_id, rank in zip(sorted(dummy_ids), ranks):
   153→            atom = rw_core.GetAtomWithIdx(dummy_id)
   154→            atom.SetAtomMapNum(rank)
   155→            atom.SetIsotope(0)
   156→    core_smiles = Chem.MolToSmiles(rw_core.GetMol())
   157→
   158→    # ReplaceCore(labelByIndex=True) labels each decoration dummy with the
   159→    # scaffold-atom position it was attached to -- relabel to the ranks.
   160→    side = Chem.ReplaceCore(mol, scaffold, labelByIndex=True)
   161→    if side is None:
   162→        return None
   163→    rank_queues = {j: list(ranks) for j, ranks in ranks_by_j.items()}
   164→    decoration_smiles: list[str] = []
   165→    for fragment in Chem.GetMolFrags(side, asMols=True):
   166→        rw_fragment = Chem.RWMol(fragment)
   167→        saw_dummy = False
   168→        for atom in rw_fragment.GetAtoms():
   169→            if atom.GetAtomicNum() != 0:
   170→                continue
   171→            queue = rank_queues.get(atom.GetIsotope())
   172→            if not queue:
   173→                return None
   174→            atom.SetAtomMapNum(queue.pop(0))
   175→            atom.SetIsotope(0)
   176→            saw_dummy = True
   177→        if not saw_dummy:
   178→            return None  # untracked fragment: refuse rather than lose atoms
   179→        decoration_smiles.append(Chem.MolToSmiles(rw_fragment.GetMol()))
   180→    return core_smiles, tuple(decoration_smiles)
   181→
   182→
   183→def _strip_dummy_label(fragment_smiles: str) -> str:
   184→    """Remove the attachment label from a single-dummy decoration SMILES."""
   185→    Chem = require_rdkit()
   186→    fragment = Chem.MolFromSmiles(fragment_smiles)
   187→    if fragment is None:  # pragma: no cover - split output always parses
   188→        return fragment_smiles
   189→    rw = Chem.RWMol(fragment)
   190→    for atom in rw.GetAtoms():
   191→        if atom.GetAtomicNum() == 0:
   192→            atom.SetAtomMapNum(0)
   193→            atom.SetIsotope(0)
   194→    return Chem.MolToSmiles(rw.GetMol())

   197→@lru_cache(maxsize=8)
   198→def _decoration_pool(corpus_key: tuple[str, ...]) -> tuple[str, ...]:
   199→    """Corpus-derived decoration vocabulary (memoised on the corpus tuple)."""
   200→    pool: list[str] = []
   201→    for smiles in corpus_key:
   202→        split = _scaffold_split(smiles)
   203→        if split is None:
   204→            continue
   205→        for fragment_smiles in split[1]:
   206→            if fragment_smiles.count("[*") != 1:
   207→                continue  # only single-attachment decorations are reusable
   208→            pool.append(fragment_smiles)
   209→    return tuple(dict.fromkeys(_strip_dummy_label(s) for s in pool))
   210→
   211→
   212→def _build_decoration_pool(corpus: Sequence[str]) -> list[str]:
   213→    """Decoration vocabulary derived from ``corpus`` (deterministic order).
   214→
   215→    Every corpus molecule is decomposed once into Murcko scaffold +
   216→    decorations (see :func:`_scaffold_split`); the single-attachment
   217→    decorations, label-stripped and de-duplicated, form the vocabulary the
   218→    scaffold mutation samples from.  Murcko decorations are always acyclic
   219→    (any ring would be part of the framework), which is what makes scaffold
   220→    preservation structural rather than lucky.
   221→    """
   222→    return list(_decoration_pool(tuple(corpus)))
   223→
   224→
   225→def _scaffold_mutate(
   226→    smiles: str,
   227→    pool: Sequence[str],
   228→    rng: random.Random,
   229→    max_smiles_length: int,
   230→) -> str | None:
   231→    """Decoration-resampling mutation that preserves the Murcko scaffold.
   232→
   233→    Keeps the parent's Murcko framework verbatim and replaces a random
   234→    non-empty subset of its decorations with entries sampled from the
   235→    corpus-derived ``pool`` (or drops them).  Returns the canonical mutant
   236→    SMILES, or ``None`` when the parent has no scaffold / no decorations,
   237→    the pool is unusable for it, the mutant would exceed
   238→    ``max_smiles_length``, or RDKit raises anywhere in the pipeline -- the
   239→    caller then falls back to the standard character mutation
   240→    (robustness first: never crash, never emit a dummy-bearing or empty
   241→    molecule).
   242→    """
   243→    Chem = require_rdkit()
   244→    from rdkit import RDLogger
   245→
   246→    # Occasional chemically invalid re-attachments are EXPECTED (they fall
   247→    # back to the character mutation); RDKit logs the sanitisation failures
   248→    # to stderr before the exception surfaces -- keep the loop quiet (the
   249→    # canonical_smiles guard-family precedent).
   250→    RDLogger.DisableLog("rdApp.error")
   251→    try:
   252→        split = _scaffold_split(smiles)
   253→        if split is None:
   254→            return None
   255→        core_smiles, fragment_smiles = split
   256→        # only single-attachment decorations can be replaced/dropped;
   257→        # bridging multi-dummy fragments are always kept as-is
   258→        replaceable = [
   259→            index for index, fragment in enumerate(fragment_smiles)
   260→            if fragment.count("[*") == 1
   261→        ]
   262→        if not replaceable:
   263→            return None
   264→        n_change = rng.randint(1, len(replaceable))
   265→        chosen = set(rng.sample(replaceable, n_change))
   266→
   267→        rw_core = Chem.RWMol(Chem.MolFromSmiles(core_smiles))
   268→        kept: list[str] = []
   269→        for index, fragment in enumerate(fragment_smiles):
   270→            if index not in chosen:
   271→                kept.append(fragment)
   272→                continue
   273→            parsed = Chem.MolFromSmiles(fragment)
   274→            if parsed is None:  # pragma: no cover - split output always parses
   275→                return None
   276→            label = next(
   277→                atom.GetAtomMapNum()
   278→                for atom in parsed.GetAtoms()
   279→                if atom.GetAtomicNum() == 0
   280→            )
   281→            if rng.random() < _DROP_PROBABILITY:
   282→                # Drop the decoration by turning its core dummy into an
   283→                # explicit hydrogen (plain RemoveAtom would leave e.g. an
   284→                # aromatic N without its H -> "Can't kekulize" on sanitise;
   285→                # an unmatched dummy would leak [*:k] into the offspring).
   286→                # RemoveHs below collapses the graph-H back to implicit.
   287→                for dummy_id in (
   288→                    atom.GetIdx()
   289→                    for atom in rw_core.GetAtoms()
   290→                    if atom.GetAtomicNum() == 0 and atom.GetAtomMapNum() == label
   291→                ):
   292→                    hydrogen = rw_core.GetAtomWithIdx(dummy_id)
   293→                    hydrogen.SetAtomicNum(1)
   294→                    hydrogen.SetAtomMapNum(0)
   295→                    hydrogen.SetIsotope(0)
   296→                    hydrogen.SetFormalCharge(0)
   297→                    hydrogen.SetNumExplicitHs(0)
   298→                    hydrogen.SetNoImplicit(True)
   299→            else:
   300→                replacement = Chem.MolFromSmiles(pool[rng.randrange(len(pool))])
   301→                if replacement is None:  # pragma: no cover - pool always parses
   302→                    return None
   303→                rw_replacement = Chem.RWMol(replacement)
   304→                for atom in rw_replacement.GetAtoms():
   305→                    if atom.GetAtomicNum() == 0:
   306→                        atom.SetAtomMapNum(label)
   307→                kept.append(Chem.MolToSmiles(rw_replacement.GetMol()))
   308→
   309→        core = rw_core.GetMol()
   310→        Chem.SanitizeMol(core)
   311→        if kept:
   312→            side = Chem.MolFromSmiles(".".join(kept))
   313→            if side is None:  # pragma: no cover - kept fragments always parse
   314→                return None
   315→            product = Chem.molzip(core, side)
   316→        else:
   317→            product = core  # every decoration dropped: the bare framework
   318→        Chem.SanitizeMol(product)
   319→        product = Chem.RemoveHs(product)  # graph-H drops -> implicit / [nH]
   320→        mutant = Chem.MolToSmiles(product)
   321→        if not mutant or len(mutant) > max_smiles_length:
   322→            return None
   323→        return mutant
   324→    except Exception:  # noqa: BLE001 - robustness-first fallback contract
   325→        return None
   326→    finally:
   327→        RDLogger.EnableLog("rdApp.error")

   330→class GeneticOptimizer:
   331→    """Tournament-selection genetic algorithm over canonical SMILES.
   332→
   333→    Parameters
   334→    ----------
   335→    objective:
   336→        Anything with ``score(smiles) -> float`` (any
   337→        :class:`~pharmaml.generative.ScoringObjective`).
   338→    population_size:
   339→        Molecules per generation (default 64).
   340→    generations:
   341→        Number of generations (default 20).
   342→    mutation_rate:
   343→        Expected fraction of characters mutated per offspring (default
   344→        0.08; at least one character whenever mutation fires).
   345→    crossover_rate:
   346→        Probability an offspring is a splice of two parents instead of a
   347→        clone of the first (default 0.7).  Ignored in ``scaffold`` mutation
   348→        mode (see ``mutation``).
   349→    elitism:
   350→        Top molecules copied verbatim into the next generation (default
   351→        4; guarantees a monotone best score).
   352→    tournament_k:
   353→        Molecules compared per selection tournament (default 3).
   354→    max_smiles_length:
   355→        Hard length cap on every genome -- offspring that exceed it are
   356→        rejected (default 120).
   357→    mutation:
   358→        ``"uniform"`` (default) -- the classic character-level
   359→        substitution / insertion / deletion.  ``"scaffold"`` --
   360→        **scaffold-preserving decoration resampling**: the offspring keeps
   361→        the parent's Murcko framework verbatim while a random subset of
   362→        its decorations is replaced with entries sampled from a
   363→        decoration vocabulary derived from the corpus (or dropped).  This
   364→        is a pragmatic lead-optimisation operator in the spirit of
   365→        LEADD's scaffold decoration: it explores the decoration space of
   366→        one fixed framework.  It is **not** a full scaffold-decorator
   367→        generative model like REINVENT 2.0 (no attachment-point
   368→        generative grammar, no learned decoration distribution -- just
   369→        corpus-derived resampling with ``Chem.molzip`` re-assembly).
   370→        Crossover is bypassed in this mode (a character splice routinely
   371→        destroys ring systems); parents without a Murcko scaffold
   372→        (acyclic molecules) or without decorations (bare ring systems)
   373→        fall back to the uniform mutation -- never an error.
   374→    seed:
   375→        Seed for the dedicated ``random.Random`` instance (default 42).
   376→        Two runs with the same seed produce identical results.
   377→
   378→    Raises
   379→    ------
   380→    DataValidationError
   381→        On out-of-range hyperparameters.
   382→
   383→    Example
   384→    -------
   385→    >>> from pharmaml.generative import GeneticOptimizer, QedObjective
   386→    >>> # lead-optimisation mode: keep aspirin's framework, vary decorations
   387→    >>> result = GeneticOptimizer(
   388→    ...     QedObjective(), population_size=16, generations=3, seed=42,
   389→    ...     mutation="scaffold",
   390→    ... ).optimize(seed_smiles=["CC(=O)Oc1ccccc1C(=O)O"],
   391→    ...              corpus=["CC(=O)Oc1ccccc1C(=O)O"])
   392→    >>> len(result) > 0
   393→    True
   394→    """
   395→
   396→    def __init__(
   397→        self,
   398→        objective: Any,
   399→        *,
   400→        population_size: int = 64,
   401→        generations: int = 20,
   402→        mutation_rate: float = 0.08,
   403→        crossover_rate: float = 0.7,
   404→        elitism: int = 4,
   405→        tournament_k: int = 3,
   406→        max_smiles_length: int = 120,
   407→        mutation: str = "uniform",
   408→        seed: int = 42,
   409→    ) -> None:
   410→        if not callable(getattr(objective, "score", None)):
   411→            raise DataValidationError(
   412→                "objective must expose a callable score(smiles) method"
   413→            )
   414→        if population_size < 4:
   415→            raise DataValidationError(
   416→                f"population_size must be >= 4, got {population_size}"
   417→            )
   418→        if generations < 1:
   419→            raise DataValidationError(f"generations must be >= 1, got {generations}")
   420→        if not 0.0 < mutation_rate <= 1.0:
   421→            raise DataValidationError(
   422→                f"mutation_rate must be in (0, 1], got {mutation_rate}"
   423→            )
   424→        if not 0.0 <= crossover_rate <= 1.0:
   425→            raise DataValidationError(
   426→                f"crossover_rate must be in [0, 1], got {crossover_rate}"
   427→            )
   428→        if not 0 <= elitism < population_size:
   429→            raise DataValidationError(
   430→                f"elitism must be in [0, population_size), got {elitism} "
   431→                f"with population_size={population_size}"
   432→            )
   433→        if tournament_k < 2:
   434→            raise DataValidationError(f"tournament_k must be >= 2, got {tournament_k}")
   435→        if max_smiles_length < 20:
   436→            raise DataValidationError(
   437→                f"max_smiles_length must be >= 20, got {max_smiles_length}"
   438→            )
   439→        if mutation not in ("uniform", "scaffold"):
   440→            raise DataValidationError(
   441→                f"mutation must be 'uniform' or 'scaffold', got {mutation!r}"
   442→            )
   443→        self.objective = objective
   444→        self.population_size = population_size
   445→        self.generations = generations
   446→        self.mutation_rate = mutation_rate
   447→        self.crossover_rate = crossover_rate
   448→        self.elitism = elitism
   449→        self.tournament_k = tournament_k
   450→        self.max_smiles_length = max_smiles_length
   451→        self.mutation = mutation
   452→        self.seed = seed
   453→
   454→    # -- internals -------------------------------------------------------
   455→    def _mutation_alphabet(self, corpus: list[str]) -> list[str]:
   456→        """Characters the mutator draws from (the corpus alphabet)."""
   457→        alphabet = smiles_alphabet(corpus)
   458→        return alphabet or list("CCcO()=")
   459→
   460→    def _mutate(
   461→        self,
   462→        smiles: str,
   463→        alphabet: list[str],
   464→        rng: random.Random,
   465→        decoration_pool: Sequence[str] = (),
   466→    ) -> str:
   467→        """Mutate ``smiles``: decoration resampling or character edits.
   468→
   469→        In ``scaffold`` mode (with a non-empty decoration pool) the
   470→        scaffold-preserving operator fires first; ``None`` returns (acyclic
   471→        parent, RDKit failure, over-length mutant) fall through to the
   472→        classic character-level mutation so evolution never stalls.
   473→        """
   474→        if decoration_pool:
   475→            mutant = _scaffold_mutate(smiles, decoration_pool, rng, self.max_smiles_length)
   476→            if mutant is not None:
   477→                return mutant
   478→        chars = list(smiles)
   479→        n_mutations = max(1, round(len(chars) * self.mutation_rate))
   480→        for _ in range(n_mutations):
   481→            if not chars:
   482→                break
   483→            position = rng.randrange(len(chars))
   484→            draw = rng.random()
   485→            if draw < 0.45:  # substitute
   486→                chars[position] = alphabet[rng.randrange(len(alphabet))]
   487→            elif draw < 0.8:  # insert
   488→                chars.insert(position, alphabet[rng.randrange(len(alphabet))])
   489→            else:  # delete
   490→                del chars[position]
   491→        return "".join(chars)
   492→
   493→    def _crossover(self, parent_a: str, parent_b: str, rng: random.Random) -> str:
   494→        """Single-point splice; over-long children fall back to parent A."""
   495→        if len(parent_a) < 2 or len(parent_b) < 2:
   496→            return parent_a
   497→        cut_a = rng.randrange(1, len(parent_a))
   498→        cut_b = rng.randrange(1, len(parent_b))
   499→        child = parent_a[:cut_a] + parent_b[cut_b:]
   500→        if len(child) > self.max_smiles_length:
   501→            return parent_a
   502→        return child
   503→
   504→    def _tournament(
   505→        self, population: list[str], fitness: dict[str, float], rng: random.Random
   506→    ) -> str:
   507→        """Pick the fittest of ``tournament_k`` random molecules."""
   508→        best = population[rng.randrange(len(population))]
   509→        for _ in range(self.tournament_k - 1):
   510→            candidate = population[rng.randrange(len(population))]
   511→            if fitness[candidate] > fitness[best]:
   512→                best = candidate
   513→        return best
   514→
   515→    def _initial_population(
   516→        self, seed_smiles: Sequence[str] | None, canonical_pool: list[str], rng: random.Random
   517→    ) -> tuple[list[str], list[str]]:
   518→        """Build the generation-0 population (seeds first, corpus padding).
   519→
   520→        Returns ``(population, canonical_seeds_used)``.
   521→        """
   522→        initial: list[str] = []
   523→        kept: list[str] = []
   524→        if seed_smiles is not None:
   525→            for smiles in list(seed_smiles):  # caller's sequence stays untouched
   526→                canonical = (
   527→                    canonical_smiles(smiles) if isinstance(smiles, str) else None
   528→                )
   529→                if canonical is None:
   530→                    _LOG.warning("dropping invalid seed SMILES %r", smiles)
   531→                elif len(canonical) > self.max_smiles_length:
   532→                    _LOG.warning(
   533→                        "dropping seed SMILES %r -- longer than max_smiles_length=%d",
   534→                        smiles,
   535→                        self.max_smiles_length,
   536→                    )
   537→                else:
   538→                    kept.append(canonical)
   539→            if not kept:
   540→                raise DataValidationError(
   541→                    "no valid seed molecule(s) provided -- invalid entries were "
   542→                    "dropped with a warning, but at least one must survive"
   543→                )
   544→            _LOG.info(
   545→                "seeded mode: %d user molecule(s) anchor the population", len(kept)
   546→            )
   547→            initial.extend(kept)
   548→        while len(initial) < self.population_size:
   549→            initial.append(canonical_pool[rng.randrange(len(canonical_pool))])
   550→        # de-duplicate while preserving order (seeds come first)
   551→        seen: set[str] = set()
   552→        population: list[str] = []
   553→        for smiles in initial:
   554→            if smiles not in seen:
   555→                seen.add(smiles)
   556→                population.append(smiles)
   557→        # top up with *unique* corpus molecules where the pool allows it...
   558→        attempts = 0
   559→        while (
   560→            len(population) < self.population_size
   561→            and attempts < 10 * self.population_size
   562→        ):
   563→            attempts += 1
   564→            candidate = canonical_pool[rng.randrange(len(canonical_pool))]
   565→            if candidate not in seen:
   566→                seen.add(candidate)
   567→                population.append(candidate)
   568→        if len(population) < self.population_size:
   569→            # ... and with duplicated members when the corpus holds fewer
   570→            # unique molecules than population_size (a tiny user corpus must
   571→            # not hang the run: evolution still works on repeated genomes).
   572→            _LOG.warning(
   573→                "corpus holds only %d unique molecule(s) -- duplicating members "
   574→                "to reach population_size=%d",
   575→                len(population),
   576→                self.population_size,
   577→            )
   578→            base = list(population)
   579→            index = 0
   580→            while len(population) < self.population_size:
   581→                population.append(base[index % len(base)])
   582→                index += 1
   583→        return population, kept
   584→
   585→    # -- public API ------------------------------------------------------
   586→    def optimize(
   587→        self,
   588→        seed_smiles: Sequence[str] | None = None,
   589→        corpus: Sequence[str] | None = None,
   590→    ) -> GenerationResult:
   591→        """Run the evolution and return the ranked :class:`GenerationResult`.
   592→
   593→        Parameters
   594→        ----------
   595→        seed_smiles:
   596→            Seeded mode: molecules the population is initialised from.
   597→            Invalid entries are warned about and dropped (an error is
   598→            raised only when *no* seed survives).  The caller's list is
   599→            never mutated.  When MORE valid seeds arrive than
   600→            ``population_size``, generation 0 scores them all and the
   601→            population contracts to ``population_size`` from the next
   602→            generation (user molecules are never silently discarded).
   603→        corpus:
   604→            De novo mode: sampling pool for the initial population
   605→            (default: :data:`~pharmaml.generative.DEFAULT_CORPUS`, an
   606→            immutable tuple).  A pool with fewer unique valid molecules
   607→            than ``population_size`` is tolerated: members are
   608→            duplicated (with a warning) rather than hanging the run.
   609→
   610→        Raises
   611→        ------
   612→        DataValidationError
   613→            If every provided seed molecule is invalid, or the corpus is
   614→            empty after validity / length filtering.
   615→        """
   616→        started = time.perf_counter()
   617→        rng = random.Random(self.seed)
   618→
   619→        # An explicitly passed *empty* corpus must raise (not silently fall
   620→        # back to the default corpus), so truthiness tricks are off limits.
   621→        pool = list(DEFAULT_CORPUS if corpus is None else corpus)
   622→        n_non_string = sum(1 for s in pool if not isinstance(s, str))
   623→        pool = [s for s in pool if isinstance(s, str)]
   624→        if not pool:
   625→            raise DataValidationError(
   626→                "corpus is empty (or holds no string entries) -- pass a "
   627→                "non-empty list of SMILES strings"
   628→                + (f" ({n_non_string} non-string entr(ies) dropped)" if n_non_string else "")
   629→            )
   630→        canonical_pool = []
   631→        for smiles in pool:
   632→            canonical = canonical_smiles(smiles)
   633→            if canonical is not None and len(canonical) <= self.max_smiles_length:
   634→                canonical_pool.append(canonical)
   635→        # de-duplicate while preserving order: duplicate spellings of the
   636→        # same molecule (e.g. "CCO" and "OCC") must not skew the initial
   637→        # sampling, and a pool with fewer UNIQUE molecules than
   638→        # population_size must not hang the unique-fill loop below.
   639→        canonical_pool = list(dict.fromkeys(canonical_pool))
   640→        if not canonical_pool:
   641→            raise DataValidationError(
   642→                "corpus holds no valid SMILES under the length cap -- nothing "
   643→                "to initialise the population from"
   644→            )
   645→        alphabet = self._mutation_alphabet(canonical_pool)
   646→        population, kept_seeds = self._initial_population(seed_smiles, canonical_pool, rng)
   647→        decoration_pool: list[str] = []
   648→        if self.mutation == "scaffold":
   649→            decoration_pool = _build_decoration_pool(canonical_pool)
   650→            if not decoration_pool:
   651→                _LOG.warning(
   652→                    "corpus yields no scaffold decorations -- scaffold "
   653→                    "mutations fall back to character mutation"
   654→                )
   655→
   656→        produced: dict[str, float] = {}
   657→        history: list[dict[str, Any]] = []
   658→        accepted_total = 0
   659→        attempts_total = 0
   660→        last_fraction = 1.0
   661→
   662→        for generation in range(self.generations):
   663→            fitness = {
   664→                smiles: _validated_score(self.objective, smiles)
   665→                for smiles in population
   666→            }
   667→            for smiles, score in fitness.items():
   668→                produced[smiles] = max(produced.get(smiles, 0.0), score)
   669→            ranked = sorted(population, key=lambda s: (-fitness[s], s))
   670→            history.append(
   671→                {
   672→                    "generation": generation,
   673→                    "best_score": fitness[ranked[0]],
   674→                    "mean_score": sum(fitness.values()) / len(fitness),
   675→                    "best_smiles": ranked[0],
   676→                    "n_unique": len(set(population)),
   677→                    "valid_offspring_fraction": last_fraction,
   678→                }
   679→            )
   680→            if generation == self.generations - 1:
   681→                break
   682→
   683→            next_population = ranked[: self.elitism]
   684→            attempts = 0
   685→            while len(next_population) < self.population_size:
   686→                attempts += 1
   687→                if attempts > 50 * self.population_size:  # pragma: no cover
   688→                    _LOG.warning(
   689→                        "mutation/crossover yield collapsed in generation %d "
   690→                        "-- padding the population with corpus molecules",
   691→                        generation,
   692→                    )
   693→                    while len(next_population) < self.population_size:
   694→                        next_population.append(
   695→                            canonical_pool[rng.randrange(len(canonical_pool))]
   696→                        )
   697→                    break
   698→                parent_a = self._tournament(population, fitness, rng)
   699→                if self.mutation == "scaffold":
   700→                    # Decoration resampling on a single parent IS the
   701→                    # variation operator: a character-level splice would
   702→                    # routinely destroy ring systems, defeating the mode's
   703→                    # purpose (crossover_rate does not apply here).
   704→                    child = parent_a
   705→                else:
   706→                    parent_b = self._tournament(population, fitness, rng)
   707→                    if rng.random() < self.crossover_rate:
   708→                        child = self._crossover(parent_a, parent_b, rng)
   709→                    else:
   710→                        child = parent_a
   711→                child = self._mutate(child, alphabet, rng, decoration_pool)
   712→                if not child or len(child) > self.max_smiles_length:
   713→                    continue
   714→                canonical = canonical_smiles(child)
   715→                if canonical is None or len(canonical) > self.max_smiles_length:
   716→                    continue
   717→                if canonical not in next_population:
   718→                    next_population.append(canonical)
   719→            accepted = self.population_size - self.elitism
   720→            attempts_total += attempts
   721→            accepted_total += accepted
   722→            if attempts:
   723→                last_fraction = accepted / attempts
   724→            history[-1]["valid_offspring_fraction"] = last_fraction
   725→            population = next_population
   726→
   727→        ranked_all = sorted(produced.items(), key=lambda kv: (-kv[1], kv[0]))
   728→        smiles_list = [smiles for smiles, _ in ranked_all]
   729→        score_list = [score for _, score in ranked_all]
   730→        breakdown: list[dict[str, float]] = []
   731→        score_with_breakdown = getattr(self.objective, "score_with_breakdown", None)
   732→        for smiles in smiles_list:
   733→            if callable(score_with_breakdown):
   734→                breakdown.append(score_with_breakdown(smiles)[1])
   735→            else:
   736→                breakdown.append({})
   737→        elapsed = time.perf_counter() - started
   738→        return GenerationResult(
   739→            smiles=smiles_list,
   740→            scores=score_list,
   741→            breakdown=breakdown,
   742→            best=smiles_list[0] if smiles_list else "",
   743→            best_score=score_list[0] if score_list else 0.0,
   744→            history=history,
   745→            metadata={
   746→                "method": "genetic_algorithm",
   747→                "seed": self.seed,
   748→                "iterations": self.generations,
   749→                "elapsed_s": elapsed,
   750→                "valid_generated": accepted_total + len(population),
   751→                "unique_generated": len(smiles_list),
   752→                "population_size": self.population_size,
   753→                "mutation": self.mutation,
   754→                "seed_molecules": kept_seeds,
   755→            },
   756→        )
   757→
   758→
   759→def run_genetic_optimization(
   760→    objective: Any,
   761→    seed_smiles: Sequence[str] | None = None,
   762→    corpus: Sequence[str] | None = None,
   763→    **kwargs: Any,
   764→) -> GenerationResult:
   765→    """One-call genetic optimisation (thin :class:`GeneticOptimizer` wrapper).
   766→
   767→    Example
   768→    -------
   769→    >>> from pharmaml.generative import QedObjective, run_genetic_optimization
   770→    >>> result = run_genetic_optimization(QedObjective(), generations=5, seed=7)
   771→    >>> len(result) > 0
   772→    True
   773→    """
   774→    optimizer = GeneticOptimizer(objective, **kwargs)
   775→    return optimizer.optimize(seed_smiles=seed_smiles, corpus=corpus)
