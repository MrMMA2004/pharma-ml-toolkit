   608→def _fail(message: str) -> None:
   609→    raise SystemExit(f"DRIVER FAILED: {message}")

   612→def apply_edits() -> int:
   613→    """Replace the fabrication cells + patch the prose.  Idempotent."""
   614→    changed = 0
   615→    notebooks: dict[str, object] = {}
   616→    for name in NOTEBOOKS:
   617→        notebooks[name] = nbformat.read(EXAMPLES / name, as_version=4)
   618→
   619→    for name, index, anchor, new_source, done in CODE_EDITS:
   620→        nb = notebooks[name]
   621→        cell = nb.cells[index]
   622→        source = cell.source
   623→        if done in source:
   624→            continue  # already converted (idempotent re-run)
   625→        if cell.cell_type != "code":
   626→            _fail(f"{name} cell {index} is {cell.cell_type}, expected code")
   627→        anchors = anchor if isinstance(anchor, tuple) else (anchor,)
   628→        if not any(a in source for a in anchors):
   629→            _fail(f"{name} cell {index} lost its anchor {anchors!r} -- "
   630→                  "cell numbering drifted; update the driver's edit table")
   631→        compile(new_source, f"{name}#{index}", "exec")  # never write broken code
   632→        cell.source = new_source
   633→        cell.outputs = []
   634→        cell.execution_count = None
   635→        changed += 1
   636→
   637→    for name, index, old, new in PROSE_EDITS:
   638→        nb = notebooks[name]
   639→        cell = nb.cells[index]
   640→        if new in cell.source:
   641→            continue  # already patched
   642→        if cell.cell_type != "markdown":
   643→            _fail(f"{name} cell {index} is {cell.cell_type}, expected markdown")
   644→        if old not in cell.source:
   645→            _fail(f"{name} cell {index} lost its prose anchor -- "
   646→                  f"update the driver's PROSE_EDITS entry starting {old[:48]!r}")
   647→        cell.source = cell.source.replace(old, new, 1)
   648→        changed += 1
   649→
   650→    for name, nb in notebooks.items():
   651→        nbformat.write(nb, EXAMPLES / name)
   652→    return changed

   655→def guard() -> None:
   656→    """Zero banned fabrication tokens across all six executed notebooks."""
   657→    hits = []
   658→    for name in NOTEBOOKS:
   659→        nb = nbformat.read(EXAMPLES / name, as_version=4)
   660→        for i, cell in enumerate(nb.cells):
   661→            if cell.cell_type != "code":
   662→                continue
   663→            for token in BANNED_TOKENS:
   664→                if token in cell.source:
   665→                    hits.append((name, i, token))
   666→    if hits:
   667→        for name, i, token in hits:
   668→            print(f"  BANNED {name} cell {i}: {token!r}")
   669→        _fail(f"{len(hits)} banned fabrication token hit(s) remain")
   670→    print(f"guard: 0 hits for all {len(NOTEBOOKS) * len(BANNED_TOKENS)} "
   671→          "notebook x token combinations")

   674→def execute_notebooks(names: tuple[str, ...] = NOTEBOOKS) -> None:
   675→    """Re-run notebooks top-to-bottom with nbclient (cwd=examples/)."""
   676→    from nbclient import NotebookClient
   677→
   678→    for name in names:
   679→        path = EXAMPLES / name
   680→        nb = nbformat.read(path, as_version=4)
   681→        for cell in nb.cells:
   682→            if cell.cell_type == "code":
   683→                cell.outputs = []
   684→                cell.execution_count = None
   685→        started = time.perf_counter()
   686→        client = NotebookClient(
   687→            nb,
   688→            timeout=2400,
   689→            kernel_name="python3",
   690→            resources={"metadata": {"path": str(EXAMPLES)}},
   691→        )
   692→        client.execute()
   693→        nbformat.write(nb, path)
   694→
   695→        code_cells = [c for c in nb.cells if c.cell_type == "code"]
   696→        executed = sum(1 for c in code_cells if c.get("execution_count"))
   697→        errors = [o for c in code_cells for o in c.get("outputs", [])
   698→                  if o.output_type == "error"]
   699→        figures = sum(1 for c in code_cells for o in c.get("outputs", [])
   700→                      if o.get("data", {}).get("image/png"))
   701→        if errors or executed != len(code_cells):
   702→            _fail(f"{name}: {len(errors)} error output(s), "
   703→                  f"{executed}/{len(code_cells)} cells executed")
   704→        print(f"executed {name}: {len(code_cells)} code cells, {figures} figures, "
   705→              f"{time.perf_counter() - started:.0f} s")

   753→def main(argv: list[str] | None = None) -> int:
   754→    args = sys.argv[1:] if argv is None else argv
   755→    stage = args[0] if args else "all"
   756→    if stage in {"--execute", "execute"}:
   757→        execute_notebooks((args[1],))
   758→        return 0
   759→    if stage in {"--twins", "twins"}:
   760→        regenerate_twins()
   761→        return 0
   762→    if stage in {"--checks", "checks"}:
   763→        spot_checks()
   764→        return 0
   765→
   766→    missing = [p for p in (
   767→        "qsar_classification.csv", "screening_library.csv",
   768→        "library_with_descriptors.csv", "qsar_multiclass.csv",
   769→        "qsar_regression.csv", "cellline_screen.csv", "cellline_expression.csv",
   770→        "pk_profile.csv", "ligand_library.csv", "lysozyme_1aki.pdb",
   771→        "docking_campaign", "seed_molecules.csv", "walkthrough_compounds.csv",
   772→        "walkthrough_classification.csv", "walkthrough_multiclass.csv",
   773→    ) if not (EXAMPLES / "data" / p).exists()]
   774→    if missing:
   775→        _fail(f"examples/data is missing {missing} -- run scratch/build_example_data.py")
   776→
   777→    print(f"edits applied : {apply_edits()}")
   778→    guard()
   779→    execute_notebooks()
   780→    regenerate_twins()
   781→    spot_checks()
   782→    print("\n26-R7 driver OK")
   783→    return 0
