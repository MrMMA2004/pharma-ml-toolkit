   175→/** Read the persisted recent-files list (client only, failure-tolerant). */
   176→function loadRecentFiles(): string[] {
   177→  if (typeof window === "undefined") return [];
   178→  try {
   179→    const raw = window.sessionStorage.getItem(RECENT_STORAGE_KEY);
   180→    if (!raw) return [];
   181→    const parsed: unknown = JSON.parse(raw);
   182→    if (!Array.isArray(parsed)) return [];
   183→    return parsed
   184→      .filter((p): p is string => typeof p === "string")
   185→      .slice(0, RECENT_LIMIT);
   186→  } catch {
   187→    return []; // private mode / quota / corrupted JSON — start fresh
   188→  }
   189→}
   190→
   191→/** Persist the recent-files list (client only, best-effort). */
   192→function saveRecentFiles(recent: string[]): void {
   193→  if (typeof window === "undefined") return;
   194→  try {
   195→    window.sessionStorage.setItem(RECENT_STORAGE_KEY, JSON.stringify(recent));
   196→  } catch {
   197→    /* quota exceeded or private mode — persistence is optional */
   198→  }

   217→/** Validate a persisted PK regimen field-by-field, falling back to defaults. */

   219→  const base = DEFAULT_PK_PARAMS;

   221→  const p = raw as Partial<PkParams>;
   222→  const num = (v: unknown, def: number, lo: number, hi: number) =>
   223→    typeof v === "number" && Number.isFinite(v) && v >= lo && v <= hi ? v : def;
   224→  return {
   225→    route: p.route === "oral" ? "oral" : "iv",
   226→    dose: num(p.dose, base.dose, 0.1, 10_000),
   227→    vd: num(p.vd, base.vd, 0.1, 1_000),
   228→    cl: num(p.cl, base.cl, 0.01, 100),
   229→    tau: num(p.tau, base.tau, 0.25, 168),
   230→    doses: Math.round(num(p.doses, base.doses, 1, 48)),
   231→    f: num(p.f, base.f, 0.05, 1),
   232→    ka: num(p.ka, base.ka, 0.01, 12),

   260→/** Validate a persisted NCA snapshot: rows are raw text, capped + typed. */
   261→function validateNca(raw: unknown): NcaState {
   262→  const base = DEFAULT_NCA_STATE;
   263→  if (typeof raw !== "object" || raw === null) return base;
   264→  const s = raw as Partial<NcaState>;
   265→  const cell = (v: unknown): string | null =>
   266→    typeof v === "string" && v.length <= 16 && /^[\d.\s-]*$/.test(v) ? v : null;
   267→  const rows: NcaRow[] = Array.isArray(s.rows)
   268→    ? s.rows
   269→        .filter(
   270→          (r): r is NcaRow =>
   271→            typeof r === "object" &&
   272→            r !== null &&
   273→            typeof (r as NcaRow).id === "number" &&
   274→            Number.isInteger((r as NcaRow).id) &&
   275→            (r as NcaRow).id >= 0 &&
   276→            (r as NcaRow).id < 100_000 &&
   277→            typeof (r as NcaRow).t === "string" &&
   278→            typeof (r as NcaRow).c === "string",
   279→        )
   280→        .slice(0, NCA_ROWS_CAP)
   281→        .map((r) => ({ id: r.id, t: cell(r.t) ?? "0", c: cell(r.c) ?? "0" }))
   282→    : [];
   283→  const terminalN =
   284→    typeof s.terminalN === "number" && Number.isFinite(s.terminalN) && s.terminalN >= 2 && s.terminalN <= 6
   285→      ? Math.round(s.terminalN)
   286→      : base.terminalN;
   287→  return {
   288→    rows: rows.length >= 2 ? rows : defaultNcaRows(),
   289→    dose: cell(s.dose) ?? base.dose,
   290→    f: cell(s.f) ?? base.f,
   291→    terminalN,
   292→  };
   293→}

   295→/** Validate a persisted ML config field-by-field, falling back to defaults. */
   296→function validateMl(raw: unknown): MlConfig {
   297→  const base = DEFAULT_ML_CONFIG;
   298→  if (typeof raw !== "object" || raw === null) return base;
   299→  const p = raw as Partial<MlConfig>;
   300→  const str = (v: unknown, def: string) =>
   301→    typeof v === "string" && v.length <= 12 ? v : def;
   302→  const str128 = (v: unknown, def: string) =>
   303→    typeof v === "string" && v.length <= 128 ? v : def;
   304→  const strArr = (v: unknown): string[] =>
   305→    Array.isArray(v)
   306→      ? v.filter((c): c is string => typeof c === "string" && c.length > 0 && c.length <= 128).slice(0, 8)
   307→      : [];
   308→  return {
   309→    task: p.task === "classification" ? "classification" : "regression",
   310→    samples:
   311→      typeof p.samples === "number" && Number.isFinite(p.samples) && p.samples >= 100 && p.samples <= 1000
   312→        ? Math.round(p.samples)
   313→        : base.samples,
   314→    noise:
   315→      typeof p.noise === "number" && Number.isFinite(p.noise) && p.noise >= 0 && p.noise <= 2
   316→        ? p.noise
   317→        : base.noise,
   318→    regModel:
   319→      p.regModel === "polynomial" || p.regModel === "ridge" ? p.regModel : "linear",
   320→    alpha: str(p.alpha, base.alpha),
   321→    l2: str(p.l2, base.l2),
   322→    testSizePct:
   323→      typeof p.testSizePct === "number" && Number.isFinite(p.testSizePct) && p.testSizePct >= 5 && p.testSizePct <= 50
   324→        ? p.testSizePct
   325→        : base.testSizePct,
   326→    cv: str(p.cv, base.cv),
   327→    seed: str(p.seed, base.seed),
   328→    source: p.source === "csv" ? "csv" : "synthetic",
   329→    featureColumn: str128(p.featureColumn, ""),
   330→    featureColumn2: str128(p.featureColumn2, ""),
   331→    targetColumn: str128(p.targetColumn, ""),
   332→    featureColumns: strArr(p.featureColumns),
   333→  };
   334→}

   336→/* ---- ML result persistence (last training run: metrics + chart series) ----
   337→ * Restored field-by-field with the same defensive validation as the rest of
   338→ * the persisted state; anything malformed falls back to "no result". */
   339→
   340→function finiteArray(raw: unknown, cap: number, validate: (v: unknown) => boolean): unknown[] | null {
   341→  if (!Array.isArray(raw) || raw.length === 0 || raw.length > cap) return null;
   342→  return raw.every(validate) ? raw : null;
   343→}
   344→
   345→const isFiniteNumber = (v: unknown): v is number =>
   346→  typeof v === "number" && Number.isFinite(v);
   347→
   348→const isPoint = (v: unknown): boolean => {
   349→  if (typeof v !== "object" || v === null) return false;
   350→  const p = v as Record<string, unknown>;
   351→  return isFiniteNumber(p.x) && isFiniteNumber(p.y);
   352→};
   353→
   354→const isCurvePoint = (v: unknown): boolean => {
   355→  if (typeof v !== "object" || v === null) return false;
   356→  const p = v as Record<string, unknown>;
   357→  return isFiniteNumber(p.x) && isFiniteNumber(p.yhat);
   358→};
   359→
   360→const isBoundaryPoint = (v: unknown): boolean => {
   361→  if (typeof v !== "object" || v === null) return false;
   362→  const p = v as Record<string, unknown>;
   363→  return isFiniteNumber(p.x) && isFiniteNumber(p.y) && isFiniteNumber(p.pred);
   364→};

   366→/** Validate a persisted ML train response; null when unusable. */
   367→function validateMlResult(raw: unknown): MLTrainResponse | null {
   368→  if (typeof raw !== "object" || raw === null) return null;
   369→  const r = raw as Record<string, unknown>;
   370→  if (r.task !== "regression" && r.task !== "classification") return null;
   371→  if (typeof r.model !== "string" || r.model.length > 40) return null;
   372→
   373→  const metricsRaw = r.metrics;
   374→  if (typeof metricsRaw !== "object" || metricsRaw === null) return null;
   375→  const metrics: Record<string, number> = {};
   376→  for (const [k, v] of Object.entries(metricsRaw as Record<string, unknown>)) {
   377→    if (isFiniteNumber(v) && k.length <= 24) metrics[k] = v;
   378→  }
   379→  if (Object.keys(metrics).length === 0) return null;
   380→
   381→  const coefsRaw = r.coefficients;
   382→  if (typeof coefsRaw !== "object" || coefsRaw === null) return null;
   383→  const c = coefsRaw as Record<string, unknown>;
   384→  const coefsArr = finiteArray(c.coefs, 512, isFiniteNumber);
   385→  if (!isFiniteNumber(c.intercept) || !coefsArr || typeof c.standardized !== "boolean") {
   386→    return null;
   387→  }
   388→
   389→  const seriesRaw = r.series;
   390→  if (typeof seriesRaw !== "object" || seriesRaw === null) return null;
   391→  const s = seriesRaw as Record<string, unknown>;
   392→  const test = finiteArray(s.test, 1200, isPoint);
   393→  if (!test) return null;
   394→  const curve = finiteArray(s.curve, 400, isCurvePoint);
   395→  const boundary = finiteArray(s.boundary, 400, isBoundaryPoint);
   396→
   397→  const metaRaw = r.datasetMeta;
   398→  if (typeof metaRaw !== "object" || metaRaw === null) return null;
   399→  const m = metaRaw as Record<string, unknown>;
   400→  const features = Array.isArray(m.features)
   401→    ? m.features.filter((f): f is string => typeof f === "string" && f.length <= 24).slice(0, 12)
   402→    : [];
   403→  if (
   404→    !isFiniteNumber(m.nTotal) ||
   405→    !isFiniteNumber(m.nTrain) ||
   406→    !isFiniteNumber(m.nTest) ||
   407→    !isFiniteNumber(m.noise) ||
   408→    !isFiniteNumber(m.seed)
   409→  ) {
   410→    return null;
   411→  }
   412→
   413→  // Optional CSV provenance — preserved when well-formed, dropped otherwise.
   414→  let source: CsvSourceMeta | undefined;
   415→  if (typeof m.source === "object" && m.source !== null) {
   416→    const s = m.source as Record<string, unknown>;
   417→    const colList = (v: unknown, cap: number): string[] =>
   418→      Array.isArray(v)
   419→        ? v.filter((c): c is string => typeof c === "string" && c.length <= 128).slice(0, cap)
   420→        : [];
   421→    const columns = colList(s.columns, 64);
   422→    const featureColumns = colList(s.featureColumns, 10);
   423→    const targetColumn = typeof s.targetColumn === "string" ? s.targetColumn.slice(0, 128) : "";
   424→    if (
   425→      s.kind === "csv" &&
   426→      columns.length > 0 &&
   427→      featureColumns.length > 0 &&
   428→      targetColumn !== "" &&
   429→      isFiniteNumber(s.nRows) &&
   430→      isFiniteNumber(s.droppedRows)
   431→    ) {
   432→      const labelMap =
   433→        Array.isArray(s.labelMap) &&
   434→        s.labelMap.length === 2 &&
   435→        s.labelMap.every((v) => typeof v === "string" && v.length <= 64)
   436→          ? (s.labelMap as [string, string])
   437→          : undefined;
   438→      source = {
   439→        kind: "csv",
   440→        fileName: typeof s.fileName === "string" ? s.fileName.slice(0, 128) : undefined,
   441→        columns,
   442→        featureColumns,
   443→        targetColumn,
   444→        nRows: s.nRows,
   445→        droppedRows: s.droppedRows,
   446→        ...(labelMap ? { labelMap } : {}),
   447→      };
   448→    }
   449→  }
   450→
   451→  // Optional cross-validation summary — dropped when malformed.
   452→  let cv: MLTrainResponse["cv"];
   453→  if (typeof r.cv === "object" && r.cv !== null) {
   454→    const cvRaw = r.cv as Record<string, unknown>;
   455→    const folds = Array.isArray(cvRaw.folds)
   456→      ? cvRaw.folds.filter(
   457→          (f): f is { fold: number; value: number } =>
   458→            typeof f === "object" &&
   459→            f !== null &&
   460→            isFiniteNumber((f as Record<string, unknown>).fold) &&
   461→            isFiniteNumber((f as Record<string, unknown>).value)
   462→        )
   463→      : [];
   464→    if (
   465→      (cvRaw.k === 3 || cvRaw.k === 5 || cvRaw.k === 10) &&
   466→      typeof cvRaw.metricName === "string" &&
   467→      folds.length === cvRaw.k &&
   468→      isFiniteNumber(cvRaw.mean) &&
   469→      isFiniteNumber(cvRaw.std)
   470→    ) {
   471→      cv = {
   472→        k: cvRaw.k,
   473→        metricName: cvRaw.metricName,
   474→        folds,
   475→        mean: cvRaw.mean,
   476→        std: cvRaw.std,
   477→      };
   478→    }
   479→  }
   480→
   481→  return {
   482→    task: r.task,
   483→    model: r.model,
   484→    metrics,
   485→    cv,
   486→    coefficients: {
   487→      intercept: c.intercept as number,
   488→      coefs: coefsArr as number[],
   489→      standardized: c.standardized,
   490→    },
   491→    series: {
   492→      test: test as { x: number; y: number; label?: number }[],
   493→      ...(curve ? { curve: curve as { x: number; yhat: number }[] } : {}),
   494→      ...(boundary ? { boundary: boundary as { x: number; y: number; pred: number }[] } : {}),
   495→    },
   496→    datasetMeta: {
   497→      nTotal: m.nTotal,
   498→      nTrain: m.nTrain,
   499→      nTest: m.nTest,
   500→      noise: m.noise,
   501→      seed: m.seed,
   502→      features,
   503→      ...(source ? { source } : {}),
   504→    },
   505→  };
   506→}

   513→    if (!raw) return null;

   516→    return null; // corrupted JSON / private mode — start fresh

   520→/** Validate a persisted run-history payload; malformed entries are skipped.
   521→ * Accepts both the current `{ currentId, runs }` shape and the legacy bare
   522→ * array (current run = newest). */
   523→function validateMlHistoryPayload(raw: unknown): {
   524→  runs: MlRunRecord[];
   525→  currentId: number | null;
   526→} {
   527→  const body = Array.isArray(raw) ? { runs: raw, currentId: undefined } : raw;
   528→  if (typeof body !== "object" || body === null || !Array.isArray((body as { runs?: unknown }).runs)) {
   529→    return { runs: [], currentId: null };
   530→  }
   531→  const runs: MlRunRecord[] = [];
   532→  for (const item of ((body as { runs: unknown[] }).runs).slice(0, ML_HISTORY_LIMIT)) {
   533→    if (typeof item !== "object" || item === null) continue;
   534→    const r = item as Record<string, unknown>;
   535→    const result = validateMlResult(r.result);
   536→    const csvMeta =
   537→      typeof r.csv === "object" && r.csv !== null ? (r.csv as Record<string, unknown>) : null;
   538→    if (
   539→      !result ||
   540→      typeof r.id !== "number" ||
   541→      !Number.isFinite(r.id) ||
   542→      typeof r.at !== "string" ||
   543→      r.at.length > 40 ||
   544→      (r.task !== "regression" && r.task !== "classification") ||
   545→      typeof r.model !== "string" ||
   546→      r.model.length > 40 ||
   547→      typeof r.keyMetric !== "string" ||
   548→      r.keyMetric.length > 16 ||
   549→      typeof r.keyValue !== "number" ||
   550→      !Number.isFinite(r.keyValue)
   551→    ) {
   552→      continue;
   553→    }
   554→    runs.push({
   555→      id: r.id,
   556→      at: r.at,
   557→      task: r.task,
   558→      model: r.model,
   559→      keyMetric: r.keyMetric,
   560→      keyValue: r.keyValue,
   561→      config: validateMl(r.config),
   562→      result,
   563→      ...(csvMeta
   564→        ? {
   565→            csv: {
   566→              fileName:
   567→                typeof csvMeta.fileName === "string"
   568→                  ? csvMeta.fileName.slice(0, 128) || "pasted.csv"
   569→                  : "pasted.csv",
   570→              featureColumns: Array.isArray(csvMeta.featureColumns)
   571→                ? (csvMeta.featureColumns as unknown[])
   572→                    .filter((c): c is string => typeof c === "string" && c.length <= 128)
   573→                    .slice(0, 8)
   574→                : [],
   575→              targetColumn:
   576→                typeof csvMeta.targetColumn === "string"
   577→                  ? csvMeta.targetColumn.slice(0, 128)
   578→                  : "",
   579→            },
   580→          }
   581→        : {}),
   582→    });
   583→  }
   584→  const rawId = (body as { currentId?: unknown }).currentId;
   585→  const currentId =
   586→    typeof rawId === "number" && Number.isFinite(rawId) && runs.some((h) => h.id === rawId)
   587→      ? rawId
   588→      : (runs[0]?.id ?? null);
   589→  return { runs, currentId };
   590→}

   592→/** Read the persisted run history (client only, failure-tolerant). */
   593→function loadMlHistory(): { runs: MlRunRecord[]; currentId: number | null } {
   594→  if (typeof window === "undefined") return { runs: [], currentId: null };
   595→  try {
   596→    const raw = window.sessionStorage.getItem(ML_HISTORY_STORAGE_KEY);
   597→    if (!raw) return { runs: [], currentId: null };
   598→    return validateMlHistoryPayload(JSON.parse(raw));
   599→  } catch {
   600→    return { runs: [], currentId: null }; // corrupted JSON / private mode — start fresh
   601→  }
   602→}

   606→  if (typeof window === "undefined") return;

   611→  } catch {
   612→    /* quota exceeded or private mode — persistence is optional */
   613→  }

   632→/** Read the persisted playground snapshot (client only, failure-tolerant). */
   633→function loadPlayground(): Partial<PlaygroundPersisted> {
   634→  if (typeof window === "undefined") return {};
   635→  try {
   636→    const raw = window.sessionStorage.getItem(PLAYGROUND_STORAGE_KEY);
   637→    if (!raw) return {};
   638→    const parsed: unknown = JSON.parse(raw);
   639→    if (typeof parsed !== "object" || parsed === null) return {};
   640→    const p = parsed as Partial<PlaygroundPersisted>;
   641→    return {
   642→      playgroundTab: typeof p.playgroundTab === "string" ? p.playgroundTab.slice(0, 32) : undefined,
   643→      pkInnerTab: typeof p.pkInnerTab === "string" ? p.pkInnerTab.slice(0, 16) : undefined,
   644→      pkParams: validatePk(p.pkParams),
   645→      pdParams: validatePd(p.pdParams),
   646→      ncaState: validateNca(p.ncaState),
   647→      mlConfig: validateMl(p.mlConfig),
   648→      mlCsv: validMlCsv(p.mlCsv),
   649→    };
   650→  } catch {
   651→    return {}; // corrupted JSON / private mode — start fresh
   652→  }
   653→}

   673→/** Validate a persisted uploaded-CSV payload; null when unusable. */
   674→function validMlCsv(raw: unknown): MlCsvState | null {
   675→  if (typeof raw !== "object" || raw === null) return null;
   676→  const c = raw as Partial<MlCsvState>;
   677→  if (
   678→    typeof c.text !== "string" ||
   679→    c.text.length === 0 ||
   680→    c.text.length > MAX_CSV_TEXT ||
   681→    typeof c.fileName !== "string" ||
   682→    c.fileName.length > 128
   683→  ) {
   684→    return null;
   685→  }
   686→  return { fileName: c.fileName, text: c.text };
   687→}

   980→      });
