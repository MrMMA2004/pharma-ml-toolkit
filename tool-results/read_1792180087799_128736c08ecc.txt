    92→def _write_if_changed(path: Path, data: bytes) -> None:
    93→    """Write bytes only when they differ from what is on disk (idempotency)."""
    94→    path.parent.mkdir(parents=True, exist_ok=True)
    95→    if path.is_file() and path.read_bytes() == data:
    96→        STATS["unchanged"] += 1
    97→        return
    98→    path.write_bytes(data)
    99→    STATS["written"] += 1

   105→    frame.to_csv(buffer, **to_csv)  # type: ignore[arg-type]
   106→    _write_if_changed(path, buffer.getvalue().encode("utf-8"))

   318→def build_cellline_data() -> None:
   319→    # --- cell 3: the screen (one rng continues into cell 17's matrix) ---
   320→    rng = np.random.default_rng(42)
   321→    egfr_level = {
   322→        name: rng.normal(9.0, 0.8) if tissue == "lung" else rng.normal(5.5, 0.9)
   323→        for name, tissue in CELL_LINES.items()
   324→    }
   325→    egfr_z = (pd.Series(egfr_level) - pd.Series(egfr_level).mean()) / pd.Series(egfr_level).std()
   326→
   327→    rows = []
   328→    for cell, tissue in CELL_LINES.items():
   329→        for drug, (base, tissue_map, egfr_coef) in DRUGS.items():
   330→            ln_ic50 = base + tissue_map.get(tissue, 0.0) + egfr_coef * egfr_z[cell]
   331→            ln_ic50 += rng.normal(0.0, 0.18)
   332→            rows.append((cell, tissue, drug, ln_ic50))
   333→    screen_frame = pd.DataFrame(rows, columns=["cell_line", "tissue", "compound", "LN_IC50"])
   334→
   335→    # One duplicate pair (assay repeat) and one missing value, planted for §2.
   336→    screen_frame = pd.concat(
   337→        [screen_frame, screen_frame.iloc[[5]]], ignore_index=True
   338→    )  # duplicate of row 5
   339→    screen_frame.loc[100, "LN_IC50"] = np.nan
   340→    _write_frame(DATA_DIR / "cellline_screen.csv", screen_frame, index=False)
   341→
   342→    # --- cell 17: the expression matrix (same rng object continues) ---

   358→def make_pk_profile(ka: float = 1.2, ke: float = 0.35, f: float = 0.9, dose: float = 250.0,
   359→                    vd: float = 35.0, duration_h: float = 24.0, step_h: float = 0.5,
   360→                    error_pct: float = 0.05, seed: int = 7) -> pd.DataFrame:
   361→    """One-compartment oral profile with proportional assay noise (cell verbatim)."""
   362→    rng = np.random.default_rng(seed)
   363→    times = np.arange(0.0, duration_h + step_h, step_h)
   364→    coefficient = f * dose * ka / (vd * (ka - ke))
   365→    conc = coefficient * (np.exp(-ke * times) - np.exp(-ka * times))
   366→    conc *= 1.0 + rng.normal(scale=error_pct, size=times.size)
   367→    conc = np.clip(conc, 0.0, None)
   368→    return pd.DataFrame({"time_h": times, "conc_ng_ml": conc})

   374→    profile = make_pk_profile()

   471→# ---------------------------------------------------------------------------
   472→# complete_docking_workflow.ipynb -- cell 29 (synthetic AutoDock4 campaign)

   474→def make_dlg(seed: int, n_clusters: int = 5, total_runs: int = 50) -> str:
   475→    """Synthetic AutoDock4 DLG ending in the 'Successful Completion' sentinel."""
   476→    rng = np.random.default_rng(seed)
   477→    lowest = np.sort(rng.uniform(-11.8, -4.5, size=n_clusters))[::-1]
   478→    sizes = rng.multinomial(total_runs, [1 / n_clusters] * n_clusters) + 1
   479→    mean = lowest + rng.uniform(0.05, 0.6, size=n_clusters)   # mean sits above the lowest
   480→    lines = [
   481→        "AutoDock4 log file (demo)",
   482→        "",
   483→        "________",
   484→        "CLUSTERING HISTOGRAM",
   485→        "__________________",
   486→        "______________________________________________________________________",
   487→        "Cluster Rank | Lowest Energy | Run | Mean Energy | # in Cluster",
   488→        "_____________|_______________|_____|_____________|________________",
   489→    ]
   490→    run = int(rng.integers(1, total_runs))
   491→    for rank in range(n_clusters):
   492→        lines.append(
   493→            f"     {rank + 1}       |     {lowest[rank]:7.2f}    |  {run:3d} |"
   494→            f"    {mean[rank]:7.2f}    |   {sizes[rank]:3d} |"
   495→        )
   496→    lines += [
   497→        "____________|_______________|_____|_____________|________________",
   498→        "",
   499→        "Successful Completion",
   500→    ]
   501→    return "\n".join(lines) + "\n"
