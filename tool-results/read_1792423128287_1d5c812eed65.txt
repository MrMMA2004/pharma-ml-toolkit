   160→/** Hyperparameters with per-model defaults applied. */
   161→interface Hyper {
   162→  alpha: number;
   163→  l1Ratio: number;
   164→  k: number;
   165→  maxDepth: number;

   175→
   176→function resolveHyper(req: TrainBody): Hyper {
   177→  return {
   178→    alpha: num(req.alpha, 1),
   179→    l1Ratio: num(req.l1Ratio, 0.5),
   180→    k: num(req.k, 5),
   181→    maxDepth: num(req.maxDepth, 4),
   182→    nEstimators: num(req.nEstimators, 50),
   183→    rounds: num(req.rounds, 50),
   184→    learningRate: num(req.learningRate, 0.1),
   185→    hidden: num(req.hidden, 16),
   186→    epochs: num(req.epochs, 200),
   187→    lr: num(req.lr, 0.05),
   188→    C: num(req.C, 1),
   189→    l2: num(req.l2, 1),
   190→  };
   191→}
   192→
   193→/** Deterministic RNG stream for model internals (bagging / MLP init).
   194→ *  Slot 0 = the main run, slots 1..k = the CV folds — every training run
   195→ *  gets its own stream so identical requests stay byte-identical. */
   196→function modelRng(effSeed: number, slot: number): () => number {
   197→  return mulberry32((effSeed + 104729 + 6151 * slot) >>> 0);
   198→}
   199→
   200→function num(value: number | undefined, fallback: number): number {
   201→  return typeof value === "number" && Number.isFinite(value) ? value : fallback;
   202→}
   203→
   204→/** Count numeric columns without importing the resolver (for mode choice). */
   205→function numericCandidates(table: { columns: string[]; rows: string[][] }): number {
   206→  let count = 0;
   207→  for (let c = 0; c < table.columns.length; c++) {
   208→    let nonEmpty = 0;
   209→    let parseable = 0;
   210→    for (const row of table.rows) {
   211→      const cell = row[c] ?? "";
   212→      if (cell === "") continue;
   213→      nonEmpty++;
   214→      if (Number.isFinite(Number(cell))) parseable++;
   215→    }
   216→    if (nonEmpty > 0 && parseable / nonEmpty >= 0.9) count++;
   217→  }
   218→  return count;
   219→}
   220→
   221→/** Min/max via loop — safe for very large CSVs (spread would overflow). */
   222→function extent(values: number[]): [number, number] {
   223→  let lo = values[0] ?? 0;
   224→  let hi = lo;
   225→  for (const v of values) {
   226→    if (v < lo) lo = v;
   227→    if (v > hi) hi = v;
   228→  }
   229→  return [lo, hi];
   230→}
   231→
   232→/** k-fold CV helper: runs `fitFold` per fold with its own RNG slot and
   233→ *  reduces fold values to mean ± population std. */
   234→function runCv(
   235→  n: number,
   236→  k: number,
   237→  effSeed: number,
   238→  metricName: string,
   239→  fitFold: (fold: number[], trainIdx: number[], slot: number) => number
   240→): MLTrainResponse["cv"] {
   241→  if (k <= 1) return undefined;
   242→  const cvRand = mulberry32((effSeed + 7919) >>> 0);
   243→  const folds = kFoldSplit(n, k, cvRand);
   244→  const foldResults: CVFold[] = folds.map((fold, i) => {
   245→    const foldSet = new Set(fold);
   246→    const trainIdx = Array.from({ length: n }, (_, j) => j).filter((j) => !foldSet.has(j));
   247→    return { fold: i + 1, value: fitFold(fold, trainIdx, i + 1) };
   248→  });
   249→  const values = foldResults.map((f) => f.value);
   250→  return {
   251→    k,
   252→    metricName,
   253→    folds: foldResults,
   254→    mean: mean(values),
   255→    std: populationStd(values),
   256→  };
   257→}

   259→export async function POST(request: NextRequest) {
   260→  let body: unknown;
   261→  try {
   262→    body = await request.json();
   263→  } catch {
   264→    return NextResponse.json({ error: "Request body must be valid JSON." }, { status: 400 });
   265→  }
   266→
   267→  const parsed = bodySchema.safeParse(body);
   268→  if (!parsed.success) {
   269→    return NextResponse.json(
   270→      {
   271→        error: "Invalid request body.",
   272→        detail: parsed.error.issues.map((i) => `${i.path.join(".") || "body"}: ${i.message}`),
   273→      },
   274→      { status: 400 }
   275→    );
   276→  }
   277→  const req = parsed.data;
   278→
   279→  /* Uploaded-CSV mode: parse + resolve columns up front so bad files/picks
   280→   * surface as actionable 400s instead of opaque 500s. */
   281→  let csvData: CsvRegressionData | CsvClassificationData | CsvMultiRegressionData | undefined;
   282→  let csvName: string | undefined;
   283→  let splitSeed = req.seed;
   284→  if (req.csv !== undefined) {
   285→    csvName = req.csvName ?? "pasted.csv";
   286→    const table = parseCsv(req.csv);
   287→    if (table.rows.length === 0) {
   288→      return NextResponse.json(
   289→        { error: "The uploaded CSV has no data rows below the header." },
   290→        { status: 400 }
   291→      );
   292→    }
   293→    try {
   294→      if (req.task === "regression") {
   295→        /* Multivariate when multiple feature columns are requested (or auto
   296→         * with ≥2 candidate features); single-feature keeps the x↔y chart. */
   297→        const multi =
   298→          (req.featureColumns?.length ?? 0) >= 2 ||
   299→          (req.featureColumns === undefined &&
   300→            !req.featureColumn &&
   301→            numericCandidates(table) >= 3);
   302→        csvData = multi
   303→          ? resolveMultiRegression(table, req.featureColumns, req.targetColumn)
   304→          : resolveRegression(table, req.featureColumn ?? req.featureColumns?.[0], req.targetColumn);
   305→      } else {
   306→        csvData = resolveClassification(
   307→          table,
   308→          req.featureColumn,
   309→          req.featureColumn2,
   310→          req.targetColumn
   311→        );
   312→      }
   313→    } catch (err) {
   314→      return NextResponse.json(
   315→        {
   316→          error: "Could not use the uploaded CSV.",
   317→          detail: [err instanceof Error ? err.message : String(err)],
   318→        },
   319→        { status: 400 }
   320→      );
   321→    }
   322→    /* Same file → same split: derive the split seed from the file content. */
   323→    splitSeed = hashCsvSeed(req.csv);
   324→  }
   325→
   326→  try {
   327→    let result: MLTrainResponse;
   328→    if (req.task === "regression") {
   329→      result =
   330→        csvData && "X" in csvData
   331→          ? runMultiRegression(req, csvData, csvName, splitSeed)
   332→          : runRegression(req, csvData as CsvRegressionData | undefined, csvName, splitSeed);
   333→    } else {
   334→      result = runClassification(
   335→        req,
   336→        csvData as CsvClassificationData | undefined,
   337→        csvName,
   338→        splitSeed
   339→      );
   340→    }
   341→    return NextResponse.json(result);
   342→  } catch (err) {
   343→    return NextResponse.json(
   344→      {
   345→        error: "Training failed.",
   346→        detail: err instanceof Error ? err.message : String(err),
   347→      },
   348→      { status: 500 }
   349→    );
   350→  }
   351→}

   357→/** What every regression fit produces: a row predictor plus the honest
   358→ *  parameter summary for the model family (coefficients for the linear
   359→ *  family, normalized impurity-reduction importances for trees/ensembles,
   360→ *  echoed hyperparameters for kNN/MLP). */
   361→interface RegFit {
   362→  predict(xsRaw: number[]): number;
   363→  coefficients?: { intercept: number; coefs: number[]; standardized: boolean };
   364→  featureImportance?: { features: string[]; importance: number[]; measure: "variance" | "gini" };
   365→  modelParams?: Record<string, number>;
   366→}
   367→
   368→/** Fit any regression model over a raw (unstandardized) feature matrix. */
   369→function fitRegressionModel(
   370→  model: string,
   371→  XTrain: number[][],
   372→  yTrain: number[],
   373→  hp: Hyper,
   374→  rand: () => number,
   375→  featureNames: string[]
   376→): RegFit {
   377→  switch (model) {
   378→    case "linear":
   379→    case "polynomial":
   380→    case "ridge": {
   381→      /* Multivariate closed-form path (degree is 1 here; the single-feature
   382→       * polynomial case is handled by the caller). */
   383→      const effAlpha = model === "ridge" ? hp.alpha : 0;
   384→      const m = fitMultivariateRegression(XTrain, yTrain, { alpha: effAlpha });
   385→      return {
   386→        predict: (xs) => m.predict(xs),
   387→        coefficients: { intercept: m.w[0] ?? 0, coefs: m.w.slice(1), standardized: true },
   388→      };
   389→    }
   390→    case "lasso":
   391→    case "elastic_net": {
   392→      const l1Ratio = model === "lasso" ? 1 : hp.l1Ratio;
   393→      const m = fitElasticNet(XTrain, yTrain, { alpha: hp.alpha, l1Ratio });
   394→      return {
   395→        predict: (xs) => m.predict(xs),
   396→        coefficients: { intercept: m.b, coefs: [...m.w], standardized: true },
   397→        modelParams:
   398→          model === "elastic_net"
   399→            ? { alpha: hp.alpha, l1Ratio }
   400→            : { alpha: hp.alpha },
   401→      };
   402→    }
   403→    case "knn": {
   404→      const m = fitKnnRegressor(XTrain, yTrain, { k: hp.k });
   405→      return { predict: (xs) => m.predict(xs), modelParams: { k: hp.k } };
   406→    }
   407→    case "decision_tree": {
   408→      const m = fitRegressionTree(XTrain, yTrain, { maxDepth: hp.maxDepth });
   409→      return {
   410→        predict: (xs) => m.predict(xs),
   411→        featureImportance: {
   412→          features: featureNames,
   413→          importance: normalizeImportance(m.rawImportance),
   414→          measure: "variance",
   415→        },
   416→      };
   417→    }
   418→    case "random_forest": {
   419→      const m = fitRandomForestRegressor(XTrain, yTrain, {
   420→        nEstimators: hp.nEstimators,
   421→        maxDepth: hp.maxDepth,
   422→        rand,
   423→      });
   424→      return {
   425→        predict: (xs) => m.predict(xs),
   426→        featureImportance: {
   427→          features: featureNames,
   428→          importance: m.importance,
   429→          measure: "variance",
   430→        },
   431→      };
   432→    }
   433→    case "gradient_boosting": {
   434→      const m = fitGradientBoostingRegressor(XTrain, yTrain, {
   435→        rounds: hp.rounds,
   436→        learningRate: hp.learningRate,
   437→      });
   438→      return {
   439→        predict: (xs) => m.predict(xs),
   440→        featureImportance: {
   441→          features: featureNames,
   442→          importance: m.importance,
   443→          measure: "variance",
   444→        },
   445→      };
   446→    }
   447→    case "mlp": {
   448→      const m = fitMlpRegressor(XTrain, yTrain, {
   449→        hidden: hp.hidden,
   450→        epochs: hp.epochs,
   451→        lr: hp.lr,
   452→        rand,
   453→      });
   454→      return {
   455→        predict: (xs) => m.predict(xs),
   456→        modelParams: { hidden: hp.hidden, epochs: hp.epochs, lr: hp.lr },
   457→      };
   458→    }
   459→    default:
   460→      throw new Error(`Unknown regression model "${model}".`);
   461→  }
   462→}

   464→function runRegression(
   465→  req: TrainBody,
   466→  csv?: CsvRegressionData,
   467→  csvName?: string,
   468→  splitSeed?: number
   469→): MLTrainResponse {
   470→  let xs: number[];
   471→  let ys: number[];
   472→  let source: CsvSourceMeta | undefined;
   473→  let features: string[];
   474→  let featureNames: string[];
   475→  if (csv) {
   476→    xs = zscore(csv.x); // standardized so coefficients stay comparable
   477→    ys = csv.y; // target kept in original units
   478→    source = {
   479→      kind: "csv",
   480→      fileName: csvName,
   481→      columns: csv.columns,
   482→      featureColumns: [csv.feature],
   483→      targetColumn: csv.target,
   484→      nRows: csv.x.length,
   485→      droppedRows: csv.droppedRows,
   486→    };
   487→    features = [`${csv.feature} (standardized)`];
   488→    featureNames = [csv.feature];
   489→  } else {
   490→    const rand = mulberry32(req.seed);
   491→    ({ xs, ys } = makeRegressionData(req.samples, req.noise, rand));
   492→    features = ["x (standardized)"];
   493→    featureNames = ["x"];
   494→  }
   495→  const effSeed = splitSeed ?? req.seed;
   496→  const rand = mulberry32(effSeed);
   497→  const n = xs.length;
   498→  const { train, test } = shuffledSplit(n, req.testSize, rand);
   499→
   500→  const hp = resolveHyper(req);
   501→  const isLegacy = req.model === "linear" || req.model === "polynomial" || req.model === "ridge";
   502→  /* Legacy single-feature models keep the exact original call (degree for
   503→   * polynomial, ridge alpha on the diagonal) so responses stay
   504→   * byte-compatible; every other model fits over the 1-column matrix. */
   505→  const degree = req.model === "polynomial" ? 2 : 1;
   506→  const effAlpha = req.model === "ridge" ? hp.alpha : 0;
   507→
   508→  const trainX = train.map((i) => [xs[i] ?? 0]);
   509→  const trainY = train.map((i) => ys[i] ?? 0);
   510→
   511→  const fit: RegFit = isLegacy
   512→    ? (() => {
   513→        const m = fitLinearRegression(trainX.map((r) => r[0] ?? 0), trainY, {
   514→          alpha: effAlpha,
   515→          degree,
   516→        });
   517→        return {
   518→          predict: (row) => m.predict(row[0] ?? 0),
   519→          coefficients: {
   520→            intercept: m.w[0] ?? 0,
   521→            coefs: m.w.slice(1),
   522→            standardized: true,
   523→          },
   524→        };
   525→      })()
   526→    : fitRegressionModel(req.model, trainX, trainY, hp, modelRng(effSeed, 0), featureNames);
   527→
   528→  const yTrue = test.map((i) => ys[i] ?? 0);
   529→  const yPred = test.map((i) => fit.predict([xs[i] ?? 0]));
   530→  const metrics = regressionMetrics(yTrue, yPred);
   531→
   532→  // Smooth fitted curve across the full observed x range.
   533→  const [xMin, xMax] = extent(xs);
   534→  const curve = Array.from({ length: 100 }, (_, i) => {
   535→    const x = xMin + ((xMax - xMin) * i) / 99;
   536→    return { x, yhat: fit.predict([x]) };
   537→  });
   538→
   539→  // k-fold CV (R² per fold) with an independent, seeded shuffle.
   540→  const cv = runCv(n, req.cv, effSeed, "R²", (fold, trainIdx, slot) => {
   541→    if (isLegacy) {
   542→      const foldModel = fitLinearRegression(
   543→        trainIdx.map((i) => xs[i] ?? 0),
   544→        trainIdx.map((i) => ys[i] ?? 0),
   545→        { alpha: effAlpha, degree }
   546→      );
   547→      const fTrue = fold.map((i) => ys[i] ?? 0);
   548→      const fPred = fold.map((i) => foldModel.predict(xs[i] ?? 0));
   549→      return regressionMetrics(fTrue, fPred).r2;
   550→    }
   551→    const foldFit = fitRegressionModel(
   552→      req.model,
   553→      trainIdx.map((i) => [xs[i] ?? 0]),
   554→      trainIdx.map((i) => ys[i] ?? 0),
   555→      hp,
   556→      modelRng(effSeed, slot),
   557→      featureNames
   558→    );
   559→    const fTrue = fold.map((i) => ys[i] ?? 0);
   560→    const fPred = fold.map((i) => foldFit.predict([xs[i] ?? 0]));
   561→    return regressionMetrics(fTrue, fPred).r2;
   562→  });
   563→
   564→  return {
   565→    task: "regression",
   566→    model: req.model,
   567→    metrics: { r2: metrics.r2, rmse: metrics.rmse, mae: metrics.mae },
   568→    cv,
   569→    ...(fit.coefficients ? { coefficients: fit.coefficients } : {}),
   570→    ...(fit.featureImportance ? { featureImportance: fit.featureImportance } : {}),
   571→    ...(fit.modelParams ? { modelParams: fit.modelParams } : {}),
   572→    series: {
   573→      test: test.map((i) => ({ x: xs[i] ?? 0, y: ys[i] ?? 0 })),
   574→      curve,
   575→    },
   576→    datasetMeta: {
   577→      nTotal: n,
   578→      nTrain: train.length,
   579→      nTest: test.length,
   580→      noise: csv ? 0 : req.noise,
   581→      seed: csv ? 0 : req.seed,
   582→      features,
   583→      ...(source ? { source } : {}),
   584→    },
   585→  };
   586→}
   587→
   588→/* ------------------------------------------------------------------ */
   589→/* Multivariate regression (uploaded CSV, k ≥ 2 features)              */
   590→/* ------------------------------------------------------------------ */
   591→
   592→/**
   593→ * Multivariate CSV regression. The chart becomes predicted-vs-actual:
   594→ * series.test carries {x: ŷ, y: actual} and `curve` is the perfect-prediction
   595→ * identity line, so the existing chart renderer works unchanged.
   596→ */
   597→function runMultiRegression(
   598→  req: TrainBody,
   599→  csv: CsvMultiRegressionData,
   600→  csvName: string | undefined,
   601→  splitSeed: number
   602→): MLTrainResponse {
   603→  const effSeed = splitSeed ?? req.seed;
   604→  const rand = mulberry32(effSeed);
   605→  const n = csv.y.length;
   606→  const { train, test } = shuffledSplit(n, req.testSize, rand);
   607→
   608→  const hp = resolveHyper(req);
   609→  const trainX = train.map((i) => csv.X[i] ?? []);
   610→  const trainY = train.map((i) => csv.y[i] ?? 0);
   611→  const fit = fitRegressionModel(req.model, trainX, trainY, hp, modelRng(effSeed, 0), csv.features);
   612→
   613→  const yTrue = test.map((i) => csv.y[i] ?? 0);
   614→  const yPred = test.map((i) => fit.predict(csv.X[i] ?? []));
   615→  const metrics = regressionMetrics(yTrue, yPred);
   616→
   617→  /* Perfect-prediction identity line over the combined ŷ/y extent. */
   618→  const all = [...yTrue, ...yPred];
   619→  const [lo, hi] = extent(all);
   620→  const curve = Array.from({ length: 100 }, (_, i) => {
   621→    const v = lo + ((hi - lo) * i) / 99;
   622→    return { x: v, yhat: v };
   623→  });
   624→
   625→  const source: CsvSourceMeta = {
   626→    kind: "csv",
   627→    fileName: csvName,
   628→    columns: csv.columns,
   629→    featureColumns: csv.features,
   630→    targetColumn: csv.target,
   631→    nRows: n,
   632→    droppedRows: csv.droppedRows,
   633→  };
   634→
   635→  // k-fold CV (R² per fold) with an independent, seeded shuffle.
   636→  const cv = runCv(n, req.cv, effSeed, "R²", (fold, trainIdx, slot) => {
   637→    const foldFit = fitRegressionModel(
   638→      req.model,
   639→      trainIdx.map((i) => csv.X[i] ?? []),
   640→      trainIdx.map((i) => csv.y[i] ?? 0),
   641→      hp,
   642→      modelRng(effSeed, slot),
   643→      csv.features
   644→    );
   645→    const fTrue = fold.map((i) => csv.y[i] ?? 0);
   646→    const fPred = fold.map((i) => foldFit.predict(csv.X[i] ?? []));
   647→    return regressionMetrics(fTrue, fPred).r2;
   648→  });
   649→
   650→  return {
   651→    task: "regression",
   652→    model: req.model,
   653→    metrics: { r2: metrics.r2, rmse: metrics.rmse, mae: metrics.mae },
   654→    cv,
   655→    ...(fit.coefficients ? { coefficients: fit.coefficients } : {}),
   656→    ...(fit.featureImportance ? { featureImportance: fit.featureImportance } : {}),
   657→    ...(fit.modelParams ? { modelParams: fit.modelParams } : {}),
   658→    series: {
   659→      test: yPred.map((yhat, i) => ({ x: yhat, y: yTrue[i] ?? 0 })),
   660→      curve,
   661→    },
   662→    datasetMeta: {
   663→      nTotal: n,
   664→      nTrain: train.length,
   665→      nTest: test.length,
   666→      noise: 0,
   667→      seed: 0,
   668→      features: csv.features.map((f) => `${f} (standardized)`),
   669→      source,
   670→    },
   671→  };
   672→}
   673→
   674→/* ------------------------------------------------------------------ */
   675→/* Classification                                                      */
   676→/* ------------------------------------------------------------------ */
   677→
   678→/** What every classifier produces: a hard 0/1 prediction over raw feature
   679→ *  pairs plus the honest parameter summary for the model family. */
   680→interface ClsFit {
   681→  predict01(xRaw: number, yRaw: number): 0 | 1;
   682→  coefficients?: { intercept: number; coefs: number[]; standardized: boolean };
   683→  featureImportance?: { features: string[]; importance: number[]; measure: "variance" | "gini" };
   684→  modelParams?: Record<string, number>;
   685→}
   686→
   687→/** Fit any classification model over raw (unstandardized) 2-feature rows. */
   688→function fitClassificationModel(
   689→  model: string,
   690→  XTrain: number[][],
   691→  labelsTrain: number[],
   692→  hp: Hyper,
   693→  rand: () => number,
   694→  featureNames: string[]
   695→): ClsFit {
   696→  switch (model) {
   697→    case "logistic": {
   698→      /* Legacy path: identical trainer invocation (batch GD, lr 0.1,
   699→       * 800 epochs, L2 with bias exempt) for byte compatibility. */
   700→      const pts = XTrain.map((row, i) => ({
   701→        x: row[0] ?? 0,
   702→        y: row[1] ?? 0,
   703→        label: labelsTrain[i] ?? 0,
   704→      }));
   705→      const m = fitLogisticRegression(pts, { l2: hp.l2, lr: 0.1, epochs: 800 });
   706→      return {
   707→        predict01: (x, y) => (m.predict(x, y) >= 0.5 ? 1 : 0),
   708→        coefficients: { intercept: m.bias, coefs: [...m.w], standardized: true },
   709→      };
   710→    }
   711→    case "knn": {
   712→      const m = fitKnnClassifier(XTrain, labelsTrain, { k: hp.k });
   713→      return {
   714→        predict01: (x, y) => m.predict([x, y]),
   715→        modelParams: { k: hp.k },
   716→      };
   717→    }
   718→    case "decision_tree": {
   719→      const m = fitClassificationTree(XTrain, labelsTrain, { maxDepth: hp.maxDepth });
   720→      return {
   721→        predict01: (x, y) => m.predict([x, y]),
   722→        featureImportance: {
   723→          features: featureNames,
   724→          importance: normalizeImportance(m.rawImportance),
   725→          measure: "gini",
   726→        },
   727→      };
   728→    }
   729→    case "random_forest": {
   730→      const m = fitRandomForestClassifier(XTrain, labelsTrain, {
   731→        nEstimators: hp.nEstimators,
   732→        maxDepth: hp.maxDepth,
   733→        rand,
   734→      });
   735→      return {
   736→        predict01: (x, y) => m.predict([x, y]),
   737→        featureImportance: {
   738→          features: featureNames,
   739→          importance: m.importance,
   740→          measure: "gini",
   741→        },
   742→      };
   743→    }
   744→    case "gaussian_nb": {
   745→      const m = fitGaussianNB(XTrain, labelsTrain);
   746→      return {
   747→        predict01: (x, y) => m.predict([x, y]),
   748→        modelParams: { varSmoothing: 1e-9 },
   749→      };
   750→    }
   751→    case "svm_linear": {
   752→      const m = fitLinearSvm(XTrain, labelsTrain, { C: hp.C });
   753→      return {
   754→        predict01: (x, y) => m.predict([x, y]),
   755→        modelParams: { C: hp.C },
   756→      };
   757→    }
   758→    case "mlp": {
   759→      const m = fitMlpClassifier(XTrain, labelsTrain, {
   760→        hidden: hp.hidden,
   761→        epochs: hp.epochs,
   762→        lr: hp.lr,
   763→        rand,
   764→      });
   765→      return {
   766→        predict01: (x, y) => m.predict([x, y]),
   767→        modelParams: { hidden: hp.hidden, epochs: hp.epochs, lr: hp.lr },
   768→      };
   769→    }
   770→    default:
   771→      throw new Error(`Unknown classification model "${model}".`);
   772→  }
   773→}
   774→
   775→function runClassification(
   776→  req: TrainBody,
   777→  csv?: CsvClassificationData,
   778→  csvName?: string,
   779→  splitSeed?: number
   780→): MLTrainResponse {
   781→  let pts: Array<{ x: number; y: number; label: number }>;
   782→  let source: CsvSourceMeta | undefined;
   783→  let features: string[];
   784→  let featureNames: string[];
   785→  if (csv) {
   786→    // Standardize both features (same space the logistic trainer expects).
   787→    const { x1, x2, label } = csv;
   788→    const sx1 = zscore(x1);
   789→    const sx2 = zscore(x2);
   790→    pts = sx1.map((x, i) => ({ x, y: sx2[i] ?? 0, label: label[i] ?? 0 }));
   791→    source = {
   792→      kind: "csv",
   793→      fileName: csvName,
   794→      columns: csv.columns,
   795→      featureColumns: [csv.feature1, csv.feature2],
   796→      targetColumn: csv.target,
   797→      nRows: csv.x1.length,
   798→      droppedRows: csv.droppedRows,
   799→      labelMap: csv.labelMap,
   800→    };
   801→    features = [`${csv.feature1} (standardized)`, `${csv.feature2} (standardized)`];
   802→    featureNames = [csv.feature1, csv.feature2];
   803→  } else {
   804→    const genRand = mulberry32(req.seed);
   805→    pts = makeMoons(req.samples, req.noise, genRand);
   806→    features = ["x1 (standardized)", "x2 (standardized)"];
   807→    featureNames = ["x1", "x2"];
   808→  }
   809→  const effSeed = splitSeed ?? req.seed;
   810→  const rand = mulberry32(effSeed);
   811→  const n = pts.length;
   812→  const { train, test } = shuffledSplit(n, req.testSize, rand);
   813→
   814→  const hp = resolveHyper(req);
   815→  const trainX = train.map((i) => [pts[i]?.x ?? 0, pts[i]?.y ?? 0]);
   816→  const trainLabels = train.map((i) => pts[i]?.label ?? 0);
   817→  const fit = fitClassificationModel(
   818→    req.model,
   819→    trainX,
   820→    trainLabels,
   821→    hp,
   822→    modelRng(effSeed, 0),
   823→    featureNames
   824→  );
   825→
   826→  const yTrue = test.map((i) => pts[i]?.label ?? 0);
   827→  const yPred = test.map((i) => fit.predict01(pts[i]?.x ?? 0, pts[i]?.y ?? 0));
   828→  const metrics = classificationMetrics(yTrue, yPred);
   829→
   830→  // Decision boundary: dense grid classified by the trained model.
   831→  const xs = pts.map((p) => p.x);
   832→  const ys = pts.map((p) => p.y);
   833→  const [xMin, xMax] = extent(xs);
   834→  const [yMin, yMax] = extent(ys);
   835→  const G = 60;
   836→  const boundary: Array<{ x: number; y: number; pred: number }> = [];
   837→  for (let gi = 0; gi < G; gi++) {
   838→    for (let gj = 0; gj < G; gj++) {
   839→      const x = (xMin - 0.5) + ((xMax + 0.5 - (xMin - 0.5)) * gi) / (G - 1);
   840→      const y = (yMin - 0.5) + ((yMax + 0.5 - (yMin - 0.5)) * gj) / (G - 1);
   841→      boundary.push({ x, y, pred: fit.predict01(x, y) });
   842→    }
   843→  }
   844→
   845→  // k-fold CV (accuracy per fold).
   846→  const cv = runCv(n, req.cv, effSeed, "Accuracy", (fold, trainIdx, slot) => {
   847→    const foldFit = fitClassificationModel(
   848→      req.model,
   849→      trainIdx.map((i) => [pts[i]?.x ?? 0, pts[i]?.y ?? 0]),
   850→      trainIdx.map((i) => pts[i]?.label ?? 0),
   851→      hp,
   852→      modelRng(effSeed, slot),
   853→      featureNames
   854→    );
   855→    let correct = 0;
   856→    for (const i of fold) {
   857→      const p = pts[i] ?? { x: 0, y: 0, label: 0 };
   858→      const pred = foldFit.predict01(p.x, p.y);
   859→      if (pred === p.label) correct++;
   860→    }
   861→    return correct / fold.length;
   862→  });
   863→
   864→  return {
   865→    task: "classification",
   866→    model: req.model,
   867→    metrics: {
   868→      accuracy: metrics.accuracy,
   869→      precision: metrics.precision,
   870→      recall: metrics.recall,
   871→      f1: metrics.f1,
   872→    },
   873→    cv,
   874→    ...(fit.coefficients ? { coefficients: fit.coefficients } : {}),
   875→    ...(fit.featureImportance ? { featureImportance: fit.featureImportance } : {}),
   876→    ...(fit.modelParams ? { modelParams: fit.modelParams } : {}),
   877→    series: {
   878→      test: test.map((i) => ({
   879→        x: pts[i]?.x ?? 0,
   880→        y: pts[i]?.y ?? 0,
   881→        label: pts[i]?.label ?? 0,
   882→      })),
   883→      boundary,
   884→    },
   885→    datasetMeta: {
   886→      nTotal: n,
   887→      nTrain: train.length,
   888→      nTest: test.length,
   889→      noise: csv ? 0 : req.noise,
   890→      seed: csv ? 0 : req.seed,
   891→      features,
   892→      ...(source ? { source } : {}),
   893→    },
   894→  };
   895→}
