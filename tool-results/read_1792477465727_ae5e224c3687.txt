     1→"""Unified model registry.
     2→
     3→The legacy codebase kept one ~150-line script per algorithm (30+ files
     4→of near-duplicate load/preprocess/tune/evaluate boilerplate).  This
     5→module collapses them into a registry: each entry provides the
     6→estimator factory, the task(s) it supports, a tuned default parameter
     7→grid and an availability probe.  One generic trainer then serves every
     8→algorithm.
     9→
    10→Optional backends (XGBoost, LightGBM, CatBoost, PyTorch deep nets) are
    11→imported lazily and reported as unavailable rather than crashing the
    12→registry import.
    13→
    14→Examples
    15→--------
    16→>>> from pharmaml.ml.models import MODEL_REGISTRY
    17→>>> spec = MODEL_REGISTRY["random_forest"]
    18→>>> spec.task
    19→'both'
    20→>>> sorted(spec.param_grid)
    21→['max_depth', 'min_samples_split', 'n_estimators']
    22→>>> spec.available            # sklearn-backed: no optional dependency
    23→True
    24→"""

    70→def _module_available(module: str) -> bool:
    71→    return importlib.util.find_spec(module) is not None

    74→@dataclass(frozen=True, slots=True)
    75→class ModelSpec:
    76→    """Registry entry for one algorithm family.
    77→
    78→    Attributes
    79→    ----------
    80→    name:
    81→        Registry key (kebab-case, e.g. ``'random_forest'``).
    82→    task:
    83→        ``'classification'``, ``'regression'`` or ``'both'``.
    84→    factory:
    85→        Zero-argument callable returning a fresh, unfitted estimator
    86→        (random_state already threaded in where the estimator accepts
    87→        one).  ``None`` for dual-task entries whose per-task estimator
    88→        is built by :data:`_TASK_FACTORIES` inside :func:`build_model`.
    89→    param_grid:
    90→        Compact default grid for hyperparameter search.
    91→    requires:
    92→        Optional third-party module needed for the factory.
    93→    notes:
    94→        Short human description used by ``list_models`` output.
    95→    """

   510→def resolve_model_name(name: str) -> str:
   511→    """Normalise a user-supplied model name to a registry key.
   512→
   513→    Applies case/whitespace/punctuation clean-up, then the
   514→    :data:`MODEL_ALIASES` short-name map, so ``'Random-Forest'``,
   515→    ``'rf'`` and ``'random_forest'`` all resolve to the same key.
   516→
   517→    Raises
   518→    ------
   519→    ModelError
   520→        When the cleaned name is neither a registry key nor an alias.
   521→    """
   522→    lowered = str(name).strip().lower()
   523→    underscored = lowered.replace(" ", "_").replace("-", "_")
   524→    for candidate in (underscored, MODEL_ALIASES.get(lowered), MODEL_ALIASES.get(underscored)):
   525→        if candidate in MODEL_REGISTRY:
   526→            return candidate
   527→    known = ", ".join(sorted(MODEL_REGISTRY))
   528→    raise ModelError(f"Unknown model {name!r}. Registered models: {known}")

   588→    """List registry entries with availability information.
   589→
   590→    ``task`` filters by kind: ``'classification'``, ``'regression'`` or
   591→    ``'both'`` (models usable for either task); ``None`` (the default)
   592→    lists every entry.
   593→
   594→    Returns a list of dicts with keys ``name``, ``task``, ``available``,
   595→    ``requires`` and ``notes`` -- ready for pretty printing or the CLI.
   596→
   597→    Raises
   598→    ------
   599→    ModelError
   600→        When ``task`` is not one of the accepted kinds.  A typo'd value
   601→        such as ``'clasification'`` used to silently return only the
   602→        dual-task models instead of complaining.
   603→    """
   604→    if task is not None and task not in {"classification", "regression", "both"}:
   605→        raise ModelError(
   606→            f"task must be 'classification', 'regression' or 'both', got {task!r}"
   607→        )
   608→    entries = []
   609→    for spec in MODEL_REGISTRY.values():
   610→        if task and spec.task != "both" and spec.task != task:
   611→            continue
   612→        if only_available and not spec.available:
   613→            continue
   614→        entries.append(
   615→            {
   616→                "name": spec.name,
   617→                "task": spec.task,
   618→                "available": spec.available,
   619→                "requires": spec.requires,
   620→                "notes": spec.notes,
   621→            }
   622→        )
   623→    return entries
   624→
   625→
   626→def _accepted_param_names(estimator: Any) -> set[str]:
   627→    """Parameter names :func:`build_model` accepts in ``params=``.
   628→
   629→    ``get_params()`` alone under-reports for estimators like CatBoost,
   630→    whose ``get_params()`` returns only the *explicitly set* constructor
   631→    kwargs — every real CatBoost hyperparameter ('iterations', 'depth',
   632→    'learning_rate', ...) would look unknown and be rejected with a
   633→    spurious :class:`ModelError`.  The constructor signature fills the
   634→    gap, but only when ``__init__`` declares no ``**kwargs``: a
   635→    kwargs-catching constructor accepts arbitrary names, so its named
   636→    parameters say nothing about the accepted set (there
   637→    :meth:`set_params` stays the rejector, and :func:`build_model`
   638→    wraps its rejections as the typed error).
   639→    """
   640→    accepted = set(estimator.get_params())
   641→    try:
   642→        parameters = inspect.signature(type(estimator).__init__).parameters
   643→    except (TypeError, ValueError):  # pragma: no cover - exotic constructors
   644→        return accepted
   645→    if any(param.kind is inspect.Parameter.VAR_KEYWORD for param in parameters.values()):
   646→        return accepted
   647→    accepted.update(
   648→        name
   649→        for name, param in parameters.items()
   650→        if name != "self"
   651→        and param.kind
   652→        in (inspect.Parameter.POSITIONAL_OR_KEYWORD, inspect.Parameter.KEYWORD_ONLY)
   653→    )
   654→    return accepted
   655→

   657→def build_model(
   658→    name: str,
   659→    task: str,
   660→    params: dict[str, Any] | None = None,
   661→) -> Any:
   662→    """Instantiate an unfitted estimator from the registry.
   663→
   664→    Parameters
   665→    ----------
   666→    name:
   667→        Registry key, e.g. ``'random_forest'``.
   668→    task:
   669→        ``'classification'`` or ``'regression'``.
   670→    params:
   671→        Estimator keyword overrides (e.g. ``{'n_estimators': 500}``).
   672→        Names are validated against ``get_params()`` ∪ the constructor
   673→        signature, so real CatBoost hyperparameters ('iterations',
   674→        'depth', ...) are accepted despite CatBoost's partial
   675→        ``get_params()``; genuinely unknown names raise
   676→        :class:`ModelError`, as do ``set_params`` rejections.
   677→
   678→    Returns
   679→    -------
   680→    sklearn-compatible estimator
   681→
   682→    Raises
   683→    ------
   684→    ModelError
   685→        Unknown name or task; ``DependencyError`` when the backend is
   686→        not installed.
   687→    """
   688→    key = resolve_model_name(name)
   689→    if task not in {"classification", "regression"}:
   690→        raise ModelError(f"task must be 'classification' or 'regression', got {task!r}")
   691→
   692→    spec = MODEL_REGISTRY[key]
   693→    if task != "both" and spec.task != "both" and spec.task != task:
   694→        raise ModelError(f"Model {key!r} supports {spec.task!r}, not {task!r}")
   695→    requires = spec.requires
   696→    # ``requires is None`` entries are always available (see
   697→    # ModelSpec.available), so this branch always has a concrete module name.
   698→    if requires is not None and not spec.available:
   699→        extra = _REQUIREMENT_EXTRAS.get(requires, requires)
   700→        raise DependencyError(
   701→            f"Model {key!r} requires the optional package '{requires}'. "
   702→            f"Install with:  pip install 'pharmaml[{extra}]'"
   703→        )
   704→
   705→    if spec.name in _TASK_FACTORIES:
   706→        estimator = _TASK_FACTORIES[spec.name][task]()
   707→    else:
   708→        factory = spec.factory
   709→        if factory is None:
   710→            # Dual-task entries carry factory=None and are dispatched through
   711→            # _TASK_FACTORIES above; a factory-less entry outside that table
   712→            # would be a registry bug, not a caller error.
   713→            raise ModelError(
   714→                f"Model {key!r} has no estimator factory registered for task {task!r}"
   715→            )
   716→        estimator = factory()
   717→
   718→    if params:
   719→        accepted = _accepted_param_names(estimator)
   720→        unknown = set(params) - accepted
   721→        if unknown:
   722→            raise ModelError(
   723→                f"Unknown parameters {sorted(unknown)} for {key!r}. "
   724→                f"Valid: {sorted(accepted)[:20]}..."
   725→            )
   726→        try:
   727→            estimator.set_params(**params)
   728→        except Exception as exc:
   729→            # Estimators with lenient constructors reject bad *values*
   730→            # (or unknown names on **kwargs classes) inside set_params;
   731→            # surface that as the documented typed error instead of a
   732→            # raw backend exception (ValueError / CatBoostError / ...).
   733→            raise ModelError(
   734→                f"Invalid parameters {sorted(params)} for {key!r}: {exc}"
   735→            ) from exc
   736→    return estimator
   737→

   739→def build_stacking(
   740→    task: str,
   741→    base_models: list[str],
   742→    meta_model: str | None = None,
   743→    passthrough: bool = False,
   744→) -> Any:
   745→    """Build a stacking ensemble (upgrades Stacking.py / Stacking reg.py).
   746→
   747→    Parameters
   748→    ----------
   749→    task:
   750→        ``'classification'`` or ``'regression'``.
   751→    base_models:
   752→        Registry names of the level-0 learners.
   753→    meta_model:
   754→        Registry name of the level-1 blender.  Defaults to
   755→        ``'logistic_regression'`` for classification and ``'ridge'``
   756→        for regression.
   757→    passthrough:
   758→        Append original features to meta-features.
   759→
   760→    Raises
   761→    ------
   762→    ModelError
   763→        When ``base_models`` is empty, a member name is unknown, or a
   764→        member does not support ``task``.  Aliases in ``base_models``
   765→        are resolved and duplicates collapsed, so ``['rf',
   766→        'random_forest']`` builds one random-forest learner, not two
   767→        identical ones.
   768→    """
   769→    if not base_models:
   770→        raise ModelError(
   771→            "base_models list is empty -- pass at least one registry name "
   772→            "(see pharmaml.ml.list_models())"
   773→        )
   774→    if meta_model is None:
   775→        meta_model = "logistic_regression" if task == "classification" else "ridge"
   776→    keys = list(dict.fromkeys(resolve_model_name(name) for name in base_models))
   777→    estimators = [(key, build_model(key, task)) for key in keys]
   778→    meta = build_model(meta_model, task)
   779→    if task == "classification":
   780→        return StackingClassifier(
   781→            estimators=estimators, final_estimator=meta, passthrough=passthrough, cv=5
   782→        )
   783→    return StackingRegressor(
   784→        estimators=estimators, final_estimator=meta, passthrough=passthrough, cv=5
   785→    )

   788→def build_voting(
   789→    task: str,
   790→    base_models: list[str],
   791→    voting: str = "soft",
   792→    weights: list[float] | None = None,
   793→) -> Any:
   794→    """Build a voting ensemble (upgrades Voting.py / Voting reg.py).
   795→
   796→    ``voting='soft'`` (probability averaging) requires all members to
   797→    expose ``predict_proba``; ``voting='hard'`` (majority vote) works
   798→    for every estimator and for **any number of classes**.  For
   799→    regression the voting rule is ignored (``VotingRegressor``
   800→    averages predictions), but the value must still be ``'soft'`` or
   801→    ``'hard'``.  Aliases in ``base_models`` are resolved and
   802→    duplicates collapsed, so ``['rf', 'random_forest']`` casts one
   803→    vote, not two.
   804→
   805→    Raises
   806→    ------
   807→    ModelError
   808→        When ``base_models`` is empty, a member name is unknown,
   809→        ``voting`` is neither ``'soft'`` nor ``'hard'``, ``weights``
   810→        does not match the number of unique members, or a soft-voting
   811→        member lacks ``predict_proba``.
   812→    """
   813→    if not base_models:
   814→        raise ModelError(
   815→            "base_models list is empty -- pass at least one registry name "
   816→            "(see pharmaml.ml.list_models())"
   817→        )
   818→    if voting not in {"soft", "hard"}:
   819→        raise ModelError(f"voting must be 'soft' or 'hard', got {voting!r}")
   820→    keys = list(dict.fromkeys(resolve_model_name(name) for name in base_models))
   821→    if weights is not None and len(weights) != len(keys):
   822→        raise ModelError(
   823→            f"weights has {len(weights)} entries for {len(keys)} unique base "
   824→            "model(s) -- alias duplicates are collapsed"
   825→        )
   826→    estimators = [(key, build_model(key, task)) for key in keys]
   827→    if task == "classification":
   828→        if voting == "soft":
   829→            unsupported = [
   830→                key for key, est in estimators if not hasattr(est, "predict_proba")
   831→            ]
   832→            if unsupported:
   833→                raise ModelError(
   834→                    f"voting='soft' needs predict_proba on every member; "
   835→                    f"{unsupported} lack it. Use voting='hard' (works for "
   836→                    "binary and multiclass) or drop those models."
   837→                )
   838→        return VotingClassifier(estimators=estimators, voting=voting, weights=weights)
   839→    return VotingRegressor(estimators=estimators, weights=weights)
