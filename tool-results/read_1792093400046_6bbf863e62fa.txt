    94→/** Read the persisted recent-files list (client only, failure-tolerant). */
    95→function loadRecentFiles(): string[] {
    96→  if (typeof window === "undefined") return [];
    97→  try {
    98→    const raw = window.sessionStorage.getItem(RECENT_STORAGE_KEY);
    99→    if (!raw) return [];
   100→    const parsed: unknown = JSON.parse(raw);
   101→    if (!Array.isArray(parsed)) return [];
   102→    return parsed
   103→      .filter((p): p is string => typeof p === "string")
   104→      .slice(0, RECENT_LIMIT);
   105→  } catch {
   106→    return []; // private mode / quota / corrupted JSON — start fresh
   107→  }
   108→}

   132→/** Validate a persisted PK regimen field-by-field, falling back to defaults. */
   133→function validatePk(raw: unknown): PkParams {
   134→  const base = DEFAULT_PK_PARAMS;
   135→  if (typeof raw !== "object" || raw === null) return base;
   136→  const p = raw as Partial<PkParams>;
   137→  const num = (v: unknown, def: number, lo: number, hi: number) =>
   138→    typeof v === "number" && Number.isFinite(v) && v >= lo && v <= hi ? v : def;
   139→  return {
   140→    route: p.route === "oral" ? "oral" : "iv",
   141→    dose: num(p.dose, base.dose, 0.1, 10_000),
   142→    vd: num(p.vd, base.vd, 0.1, 1_000),
   143→    cl: num(p.cl, base.cl, 0.01, 100),
   144→    tau: num(p.tau, base.tau, 0.25, 168),
   145→    doses: Math.round(num(p.doses, base.doses, 1, 48)),
   146→    f: num(p.f, base.f, 0.05, 1),
   147→    ka: num(p.ka, base.ka, 0.01, 12),
   148→  };
   149→}

   151→/** Validate a persisted ML config field-by-field, falling back to defaults. */
   152→function validateMl(raw: unknown): MlConfig {
   153→  const base = DEFAULT_ML_CONFIG;
   154→  if (typeof raw !== "object" || raw === null) return base;
   155→  const p = raw as Partial<MlConfig>;
   156→  const str = (v: unknown, def: string) =>
   157→    typeof v === "string" && v.length <= 12 ? v : def;
   158→  const str128 = (v: unknown, def: string) =>
   159→    typeof v === "string" && v.length <= 128 ? v : def;
   160→  const strArr = (v: unknown): string[] =>
   161→    Array.isArray(v)
   162→      ? v.filter((c): c is string => typeof c === "string" && c.length > 0 && c.length <= 128).slice(0, 8)
   163→      : [];
   164→  return {
   165→    task: p.task === "classification" ? "classification" : "regression",
   166→    samples:
   167→      typeof p.samples === "number" && Number.isFinite(p.samples) && p.samples >= 100 && p.samples <= 1000
   168→        ? Math.round(p.samples)
   169→        : base.samples,
   170→    noise:
   171→      typeof p.noise === "number" && Number.isFinite(p.noise) && p.noise >= 0 && p.noise <= 2
   172→        ? p.noise
   173→        : base.noise,
   174→    regModel:
   175→      p.regModel === "polynomial" || p.regModel === "ridge" ? p.regModel : "linear",
   176→    alpha: str(p.alpha, base.alpha),
   177→    l2: str(p.l2, base.l2),
   178→    testSizePct:
   179→      typeof p.testSizePct === "number" && Number.isFinite(p.testSizePct) && p.testSizePct >= 5 && p.testSizePct <= 50
   180→        ? p.testSizePct
   181→        : base.testSizePct,
   182→    cv: str(p.cv, base.cv),
   183→    seed: str(p.seed, base.seed),
   184→    source: p.source === "csv" ? "csv" : "synthetic",
   185→    featureColumn: str128(p.featureColumn, ""),
   186→    featureColumn2: str128(p.featureColumn2, ""),
   187→    targetColumn: str128(p.targetColumn, ""),
   188→    featureColumns: strArr(p.featureColumns),
   189→  };
   190→}

   192→/* ---- ML result persistence (last training run: metrics + chart series) ----
   193→ * Restored field-by-field with the same defensive validation as the rest of
   194→ * the persisted state; anything malformed falls back to "no result". */
   195→
   196→function finiteArray(raw: unknown, cap: number, validate: (v: unknown) => boolean): unknown[] | null {
   197→  if (!Array.isArray(raw) || raw.length === 0 || raw.length > cap) return null;
   198→  return raw.every(validate) ? raw : null;
   199→}
   200→
   201→const isFiniteNumber = (v: unknown): v is number =>
   202→  typeof v === "number" && Number.isFinite(v);
   203→
   204→const isPoint = (v: unknown): boolean => {
   205→  if (typeof v !== "object" || v === null) return false;
   206→  const p = v as Record<string, unknown>;
   207→  return isFiniteNumber(p.x) && isFiniteNumber(p.y);
   208→};
   209→
   210→const isCurvePoint = (v: unknown): boolean => {
   211→  if (typeof v !== "object" || v === null) return false;
   212→  const p = v as Record<string, unknown>;
   213→  return isFiniteNumber(p.x) && isFiniteNumber(p.yhat);
   214→};
   215→
   216→const isBoundaryPoint = (v: unknown): boolean => {
   217→  if (typeof v !== "object" || v === null) return false;
   218→  const p = v as Record<string, unknown>;
   219→  return isFiniteNumber(p.x) && isFiniteNumber(p.y) && isFiniteNumber(p.pred);
   220→};

   222→/** Validate a persisted ML train response; null when unusable. */
   223→function validateMlResult(raw: unknown): MLTrainResponse | null {
   224→  if (typeof raw !== "object" || raw === null) return null;
   225→  const r = raw as Record<string, unknown>;
   226→  if (r.task !== "regression" && r.task !== "classification") return null;
   227→  if (typeof r.model !== "string" || r.model.length > 40) return null;
   228→
   229→  const metricsRaw = r.metrics;
   230→  if (typeof metricsRaw !== "object" || metricsRaw === null) return null;
   231→  const metrics: Record<string, number> = {};
   232→  for (const [k, v] of Object.entries(metricsRaw as Record<string, unknown>)) {
   233→    if (isFiniteNumber(v) && k.length <= 24) metrics[k] = v;
   234→  }
   235→  if (Object.keys(metrics).length === 0) return null;
   236→
   237→  const coefsRaw = r.coefficients;
   238→  if (typeof coefsRaw !== "object" || coefsRaw === null) return null;
   239→  const c = coefsRaw as Record<string, unknown>;
   240→  const coefsArr = finiteArray(c.coefs, 512, isFiniteNumber);
   241→  if (!isFiniteNumber(c.intercept) || !coefsArr || typeof c.standardized !== "boolean") {
   242→    return null;
   243→  }
   244→
   245→  const seriesRaw = r.series;
   246→  if (typeof seriesRaw !== "object" || seriesRaw === null) return null;
   247→  const s = seriesRaw as Record<string, unknown>;
   248→  const test = finiteArray(s.test, 1200, isPoint);
   249→  if (!test) return null;
   250→  const curve = finiteArray(s.curve, 400, isCurvePoint);
   251→  const boundary = finiteArray(s.boundary, 400, isBoundaryPoint);
   252→
   253→  const metaRaw = r.datasetMeta;
   254→  if (typeof metaRaw !== "object" || metaRaw === null) return null;
   255→  const m = metaRaw as Record<string, unknown>;
   256→  const features = Array.isArray(m.features)
   257→    ? m.features.filter((f): f is string => typeof f === "string" && f.length <= 24).slice(0, 12)
   258→    : [];
   259→  if (
   260→    !isFiniteNumber(m.nTotal) ||
   261→    !isFiniteNumber(m.nTrain) ||
   262→    !isFiniteNumber(m.nTest) ||
   263→    !isFiniteNumber(m.noise) ||
   264→    !isFiniteNumber(m.seed)
   265→  ) {
   266→    return null;
   267→  }
   268→
   269→  // Optional CSV provenance — preserved when well-formed, dropped otherwise.
   270→  let source: CsvSourceMeta | undefined;
   271→  if (typeof m.source === "object" && m.source !== null) {
   272→    const s = m.source as Record<string, unknown>;
   273→    const colList = (v: unknown, cap: number): string[] =>
   274→      Array.isArray(v)
   275→        ? v.filter((c): c is string => typeof c === "string" && c.length <= 128).slice(0, cap)
   276→        : [];
   277→    const columns = colList(s.columns, 64);
   278→    const featureColumns = colList(s.featureColumns, 10);
   279→    const targetColumn = typeof s.targetColumn === "string" ? s.targetColumn.slice(0, 128) : "";
   280→    if (
   281→      s.kind === "csv" &&
   282→      columns.length > 0 &&
   283→      featureColumns.length > 0 &&
   284→      targetColumn !== "" &&
   285→      isFiniteNumber(s.nRows) &&
   286→      isFiniteNumber(s.droppedRows)
   287→    ) {
   288→      const labelMap =
   289→        Array.isArray(s.labelMap) &&
   290→        s.labelMap.length === 2 &&
   291→        s.labelMap.every((v) => typeof v === "string" && v.length <= 64)
   292→          ? (s.labelMap as [string, string])
   293→          : undefined;
   294→      source = {
   295→        kind: "csv",
   296→        fileName: typeof s.fileName === "string" ? s.fileName.slice(0, 128) : undefined,
   297→        columns,
   298→        featureColumns,
   299→        targetColumn,
   300→        nRows: s.nRows,
   301→        droppedRows: s.droppedRows,
   302→        ...(labelMap ? { labelMap } : {}),
   303→      };
   304→    }
   305→  }
   306→
   307→  // Optional cross-validation summary — dropped when malformed.
   308→  let cv: MLTrainResponse["cv"];
   309→  if (typeof r.cv === "object" && r.cv !== null) {
   310→    const cvRaw = r.cv as Record<string, unknown>;
   311→    const folds = Array.isArray(cvRaw.folds)
   312→      ? cvRaw.folds.filter(
   313→          (f): f is { fold: number; value: number } =>
   314→            typeof f === "object" &&
   315→            f !== null &&
   316→            isFiniteNumber((f as Record<string, unknown>).fold) &&
   317→            isFiniteNumber((f as Record<string, unknown>).value)
   318→        )
   319→      : [];
   320→    if (
   321→      (cvRaw.k === 3 || cvRaw.k === 5 || cvRaw.k === 10) &&
   322→      typeof cvRaw.metricName === "string" &&
   323→      folds.length === cvRaw.k &&
   324→      isFiniteNumber(cvRaw.mean) &&
   325→      isFiniteNumber(cvRaw.std)
   326→    ) {
   327→      cv = {
   328→        k: cvRaw.k,
   329→        metricName: cvRaw.metricName,
   330→        folds,
   331→        mean: cvRaw.mean,
   332→        std: cvRaw.std,
   333→      };
   334→    }
   335→  }
   336→
   337→  return {
   338→    task: r.task,
   339→    model: r.model,
   340→    metrics,
   341→    cv,
   342→    coefficients: {
   343→      intercept: c.intercept as number,
   344→      coefs: coefsArr as number[],
   345→      standardized: c.standardized,
   346→    },
   347→    series: {
   348→      test: test as { x: number; y: number; label?: number }[],
   349→      ...(curve ? { curve: curve as { x: number; yhat: number }[] } : {}),
   350→      ...(boundary ? { boundary: boundary as { x: number; y: number; pred: number }[] } : {}),
   351→    },
   352→    datasetMeta: {
   353→      nTotal: m.nTotal,
   354→      nTrain: m.nTrain,
   355→      nTest: m.nTest,
   356→      noise: m.noise,
   357→      seed: m.seed,
   358→      features,
   359→      ...(source ? { source } : {}),
   360→    },
   361→  };
   362→}

   376→/** Validate a persisted run-history payload; malformed entries are skipped.
   377→ * Accepts both the current `{ currentId, runs }` shape and the legacy bare
   378→ * array (current run = newest). */
   379→function validateMlHistoryPayload(raw: unknown): {
   380→  runs: MlRunRecord[];
   381→  currentId: number | null;
   382→} {
   383→  const body = Array.isArray(raw) ? { runs: raw, currentId: undefined } : raw;
   384→  if (typeof body !== "object" || body === null || !Array.isArray((body as { runs?: unknown }).runs)) {
   385→    return { runs: [], currentId: null };
   386→  }
   387→  const runs: MlRunRecord[] = [];
   388→  for (const item of ((body as { runs: unknown[] }).runs).slice(0, ML_HISTORY_LIMIT)) {
   389→    if (typeof item !== "object" || item === null) continue;
   390→    const r = item as Record<string, unknown>;
   391→    const result = validateMlResult(r.result);
   392→    const csvMeta =
   393→      typeof r.csv === "object" && r.csv !== null ? (r.csv as Record<string, unknown>) : null;
   394→    if (
   395→      !result ||
   396→      typeof r.id !== "number" ||
   397→      !Number.isFinite(r.id) ||
   398→      typeof r.at !== "string" ||
   399→      r.at.length > 40 ||
   400→      (r.task !== "regression" && r.task !== "classification") ||
   401→      typeof r.model !== "string" ||
   402→      r.model.length > 40 ||
   403→      typeof r.keyMetric !== "string" ||
   404→      r.keyMetric.length > 16 ||
   405→      typeof r.keyValue !== "number" ||
   406→      !Number.isFinite(r.keyValue)
   407→    ) {
   408→      continue;
   409→    }
   410→    runs.push({
   411→      id: r.id,
   412→      at: r.at,
   413→      task: r.task,
   414→      model: r.model,
   415→      keyMetric: r.keyMetric,
   416→      keyValue: r.keyValue,
   417→      config: validateMl(r.config),
   418→      result,
   419→      ...(csvMeta
   420→        ? {
   421→            csv: {
   422→              fileName:
   423→                typeof csvMeta.fileName === "string"
   424→                  ? csvMeta.fileName.slice(0, 128) || "pasted.csv"
   425→                  : "pasted.csv",
   426→              featureColumns: Array.isArray(csvMeta.featureColumns)
   427→                ? (csvMeta.featureColumns as unknown[])
   428→                    .filter((c): c is string => typeof c === "string" && c.length <= 128)
   429→                    .slice(0, 8)
   430→                : [],
   431→              targetColumn:
   432→                typeof csvMeta.targetColumn === "string"
   433→                  ? csvMeta.targetColumn.slice(0, 128)
   434→                  : "",
   435→            },
   436→          }
   437→        : {}),
   438→    });
   439→  }
   440→  const rawId = (body as { currentId?: unknown }).currentId;
   441→  const currentId =
   442→    typeof rawId === "number" && Number.isFinite(rawId) && runs.some((h) => h.id === rawId)
   443→      ? rawId
   444→      : (runs[0]?.id ?? null);
   445→  return { runs, currentId };
   446→}

   448→/** Read the persisted run history (client only, failure-tolerant). */
   449→function loadMlHistory(): { runs: MlRunRecord[]; currentId: number | null } {
   450→  if (typeof window === "undefined") return { runs: [], currentId: null };
   451→  try {
   452→    const raw = window.sessionStorage.getItem(ML_HISTORY_STORAGE_KEY);
   453→    if (!raw) return { runs: [], currentId: null };
   454→    return validateMlHistoryPayload(JSON.parse(raw));
   455→  } catch {
   456→    return { runs: [], currentId: null }; // corrupted JSON / private mode — start fresh
   457→  }
   458→}

   468→    /* quota exceeded or private mode — persistence is optional */

   478→      return;
   479→    }
   480→    const json = JSON.stringify(result);
   481→    if (json.length > ML_RESULT_MAX_BYTES) return; // oversized — persistence is optional
   482→    window.sessionStorage.setItem(ML_RESULT_STORAGE_KEY, json);
   483→  } catch {
   484→    /* quota exceeded or private mode — persistence is optional */
   485→  }

   489→function loadPlayground(): Partial<PlaygroundPersisted> {
   490→  if (typeof window === "undefined") return {};

   523→/** Validate a persisted uploaded-CSV payload; null when unusable. */
   524→function validMlCsv(raw: unknown): MlCsvState | null {
   525→  if (typeof raw !== "object" || raw === null) return null;
   526→  const c = raw as Partial<MlCsvState>;
   527→  if (
   528→    typeof c.text !== "string" ||
   529→    c.text.length === 0 ||
   530→    c.text.length > MAX_CSV_TEXT ||
   531→    typeof c.fileName !== "string" ||
   532→    c.fileName.length > 128
   533→  ) {
   534→    return null;
   535→  }
   536→  return { fileName: c.fileName, text: c.text };
   537→}

   591→interface ToolkitStore {
   592→  tab: TabId;
   593→  setTab: (tab: TabId) => void;
   594→
   595→  /** Command palette open state (Ctrl/Cmd+K or the header button). */
   596→  paletteOpen: boolean;
   597→  setPaletteOpen: (open: boolean) => void;
   598→
   599→  /** Currently selected file (relative path) in the code explorer. */
   600→  selectedFile: string;
   601→  setSelectedFile: (path: string) => void;
   602→
   603→  /** Definition line to highlight + scroll to (API-reference deep links). */
   604→  targetLine: number | null;
   605→  setTargetLine: (line: number | null) => void;
   606→  /** Open a file and (optionally) jump to a definition line. */
   607→  openFileAtLine: (path: string, line: number | null) => void;
   608→
   609→  /** Most-recently-viewed files (newest first, deduplicated). */
   610→  recentFiles: string[];
   611→  /** Clear the recent-files strip. */
   612→  clearRecentFiles: () => void;
   613→
   614→  /** Active playground sub-tab (restored when returning to Playground). */
   615→  playgroundTab: string;
   616→  setPlaygroundTab: (tab: string) => void;
   617→
   618→  /** PK simulator regimen — persisted so sliders survive tab switches. */
   619→  pkParams: PkParams;
   620→  setPkParam: <K extends keyof PkParams>(key: K, value: PkParams[K]) => void;
   621→  resetPkParams: () => void;

   623→  /** ML playground experiment config — persisted so the setup survives tab switches. */
   624→  mlConfig: MlConfig;

   627→  /** User-uploaded CSV (file name + full text); null = not loaded. */

   634→  mlPreview: DatasetPreviewResponse | null;

   637→  /** Recent training runs (newest first) — persisted so the strip survives reloads. */

   639→  pushMlRun: (record: MlRunRecord) => void;

   645→  tree: TreeNode | null;

   653→/** Newest-first, deduplicated, capped recent list (pure helper). */
   654→function pushRecent(recent: string[], path: string): string[] {
   655→  const next = [path, ...recent.filter((p) => p !== path)];
   656→  return next.slice(0, RECENT_LIMIT);
   657→}

   665→export const useToolkitStore = create<ToolkitStore>((set, get) => ({
   666→  tab: "overview",
   667→  setTab: (tab) => set({ tab }),
   668→
   669→  paletteOpen: false,
   670→  setPaletteOpen: (open) => set({ paletteOpen: open }),
   671→
   672→  selectedFile: "README.md",
   673→  setSelectedFile: (path) =>
   674→    set((s) => {
   675→      const recentFiles = pushRecent(s.recentFiles, path);
   676→      saveRecentFiles(recentFiles);
   677→      return { selectedFile: path, targetLine: null, recentFiles };
   678→    }),
   679→
   680→  targetLine: null,
   681→  setTargetLine: (line) => set({ targetLine: line }),
   682→  openFileAtLine: (path, line) =>
   683→    set((s) => {
   684→      const recentFiles = pushRecent(s.recentFiles, path);
   685→      saveRecentFiles(recentFiles);
   686→      return { selectedFile: path, targetLine: line, recentFiles };
   687→    }),
   688→
   689→  recentFiles: loadRecentFiles(),
   690→  clearRecentFiles: () => {
   691→    saveRecentFiles([]);
   692→    set({ recentFiles: [] });
   693→  },
   694→
   695→  playgroundTab: persistedPlayground.playgroundTab ?? "calculators",
   696→  setPlaygroundTab: (tab) => {
   697→    set({ playgroundTab: tab });
   698→    savePlayground({
   699→      playgroundTab: tab,
   700→      pkParams: get().pkParams,
   701→      mlConfig: get().mlConfig,
   702→      mlCsv: get().mlCsv,
   703→    });
   704→  },
   705→
   706→  pkParams: persistedPlayground.pkParams ?? DEFAULT_PK_PARAMS,
   707→  setPkParam: (key, value) => {
   708→    const pkParams = { ...get().pkParams, [key]: value };
   709→    set({ pkParams });
   710→    savePlayground({
   711→      playgroundTab: get().playgroundTab,
   712→      pkParams,
   713→      mlConfig: get().mlConfig,
   714→      mlCsv: get().mlCsv,
   715→    });
   716→  },
   717→  resetPkParams: () => {
   718→    set({ pkParams: DEFAULT_PK_PARAMS });
   719→    savePlayground({
   720→      playgroundTab: get().playgroundTab,
   721→      pkParams: DEFAULT_PK_PARAMS,
   722→      mlConfig: get().mlConfig,
   723→      mlCsv: get().mlCsv,
   724→    });
   725→  },
   726→
   727→  mlConfig: persistedPlayground.mlConfig ?? DEFAULT_ML_CONFIG,
   728→  setMlConfig: (patch) => {
   729→    const mlConfig = { ...get().mlConfig, ...patch };
   730→    set({ mlConfig });
   731→    savePlayground({
   732→      playgroundTab: get().playgroundTab,
   733→      pkParams: get().pkParams,
   734→      mlConfig,
   735→      mlCsv: get().mlCsv,
   736→    });
   737→  },
   738→  resetMlConfig: () => {
   739→    set({ mlConfig: DEFAULT_ML_CONFIG });
   740→    savePlayground({
   741→      playgroundTab: get().playgroundTab,
   742→      pkParams: get().pkParams,
   743→      mlConfig: DEFAULT_ML_CONFIG,
   744→      mlCsv: get().mlCsv,
   745→    });
   746→  },
   747→
   748→  mlCsv: persistedPlayground.mlCsv ?? null,
   749→  setMlCsv: (csv) => {
   750→    set({ mlCsv: csv });
   751→    savePlayground({
   752→      playgroundTab: get().playgroundTab,
   753→      pkParams: get().pkParams,
   754→      mlConfig: get().mlConfig,
   755→      mlCsv: csv,
   756→    });
   757→  },
   758→  mlResult: loadMlResult(),
   759→  setMlResult: (result) => {
   760→    set({ mlResult: result });
   761→    saveMlResult(result);
   762→  },
   763→  mlPreview: null,
   764→  setMlPreview: (preview) => set({ mlPreview: preview }),
   765→
   766→  mlHistory: persistedMlHistory.runs,
   767→  pushMlRun: (record) => {
   768→    const mlHistory = [
   769→      record,
   770→      ...get().mlHistory.filter((h) => h.id !== record.id),
   771→    ].slice(0, ML_HISTORY_LIMIT);
   772→    set({ mlHistory, mlCurrentRunId: record.id });
   773→    saveMlHistory(mlHistory, record.id);
   774→  },
   775→  clearMlHistory: () => {
   776→    saveMlHistory([], null);
   777→    set({ mlHistory: [], mlCurrentRunId: null });
   778→  },
   779→  mlCurrentRunId: persistedMlHistory.currentId,
   780→  setMlCurrentRun: (id) => {
   781→    set({ mlCurrentRunId: id });
   782→    saveMlHistory(get().mlHistory, id);
   783→  },
   784→
   785→  tree: null,
   786→  stats: null,
   787→  treeStatus: "idle",
   788→  treeError: null,
   789→  ensureTree: async () => {
   790→    const { treeStatus, tree } = get();
   791→    if (treeStatus === "loading" || tree) return;
   792→    set({ treeStatus: "loading", treeError: null });
   793→    try {
   794→      const res = await fetch("/api/toolkit/tree");
   795→      if (!res.ok) {
   796→        throw new Error(`Tree API responded with HTTP ${res.status}`);
   797→      }
   798→      const data = (await res.json()) as TreeResponse;
   799→      set({ tree: data.root, stats: data.stats, treeStatus: "ready" });
   800→    } catch (err) {
   801→      set({
   802→        treeStatus: "error",
   803→        treeError:
   804→          err instanceof Error ? err.message : "Unknown error while loading the file tree.",
   805→      });
   806→    }
   807→  },
   808→}));
