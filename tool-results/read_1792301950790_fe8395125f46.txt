    62→# --------------------------------------------------------------------- #
    63→# toolkit version plumbing (pyproject [project] -> VERSION -> 0.0.0)
    64→# --------------------------------------------------------------------- #
    65→class TestToolkitVersion:
    66→    @staticmethod
    67→    def _pyproject_project_version() -> str:
    68→        """Independent re-derivation of the pyproject ``[project]`` version."""
    69→        in_project = False
    70→        for raw_line in (TOOLKIT_ROOT / "pyproject.toml").read_text(
    71→            encoding="utf-8"
    72→        ).splitlines():
    73→            line = raw_line.strip()
    74→            if line.startswith("[") and line.endswith("]"):
    75→                in_project = line == "[project]"
    76→                continue
    77→            if not in_project or not line.startswith("version"):
    78→                continue
    79→            key, sep, value = line.partition("=")
    80→            if sep and key.strip() == "version":
    81→                return value.split("#", 1)[0].strip().strip('"').strip("'")
    82→        return ""
    83→
    84→    def test_pyproject_version_file_and_playground_agree(self, playground: Any) -> None:
    85→        declared = self._pyproject_project_version()
    86→        assert declared, "no [project] version line found in pyproject.toml"
    87→        version_file = (TOOLKIT_ROOT / "VERSION").read_text(encoding="utf-8").strip()
    88→        # the drift guard: all three sources must name the same version
    89→        assert playground.toolkit_version() == declared
    90→        assert version_file == declared
    91→        assert playground.__version__ == declared
    92→
    93→    def test_fallback_chain_version_file_then_placeholder(
    94→        self, playground: Any, monkeypatch: Any, tmp_path: Any
    95→    ) -> None:
    96→        only_version = tmp_path / "only_version"
    97→        only_version.mkdir()
    98→        (only_version / "VERSION").write_text("9.9.9\n", encoding="utf-8")
    99→        monkeypatch.setattr(playground, "TOOLKIT_ROOT", only_version)
   100→        assert playground.toolkit_version() == "9.9.9"
   101→
   102→        neither = tmp_path / "neither"
   103→        neither.mkdir()
   104→        monkeypatch.setattr(playground, "TOOLKIT_ROOT", neither)
   105→        assert playground.toolkit_version() == "0.0.0"
   106→
   107→    def test_fallback_ignores_versions_outside_project_section(
   108→        self, playground: Any, monkeypatch: Any, tmp_path: Any
   109→    ) -> None:
   110→        sneaky = tmp_path / "sneaky"
   111→        sneaky.mkdir()
   112→        (sneaky / "pyproject.toml").write_text(
   113→            '[tool.something]\nversion = "7.7.7"\n\n[project]\nname = "x"\n',
   114→            encoding="utf-8",
   115→        )
   116→        (sneaky / "VERSION").write_text("8.8.8\n", encoding="utf-8")
   117→        monkeypatch.setattr(playground, "TOOLKIT_ROOT", sneaky)
   118→        # a version line outside [project] must not be picked up
   119→        assert playground.toolkit_version() == "8.8.8"
