   185→   185→interface TorsionAudit {
   186→   186→  torsdof: number;
   187→   187→  activeRemarks: number;
   188→   188→  branches: number;
   189→   189→  /** Rule actually applied once "auto" is resolved from the remark count. */
   190→   190→  rule: "mgltools" | "meeko";
   191→   191→  ok: boolean;
   192→   192→  /** One-line invariant report (failed invariant when mismatched). */
   193→   193→  explanation: string;
   194→   194→}
   195→   195→
   196→   196→/** Mirror of pharmaml.docking.check_torsion_consistency, line by line:
   197→   197→ * per trimmed line the last well-formed `TORSDOF <int>` wins (malformed
   198→   198→ * values are ignored), `REMARK … A …` lines mark active torsions and
   199→   199→ * `BRANCH` lines grow the torsion tree; the convention then applies the
   200→   200→ * MGLTools three-way invariant or the Meeko two-way one. */
   201→   201→function auditTorsionTree(content: string, convention: TorsionConvention): TorsionAudit {
   202→   202→  let torsdof = 0;
   203→   203→  let activeRemarks = 0;
   204→   204→  let branches = 0;
   205→   205→  for (const raw of content.split("\n")) {
   206→   206→    const line = raw.trim();
   207→   207→    if (line.startsWith("TORSDOF")) {
   208→   208→      const value = line.split(/\s+/)[1];
   209→   209→      // Python int() tolerates an optional sign; anything else raises → ignored.
   210→   210→      if (value !== undefined && /^[+-]?\d+$/.test(value)) torsdof = Number(value);
   211→   211→    } else if (line.startsWith("REMARK") && line.includes(" A ")) {
   212→   212→      activeRemarks += 1;
   213→   213→    } else if (line.startsWith("BRANCH")) {
   214→   214→      branches += 1;
   215→   215→    }
   216→   216→  }
   217→   217→  const rule: "mgltools" | "meeko" =
   218→   218→    convention === "auto" ? (activeRemarks > 0 ? "mgltools" : "meeko") : convention;
   219→   219→  const ok =
   220→   220→    rule === "mgltools"
   221→   221→      ? torsdof === activeRemarks && activeRemarks === branches
   222→   222→      : torsdof === branches;
   223→   223→  return {
   224→   224→    torsdof,
   225→   225→    activeRemarks,
   226→   226→    branches,
   227→   227→    rule,
   228→   228→    ok,
   229→   229→    explanation: torsionExplanation(torsdof, activeRemarks, branches, rule, ok),
   230→   230→  };
   231→   231→}
   232→   232→
   233→   233→/** Human one-liner naming the invariant that held or failed. */
   234→   234→function torsionExplanation(
   235→   235→  torsdof: number,
   236→   236→  activeRemarks: number,
   237→   237→  branches: number,
   238→   238→  rule: "mgltools" | "meeko",
   239→   239→  ok: boolean
   240→   240→): string {
   241→   241→  if (ok) {
   242→   242→    return rule === "mgltools"
   243→   243→      ? `TORSDOF ${torsdof} = ${activeRemarks} active torsion remarks = ${branches} BRANCH records — the torsion tree is consistent.`
   244→   244→      : `TORSDOF ${torsdof} = BRANCH count ${branches} — the torsion tree is consistent under the Meeko convention${
   245→   245→          activeRemarks > 0 && activeRemarks !== torsdof
   246→   246→            ? " (the REMARK count is not part of this invariant)"
   247→   247→            : ""
   248→   248→        }.`;
   249→   249→  }
   250→   250→  if (rule === "mgltools") {
   251→   251→    if (torsdof !== activeRemarks && torsdof !== branches && activeRemarks !== branches) {
   252→   252→      return `TORSDOF ${torsdof}, ${activeRemarks} active torsion remarks and ${branches} BRANCH records all disagree — the torsion tree is inconsistent.`;
   253→   253→    }
   254→   254→    if (torsdof !== activeRemarks) {
   255→   255→      return `TORSDOF ${torsdof} ≠ ${activeRemarks} active torsion remarks — the torsion tree is inconsistent.`;
   256→   256→    }
   257→   257→    return `TORSDOF ${torsdof} ≠ ${branches} BRANCH records — the torsion tree is inconsistent.`;
   258→   258→  }
   259→   259→  return `TORSDOF ${torsdof} ≠ ${branches} BRANCH records — the torsion tree is inconsistent (Meeko convention).`;
   260→   260→}

   264→   264→  return <span className="tabular-nums">{fmtNum(value, sig)}</span>;
   265→   265→}
