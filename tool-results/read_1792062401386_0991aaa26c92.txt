     1→"""Command-line interface — the scaffolding entry point of PharmaML.
     2→
     3→Replaces the legacy workflow of "open a per-algorithm .py file, edit
     4→the hardcoded CSV filename at the bottom, run it".  One command covers
     5→training, comparison, splitting, conversion, PK analysis and docking
     6→scans::
     7→
     8→    pharmaml ml train    --data qsar.csv --target Target --model rf --plots
     9→    pharmaml ml train    --data qsar.csv --task auto        # infer classification/regression
    10→    pharmaml ml train    --data qsar.csv --keep-categorical  # one-hot string features
    11→    pharmaml ml compare  --data qsar.csv --target Target --task classification --plot
    12→    pharmaml data inspect --data qsar.csv                  # audit your own file
    13→    pharmaml data split  --data qsar.csv --test-size 0.2 --val-size 0.1
    14→    pharmaml data convert --data ligands.csv --to sdf      # .smi/.csv/.sdf round-trip
    15→    pharmaml chem pic50  --data activity.xlsx --column "IC50 (nM)"
    16→    pharmaml pharma pk   --data profile.csv --time time_h --conc conc_ng_ml
    17→    pharmaml docking prep-ligands --ligands ligands.csv --out ligand_folders/
    18→    pharmaml docking prep-protein --protein protein.pdb --out r.pdbqt
    19→    pharmaml docking scan --directory docking_run/
    20→    pharmaml docking dpf --ligand ligand_folders/1/l.pdbqt --ga-runs 50
    21→    pharmaml docking collect --dir docking_run/ --pattern '^\\d+\\.pdbqt$'
    22→    pharmaml docking torsion --dir ligand_folders/
    23→    pharmaml docking gpf-scan --dir docking_run/
    24→    pharmaml docking vina-scan --dir vina_results/
    25→    pharmaml models list
    26→
    27→``--model`` accepts registry names and their aliases (``rf``,
    28→``logistic``, ``xgb`` ... — see ``pharmaml.ml.MODEL_ALIASES``).
    29→"""

   470→def _cmd_models(args: argparse.Namespace) -> int:
   471→    from pharmaml.ml.models import list_models
   472→
   473→    entries = list_models(only_available=False)
   474→    print(f"{'model':<22} {'task':<16} {'available':<10} notes")
   475→    print("-" * 78)
   476→    for entry in entries:
   477→        flag = "yes" if entry["available"] else f"needs {entry['requires']}"
   478→        print(f"{entry['name']:<22} {entry['task']:<16} {flag:<10} {entry['notes']}")
   479→    return 0

   482→def _load_cli_dataset(args: argparse.Namespace):
   483→    """Load the ``--data`` file via :mod:`pharmaml.data` with task inference.
   484→
   485→    Both ``ml train`` and ``ml compare`` accept ``--task auto`` (the
   486→    default), which lets :func:`pharmaml.data.load_dataset` decide
   487→    classification (2+ classes) vs regression from the target column,
   488→    and ``--keep-categorical`` to keep string feature columns for
   489→    one-hot encoding.
   490→    """
   491→    from pharmaml.data import load_dataset

   531→def _smi_bridge_frame(path: Path):
   532→    """Read a ``.smi`` file into a SMILES/ID table (CLI-internal bridge).
   533→
   534→    Lets ``--data x.smi --to sdf|smi`` reuse the table converters
   535→    without a temporary CSV on disk (the legacy two-step workflow).
   536→    """
   537→    import pandas as pd
   538→
   539→    from pharmaml.exceptions import DataValidationError
   540→
   541→    rows: list[tuple[str, str]] = []
   542→    with open(path, "r", encoding="utf-8", errors="ignore") as handle:
   543→        for line in handle:
   544→            parts = line.strip().split()
   545→            if not parts:
   546→                continue
   547→            rows.append((parts[0], " ".join(parts[1:])))
   548→    frame = pd.DataFrame(rows, columns=["SMILES", "ID"])
   549→    if frame.empty:
   550→        raise DataValidationError(f"no SMILES lines found in {path}")
   551→    return frame

   554→def _cmd_data_convert(args: argparse.Namespace) -> int:
   555→    from pharmaml.chem.converters import (
   556→        sdf_to_table,
   557→        smi_to_csv,
   558→        table_to_sdf,
   559→        table_to_smi,
   560→    )
   561→    from pharmaml.exceptions import DataValidationError
   562→    from pharmaml.utils.io import read_table, write_table
   563→
   564→    src = Path(args.data)
   565→    target = args.to
   566→    suffix = src.suffix.lower()
   567→    out = Path(args.out) if args.out else src.with_suffix(f".{target}")
   568→
   569→    if suffix == ".smi":
   570→        if target == "csv":
   571→            written = smi_to_csv(src, out)
   572→            print(f"converted {src.name} -> {written}")
   573→            return 0
   574→        if target == "smi":
   575→            raise DataValidationError(
   576→                f"input {src.name} is already a .smi file -- nothing to convert"
   577→            )
   578→        frame = _smi_bridge_frame(src)
   579→        report = table_to_sdf(
   580→            frame,
   581→            out,
   582→            smiles_col=args.smiles_col,
   583→            name_col=args.name_col,
   584→            embed3d=not args.no_3d,
   585→        )
   586→    elif suffix == ".sdf":
   587→        if target == "sdf":
   588→            raise DataValidationError(
   589→                f"input {src.name} is already an SDF file -- use --to csv or --to smi"
   590→            )
   591→        if target == "csv":
   592→            frame = sdf_to_table(src, out=out, smiles_col=args.smiles_col)
   593→            print(f"converted {src.name} -> {out} ({len(frame)} molecules)")
   594→            return 0
   595→        frame = sdf_to_table(src, smiles_col=args.smiles_col)
   596→        name_col = args.name_col or ("name" if "name" in frame.columns else None)
   597→        written = table_to_smi(
   598→            frame, out, smiles_col=args.smiles_col, name_col=name_col
   599→        )
   600→        print(f"converted {src.name} -> {written} ({len(frame)} molecules)")
   601→        return 0
   602→    else:  # CSV/TSV/Excel table input
   603→        if target == "smi":
   604→            written = table_to_smi(
   605→                src, out, smiles_col=args.smiles_col, name_col=args.name_col
   606→            )
   607→            print(f"converted {src.name} -> {written}")
   608→            return 0
   609→        if target == "csv":
   610→            frame = read_table(src)
   611→            written = write_table(frame, out)
   612→            print(f"converted {src.name} -> {written} ({len(frame)} rows)")
   613→            return 0
   614→        report = table_to_sdf(
   615→            src,
   616→            out,
   617→            smiles_col=args.smiles_col,
   618→            name_col=args.name_col,
   619→            embed3d=not args.no_3d,
   620→        )
   621→
   622→    # --to sdf: full audit + output path; invalid SMILES -> exit 1
   623→    print(report)
   624→    print(f"output -> {report.output_sdf}")
   625→    if report.failures:
   626→        print(
   627→            f"{report.failures} invalid SMILES skipped: "
   628→            f"{report.failed_smiles}",
   629→            file=sys.stderr,
   630→        )
   631→        return 1
   632→    return 0

   635→def _train_figure_bundle(result, report, dataset, X_test, y_test):
   636→    """Collect the standard figure/table set for ``ml train --plots``."""
   637→    import pandas as pd
   638→
   639→    from pharmaml.viz import metrics_to_dataframe, ml_plots
   640→
   641→    figures: dict[str, Any] = {}
   642→    tables: dict[str, Any] = {}
   643→
   644→    if dataset.task == "classification":
   645→        figures["confusion_matrix"] = ml_plots.plot_confusion_matrix(report)
   646→        if hasattr(result.model, "predict_proba"):
   647→            figures["roc_curves"] = ml_plots.plot_roc_curves(
   648→                result.model, X_test, y_test
   649→            )
   650→        if dataset.class_counts:
   651→            figures["class_distribution"] = ml_plots.plot_class_distribution(
   652→                dataset.class_counts
   653→            )
   654→            tables["class_counts"] = pd.DataFrame(
   655→                {"class": list(dataset.class_counts),
   656→                 "count": list(dataset.class_counts.values())}
   657→            )
   658→    else:
   659→        y_pred = result.model.predict(X_test)
   660→        figures["predicted_vs_actual"] = ml_plots.plot_predicted_vs_actual(
   661→            y_test, y_pred, metrics=report.as_dict()
   662→        )
   663→        figures["residuals"] = ml_plots.plot_residuals(y_test, y_pred)
   664→
   665→    if result.cv_scores is not None:
   666→        figures["cv_scores"] = ml_plots.plot_cv_scores(result)
   667→    try:
   668→        figures["feature_importance"] = ml_plots.plot_feature_importance(result.model)
   669→    except Exception:  # noqa: BLE001 - estimator exposes no importances
   670→        pass  # e.g. kernel SVM / KNN — importance plot is simply skipped
   671→
   672→    tables.setdefault(
   673→        "held_out_metrics", metrics_to_dataframe(report)
   674→    )
   675→    return figures, tables

   678→def _cmd_ml_train(args: argparse.Namespace) -> int:
   679→    from pharmaml.config import PipelineConfig
   680→    from pharmaml.ml.evaluation import evaluate_model
   681→    from pharmaml.ml.persistence import save_model
   682→    from pharmaml.ml.preprocessing import make_train_test_split
   683→    from pharmaml.ml.training import train_model
   684→
   685→    dataset = _load_cli_dataset(args)
   686→    X_train, X_test, y_train, y_test = make_train_test_split(
   687→        dataset.X, dataset.y, test_size=args.test_size
   688→    )
   689→
   690→    result = train_model(
   691→        args.model, dataset.task, X_train, y_train, tune=not args.no_tune, cv=args.cv
   692→    )
   693→    report = evaluate_model(result.model, X_test, y_test, dataset.task)
   694→
   695→    config = PipelineConfig()
   696→    model_path = config.output_path(f"{result.name}_{dataset.task}.joblib")
   697→    save_model(
   698→        result.model,
   699→        model_path,
   700→        name=result.name,
   701→        task=dataset.task,
   702→        features=dataset.feature_names,
   703→        best_params=result.best_params,
   704→        metrics=report.as_dict(),
   705→    )
   706→    print(f"dataset:  {dataset.n_samples} rows x {dataset.n_features} features")
   707→    if dataset.class_counts:
   708→        print(f"classes:  {dataset.class_counts}")
   709→    print(f"task:     {dataset.task}")
   710→    print(f"model:    {result.name}")
   711→    print(f"best:     {result.best_params or '(defaults)'}")
   712→    print(f"held-out: {report}")
   713→    print(f"saved:    {model_path}")
   714→
   715→    if args.plots:
   716→        from pharmaml.viz import save_visual_bundle
   717→
   718→        figures, tables = _train_figure_bundle(result, report, dataset, X_test, y_test)
   719→        bundle_dir = Path(args.out) / "report"
   720→        written = save_visual_bundle(
   721→            figures,
   722→            tables,
   723→            metrics=report,
   724→            out_dir=bundle_dir,
   725→            title=f"pharmaml ml train — {result.name} ({dataset.task})",
   726→            subtitle=str(args.data),
   727→        )
   728→        print(f"plots:    {bundle_dir / 'report.html'} (+{len(written) - 1} files)")
   729→    return 0
