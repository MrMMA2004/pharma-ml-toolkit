   186→   186→function countFiles(node: TreeNode): number {
   187→   187→  if (node.type === "file") return 1;
   188→   188→  return (node.children ?? []).reduce((acc, child) => acc + countFiles(child), 0);

   191→   191→/** Look up a node in the tree by its toolkit-relative path. */
   192→   192→function findNode(node: TreeNode, path: string): TreeNode | null {
   193→   193→  if (node.path === path) return node;
   194→   194→  for (const child of node.children ?? []) {
   195→   195→    const hit = findNode(child, path);
   196→   196→    if (hit) return hit;
   197→   197→  }
   198→   198→  return null;
   199→   199→}

   260→   260→function collectDirSummary(
   261→   261→  root: TreeNode
   262→   262→): Array<{ name: string; files: number; size: number; note: string }> {
   263→   263→  const notes: Record<string, string> = {
   264→   264→    src: "the pharmaml package (data, chem, ml, pharma, docking, viz, utils + CLI)",
   265→   265→    tests: "1478 pytest cases across all domains",
   266→   266→    examples: "six executed notebooks + five no-output twins + runnable demos",
   267→   267→    ".github": "GitHub Actions CI matrix (Python 3.10–3.12)",
   268→   268→    docs: "module index — auto-generated API reference",
   269→   269→    scratch: "idempotent build & patch scripts (notebooks, docs index)",
   270→   270→  };
   271→   271→  const out: Array<{ name: string; files: number; size: number; note: string }> = [];
   272→   272→  let rootFiles = 0;
   273→   273→  let rootBytes = 0;
   274→   274→  for (const child of root.children ?? []) {
   275→   275→    if (child.type === "dir") {
   276→   276→      out.push({
   277→   277→        name: child.name,
   278→   278→        files: countFiles(child),
   279→   279→        size: child.size,
   280→   280→        note: notes[child.name] ?? "",
   281→   281→      });
   282→   282→    } else {
   283→   283→      rootFiles++;
   284→   284→      rootBytes += child.size;
   285→   285→    }
   286→   286→  }
   287→   287→  if (rootFiles > 0) {
   288→   288→    out.push({
   289→   289→      name: "(root)",
   290→   290→      files: rootFiles,
   291→   291→      size: rootBytes,
   292→   292→      note: "README, pyproject.toml, Dockerfile, requirements.txt, .gitignore",
   293→   293→    });
   294→   294→  }
   295→   295→  return out;
   296→   296→}
