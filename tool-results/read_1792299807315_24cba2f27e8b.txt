  2015→    def _bio_call_sensitive(self) -> None:
  2016→        if not self._require_pharmaml() or not self._require("pandas", "sensitivity calls"):
  2017→            return
  2018→        if self._bio_screen is None:
  2019→            messagebox.showwarning("No screen", "Load a response screen first.")
  2020→            return
  2021→        compound = self._bio_compound_var.get()
  2022→        if not compound:
  2023→            messagebox.showwarning("No compound", "Pick the compound to call.")
  2024→            return
  2025→        try:
  2026→            sensitive_q = parse_float(self._bio_q1.get(), "sensitive quantile",
  2027→                                      minimum=0.0, maximum=1.0)
  2028→            resistant_q = parse_float(self._bio_q2.get(), "resistant quantile",
  2029→                                      minimum=0.0, maximum=1.0)
  2030→            if sensitive_q >= resistant_q:
  2031→                raise ValueError(
  2032→                    "the sensitive quantile must be below the resistant quantile"
  2033→                )
  2034→        except ValueError as exc:
  2035→            messagebox.showerror("Input", str(exc))
  2036→            return
  2037→        lower = bool(self._bio_lower_var.get())
  2038→        metric = self._bio_metric_combo.get()
  2039→
  2040→        def work() -> Any:
  2041→            from pharmaml.biology import call_sensitive_resistant, response_matrix
  2042→
  2043→            matrix = self._bio_matrix
  2044→            built = matrix is None or self._bio_matrix_metric != metric
  2045→            if built:
  2046→                matrix = response_matrix(self._bio_screen, metric)
  2047→            calls = call_sensitive_resistant(
  2048→                matrix[compound], sensitive_q, resistant_q, lower_is_sensitive=lower
  2049→            )
  2050→            return matrix, metric, built, calls, compound
  2051→
  2052→        def render(payload: Any) -> None:
  2053→            matrix, used_metric, built, calls, used_compound = payload
  2054→            if built:
  2055→                self._bio_render_matrix(matrix, used_metric)
  2056→            counts = calls.value_counts()
  2057→            labels = ("sensitive", "intermediate", "resistant", "not tested")
  2058→            summary = " · ".join(f"{int(counts.get(label, 0))} {label}"
  2059→                                 for label in labels)
  2060→            self._bio_calls_summary.configure(
  2061→                text=f"{used_compound}: {summary}", style="Field.TLabel"
  2062→            )
  2063→            self._bio_calls_tree.delete(*self._bio_calls_tree.get_children())
  2064→            rank = {label: index for index, label in enumerate(labels)}
  2065→            for cell_line, call in sorted(
  2066→                calls.items(), key=lambda item: (rank.get(item[1], 99), str(item[0]))
  2067→            ):
  2068→                self._bio_calls_tree.insert("", "end", text=str(cell_line),
  2069→                                            values=(call,))
  2070→
  2071→        self._start_job(f"sensitivity calls ({compound})", work, render)
