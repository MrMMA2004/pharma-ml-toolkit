   112→/** Qualitative label for a Pearson correlation coefficient. */
   113→function corrStrength(r: number): string {
   114→  const a = Math.abs(r);
   115→  if (a >= 0.9) return "very strong";
   116→  if (a >= 0.7) return "strong";
   117→  if (a >= 0.5) return "moderate";
   118→  return "weak";
   119→}

   121→/** Compact "time ago" label for run-history rows (client-side only). */
   122→function timeAgo(iso: string): string {
   123→  const t = Date.parse(iso);
   124→  if (!Number.isFinite(t)) return "";
   125→  const s = Math.max(0, Math.floor((Date.now() - t) / 1000));
   126→  if (s < 60) return `${s}s ago`;
   127→  if (s < 3600) return `${Math.floor(s / 60)}m ago`;
   128→  if (s < 86400) return `${Math.floor(s / 3600)}h ago`;
   129→  return `${Math.floor(s / 86400)}d ago`;
   130→}

   185→    const ys = ySource.map((p) => p.y);
   186→    // Pad the window slightly so edge dots aren't cut by the plot boundary.

   188→
   189→    const inWin = (p: { x: number }) => !zoomed || (p.x >= win[0] && p.x <= win[1]);

   191→      fullX,
   192→      win,
   193→      zoomed,
   194→      xDomain: [win[0] - xPad, win[1] + xPad] as [number, number],
   195→      yDomain: [Math.min(...ys) - yPad, Math.max(...ys) + yPad] as [number, number],
   196→      points: inside,
   197→      cls0: previewClasses ? previewClasses.cls0.filter(inWin) : [],

   203→  /** Export the full deterministic dataset (all rows) as a CSV file. */
   204→  async function downloadCsv() {
   205→    if (!seedValid) {
   206→      setError("Seed must be a non-negative integer (0 – 2147483647).");
   207→      return;
   208→    }
   209→    setCsvLoading(true);
   210→    setError(null);
   211→    const seedNum = Number.parseInt(seed, 10) || 0;
   212→    const fileName = `pharmaml_${task}_n${samples}_seed${seedNum}.csv`;
   213→    try {
   214→      const qs = new URLSearchParams({
   215→        task,
   216→        samples: String(samples),
   217→        noise: String(noise),
   218→        seed: String(seedNum),
   219→      });
   220→      const res = await fetch(`/api/ml/dataset/csv?${qs.toString()}`);
   221→      if (!res.ok) {
   222→        let message = `HTTP ${res.status}`;
   223→        try {
   224→          const data: unknown = await res.json();
   225→          if (typeof data === "object" && data !== null && "error" in data) {
   226→            message = String((data as { error: unknown }).error);
   227→          }
   228→        } catch {
   229→          /* body wasn't JSON — keep the status message */
   230→        }
   231→        throw new Error(message);
   232→      }
   233→      const blob = await res.blob();
   234→      const url = URL.createObjectURL(blob);
   235→      const anchor = document.createElement("a");
   236→      anchor.href = url;
   237→      anchor.download = fileName;
   238→      document.body.appendChild(anchor);
   239→      anchor.click();
   240→      document.body.removeChild(anchor);
   241→      URL.revokeObjectURL(url);
   242→      toast.success("Dataset exported", {
   243→        description: `${fileName} · ${samples} rows · ${fmtBytes(blob.size)}`,
   244→      });
   245→    } catch (err) {
   246→      setError(err instanceof Error ? err.message : "CSV export failed.");
   247→    } finally {
   248→      setCsvLoading(false);
   249→    }
   250→  }

   282→  async function run() {
   283→    if (!seedValid) {
   284→      setError("Seed must be a non-negative integer (0 – 2147483647).");
   285→      return;
   286→    }
   287→    if (task === "regression" && regModel === "ridge" && !alphaValid) {
   288→      setError("Ridge alpha must be a number ≥ 0.");
   289→      return;
   290→    }
   291→    if (task === "classification" && !l2Valid) {
   292→      setError("L2 strength must be a number ≥ 0.");
   293→      return;
   294→    }
   295→    setStatus("running");
   296→    setError(null);
   297→    const body: MLTrainRequest = {
   298→      task,
   299→      samples,
   300→      noise,
   301→      model: task === "regression" ? regModel : "logistic",
   302→      testSize: testSizePct / 100,
   303→      cv: (Number.parseInt(cv, 10) || 0) as 0 | 3 | 5 | 10,
   304→      seed: Number.parseInt(seed, 10) || 0,
   305→    };
   306→    if (task === "regression" && regModel === "ridge") body.alpha = Number.parseFloat(alpha);
   307→    if (task === "classification") body.l2 = Number.parseFloat(l2);
   308→
   309→    try {
   310→      const res = await fetch("/api/ml/train", {
   311→        method: "POST",
   312→        headers: { "Content-Type": "application/json" },
   313→        body: JSON.stringify(body),
   314→      });
   315→      const data: unknown = await res.json();
   316→      if (!res.ok) {
   317→        const message =
   318→          typeof data === "object" && data !== null && "error" in data
   319→            ? `${String((data as { error: unknown }).error)}${
   320→                "detail" in data && Array.isArray((data as { detail: unknown }).detail)
   321→                  ? ` ${((data as { detail: unknown[] }).detail as string[]).join("; ")}`
   322→                  : ""
   323→              }`
   324→            : `HTTP ${res.status}`;
   325→        throw new Error(message);
   326→      }
   327→      const payload = data as MLTrainResponse;
   328→      setResult(payload);
   329→      setStatus("done");
   330→      /* Record the completed run in the history strip (restorable). */
   331→      const key = payload.task === "regression" ? "r2" : "accuracy";
   332→      const keyValue = payload.metrics[key];
   333→      if (typeof keyValue === "number" && Number.isFinite(keyValue)) {
   334→        pushMlRun({
   335→          id: Date.now(),
   336→          at: new Date().toISOString(),
   337→          task: payload.task,
   338→          model: payload.model,
   339→          keyMetric: payload.task === "regression" ? "R²" : "Accuracy",
   340→          keyValue,
   341→          config: { ...mlConfig },
   342→          result: payload,
   343→        });
   344→      }
   345→    } catch (err) {
   346→      setStatus("error");
   347→      const message = err instanceof Error ? err.message : "Training request failed.";
   348→      setError(message);
   349→    }
   350→  }
