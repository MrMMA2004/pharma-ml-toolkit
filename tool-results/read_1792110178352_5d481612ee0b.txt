   161→        assert _band_score(0.31, 0.31, 2.31) == pytest.approx(math.exp(-0.5))

   231→                mol = self._parse(smiles)

   240→        assert objective.score_many([ASPIRIN, "CCO"]) == [
   241→            pytest.approx(13 / 30),
   242→            pytest.approx(3 / 30),
