   125→/** Qualitative label for a Pearson correlation coefficient. */
   126→function corrStrength(r: number): string {
   127→  const a = Math.abs(r);
   128→  if (a >= 0.9) return "very strong";
   129→  if (a >= 0.7) return "strong";
   130→  if (a >= 0.5) return "moderate";
   131→  return "weak";
   132→}

   134→/** Compact "time ago" label for run-history rows (client-side only). */
   135→function timeAgo(iso: string): string {
   136→  const t = Date.parse(iso);
   137→  if (!Number.isFinite(t)) return "";
   138→  const s = Math.max(0, Math.floor((Date.now() - t) / 1000));
   139→  if (s < 60) return `${s}s ago`;
   140→  if (s < 3600) return `${Math.floor(s / 60)}m ago`;
   141→  if (s < 86400) return `${Math.floor(s / 3600)}h ago`;
   142→  return `${Math.floor(s / 86400)}d ago`;
   143→}

   270→  /** Export the full deterministic dataset (all rows) as a CSV file. */
   271→  async function downloadCsv() {
   272→    if (!seedValid) {
   273→      setError("Seed must be a non-negative integer (0 – 2147483647).");
   274→      return;
   275→    }
   276→    setCsvLoading(true);
   277→    setError(null);
   278→    const seedNum = Number.parseInt(seed, 10) || 0;
   279→    const fileName = `pharmaml_${task}_n${samples}_seed${seedNum}.csv`;
   280→    try {
   281→      const qs = new URLSearchParams({
   282→        task,
   283→        samples: String(samples),
   284→        noise: String(noise),
   285→        seed: String(seedNum),
   286→      });
   287→      const res = await fetch(`/api/ml/dataset/csv?${qs.toString()}`);
   288→      if (!res.ok) {
   289→        let message = `HTTP ${res.status}`;
   290→        try {
   291→          const data: unknown = await res.json();
   292→          if (typeof data === "object" && data !== null && "error" in data) {
   293→            message = String((data as { error: unknown }).error);
   294→          }
   295→        } catch {
   296→          /* body wasn't JSON — keep the status message */
   297→        }
   298→        throw new Error(message);
   299→      }
   300→      const blob = await res.blob();
   301→      const url = URL.createObjectURL(blob);
   302→      const anchor = document.createElement("a");
   303→      anchor.href = url;
   304→      anchor.download = fileName;
   305→      document.body.appendChild(anchor);
   306→      anchor.click();
   307→      document.body.removeChild(anchor);
   308→      URL.revokeObjectURL(url);
   309→      toast.success("Dataset exported", {
   310→        description: `${fileName} · ${samples} rows · ${fmtBytes(blob.size)}`,
   311→      });
   312→    } catch (err) {
   313→      setError(err instanceof Error ? err.message : "CSV export failed.");
   314→    } finally {
   315→      setCsvLoading(false);
   316→    }

   347→  /** Load CSV text into the store, reset column picks, drop stale results.
   348→   *  Returns false (with a toast) when the text is unusable. */
   349→  function loadCsv(fileName: string, text: string): boolean {
   350→    if (text.trim().length < 4 || text.length > MAX_CSV_TEXT) {
   351→      toast.error("CSV not loaded", {
   352→        description: "Expected a non-empty file of at most 256 KB.",
   353→      });
   354→      return false;
   355→    }
   356→    const table = parseCsv(text);
   357→    if (table.columns.length === 0 || table.rows.length === 0) {
   358→      toast.error("Could not parse the CSV", {
   359→        description: "Expected a header row followed by data rows.",
   360→      });
   361→      return false;
   362→    }
   363→    setMlCsv({ fileName, text });
   364→    setMlConfig({ featureColumn: "", featureColumn2: "", targetColumn: "" });
   365→    setStatus("idle");
   366→    setResult(null);
   367→    setPreview(null);
   368→    setError(null);
   369→    setXZoom(null);
   370→    toast.success("CSV loaded", {
   371→      description: `${fileName} — ${table.rows.length} rows × ${table.columns.length} columns · ${numericColumnNames(table).length} numeric`,
   372→    });
   373→    return true;
   374→  }

   436→  async function run() {
   437→    if (source === "synthetic" && !seedValid) {
   438→      setError("Seed must be a non-negative integer (0 – 2147483647).");
   439→      return;
   440→    }
   441→    if (source === "csv" && (!mlCsv || !csvUsable)) {
   442→      setError(
   443→        `Load a CSV with at least ${MIN_CSV_ROWS} rows and 2 numeric columns before training.`
   444→      );
   445→      return;
   446→    }
   447→    if (task === "regression" && regModel === "ridge" && !alphaValid) {
   448→      setError("Ridge alpha must be a number ≥ 0.");
   449→      return;
   450→    }
   451→    if (task === "classification" && !l2Valid) {
   452→      setError("L2 strength must be a number ≥ 0.");
   453→      return;
   454→    }
   455→    setStatus("running");
   456→    setError(null);
   457→    const body: MLTrainRequest = {
   458→      task,
   459→      samples,
   460→      noise,
   461→      model: task === "regression" ? regModel : "logistic",
   462→      testSize: testSizePct / 100,
   463→      cv: (Number.parseInt(cv, 10) || 0) as 0 | 3 | 5 | 10,
   464→      seed: Number.parseInt(seed, 10) || 0,
   465→    };
   466→    if (task === "regression" && regModel === "ridge") body.alpha = Number.parseFloat(alpha);
   467→    if (task === "classification") body.l2 = Number.parseFloat(l2);
   468→    if (source === "csv" && mlCsv) {
   469→      body.csv = mlCsv.text;
   470→      body.csvName = mlCsv.fileName;
   471→      if (task === "regression") {
   472→        if (featureColumns.length > 0) body.featureColumns = featureColumns;
   473→        if (targetColumn) body.targetColumn = targetColumn;
   474→      } else {
   475→        if (featureColumn) body.featureColumn = featureColumn;
   476→        if (featureColumn2) body.featureColumn2 = featureColumn2;
   477→        if (targetColumn) body.targetColumn = targetColumn;
   478→      }
   479→    }
   480→
   481→    try {
   482→      const res = await fetch("/api/ml/train", {
   483→        method: "POST",
   484→        headers: { "Content-Type": "application/json" },
   485→        body: JSON.stringify(body),
   486→      });
   487→      const data: unknown = await res.json();
   488→      if (!res.ok) {
   489→        const message =
   490→          typeof data === "object" && data !== null && "error" in data
   491→            ? `${String((data as { error: unknown }).error)}${
   492→                "detail" in data && Array.isArray((data as { detail: unknown }).detail)
   493→                  ? ` ${((data as { detail: unknown[] }).detail as string[]).join("; ")}`
   494→                  : ""
   495→              }`
   496→            : `HTTP ${res.status}`;
   497→        throw new Error(message);
   498→      }
   499→      const payload = data as MLTrainResponse;
   500→      setResult(payload);
   501→      setStatus("done");
   502→      /* Record the completed run in the history strip (restorable). */
   503→      const key = payload.task === "regression" ? "r2" : "accuracy";
   504→      const keyValue = payload.metrics[key];
   505→      if (typeof keyValue === "number" && Number.isFinite(keyValue)) {
   506→        pushMlRun({
   507→          id: Date.now(),
   508→          at: new Date().toISOString(),
   509→          task: payload.task,
   510→          model: payload.model,
   511→          keyMetric: payload.task === "regression" ? "R²" : "Accuracy",
   512→          keyValue,
   513→          config: { ...mlConfig },
   514→          result: payload,
   515→          ...(source === "csv" && mlCsv
   516→            ? {
   517→                csv: {
   518→                  fileName: mlCsv.fileName,
   519→                  featureColumns:
   520→                    task === "regression"
   521→                      ? featureColumns
   522→                      : [featureColumn, featureColumn2].filter(Boolean),
   523→                  targetColumn,
   524→                },
   525→              }
   526→            : {}),
   527→        });
   528→      }
   529→    } catch (err) {
   530→      setStatus("error");
   531→      const message = err instanceof Error ? err.message : "Training request failed.";
   532→      setError(message);
   533→    }
   534→  }
