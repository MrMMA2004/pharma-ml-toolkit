     1→"""Unified model registry.
     2→
     3→The legacy codebase kept one ~150-line script per algorithm (30+ files
     4→of near-duplicate load/preprocess/tune/evaluate boilerplate).  This
     5→module collapses them into a registry: each entry provides the
     6→estimator factory, the task(s) it supports, a tuned default parameter
     7→grid and an availability probe.  One generic trainer then serves every
     8→algorithm.
     9→
    10→Optional backends (XGBoost, LightGBM, CatBoost, PyTorch deep nets) are
    11→imported lazily and reported as unavailable rather than crashing the
    12→registry import.
    13→
    14→Examples
    15→--------
    16→>>> from pharmaml.ml.models import MODEL_REGISTRY
    17→>>> spec = MODEL_REGISTRY["random_forest"]
    18→>>> spec.task
    19→'both'
    20→>>> sorted(spec.param_grid)
    21→['max_depth', 'min_samples_split', 'n_estimators']
    22→>>> spec.available            # sklearn-backed: no optional dependency
    23→True
    24→"""

    72→    """Registry entry for one algorithm family.
    73→
    74→    Attributes
    75→    ----------
    76→    name:
    77→        Registry key (kebab-case, e.g. ``'random_forest'``).
    78→    task:
    79→        ``'classification'``, ``'regression'`` or ``'both'``.

   448→def resolve_model_name(name: str) -> str:
   449→    """Normalise a user-supplied model name to a registry key.
   450→
   451→    Applies case/whitespace/punctuation clean-up, then the
   452→    :data:`MODEL_ALIASES` short-name map, so ``'Random-Forest'``,
   453→    ``'rf'`` and ``'random_forest'`` all resolve to the same key.
   454→
   455→    Raises
   456→    ------
   457→    ModelError
   458→        When the cleaned name is neither a registry key nor an alias.
   459→    """
   460→    lowered = str(name).strip().lower()
   461→    underscored = lowered.replace(" ", "_").replace("-", "_")
   462→    for candidate in (underscored, MODEL_ALIASES.get(lowered), MODEL_ALIASES.get(underscored)):
   463→        if candidate in MODEL_REGISTRY:
   464→            return candidate
   465→    known = ", ".join(sorted(MODEL_REGISTRY))
   466→    raise ModelError(f"Unknown model {name!r}. Registered models: {known}")

   517→def list_models(task: str | None = None, only_available: bool = True) -> list[dict[str, Any]]:
   518→    """List registry entries with availability information.
   519→
   520→    ``task`` filters by kind: ``'classification'``, ``'regression'`` or
   521→    ``'both'`` (models usable for either task); ``None`` (the default)
   522→    lists every entry.
   523→
   524→    Returns a list of dicts with keys ``name``, ``task``, ``available``,
   525→    ``requires`` and ``notes`` -- ready for pretty printing or the CLI.
   526→
   527→    Raises
   528→    ------
   529→    ModelError
   530→        When ``task`` is not one of the accepted kinds.  A typo'd value
   531→        such as ``'clasification'`` used to silently return only the
   532→        dual-task models instead of complaining.
   533→    """
   534→    if task is not None and task not in {"classification", "regression", "both"}:
   535→        raise ModelError(
   536→            f"task must be 'classification', 'regression' or 'both', got {task!r}"
   537→        )
   538→    entries = []
   539→    for spec in MODEL_REGISTRY.values():
   540→        if task and spec.task != "both" and spec.task != task:
   541→            continue
   542→        if only_available and not spec.available:
   543→            continue
   544→        entries.append(
   545→            {
   546→                "name": spec.name,
   547→                "task": spec.task,
   548→                "available": spec.available,
   549→                "requires": spec.requires,
   550→                "notes": spec.notes,
   551→            }
   552→        )
   553→    return entries

   556→def _accepted_param_names(estimator: Any) -> set[str]:
   557→    """Parameter names :func:`build_model` accepts in ``params=``.
   558→
   559→    ``get_params()`` alone under-reports for estimators like CatBoost,
   560→    whose ``get_params()`` returns only the *explicitly set* constructor
   561→    kwargs — every real CatBoost hyperparameter ('iterations', 'depth',
   562→    'learning_rate', ...) would look unknown and be rejected with a
   563→    spurious :class:`ModelError`.  The constructor signature fills the
   564→    gap, but only when ``__init__`` declares no ``**kwargs``: a
   565→    kwargs-catching constructor accepts arbitrary names, so its named
   566→    parameters say nothing about the accepted set (there
   567→    :meth:`set_params` stays the rejector, and :func:`build_model`
   568→    wraps its rejections as the typed error).
   569→    """
   570→    accepted = set(estimator.get_params())
   571→    try:
   572→        parameters = inspect.signature(type(estimator).__init__).parameters
   573→    except (TypeError, ValueError):  # pragma: no cover - exotic constructors
   574→        return accepted
   575→    if any(param.kind is inspect.Parameter.VAR_KEYWORD for param in parameters.values()):
   576→        return accepted
   577→    accepted.update(
   578→        name
   579→        for name, param in parameters.items()
   580→        if name != "self"
   581→        and param.kind
   582→        in (inspect.Parameter.POSITIONAL_OR_KEYWORD, inspect.Parameter.KEYWORD_ONLY)
   583→    )
   584→    return accepted

   587→def build_model(
   588→    name: str,
   589→    task: str,
   590→    params: dict[str, Any] | None = None,
   591→) -> Any:
   592→    """Instantiate an unfitted estimator from the registry.
   593→
   594→    Parameters
   595→    ----------
   596→    name:
   597→        Registry key, e.g. ``'random_forest'``.
   598→    task:
   599→        ``'classification'`` or ``'regression'``.
   600→    params:
   601→        Estimator keyword overrides (e.g. ``{'n_estimators': 500}``).
   602→        Names are validated against ``get_params()`` ∪ the constructor
   603→        signature, so real CatBoost hyperparameters ('iterations',
   604→        'depth', ...) are accepted despite CatBoost's partial
   605→        ``get_params()``; genuinely unknown names raise
   606→        :class:`ModelError`, as do ``set_params`` rejections.
   607→
   608→    Returns
   609→    -------
   610→    sklearn-compatible estimator
   611→
   612→    Raises
   613→    ------
   614→    ModelError
   615→        Unknown name or task; ``DependencyError`` when the backend is
   616→        not installed.
   617→    """
   618→    key = resolve_model_name(name)
   619→    if task not in {"classification", "regression"}:
   620→        raise ModelError(f"task must be 'classification' or 'regression', got {task!r}")
   621→
   622→    spec = MODEL_REGISTRY[key]
   623→    if task != "both" and spec.task != "both" and spec.task != task:
   624→        raise ModelError(f"Model {key!r} supports {spec.task!r}, not {task!r}")
   625→    requires = spec.requires
   626→    # ``requires is None`` entries are always available (see
   627→    # ModelSpec.available), so this branch always has a concrete module name.
   628→    if requires is not None and not spec.available:
   629→        extra = _REQUIREMENT_EXTRAS.get(requires, requires)
   630→        raise DependencyError(
   631→            f"Model {key!r} requires the optional package '{requires}'. "
   632→            f"Install with:  pip install 'pharmaml[{extra}]'"
   633→        )
   634→
   635→    if spec.name in _TASK_FACTORIES:
   636→        estimator = _TASK_FACTORIES[spec.name][task]()
   637→    else:
   638→        factory = spec.factory
   639→        if factory is None:
   640→            # Dual-task entries carry factory=None and are dispatched through
   641→            # _TASK_FACTORIES above; a factory-less entry outside that table
   642→            # would be a registry bug, not a caller error.
   643→            raise ModelError(
   644→                f"Model {key!r} has no estimator factory registered for task {task!r}"
   645→            )
   646→        estimator = factory()
   647→
   648→    if params:
   649→        accepted = _accepted_param_names(estimator)
   650→        unknown = set(params) - accepted
   651→        if unknown:
   652→            raise ModelError(
   653→                f"Unknown parameters {sorted(unknown)} for {key!r}. "
   654→                f"Valid: {sorted(accepted)[:20]}..."
   655→            )
   656→        try:
   657→            estimator.set_params(**params)
   658→        except Exception as exc:
   659→            # Estimators with lenient constructors reject bad *values*
   660→            # (or unknown names on **kwargs classes) inside set_params;
   661→            # surface that as the documented typed error instead of a
   662→            # raw backend exception (ValueError / CatBoostError / ...).
   663→            raise ModelError(
   664→                f"Invalid parameters {sorted(params)} for {key!r}: {exc}"
   665→            ) from exc
   666→    return estimator

   669→def build_stacking(
   670→    task: str,
   671→    base_models: list[str],
   672→    meta_model: str | None = None,
   673→    passthrough: bool = False,
   674→) -> Any:
   675→    """Build a stacking ensemble (upgrades Stacking.py / Stacking reg.py).
   676→
   677→    Parameters
   678→    ----------
   679→    task:
   680→        ``'classification'`` or ``'regression'``.
   681→    base_models:
   682→        Registry names of the level-0 learners.
   683→    meta_model:
   684→        Registry name of the level-1 blender.  Defaults to
   685→        ``'logistic_regression'`` for classification and ``'ridge'``
   686→        for regression.
   687→    passthrough:
   688→        Append original features to meta-features.
   689→
   690→    Raises
   691→    ------
   692→    ModelError
   693→        When ``base_models`` is empty, a member name is unknown, or a
   694→        member does not support ``task``.  Aliases in ``base_models``
   695→        are resolved and duplicates collapsed, so ``['rf',
   696→        'random_forest']`` builds one random-forest learner, not two
   697→        identical ones.
   698→    """
   699→    if not base_models:
   700→        raise ModelError(
   701→            "base_models list is empty -- pass at least one registry name "
   702→            "(see pharmaml.ml.list_models())"
   703→        )
   704→    if meta_model is None:
   705→        meta_model = "logistic_regression" if task == "classification" else "ridge"
   706→    keys = list(dict.fromkeys(resolve_model_name(name) for name in base_models))
   707→    estimators = [(key, build_model(key, task)) for key in keys]
   708→    meta = build_model(meta_model, task)
   709→    if task == "classification":
   710→        return StackingClassifier(
   711→            estimators=estimators, final_estimator=meta, passthrough=passthrough, cv=5
   712→        )
   713→    return StackingRegressor(
   714→        estimators=estimators, final_estimator=meta, passthrough=passthrough, cv=5
   715→    )

   718→def build_voting(
   719→    task: str,
   720→    base_models: list[str],
   721→    voting: str = "soft",
   722→    weights: list[float] | None = None,
   723→) -> Any:
   724→    """Build a voting ensemble (upgrades Voting.py / Voting reg.py).
   725→
   726→    ``voting='soft'`` (probability averaging) requires all members to
   727→    expose ``predict_proba``; ``voting='hard'`` (majority vote) works
   728→    for every estimator and for **any number of classes**.  For
   729→    regression the voting rule is ignored (``VotingRegressor``
   730→    averages predictions), but the value must still be ``'soft'`` or
   731→    ``'hard'``.  Aliases in ``base_models`` are resolved and
   732→    duplicates collapsed, so ``['rf', 'random_forest']`` casts one
   733→    vote, not two.
   734→
   735→    Raises
   736→    ------
   737→    ModelError
   738→        When ``base_models`` is empty, a member name is unknown,
   739→        ``voting`` is neither ``'soft'`` nor ``'hard'``, ``weights``
   740→        does not match the number of unique members, or a soft-voting
   741→        member lacks ``predict_proba``.
   742→    """
   743→    if not base_models:
   744→        raise ModelError(
   745→            "base_models list is empty -- pass at least one registry name "
   746→            "(see pharmaml.ml.list_models())"
   747→        )
   748→    if voting not in {"soft", "hard"}:
   749→        raise ModelError(f"voting must be 'soft' or 'hard', got {voting!r}")
   750→    keys = list(dict.fromkeys(resolve_model_name(name) for name in base_models))
   751→    if weights is not None and len(weights) != len(keys):
   752→        raise ModelError(
   753→            f"weights has {len(weights)} entries for {len(keys)} unique base "
   754→            "model(s) -- alias duplicates are collapsed"
   755→        )
   756→    estimators = [(key, build_model(key, task)) for key in keys]
   757→    if task == "classification":
   758→        if voting == "soft":
   759→            unsupported = [
   760→                key for key, est in estimators if not hasattr(est, "predict_proba")
   761→            ]
   762→            if unsupported:
   763→                raise ModelError(
   764→                    f"voting='soft' needs predict_proba on every member; "
   765→                    f"{unsupported} lack it. Use voting='hard' (works for "
   766→                    "binary and multiclass) or drop those models."
   767→                )
   768→        return VotingClassifier(estimators=estimators, voting=voting, weights=weights)
   769→    return VotingRegressor(estimators=estimators, weights=weights)
