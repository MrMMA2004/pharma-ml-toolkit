     5→
     6→  Level 1  (per offspring)   : crossover vs clone        (replaces the fixed
     7→                               ``rng.random() < crossover_rate`` Bernoulli)
     8→  Level 2  (per recombination): the seven scaffold-hop geometries
     9→                               (bridge / merge / transplant / transplant_path /

    11→                               uniform ``candidates[randrange]`` draw

    19→  * a drawn operator whose own product survives into the final child earns
    20→    reward = (valid_child ? score(child) : 0.0) - mid_parent

    22→    None / fell back) earns the null reward 0.0 - mid_parent

    24→    drawn before the level-1 branch in the current loop, so the baseline is
    25→    available for both arms)

    80→def _scan_candidates_unlocked(parent_a: str, parent_b: str):
    81→    """Replicate the deterministic candidate scan of scaffold_hop_crossover.
    82→
    83→    Returns the candidates list (same order as the original), or None when
    84→    either parent does not parse / is dot-disconnected (the original's early
    85→    None path).
    86→    """
    87→    Chem = sh.require_rdkit()
    88→    mol_a = Chem.MolFromSmiles(parent_a)
    89→    mol_b = Chem.MolFromSmiles(parent_b)
    90→    if mol_a is None or mol_b is None:
    91→        return None
    92→    if len(Chem.GetMolFrags(mol_a)) != 1 or len(Chem.GetMolFrags(mol_b)) != 1:
    93→        return None
    94→
    95→    sig_a = {atom.GetIdx(): sh._atom_signature(atom) for atom in mol_a.GetAtoms()}
    96→    sig_b = {atom.GetIdx(): sh._atom_signature(atom) for atom in mol_b.GetAtoms()}
    97→
    98→    candidates: list[tuple] = []
    99→    for bond_a in mol_a.GetBonds():
   100→        if bond_a.IsInRing() or bond_a.GetBondType() != Chem.BondType.SINGLE:
   101→            continue
   102→        a1, a2 = bond_a.GetBeginAtomIdx(), bond_a.GetEndAtomIdx()
   103→        for bond_b in mol_b.GetBonds():
   104→            if bond_b.IsInRing() or bond_b.GetBondType() != Chem.BondType.SINGLE:
   105→                continue
   106→            b1, b2 = bond_b.GetBeginAtomIdx(), bond_b.GetEndAtomIdx()
   107→            direct = sig_a[a1] == sig_b[b1] and sig_a[a2] == sig_b[b2]
   108→            flipped = sig_a[a1] == sig_b[b2] and sig_a[a2] == sig_b[b1]
   109→            if direct or flipped:
   110→                candidates.append(
   111→                    ("bridge", bond_a.GetIdx(), bond_b.GetIdx(), direct, flipped)
   112→                )
   113→
   114→    ring_bonds_a = [
   115→        bond for bond in mol_a.GetBonds()
   116→        if bond.IsInRing() and bond.GetBondType() == Chem.BondType.SINGLE
   117→    ]
   118→    ring_bonds_b = [
   119→        bond for bond in mol_b.GetBonds()
   120→        if bond.IsInRing() and bond.GetBondType() == Chem.BondType.SINGLE
   121→    ]
   122→    for bond_a in ring_bonds_a:
   123→        a1, a2 = bond_a.GetBeginAtomIdx(), bond_a.GetEndAtomIdx()
   124→        for bond_b in ring_bonds_b:
   125→            b1, b2 = bond_b.GetBeginAtomIdx(), bond_b.GetEndAtomIdx()
   126→            direct = sig_a[a1] == sig_b[b1] and sig_a[a2] == sig_b[b2]
   127→            flipped = sig_a[a1] == sig_b[b2] and sig_a[a2] == sig_b[b1]
   128→            if direct or flipped:
   129→                candidates.append(("merge", bond_a.GetIdx(), bond_b.GetIdx(), direct))
   130→
   131→    for ring_side, other, ring_bonds, sig_ring in (
   132→        ("a", mol_b, ring_bonds_a, sig_a),
   133→        ("b", mol_a, ring_bonds_b, sig_b),
   134→    ):
   135→        for bond in ring_bonds:
   136→            r1, r2 = bond.GetBeginAtomIdx(), bond.GetEndAtomIdx()
   137→            for atom in other.GetAtoms():
   138→                if atom.GetDegree() != 2 or atom.IsInRing():
   139→                    continue
   140→                excision = sh._excise_linker_path(other, (atom.GetIdx(),))
   141→                if excision is None:
   142→                    continue
   143→                _, host_sigs = excision
   144→                direct = host_sigs[0] == sig_ring[r1] and host_sigs[1] == sig_ring[r2]
   145→                flipped = host_sigs[0] == sig_ring[r2] and host_sigs[1] == sig_ring[r1]
   146→                if direct or flipped:
   147→                    candidates.append(
   148→                        ("transplant", ring_side, bond.GetIdx(), atom.GetIdx())
   149→                    )
   150→        if ring_bonds:
   151→            for path, host_sigs in sh._linker_path_units(other):
   152→                for bond in ring_bonds:
   153→                    r1, r2 = bond.GetBeginAtomIdx(), bond.GetEndAtomIdx()
   154→                    direct = host_sigs[0] == sig_ring[r1] and host_sigs[1] == sig_ring[r2]
   155→                    flipped = host_sigs[0] == sig_ring[r2] and host_sigs[1] == sig_ring[r1]
   156→                    if direct or flipped:
   157→                        candidates.append(
   158→                            ("transplant_path", ring_side, bond.GetIdx(), path, direct)
   159→                        )
   160→            for ring, a1, a2, host_sigs in sh._ring_units(other):
   161→                for bond in ring_bonds:
   162→                    r1, r2 = bond.GetBeginAtomIdx(), bond.GetEndAtomIdx()
   163→                    direct = host_sigs[0] == sig_ring[r1] and host_sigs[1] == sig_ring[r2]
   164→                    flipped = host_sigs[0] == sig_ring[r2] and host_sigs[1] == sig_ring[r1]
   165→                    if direct or flipped:
   166→                        candidates.append(
   167→                            ("fuse", ring_side, bond.GetIdx(), ring, a1, a2, direct)
   168→                        )
   169→
   170→    for donor_side, donor, recipient, sig_recipient in (
   171→        ("a", mol_a, mol_b, sig_b),
   172→        ("b", mol_b, mol_a, sig_a),
   173→    ):
   174→        acyclic_bonds = [
   175→            bond for bond in recipient.GetBonds()
   176→            if not bond.IsInRing() and bond.GetBondType() == Chem.BondType.SINGLE
   177→        ]
   178→        if not acyclic_bonds:
   179→            continue
   180→        for ring, a1, a2, host_sigs in sh._ring_units(donor):
   181→            for bond in acyclic_bonds:
   182→                b1, b2 = bond.GetBeginAtomIdx(), bond.GetEndAtomIdx()
   183→                direct = (
   184→                    host_sigs[0] == sig_recipient[b1]
   185→                    and host_sigs[1] == sig_recipient[b2]
   186→                )
   187→                flipped = (
   188→                    host_sigs[0] == sig_recipient[b2]
   189→                    and host_sigs[1] == sig_recipient[b1]
   190→                )
   191→                if direct or flipped:
   192→                    candidates.append(
   193→                        ("insert", donor_side, bond.GetIdx(), ring, a1, a2, direct)
   194→                    )
   195→
   196→    middle_units_a = sh._middle_units(mol_a)
   197→    middle_units_b = sh._middle_units(mol_b)
   198→    for flank_side, flank_units, piece_units in (
   199→        ("a", middle_units_a, middle_units_b),
   200→        ("b", middle_units_b, middle_units_a),
   201→    ):
   202→        for flank_path, flank_sigs in flank_units:
   203→            for piece_path, piece_sigs in piece_units:
   204→                direct = flank_sigs == piece_sigs
   205→                flipped = (
   206→                    flank_sigs[0] == piece_sigs[1]
   207→                    and flank_sigs[1] == piece_sigs[0]
   208→                )
   209→                if direct or flipped:
   210→                    candidates.append(
   211→                        ("swap", flank_side, flank_path, piece_path, direct)
   212→                    )
   213→    return candidates

   216→class RecordingRng:
   217→    """Delegate rng that logs randrange/shuffle calls (scan validation)."""
   218→
   219→    def __init__(self, base: random.Random) -> None:
   220→        self.base = base
   221→        self.randrange_calls: list[tuple] = []
   222→        self.shuffle_calls = 0
   223→
   224→    def randrange(self, *args):
   225→        self.randrange_calls.append(args)
   226→        return self.base.randrange(*args)
   227→
   228→    def shuffle(self, seq):
   229→        self.shuffle_calls += 1
   230→        return self.base.shuffle(seq)

   232→
   233→class RiggedRng:
   234→    """Delegate rng forcing ``randrange(n) -> forced`` when n == target.
   235→
   236→    The original scaffold_hop_crossover consumes exactly one randrange (the
   237→    candidate draw) plus an optional shuffle (bridge geometry); the shuffle is
   238→    delegated to the wrapped GA rng so the TOTAL rng consumption matches the
   239→    production call (``noop_shuffle=True`` freezes the bridge pairing order
   240→    for the scan-consistency validation below).
   241→    """
   242→
   243→    def __init__(self, base: random.Random, forced: int, target: int,
   244→                 *, noop_shuffle: bool = False) -> None:
   245→        self.base = base
   246→        self.forced = forced
   247→        self.target = target
   248→        self.noop_shuffle = noop_shuffle
   249→
   250→    def randrange(self, *args):
   251→        if len(args) == 1 and args[0] == self.target:
   252→            return self.forced
   253→        return self.base.randrange(*args)
   254→
   255→    def shuffle(self, seq):
   256→        if self.noop_shuffle:
   257→            return
   258→        return self.base.shuffle(seq)
   259→
   260→
   261→class NoShuffleRng:
   262→    """Recording rng with a no-op shuffle (validation only).
   263→
   264→    The candidate draw is the function's only 1-arg randrange, so its return
   265→    value IS the drawn candidate index.
   266→    """
   267→
   268→    def __init__(self, base: random.Random) -> None:
   269→        self.base = base
   270→        self.drawn_index: int | None = None
   271→
   272→    def randrange(self, *args):
   273→        value = self.base.randrange(*args)
   274→        if len(args) == 1:
   275→            self.drawn_index = value
   276→        return value
   277→
   278→    def shuffle(self, seq):
   279→        return  # freeze the bridge pairing order (construction order)

   282→def validate_scan_copy(pairs: list[tuple[str, str]], log) -> None:
   283→    """Prove the scan replication matches the original function's pool.
   284→
   285→    Two checks per pair: (1) a recording rng reveals the original's candidate
   286→    pool size -- it must equal ``len(_scan_candidates)``; (2) with the bridge
   287→    shuffle frozen, forcing candidate index i through the ORIGINAL function
   288→    must reproduce exactly the child a plain (shuffle-frozen) draw of index i
   289→    produces -- same index, same child, so the replication is
   290→    order-consistent with the production pool.
   291→    """
   292→    for a, b in pairs:
   293→        mine = _scan_candidates(a, b)
   294→        rec = RecordingRng(random.Random(0))
   295→        child = sh.scaffold_hop_crossover(a, b, rec, max_smiles_length=120)
   296→        if mine is None or not mine:
   297→            assert child is None or rec.randrange_calls == [], (a, b, child)
   298→            log(f"  scan OK: {a[:18]:>18} x {b[:18]:<18} -> no candidates "
   299→                f"(original returns None: {child is None})")
   300→            continue
   301→        assert rec.randrange_calls and rec.randrange_calls[0] == (len(mine),), (
   302→            a, b, rec.randrange_calls, len(mine)
   303→        )
   304→        for seed in range(12):
   305→            plain_rng = NoShuffleRng(random.Random(seed))
   306→            plain = sh.scaffold_hop_crossover(a, b, plain_rng,
   307→                                              max_smiles_length=120)
   308→            forced = sh.scaffold_hop_crossover(
   309→                a, b,
   310→                RiggedRng(random.Random(seed), plain_rng.drawn_index, len(mine),
   311→                          noop_shuffle=True),
   312→                max_smiles_length=120,
   313→            )
   314→            assert plain == forced, (a, b, seed, plain_rng.drawn_index, plain,
   315→                                     forced)
   316→        log(f"  scan OK: {a[:18]:>18} x {b[:18]:<18} -> {len(mine):3d} candidates, "
   317→            f"{len({c[0] for c in mine})} geometries, index-consistent x12")

   320→# ---------------------------------------------------------------------------
   321→# 2. The AOS machinery (the production design, verbatim semantics)
   322→# ---------------------------------------------------------------------------
   323→class OperatorBook:
   324→    """Sliding-window credit + softmax-with-floor selection for one level."""
   325→
   326→    def __init__(self, operators: list[str], window: int, temperature: float,
   327→                 label: str) -> None:
   328→        self.operators = list(operators)
   329→        self.window = window
   330→        self.temperature = temperature
   331→        self.label = label
   332→        self.rewards: dict[str, deque] = {op: deque(maxlen=window) for op in operators}
   333→        self.draws: dict[str, int] = {op: 0 for op in operators}
   334→        self.successes: dict[str, int] = {op: 0 for op in operators}
   335→        self.best: dict[str, tuple[float, str]] = {}
   336→
   337→    def credit(self, op: str) -> float:
   338→        window = self.rewards[op]
   339→        if not window:
   340→            return 0.0  # empty window -> neutral prior (never NaN)
   341→        return sum(window) / len(window)
   342→
   343→    def probabilities(self, present: list[str] | None = None) -> dict[str, float]:
   344→        ops = [op for op in self.operators if present is None or op in present]
   345→        if not ops:
   346→            return {}
   347→        if len(ops) == 1:
   348→            return {ops[0]: 1.0}
   349→        k = len(ops)
   350→        floor = 1.0 / (k * FLOOR_MARGIN)
   351→        credits = [self.credit(op) for op in ops]
   352→        top = max(credits)
   353→        exps = [math.exp((c - top) / self.temperature) for c in credits]
   354→        total = sum(exps)
   355→        return {
   356→            op: floor + (1.0 - k * floor) * (e / total)
   357→            for op, e in zip(ops, exps)
   358→        }
   359→
   360→    def draw(self, rng: random.Random, present: list[str] | None = None) -> str:
   361→        probs = self.probabilities(present)
   362→        ops = list(probs)
   363→        pick = rng.random()
   364→        cumulative = 0.0
   365→        for op in ops:
   366→            cumulative += probs[op]
   367→            if pick < cumulative:
   368→                return op
   369→        return ops[-1]  # float-tail guard (deterministic)
   370→
   371→    def push(self, op: str, reward: float, success: bool,
   372→             child: str | None, child_score: float | None) -> None:
   373→        if op not in self.rewards:  # lazily discovered operator
   374→            self.rewards[op] = deque(maxlen=self.window)
   375→            self.draws[op] = 0
   376→            self.successes[op] = 0
   377→            self.operators.append(op)
   378→        self.rewards[op].append(reward)
   379→        self.draws[op] += 1
   380→        if success:
   381→            self.successes[op] += 1
   382→            if child is not None and (
   383→                op not in self.best or child_score > self.best[op][0]
   384→            ):
   385→                self.best[op] = (child_score, child)
   386→
   387→    def table(self) -> list[dict]:
   388→        probs = self.probabilities(None)
   389→        rows = []
   390→        for op in self.operators:
   391→            credit = self.credit(op)
   392→            assert math.isfinite(credit)  # the NaN guard, asserted per row
   393→            rows.append({
   394→                "level": self.label,
   395→                "operator": op,
   396→                "draws": self.draws[op],
   397→                "successes": self.successes[op],
   398→                "credit": round(credit, 4),
   399→                "final_probability": round(probs.get(op, 0.0), 4),
   400→                "best_offspring": self.best.get(op, ("", None))[1],
   401→                "best_offspring_score": round(self.best.get(op, (0.0, None))[0], 4),
   402→            })
   403→        return rows
   404→

   406→# ---------------------------------------------------------------------------
   407→# 3. The probe GA loop (faithful copy of GeneticOptimizer.optimize, reusing
   408→#    the real optimizer's private methods; AOS hooks at the three levels)
   409→# ---------------------------------------------------------------------------
   410→def probe_ga(
   411→    objective,
   412→    *,
   413→    corpus,
   414→    seed: int = 42,
   415→    population_size: int = 16,
   416→    generations: int = 16,
   417→    mutation: str = "uniform",
   418→    crossover: str = "scaffold_hop",
   419→    operator_selection: str = "uniform",
   420→    aos_window: int = 20,
   421→    aos_temperature: float = 0.1,
   422→    crossover_rate: float = 0.7,
   423→    mutation_rate: float = 0.08,
   424→    elitism: int = 4,
   425→    tournament_k: int = 3,
   426→    max_smiles_length: int = 120,
   427→    fail_geometry: str | None = None,
   428→):
   429→    adaptive = operator_selection == "adaptive"
   430→    opt = GeneticOptimizer(
   431→        objective,
   432→        population_size=population_size,
   433→        generations=generations,
   434→        mutation_rate=mutation_rate,
   435→        crossover_rate=crossover_rate,
   436→        elitism=elitism,
   437→        tournament_k=tournament_k,
   438→        max_smiles_length=max_smiles_length,
   439→        mutation=mutation,
   440→        crossover=crossover,
   441→        seed=seed,
   442→    )
   443→    rng = random.Random(seed)
   444→
   445→    # -- faithful preamble (optimize lines 750-803) ------------------------
   446→    pool = list(corpus)
   447→    canonical_pool = []
   448→    for smiles in pool:
   449→        canonical = canonical_smiles(smiles)
   450→        if canonical is not None and len(canonical) <= max_smiles_length:
   451→            canonical_pool.append(canonical)
   452→    canonical_pool = list(dict.fromkeys(canonical_pool))
   453→    assert canonical_pool
   454→    alphabet = opt._mutation_alphabet(canonical_pool)
   455→    population, _kept = opt._initial_population(None, canonical_pool, rng)
   456→    decoration_pool: list[str] = []
   457→    if mutation == "scaffold":
   458→        decoration_pool = _build_decoration_pool(canonical_pool)
   459→
   460→    # -- AOS state ----------------------------------------------------------
   461→    level1 = OperatorBook(["crossover", "clone"], aos_window, aos_temperature,
   462→                          "level1")
   463→    geometry = OperatorBook(list(GEOMETRY_ORDER), aos_window, aos_temperature,
   464→                            "geometry")
   465→    if mutation == "topology":
   466→        family = OperatorBook(["topology", "character"], aos_window,
   467→                              aos_temperature, "family")
   468→    elif mutation == "scaffold":
   469→        family = OperatorBook(["scaffold", "character"], aos_window,
   470→                              aos_temperature, "family")
   471→    else:
   472→        family = OperatorBook(["character"], aos_window, aos_temperature, "family")
   473→
   474→    produced: dict[str, float] = {}
   475→    history: list[dict] = []
   476→    last_fraction = 1.0
   477→
   478→    def character_mutate(smiles: str) -> str:
   479→        # verbatim copy of the _mutate character block (rng-identical)
   480→        chars = list(smiles)
   481→        n_mutations = max(1, round(len(chars) * opt.mutation_rate))
   482→        for _ in range(n_mutations):
   483→            if not chars:
   484→                break
   485→            position = rng.randrange(len(chars))
   486→            draw = rng.random()
   487→            if draw < 0.45:  # substitute
   488→                chars[position] = alphabet[rng.randrange(len(alphabet))]
   489→            elif draw < 0.8:  # insert
   490→                chars.insert(position, alphabet[rng.randrange(len(alphabet))])
   491→            else:  # delete
   492→                del chars[position]
   493→        return "".join(chars)
   494→
   495→    for generation in range(generations):
   496→        fitness = {s: _validated_score(objective, s) for s in population}
   497→        for smiles, score in fitness.items():
   498→            produced[smiles] = max(produced.get(smiles, 0.0), score)
   499→        ranked = sorted(population, key=lambda s: (-fitness[s], s))
   500→        history.append({
   501→            "generation": generation,
   502→            "best_score": fitness[ranked[0]],
   503→            "mean_score": sum(fitness.values()) / len(fitness),
   504→            "best_smiles": ranked[0],
   505→            "n_unique": len(set(population)),
   506→            "valid_offspring_fraction": last_fraction,
   507→        })
   508→        if generation == generations - 1:
   509→            break
   510→
   511→        next_population = ranked[:elitism]
   512→        attempts = 0
   513→        while len(next_population) < population_size:
   514→            attempts += 1
   515→            if attempts > 50 * population_size:
   516→                while len(next_population) < population_size:
   517→                    next_population.append(
   518→                        canonical_pool[rng.randrange(len(canonical_pool))]
   519→                    )
   520→                break
   521→            parent_a = opt._tournament(population, fitness, rng)
   522→            drawn_l1: str | None = None
   523→            drawn_geometry: str | None = None
   524→            drawn_family: str | None = None
   525→            recombined = False  # level-1 crossover product survived?
   526→            geometry_survived = False
   527→            family_survived = False
   528→            mid_parent = 0.0
   529→            if mutation == "scaffold" and crossover == "splice":
   530→                child = parent_a  # the documented bypass (no level-1 draw)
   531→            else:
   532→                parent_b = opt._tournament(population, fitness, rng)
   533→                mid_parent = (fitness[parent_a] + fitness[parent_b]) / 2.0
   534→                if adaptive:
   535→                    drawn_l1 = level1.draw(rng)
   536→                    use_cross = drawn_l1 == "crossover"
   537→                else:
   538→                    use_cross = rng.random() < opt.crossover_rate
   539→                if use_cross:
   540→                    if crossover == "scaffold_hop":
   541→                        if adaptive:
   542→                            candidates = _scan_candidates(parent_a, parent_b)
   543→                            if not candidates:
   544→                                child = None  # no viable cut: fallback below
   545→                            else:
   546→                                buckets: dict[str, list[int]] = {}
   547→                                for idx, cand in enumerate(candidates):
   548→                                    buckets.setdefault(cand[0], []).append(idx)
   549→                                present = [g for g in GEOMETRY_ORDER
   550→                                           if g in buckets]
   551→                                drawn_geometry = geometry.draw(rng, present)
   552→                                if drawn_geometry == fail_geometry:
   553→                                    child = None  # injected always-fail operator
   554→                                else:
   555→                                    bucket = buckets[drawn_geometry]
   556→                                    idx = bucket[rng.randrange(len(bucket))]
   557→                                    child = sh.scaffold_hop_crossover(
   558→                                        parent_a, parent_b,
   559→                                        RiggedRng(rng, idx, len(candidates)),
   560→                                        max_smiles_length=max_smiles_length,
   561→                                    )
   562→                        else:
   563→                            child = opt._crossover(parent_a, parent_b, rng)
   564→                        if child is not None:
   565→                            recombined = True
   566→                            geometry_survived = True
   567→                        child = child if child is not None else parent_a
   568→                    else:  # splice: inline replica (rng-identical to _crossover)
   569→                        if len(parent_a) < 2 or len(parent_b) < 2:
   570→                            child = parent_a
   571→                        else:
   572→                            cut_a = rng.randrange(1, len(parent_a))
   573→                            cut_b = rng.randrange(1, len(parent_b))
   574→                            spliced = parent_a[:cut_a] + parent_b[cut_b:]
   575→                            if len(spliced) > max_smiles_length:
   576→                                child = parent_a
   577→                            else:
   578→                                child = spliced
   579→                                recombined = True
   580→                                geometry_survived = True
   581→                        if adaptive:
   582→                            drawn_geometry = "splice"
   583→                else:
   584→                    child = parent_a
   585→            # -- mutation stage ------------------------------------------------
   586→            if not adaptive:
   587→                child = opt._mutate(child, alphabet, rng, decoration_pool)
   588→            elif mutation == "topology":
   589→                drawn_family = family.draw(rng)
   590→                if drawn_family == "topology":
   591→                    mutant = ring_topology_mutate(
   592→                        child, rng, max_smiles_length=max_smiles_length
   593→                    )
   594→                    if mutant is not None:
   595→                        child = mutant
   596→                        family_survived = True
   597→                    else:
   598→                        child = character_mutate(child)
   599→                else:
   600→                    child = character_mutate(child)
   601→                    family_survived = True
   602→            else:
   603→                child = character_mutate(child)
   604→                drawn_family = "character"
   605→                family_survived = True
   606→
   607→            # -- the three gates -------------------------------------------------
   608→            valid = False
   609→            score_child: float | None = None
   610→            canonical: str | None = None
   611→            if child and len(child) <= max_smiles_length:
   612→                canonical = canonical_smiles(child)
   613→                if canonical is not None and len(canonical) <= max_smiles_length:
   614→                    valid = True
   615→                    score_child = _validated_score(objective, canonical)
   616→                    if canonical not in next_population:
   617→                        next_population.append(canonical)
   618→
   619→            # -- credit assignment (adaptive arms only) ---------------------------
   620→            if adaptive and drawn_l1 is not None:
   621→                end_to_end = (score_child if valid else 0.0) - mid_parent
   622→                null_reward = 0.0 - mid_parent
   623→                level1.push(drawn_l1,
   624→                            end_to_end if (drawn_l1 == "clone" or recombined)
   625→                            else null_reward,
   626→                            valid, canonical if valid else None, score_child)
   627→                if drawn_geometry is not None:
   628→                    geometry.push(drawn_geometry,
   629→                                  end_to_end if geometry_survived else null_reward,
   630→                                  valid and geometry_survived,
   631→                                  canonical if valid else None, score_child)
   632→                if drawn_family is not None:
   633→                    family.push(drawn_family,
   634→                                end_to_end if family_survived else null_reward,
   635→                                valid and family_survived,
   636→                                canonical if valid else None, score_child)
   637→
   638→        accepted = population_size - elitism
   639→        if attempts:
   640→            last_fraction = accepted / attempts
   641→        history[-1]["valid_offspring_fraction"] = last_fraction
   642→        population = next_population
   643→
   644→    ranked_all = sorted(produced.items(), key=lambda kv: (-kv[1], kv[0]))
   645→    stats = {
   646→        "level1": level1.table(),
   647→        "geometry": geometry.table(),
   648→        "family": family.table(),
   649→    }
   650→    return {
   651→        "smiles": [s for s, _ in ranked_all],
   652→        "scores": [v for _, v in ranked_all],
   653→        "history": history,
   654→        "stats": stats,
   655→        "best_score": ranked_all[0][1] if ranked_all else 0.0,
   656→        "best_smiles": ranked_all[0][0] if ranked_all else "",
   657→    }

   666→def sat_ring_sizes(mol):
   667→    sizes = []
   668→    for ring in mol.GetRingInfo().AtomRings():
   669→        bonds = [
   670→            mol.GetBondBetweenAtoms(ring[i], ring[(i + 1) % len(ring)])
   671→            for i in range(len(ring))
   672→        ]
   673→        if all(
   674→            b.GetBondType() == Chem.BondType.SINGLE and not b.GetIsAromatic()
   675→            for b in bonds
   676→        ):
   677→            sizes.append(len(ring))
   678→    return sizes
   679→
   680→
   681→class RingSizeBandObjective(ScoringObjective):
   682→    name = "ring_size_band"
   683→
   684→    def __init__(self, lo, hi):
   685→        super().__init__()
   686→        self.centre = (lo + hi) / 2.0
   687→        self.sigma = max((hi - lo) / 2.0, 1e-6)
   688→
   689→    def score(self, smiles):
   690→        mol = self._parse(smiles)
   691→        if mol is None:
   692→            return 0.0
   693→        sizes = sat_ring_sizes(mol)
   694→        if not sizes:
   695→            return 0.0
   696→        z = (sum(sizes) / len(sizes) - self.centre) / self.sigma
   697→        return math.exp(-0.5 * z * z)
   698→

   729→def fmt_table(rows: list[dict]) -> str:
   730→    header = (f"{'level':<8} {'operator':<16} {'draws':>5} {'ok':>5} {'rate':>5} "
   731→              f"{'credit':>8} {'p_fin':>6}  best offspring (score)")
   732→    lines = [header, "-" * len(header)]
   733→    for r in rows:
   734→        rate = r["successes"] / r["draws"] if r["draws"] else 0.0
   735→        best = r["best_offspring"] or "-"
   736→        best_s = f" ({r['best_offspring_score']})" if r["best_offspring"] else ""
   737→        if len(best) > 40:
   738→            best = best[:37] + "..."
   739→        lines.append(
   740→            f"{r['level']:<8} {r['operator']:<16} {r['draws']:>5} "
   741→            f"{r['successes']:>5} {rate:>5.2f} {r['credit']:>8.3f} "
   742→            f"{r['final_probability']:>6.3f}  {best}{best_s}"
   743→        )
   744→    return "\n".join(lines)
