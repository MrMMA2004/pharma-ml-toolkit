    38→def atom_line(
    39→    serial: int,
    40→    name: str,
    41→    resname: str,
    42→    chain: str,
    43→    resseq: int,
    44→    x: float,
    45→    y: float,
    46→    z: float,
    47→    element: str,
    48→    record: str = "ATOM  ",
    49→    altloc: str = " ",
    50→) -> str:
    51→    """Column-exact PDB ATOM/HETATM record (altloc at col 17)."""
    52→    return (
    53→        f"{record}{serial:5d} {name:>4s}{altloc}{resname:>3s} {chain}"
    54→        f"{resseq:4d}    {x:8.3f}{y:8.3f}{z:8.3f}"
    55→        f"{1.0:6.2f}{0.0:6.2f}          {element:>2s}"
    56→    )
    57→
    58→
    59→def pdbqt_line(serial: int, name: str, resname: str, chain: str, resseq: int,
    60→               charge: float, atom_type: str) -> str:
    61→    """Column-exact PDBQT record (charge at cols 71-76, type last)."""
    62→    return (
    63→        f"ATOM  {serial:5d} {name:>4s} {resname:>3s} {chain}{resseq:4d}    "
    64→        f"{1.0:8.3f}{2.0:8.3f}{3.0:8.3f}{1.0:6.2f}{0.0:6.2f}"
    65→        f"    {charge:+6.3f}{atom_type:>3s}"
    66→    )

   306→class TestProteinPdbqtPassthrough:
   307→    def _receptor(self, tmp_path: Path, extra: list[str] | None = None) -> Path:
   308→        lines = [
   309→            "REMARK  prepared receptor",
   310→            pdbqt_line(1, "CA", "ALA", "A", 1, 0.24, "C"),
   311→            pdbqt_line(2, "O", "ALA", "A", 1, -0.27, "OA"),
   312→            pdbqt_line(3, "HD", "ALA", "A", 1, 0.42, "HD"),
   313→            "TER",
   314→            "END",
   315→        ]
   316→        if extra:
   317→            lines = lines[:1] + extra + lines[1:]
   318→        path = tmp_path / "r.pdbqt"

   331→        assert report.charge_min == pytest.approx(-0.27)
   332→        assert report.charge_max == pytest.approx(0.42)
