   201→function countFiles(node: TreeNode): number {
   202→  if (node.type === "file") return 1;
   203→  return (node.children ?? []).reduce((acc, child) => acc + countFiles(child), 0);
   204→}

   206→/** Look up a node in the tree by its toolkit-relative path. */
   207→function findNode(node: TreeNode, path: string): TreeNode | null {
   208→  if (node.path === path) return node;
   209→  for (const child of node.children ?? []) {
   210→    const hit = findNode(child, path);
   211→    if (hit) return hit;
   212→  }
   213→  return null;
   214→}

   307→function collectDirSummary(
   308→  root: TreeNode
   309→): Array<{ name: string; files: number; size: number; note: string }> {
   310→  const notes: Record<string, string> = {
   311→    src: "the pharmaml package (data, chem, ml, pharma, docking, viz, utils + CLI)",
   312→    tests: "2136 pytest cases across all domains",
   313→    examples: "eight executed notebooks + seven no-output twins + runnable demos",
   314→    ".github": "GitHub Actions CI matrix (Python 3.10–3.12)",
   315→    docs: "module index — auto-generated API reference",
   316→    scratch: "idempotent build & patch scripts (notebooks, docs index)",
   317→  };
   318→  const out: Array<{ name: string; files: number; size: number; note: string }> = [];
   319→  let rootFiles = 0;
   320→  let rootBytes = 0;
   321→  for (const child of root.children ?? []) {
   322→    if (child.type === "dir") {
   323→      out.push({
   324→        name: child.name,
   325→        files: countFiles(child),
   326→        size: child.size,
   327→        note: notes[child.name] ?? "",
   328→      });
   329→    } else {
   330→      rootFiles++;
   331→      rootBytes += child.size;
   332→    }
   333→  }
   334→  if (rootFiles > 0) {
   335→    out.push({
   336→      name: "(root)",
   337→      files: rootFiles,
   338→      size: rootBytes,
   339→      note: "README, CHANGELOG, VERSION, playground.py, pyproject.toml, Dockerfile, requirements.txt",
   340→    });
   341→  }
   342→  return out;
   343→}

   385→  async function download() {
   386→    setDownloading(true);
   387→    try {
   388→      const res = await fetch("/api/toolkit/download");
   389→      if (!res.ok) {
   390→        let message = `HTTP ${res.status}`;
   391→        try {
   392→          const data: unknown = await res.json();
   393→          if (typeof data === "object" && data !== null && "error" in data) {
   394→            message = String((data as { error: unknown }).error);
   395→          }
   396→        } catch {
   397→          /* body wasn't JSON — keep the status message */
   398→        }
   399→        throw new Error(message);
   400→      }
   401→      const blob = await res.blob();
   402→      if (blob.size < 1024) {
   403→        throw new Error("The archive came back suspiciously small — try again.");
   404→      }
   405→      const url = URL.createObjectURL(blob);
   406→      const anchor = document.createElement("a");
   407→      anchor.href = url;
   408→      anchor.download = zipName;
   409→      document.body.appendChild(anchor);
   410→      anchor.click();
   411→      document.body.removeChild(anchor);
   412→      URL.revokeObjectURL(url);
   413→      toast.success("Download started", {
   414→        description: `${zipName} · ${fmtBytes(blob.size)}`,
   415→      });
   416→    } catch (err) {
   417→      toast.error("Download failed", {
   418→        description: err instanceof Error ? err.message : "Unknown error.",
   419→      });
   420→    } finally {
   421→      setDownloading(false);
