    40→/** Depth-first lookup of a node by exact path (hero notebook metadata). */
    41→function findNode(node: TreeNode, path: string): TreeNode | null {
    42→  if (node.path === path) return node;
    43→  for (const child of node.children ?? []) {
    44→    const hit = findNode(child, path);
    45→    if (hit) return hit;
    46→  }
    47→  return null;
    48→}
